IVN_marches_category = {
	# A paid construction programme. The site is checked again on delivery;
	# without one the price is returned instead of a factory.
	IVN_march_workshops = {
		icon = generic_construction
		allowed = { tag = IVN }
		visible = { has_completed_focus = IVN_march_reconstruction }
		available = {
			custom_trigger_tooltip = {
				tooltip = IVN_march_workshops_limit_tt
				check_variable = { var = IVN_march_workshops_built value = 3 compare = less_than }
			}
			custom_trigger_tooltip = {
				tooltip = IVN_march_workshops_site_tt
				IVN_march_workshop_site_exists = yes
			}
		}
		cost = 50
		days_remove = 60
		days_re_enable = 30
		complete_effect = {
			custom_effect_tooltip = IVN_march_workshops_tt
		}
		remove_effect = {
			if = {
				limit = { IVN_march_workshop_site_exists = yes }
				random_owned_controlled_state = {
					limit = { IVN_march_workshop_site = yes }
					add_building_construction = {
						type = industrial_complex
						level = 1
						instant_build = yes
					}
				}
				add_to_variable = { var = IVN_march_workshops_built value = 1 }
				ADISCORD_economy_mark_dirty = yes
			}
			else = {
				add_political_power = 50
			}
		}
		ai_will_do = {
			base = 10
			modifier = {
				factor = 0
				has_political_power < 100
			}
		}
	}

	# The trained reserve is called up for a chosen moment of the war.
	IVN_reserve_call_up = {
		icon = generic_manpower
		allowed = { tag = IVN }
		visible = { has_completed_focus = IVN_citizen_reserve }
		cost = 25
		days_remove = 45
		days_re_enable = 120
		modifier = {
			training_time_army_factor = -0.10
		}
		remove_effect = {
			add_manpower = 15000
		}
		ai_will_do = {
			base = 2
			modifier = {
				add = 20
				has_war = yes
			}
		}
	}
}
