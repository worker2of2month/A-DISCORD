	# The only Ivanland intervention control exposed to the player. Starting it
	# opens two limited border wars; holding the full corridor before the deadline
	# creates Matvey Mateusk's single mandate. Timing out forces a retreat.
	ADISCORD_ivanland_limited_intervention = {
		icon = generic_border_conflict
		allowed = { tag = IVN }
		visible = {
			has_global_flag = ADISCORD_vorkerland_ivanland_intervention_available
			NOT = { has_global_flag = ADISCORD_vorkerland_ivanland_intervention_resolved }
		}
		available = {
			has_war = no
			IBL = {
				exists = yes
				is_subject = no
				is_neighbor_of = ROOT
			}
			PWR = {
				exists = yes
				is_subject = no
				is_neighbor_of = ROOT
			}
		}
		cost = 0
		selectable_mission = yes
		fire_only_once = yes
		days_mission_timeout = 240
		complete_effect = {
			set_global_flag = ADISCORD_vorkerland_ivanland_intervention_active
			declare_war_on = {
				target = IBL
				type = take_state_focus
				generator = { 91 }
			}
			declare_war_on = {
				target = PWR
				type = take_state_focus
				generator = { 90 }
			}
		}
		cancel_trigger = {
			has_global_flag = ADISCORD_vorkerland_ivanland_intervention_active
			OR = {
				AND = { controls_state = 90 controls_state = 91 }
				AND = {
					has_global_flag = ADISCORD_vorkerland_ivanland_mandate_proclaimed
					IBA = { exists = yes controls_state = 90 controls_state = 91 }
				}
			}
		}
		cancel_effect = { ADISCORD_vorkerland_ivanland_intervention_success = yes }
		timeout_effect = { ADISCORD_vorkerland_ivanland_intervention_failure = yes }
		ai_will_do = { factor = 100 }
	}
