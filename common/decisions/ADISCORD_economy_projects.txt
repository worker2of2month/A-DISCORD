	ADISCORD_economy_precision_tooling = {
		icon = generic_construction
		visible = {
			ADISCORD_economy_projects_unlocked = yes
			OR = {
				has_completed_focus = VAL_econ_precision_industry
				has_completed_focus = STP_pc_investment_precision_industry
			}
			NOT = {
				has_idea = ADISCORD_economy_precision_tooling
			}
		}
		available = {
			custom_trigger_tooltip = { tooltip = ADISCORD_economy_project_country_ready_tt ADISCORD_economy_projects_country_ready = yes }
			custom_trigger_tooltip = { tooltip = ADISCORD_economy_project_slot_available_tt NOT = { has_variable = ADISCORD_economy_project_id } }
			NOT = { num_of_available_civilian_factories < 3 }
		}
		cost = 0
		custom_cost_trigger = {
			check_variable = { var = ADISCORD_economy_treasury value = 800 compare = greater_than_or_equals }
			check_variable = { var = resource@rare_components value = 6 compare = greater_than_or_equals }
			check_variable = { var = resource@rare_alloys value = 3 compare = greater_than_or_equals }
		}
		custom_cost_text = ADISCORD_economy_precision_tooling_cost
		days_remove = 90
		days_re_enable = 90
		complete_effect = {
			custom_effect_tooltip = ADISCORD_economy_precision_tooling_delivery_tt
			effect_tooltip = { add_ideas = ADISCORD_economy_precision_tooling }
			hidden_effect = { ADISCORD_economy_start_precision_tooling = yes }
		}
		cancel_trigger = {
			OR = {
				NOT = { check_variable = { var = ADISCORD_economy_project_id value = 1 compare = equals } }
				ADISCORD_economy_project_can_continue = no
				ADISCORD_economy_precision_tooling_requirements = no
			}
		}
		cancel_effect = {
			if = {
				limit = { check_variable = { var = ADISCORD_economy_project_id value = 1 compare = equals } }
				custom_effect_tooltip = ADISCORD_economy_project_cancelled_tt
				hidden_effect = { ADISCORD_economy_cancel_project = yes }
			}
		}
		remove_effect = { ADISCORD_economy_finish_precision_tooling = yes }
		ai_will_do = {
			base = 4
			modifier = { factor = 0 ADISCORD_economy_ai_is_healthy = no }
			modifier = { factor = 0 NOT = { check_variable = { var = ADISCORD_economy_treasury value = 1100 compare = greater_than_or_equals } } }
		}
	}
