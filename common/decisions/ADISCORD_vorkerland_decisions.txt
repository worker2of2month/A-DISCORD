	# The only route by which a regional or district winner reaches the central
	# war. It is gated on the local bracket actually being cleared, and it splices
	# the winner into the host's existing war through faction membership plus
	# add_to_war rather than opening a second war object. One targeted decision
	# covers all three claimants instead of a copied decision per host.
ADISCORD_vorkerland_join_claimant_coalition = {
		icon = generic_diplomatic_treaty
		allowed = {
			OR = {
				ADISCORD_vorkerland_is_central_bracket_member = yes
				ADISCORD_vorkerland_is_regional_combatant = yes
			}
		}
		# Only the three claimants are ever candidate hosts, so the target list is
		# closed and the expensive host trigger runs at most three times. The root
		# gate keeps it from running at all outside the collapse window.
		targets = { WKR VAD TVA }
		target_root_trigger = {
			has_global_flag = ADISCORD_vorkerland_collapse_wars_started
			NOT = { has_global_flag = ADISCORD_vorkerland_collapse_finished }
			NOT = { has_country_flag = ADISCORD_vorkerland_regional_auxiliary }
			NOT = { has_country_flag = ADISCORD_vorkerland_coalition_join_paid }
		}
target_trigger = { FROM = { ADISCORD_vorkerland_is_main_claimant = yes } }
		visible = {
			is_subject = no
			is_in_faction = no
			NOT = { has_country_flag = ADISCORD_vorkerland_regional_auxiliary }
			NOT = { has_country_flag = ADISCORD_vorkerland_coalition_join_paid }
		}
		available = {
			has_war = no
			NOT = { has_capitulated = yes }
			capital_scope = { is_controlled_by = ROOT }
			ADISCORD_vorkerland_has_live_local_rival = no
			FROM = { ADISCORD_vorkerland_is_coalition_host_for_ROOT = yes }
		}
		cost = 25

		fire_only_once = no
		days_re_enable = 7
		complete_effect = {
			custom_effect_tooltip = ADISCORD_vorkerland_join_claimant_coalition_tt
			hidden_effect = {
				set_country_flag = ADISCORD_vorkerland_coalition_join_paid
				ADISCORD_vorkerland_join_coalition_of_FROM = yes
			}
		}
		ai_will_do = {
			base = 200
			modifier = { factor = 3 has_country_flag = ADISCORD_vorkerland_compact_supply_received FROM = { tag = WKR } }
			modifier = { factor = 0 ADISCORD_vorkerland_has_live_local_rival = yes }
			modifier = { factor = 0 FROM = { tag = VAD } }
			modifier = { factor = 2 num_divisions > 8 }
		}
	}

	# A victorious and independent Frealor may open a limited northern operation.
	# It receives the corridor only after physically controlling all three states;
	# a timeout or the end of the central war closes both wars without transfers.
	ADISCORD_vorkerland_rom_northern_intervention = {
		icon = generic_border_conflict
		allowed = { tag = ROM }
		# Keep this player/AI-selected mission from auto-starting on a transient
		# availability tick; its click remains the only 240-day timer authority.
		activation = { always = no }
		visible = {
			ADISCORD_vorkerland_rom_northern_intervention_start_ready = yes
			NOT = { has_global_flag = ADISCORD_vorkerland_central_war_finished }
			NOT = { has_global_flag = ADISCORD_vorkerland_rom_northern_intervention_resolved }
		}
		available = {
			ADISCORD_vorkerland_rom_northern_intervention_start_ready = yes
			is_subject = no
			has_war = no
			ZAO = { exists = yes is_subject = no controls_state = 72 }
			WPS = { exists = yes is_subject = no controls_state = 196 controls_state = 322 }
		}
		cost = 50
		selectable_mission = yes
		fire_only_once = yes
		days_mission_timeout = 240
		complete_effect = {
			set_global_flag = ADISCORD_vorkerland_rom_northern_intervention_active
			clr_global_flag = ADISCORD_vorkerland_rom_northern_intervention_resolved
			clr_global_flag = ADISCORD_vorkerland_rom_northern_intervention_succeeded
			clr_global_flag = ADISCORD_vorkerland_rom_northern_intervention_failed
			declare_war_on = {
				target = ZAO
				type = take_state_focus
				generator = { 72 }
			}
			declare_war_on = {
				target = WPS
				type = take_state_focus
				generator = { 196 322 }
			}
			# The mission UI is not the lifecycle authority. A one-day assertion catches
			# rejected declarations, while the bounded watchdog guarantees a terminal
			# outcome even if Clausewitz drops the selectable-mission timeout effect.
			set_country_flag = { flag = ADISCORD_vorkerland_rom_northern_watchdog_scheduled days = 241 }
			country_event = { id = ADISCORD_vorkerland_collapse.46 days = 1 }
			country_event = { id = ADISCORD_vorkerland_collapse.45 days = 240 }
		}
		cancel_trigger = {
			OR = {
				AND = { controls_state = 72 controls_state = 196 controls_state = 322 }
				has_global_flag = ADISCORD_vorkerland_central_war_finished
			}
		}
		cancel_effect = {
			if = {
				limit = { controls_state = 72 controls_state = 196 controls_state = 322 }
				ADISCORD_vorkerland_rom_northern_intervention_success = yes
			}
			else = { ADISCORD_vorkerland_rom_northern_intervention_failure = yes }
		}
		timeout_effect = { ADISCORD_vorkerland_resolve_rom_northern_intervention_timeout = yes }
		ai_will_do = { factor = 225 }
	}
