		always = no
	}

NOD_crisis_posture = {
	allowed = {
		tag = NOD
	}

	visible = {
		STP = {
			exists = yes
			has_country_flag = STP_main_campaign_side
			NOT = { has_country_flag = STP_internal_outcome_finalized }
		}
	}
}

STP_VAL_war_countdown_category = {
	allowed = {
		OR = {
			tag = VAL
			STP_is_postwar_country = yes
		}
	}

	visible = {
		OR = {
			has_active_mission = STP_VAL_war_countdown_120
			has_active_mission = STP_VAL_war_countdown_180
			has_active_mission = STP_VAL_war_countdown_300
			has_active_mission = STP_VAL_war_countdown_450
			has_active_mission = STP_VAL_war_countdown_breached
		}
	}
}
