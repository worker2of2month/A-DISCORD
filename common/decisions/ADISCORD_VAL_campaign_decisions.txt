		icon = generic_production

		cost = 10
		days_remove = 21

		complete_effect = {
			ADISCORD_economy_spend_50 = yes
			set_country_flag = VAL_final_preparation_active
			set_country_flag = VAL_preparation_contractor_depot_active
		}
		remove_effect = {
			clr_country_flag = VAL_final_preparation_active
			clr_country_flag = VAL_preparation_contractor_depot_active
			if = {
				limit = { NOT = { has_country_flag = VAL_final_contract_war_started } }
				set_country_flag = VAL_asset_contractor_depot
			}
		}
		cancel_effect = {
			clr_country_flag = VAL_final_preparation_active
			clr_country_flag = VAL_preparation_contractor_depot_active
		}
		ai_will_do = {
			base = 5
			modifier = { factor = 0 NOT = { ADISCORD_economy_can_spend_50 = yes } }
		}
	}
