STP_hegemony_administration = {
	allowed = { tag = STS }
	visible = {
		has_country_flag = STP_cw_postwar
		has_completed_focus = STP_pc_heg_administrations
		check_variable = { var = STP_pc_course value = 1 compare = equals }
	}

	STP_heg_establish_nod_administration = {
		icon = generic_political_pressure
		visible = {
			NOT = { has_country_flag = STP_heg_nod_integrated }
			OR = {
				NOD = { exists = yes }
				any_owned_state = { is_core_of = NOD }
			}
			NOT = {
				NOD = {
					is_subject_of = STS
					has_autonomy_state = autonomy_STP_provisional_administration
				}
			}
		}
		available = { STP_heg_nod_administration_available = yes }
		cost = 75
		fire_only_once = no
		complete_effect = {
			STP_heg_prepare_nod_administration = yes
		}
		ai_will_do = { base = 8 }
	}
	STP_heg_annex_nod_administration = {
		icon = generic_nationalism
		visible = {
			NOD = {
				exists = yes
				is_subject_of = STS
				OR = {
					has_autonomy_state = autonomy_STP_provisional_administration
					has_autonomy_state = autonomy_puppet
				}
			}
		}
		available = {
			has_war = no
			NOD = {
				has_war = no
				is_subject_of = STS
				OR = {
					has_autonomy_state = autonomy_STP_provisional_administration
					has_autonomy_state = autonomy_puppet
				}
			}
			NOT = { has_country_flag = STP_heg_annex_nod_pending }
		}
		cost = 100
		days_remove = 90
		days_re_enable = 1
		cancel_if_not_visible = yes
		complete_effect = { set_country_flag = STP_heg_annex_nod_pending }
		remove_effect = {
			if = {
				limit = {
					has_country_flag = STP_heg_annex_nod_pending
					NOD = {
						exists = yes
						is_subject_of = STS
						OR = {
							has_autonomy_state = autonomy_STP_provisional_administration
							has_autonomy_state = autonomy_puppet
						}
						has_war = no
					}
				}
				clr_country_flag = STP_heg_annex_nod_pending
				set_country_flag = STP_heg_nod_integrated
				annex_country = { target = NOD transfer_troops = yes }
				add_stability = 0.01
				custom_effect_tooltip = STP_heg_annex_complete_tt
				hidden_effect = { ADISCORD_economy_mark_dirty = yes }
			}
			else = {
				clr_country_flag = STP_heg_annex_nod_pending
				add_political_power = 100
			}
		}
		cancel_effect = {
			if = {
				limit = { has_country_flag = STP_heg_annex_nod_pending }
				clr_country_flag = STP_heg_annex_nod_pending
				add_political_power = 100
			}
		}
		ai_will_do = { base = 4 }
	}
