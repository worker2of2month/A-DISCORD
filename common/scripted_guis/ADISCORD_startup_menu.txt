scripted_gui = {
	ADISCORDStartupLauncher = {
		context_type = player_context
		parent_window_name = "countrypoliticsview"
		window_name = "ADISCORD_startup_launcher"
		ai_enabled = { always = no }
		visible = { is_ai = no OR = { tag = STP tag = VAL } }
		effects = {
			ADISCORD_startup_open_click = {
				set_variable = { var = ADISCORD_startup_open value = 1 }
				clear_variable = ADISCORD_startup_tab
				clear_variable = ADISCORD_startup_page
				clear_variable = ADISCORD_startup_spoilers
			}
		}
	}

	ADISCORDStartupMenu = {
		context_type = player_context
		window_name = "ADISCORD_startup_window"
		ai_enabled = { always = no }
		visible = { is_ai = no OR = { tag = STP tag = VAL } check_variable = { var = ADISCORD_startup_open value = 1 compare = equals } }
		effects = {
			ADISCORD_startup_close_click = { clear_variable = ADISCORD_startup_open }
			ADISCORD_startup_play_click = { clear_variable = ADISCORD_startup_open }
			ADISCORD_startup_country_click = {
				set_variable = { var = ADISCORD_startup_tab value = 0 }
				clear_variable = ADISCORD_startup_page
				clear_variable = ADISCORD_startup_spoilers
			}
			ADISCORD_startup_guide_click = {
				set_variable = { var = ADISCORD_startup_tab value = 1 }
				clear_variable = ADISCORD_startup_page
				clear_variable = ADISCORD_startup_spoilers
			}
			ADISCORD_startup_paths_click = {
				set_variable = { var = ADISCORD_startup_tab value = 2 }
				clear_variable = ADISCORD_startup_page
				clear_variable = ADISCORD_startup_spoilers
			}
			ADISCORD_startup_page_0_click = {
				set_variable = { var = ADISCORD_startup_page value = 0 }
				clear_variable = ADISCORD_startup_spoilers
			}
			ADISCORD_startup_page_1_click = {
				set_variable = { var = ADISCORD_startup_page value = 1 }
				clear_variable = ADISCORD_startup_spoilers
			}
			ADISCORD_startup_page_2_click = {
				set_variable = { var = ADISCORD_startup_page value = 2 }
				clear_variable = ADISCORD_startup_spoilers
			}
			ADISCORD_startup_page_3_click = {
				set_variable = { var = ADISCORD_startup_page value = 3 }
				clear_variable = ADISCORD_startup_spoilers
			}
			ADISCORD_startup_spoilers_click = {
				if = { limit = { check_variable = { var = ADISCORD_startup_spoilers value = 1 compare = equals } } clear_variable = ADISCORD_startup_spoilers }
				else = { set_variable = { var = ADISCORD_startup_spoilers value = 1 } }
			}
		}
		triggers = {
			ADISCORD_startup_country_click_enabled = { NOT = { check_variable = { var = ADISCORD_startup_tab value = 0 compare = equals } } }
			ADISCORD_startup_guide_click_enabled = { NOT = { check_variable = { var = ADISCORD_startup_tab value = 1 compare = equals } } }
			ADISCORD_startup_paths_click_enabled = { NOT = { check_variable = { var = ADISCORD_startup_tab value = 2 compare = equals } } }
			ADISCORD_startup_page_0_visible = { OR = { check_variable = { var = ADISCORD_startup_tab value = 1 compare = equals } check_variable = { var = ADISCORD_startup_tab value = 2 compare = equals } } }
			ADISCORD_startup_page_0_click_enabled = { NOT = { check_variable = { var = ADISCORD_startup_page value = 0 compare = equals } } }
			ADISCORD_startup_page_1_visible = { OR = { check_variable = { var = ADISCORD_startup_tab value = 1 compare = equals } check_variable = { var = ADISCORD_startup_tab value = 2 compare = equals } } }
			ADISCORD_startup_page_1_click_enabled = { NOT = { check_variable = { var = ADISCORD_startup_page value = 1 compare = equals } } }
			ADISCORD_startup_page_2_visible = { check_variable = { var = ADISCORD_startup_tab value = 1 compare = equals } }
			ADISCORD_startup_page_2_click_enabled = { NOT = { check_variable = { var = ADISCORD_startup_page value = 2 compare = equals } } }
			ADISCORD_startup_page_3_visible = { check_variable = { var = ADISCORD_startup_tab value = 1 compare = equals } }
			ADISCORD_startup_page_3_click_enabled = { NOT = { check_variable = { var = ADISCORD_startup_page value = 3 compare = equals } } }
			ADISCORD_startup_spoilers_visible = { check_variable = { var = ADISCORD_startup_tab value = 2 compare = equals } }
			ADISCORD_startup_stp_picture_visible = { tag = STP check_variable = { var = ADISCORD_startup_tab value = 0 compare = equals } }
			ADISCORD_startup_val_picture_visible = { tag = VAL check_variable = { var = ADISCORD_startup_tab value = 0 compare = equals } }
			ADISCORD_startup_country_body_visible = { check_variable = { var = ADISCORD_startup_tab value = 0 compare = equals } }
			ADISCORD_startup_body_visible = { NOT = { check_variable = { var = ADISCORD_startup_tab value = 0 compare = equals } } }
			ADISCORD_startup_intro_note_visible = { check_variable = { var = ADISCORD_startup_tab value = 0 compare = equals } }
			ADISCORD_startup_guide_note_visible = { check_variable = { var = ADISCORD_startup_tab value = 1 compare = equals } }
			ADISCORD_startup_paths_note_visible = { check_variable = { var = ADISCORD_startup_tab value = 2 compare = equals } }
		}
	}
}
