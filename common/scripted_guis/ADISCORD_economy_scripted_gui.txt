		parent_window_token = top_bar

		# Observer mode may inspect and open the dashboard; mutation buttons are
		# separately gated to the human-controlled country below.
		visible = { always = yes }
		triggers = {
			ADISCORD_economy_topbar_button_visible = {
				# The native toolbar removes one 55px slot without the intelligence DLC.
				set_temp_variable = { var = ADISCORD_economy_topbar_button_x value = 0 }

					limit = { NOT = { has_dlc = "La Resistance" } }
					set_temp_variable = { var = ADISCORD_economy_topbar_button_x value = -55 }
				}
				always = yes
			}
		}
		properties = {

		}
		effects = {
			ADISCORD_economy_topbar_button_click = { ADISCORD_economy_open_window = yes }
		}
