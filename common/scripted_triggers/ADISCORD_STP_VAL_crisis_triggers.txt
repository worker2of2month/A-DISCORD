	has_country_flag = STP_capital_guard_loyal_to_resistance
	check_variable = { var = STP_node_officers value = 2 compare = greater_than_or_equals }
	check_variable = { var = STP_node_market value = 2 compare = equals }
	check_variable = { var = STP_resistance_readiness value = 85 compare = greater_than_or_equals }
	check_variable = { var = STP_party_suspicion value = 20 compare = less_than_or_equals }

}

		check_variable = { var = STP_node_officers value = 0 compare = greater_than }
		check_variable = { var = STP_node_mountains value = 0 compare = greater_than }
		check_variable = { var = STP_node_market value = 0 compare = greater_than }
		check_variable = { var = STP_node_street value = 0 compare = greater_than }

	OR = {

}
