VAL_has_active_stp_civil_war = {
	has_event_target = STP_crisis_party_side
	has_event_target = STP_crisis_resistance_side
	country_exists = event_target:STP_crisis_party_side
	country_exists = event_target:STP_crisis_resistance_side
	event_target:STP_crisis_party_side = {
		has_country_flag = STP_crisis_party_side
		has_war_with = event_target:STP_crisis_resistance_side
	}
	event_target:STP_crisis_resistance_side = {
		has_country_flag = STP_crisis_resistance_side
	}
}

	VAL_has_active_stp_civil_war = yes
	event_target:STP_crisis_party_side = { surrender_progress < 0.35 }
	event_target:STP_crisis_resistance_side = { surrender_progress < 0.35 }

	num_equipment@infantry_equipment > 1499

VAL_can_start_northern_campaign = {
	tag = VAL
	NOT = { has_country_flag = VAL_northern_campaign_attempted }
	NOT = { has_country_flag = VAL_northern_campaign_active }
	NOT = { has_country_flag = VAL_STP_final_warning_active }
	NOT = { has_country_flag = VAL_STP_final_war_active }
	OR = {
		VAL_northern_full_mode_is_eligible = yes
		VAL_northern_partial_cin_is_eligible = yes
		VAL_northern_partial_osf_is_eligible = yes
	}
	OR = {
		AND = {
			is_ai = no
			OR = {
				VAL_northern_balanced_civil_war_human = yes
				VAL_northern_fresh_long_deal = yes
			}
		}
		AND = {
			is_ai = yes
			NOT = { has_country_flag = VAL_STP_countdown_120 }
			NOT = { has_country_flag = VAL_STP_countdown_180 }
			OR = {
				VAL_northern_balanced_civil_war_ai = yes
				VAL_northern_fresh_long_deal = yes
			}
		}
	}
}
