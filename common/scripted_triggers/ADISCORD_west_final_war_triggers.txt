# Western final war: reunified Vorkerland against Itora.
# Side numbers: 1 = Vorkerland (WRK), 2 = Itora (IVN).

# GLOBAL. Only a verified reunified WRK can claim the west.
ADISCORD_west_wrk_reunified = {
	has_global_flag = ADISCORD_vorkerland_collapse_finished
	WRK = {
		exists = yes
		has_country_flag = ADISCORD_vorkerland_central_unifier
	}
}

# GLOBAL. The western clock runs from reunification to the final campaign.
ADISCORD_west_crisis_open = {
	NOT = { has_global_flag = RUS_crisis_world_ended }

	has_global_flag = ADISCORD_fresh_campaign_contract_v1
	ADISCORD_west_wrk_reunified = yes
	NOT = { has_global_flag = ADISCORD_west_final_started }
	NOT = { has_global_flag = ADISCORD_west_final_resolved }
}

ADISCORD_west_final_active = {
	has_global_flag = ADISCORD_west_final_started
	NOT = { has_global_flag = ADISCORD_west_final_resolved }
}

# COUNTRY. A sovereign country able to lead one side of the final war.
ADISCORD_west_champion_viable = {
	exists = yes
	is_subject = no
	has_capitulated = no
}

# GLOBAL. The existing completion flag dates the verified reunification.
ADISCORD_west_reconstruction_grace_elapsed = {
	has_global_flag = {
		flag = ADISCORD_vorkerland_collapse_finished
		days > 179
	}
}

# GLOBAL. The Itoran intervention keeps its own settlement; the final campaign waits for it.
ADISCORD_west_final_can_start = {
	ADISCORD_west_crisis_open = yes
	ADISCORD_west_reconstruction_grace_elapsed = yes
	WRK = { ADISCORD_west_champion_viable = yes }
	IVN = {
		ADISCORD_west_champion_viable = yes
		NOT = { has_war_with = WRK }
	}
	NOT = { has_global_flag = ADISCORD_vorkerland_ivanland_intervention_active }
	# Itora's revolt of the marches pauses the western question for both sides.
	NOT = { has_global_flag = IVN_marches_revolt_active }
}

# COUNTRY.
ADISCORD_west_on_wrk_side = {
	check_variable = { var = ADISCORD_west_side value = 1 compare = equals }
}

ADISCORD_west_on_ivn_side = {
	check_variable = { var = ADISCORD_west_side value = 2 compare = equals }
}

# COUNTRY. One of the western states that must take a side.
ADISCORD_west_regional_country = {
	OR = {
		tag = WIT
		tag = IIA
		tag = IBA
		tag = IBL
		tag = PWR
		tag = PSD
		tag = ZAO
		tag = WPA
		tag = WPS
		tag = ROM
		tag = TRU
	}
}

# COUNTRY. Regional faction followers enter with their leader instead of
# choosing alone; a member of an outside faction still chooses for itself.
ADISCORD_west_needs_side_choice = {
	exists = yes
	is_subject = no
	NOT = { has_variable = ADISCORD_west_side }
	OR = {
		is_in_faction = no
		is_faction_leader = yes
		faction_leader = { NOT = { ADISCORD_west_regional_country = yes } }
	}
}

# GLOBAL. Either champion has lost its sovereignty or the two no longer fight.
ADISCORD_west_final_broken = {
	OR = {
		NOT = { WRK = { ADISCORD_west_champion_viable = yes } }
		NOT = { IVN = { ADISCORD_west_champion_viable = yes } }
		NOT = { WRK = { has_war_with = IVN } }
	}
}

# STATE. Controlled by a participant of the given side.
ADISCORD_west_state_held_by_wrk_side = {
	controller = { ADISCORD_west_on_wrk_side = yes }
}

ADISCORD_west_state_held_by_ivn_side = {
	controller = { ADISCORD_west_on_ivn_side = yes }
}

# COUNTRY IVN. The three Itoran courses of the western question.
ADISCORD_west_ivn_shield_course = {
	has_completed_focus = IVN_humanist_shield
}

ADISCORD_west_ivn_emergency_course = {
	has_completed_focus = IVN_emergency_order
}

ADISCORD_west_ivn_revanche_course = {
	has_completed_focus = IVN_revanche_of_the_marches
}

# COUNTRY invited state; ROOT is the inviting champion. The pending flag is
# shared, so one state never weighs offers from both champions at once.
ADISCORD_west_league_target_valid = {
	exists = yes
	is_subject = no
	has_capitulated = no
	is_in_faction = no
	NOT = { has_war_with = ROOT }
	NOT = { has_country_flag = ADISCORD_west_league_offer_pending }
	NOT = { has_global_flag = ADISCORD_west_final_started }
}

# COUNTRY WRK. A postwar political programme opens Vorkerland's own treaty
# offers, the counterpart of Itora's League invitations.
ADISCORD_west_wrk_treaties_unlocked = {
	OR = {
		has_completed_focus = WRK_worker_west_mandate_offices
		has_completed_focus = WRK_joint_pw_territorial_council
		has_completed_focus = WRK_utilitarian_west_corridor_survey
		has_completed_focus = WRK_empire_military_conventions
	}
}

# ----- Influence over the western states -----
# ADISCORD_west_lean: +100 Vorkerland, -100 Itora.

# GLOBAL. Champions court the West from reunification until the settlement.
ADISCORD_west_influence_open = {
	OR = {
		ADISCORD_west_crisis_open = yes
		ADISCORD_west_final_active = yes
	}
}

# COUNTRY regional state; ROOT is the acting champion. A state already on a
# side of the war no longer weighs offers.
ADISCORD_west_influence_target_valid = {
	exists = yes
	is_subject = no
	NOT = { has_variable = ADISCORD_west_side }
	NOT = { has_war_with = ROOT }
}

# COUNTRY regional state.
ADISCORD_west_leans_to_wrk_ally = {
	check_variable = { var = ADISCORD_west_lean value = 50 compare = greater_than_or_equals }
}

ADISCORD_west_leans_to_ivn_ally = {
	check_variable = { var = ADISCORD_west_lean value = -50 compare = less_than_or_equals }
}

# COUNTRY regional state; ROOT is a champion.
ADISCORD_west_leans_to_root_ally = {
	OR = {
		AND = {
			ROOT = { tag = WRK }
			ADISCORD_west_leans_to_wrk_ally = yes
		}
		AND = {
			ROOT = { tag = IVN }
			ADISCORD_west_leans_to_ivn_ally = yes
		}
	}
}

ADISCORD_west_grants_wrk_transit = {
	check_variable = { var = ADISCORD_west_lean value = 20 compare = greater_than_or_equals }
}

ADISCORD_west_grants_ivn_transit = {
	check_variable = { var = ADISCORD_west_lean value = -20 compare = less_than_or_equals }
}

# COUNTRY regional state; ROOT is a champion. The state can still be won over:
# it is not yet an ally of ROOT and leans no further than 40 toward the rival.
ADISCORD_west_lean_contested_by_root = {
	OR = {
		AND = {
			ROOT = { tag = WRK }
			check_variable = { var = ADISCORD_west_lean value = -40 compare = greater_than_or_equals }
			check_variable = { var = ADISCORD_west_lean value = 50 compare = less_than }
		}
		AND = {
			ROOT = { tag = IVN }
			check_variable = { var = ADISCORD_west_lean value = 40 compare = less_than_or_equals }
			check_variable = { var = ADISCORD_west_lean value = -50 compare = greater_than }
		}
	}
}

# COUNTRY. Owns part of the corridor, the only land route between the champions.
ADISCORD_west_corridor_owner = {
	OR = {
		owns_state = 90
		owns_state = 91
		owns_state = 93
		owns_state = 94
	}
}

ADISCORD_west_ivn_league_unlocked = {

		has_completed_focus = IVN_emergency_protectorates
		has_completed_focus = IVN_revanche_iron_protectorate
	}
}
