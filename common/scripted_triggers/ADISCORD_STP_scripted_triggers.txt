# COUNTRY: each public campaign needs a new election stage and living local institutions.

	has_active_mission = STP_cw_election_window
	OR = {
		check_variable = { var = STP_cw_completed_public_campaigns value = 1 compare = less_than }
		AND = {
			check_variable = { var = STP_cw_completed_public_campaigns value = 1 compare = equals }
			check_variable = { var = days_mission_timeout@STP_cw_election_window value = 105 compare = less_than_or_equals }

				has_country_flag = STP_sided_with_the_party_flag
				2 = { is_owned_by = STP is_controlled_by = STP has_state_flag = { flag = STP_resistance_administration_asset days > 20 } }
				3 = { is_owned_by = STP is_controlled_by = STP has_state_flag = { flag = STP_resistance_administration_asset days > 20 } }
			}
		}
		AND = {
			check_variable = { var = STP_cw_completed_public_campaigns value = 2 compare = equals }
			check_variable = { var = days_mission_timeout@STP_cw_election_window value = 70 compare = less_than_or_equals }
			OR = {
				has_country_flag = STP_sided_with_the_party_flag
				AND = {
					2 = { is_owned_by = STP is_controlled_by = STP has_state_flag = { flag = STP_resistance_administration_asset days > 20 } }
					3 = { is_owned_by = STP is_controlled_by = STP has_state_flag = { flag = STP_resistance_administration_asset days > 20 } }

		}
	}
