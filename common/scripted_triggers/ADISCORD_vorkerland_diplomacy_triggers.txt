# The Solar intervention remains unavailable until VAD has physically opened a
# front through the former EGC belt. This prevents an unreachable remote war.
ADISCORD_vorkerland_vad_has_solar_intervention_border = {
	OR = {
		AND = {
			owns_state = 81
			controls_state = 81
			OR = {
				AND = { has_global_flag = ADISCORD_vorkerland_solar_winner_sra SRA = { owns_state = 307 controls_state = 307 } }
				AND = { has_global_flag = ADISCORD_vorkerland_solar_winner_csl CSL = { owns_state = 307 controls_state = 307 } }
			}
		}
		AND = {
			owns_state = 110
			controls_state = 110
			OR = {
				AND = { has_global_flag = ADISCORD_vorkerland_solar_winner_sra SRA = { OR = { AND = { owns_state = 198 controls_state = 198 } AND = { owns_state = 307 controls_state = 307 } } } }
				AND = { has_global_flag = ADISCORD_vorkerland_solar_winner_csl CSL = { OR = { AND = { owns_state = 198 controls_state = 198 } AND = { owns_state = 307 controls_state = 307 } } } }
			}
		}
	}
}
