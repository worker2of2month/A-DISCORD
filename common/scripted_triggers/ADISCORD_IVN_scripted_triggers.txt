# STATE. A controlled Itoran core state with room for one more civilian factory.
IVN_march_workshop_site = {
	is_core_of = ROOT
	is_controlled_by = ROOT
	free_building_slots = {
		building = industrial_complex
		size > 0
		include_locked = yes
	}
}

# COUNTRY IVN.
IVN_march_workshop_site_exists = {
	any_owned_state = {
		IVN_march_workshop_site = yes
	}
}

# COUNTRY IVN. The northern expedition is over or can no longer happen, so the
# committee knows which course the marches will accept.
IVN_northern_question_settled = {
	NOT = { has_global_flag = ADISCORD_vorkerland_ivanland_intervention_active }
	OR = {
		has_global_flag = ADISCORD_vorkerland_ivanland_intervention_resolved
		has_global_flag = ADISCORD_vorkerland_collapse_finished
	}
}

# COUNTRY IVN. The legion can rise only in a sovereign Itora at peace that still
# holds Vedrina, the legion's capital.
IVN_marches_revolt_ready = {
	exists = yes
	is_subject = no
	has_war = no
	owns_state = 95
	controls_state = 95
	NOT = { has_global_flag = IVN_marches_revolt_active }
	NOT = { has_global_flag = IVN_marches_revolt_resolved }
	NOT = { country_exists = IVC }
}

# STATE. An inner frontier march that follows a neighbouring legion march.
IVN_march_follows_the_legion = {
	check_variable = { var = IVN_march_trust value = 40 compare = less_than }
}

# COUNTRY IVN. The Committee holds the balance of command.
IVN_committee_leads = {
	power_balance_value = {
		id = IVN_command_bop
		value < 0
	}
}

# COUNTRY IVN. The Staff of the marches holds the balance of command.
IVN_staff_leads = {
	power_balance_value = {
		id = IVN_command_bop
		value > 0
	}
}

# STATE, ROOT = a side of the revolt. The march touches land ROOT holds, so a
# defection does not create an island behind the front.
IVN_march_borders_root_control = {
	any_neighbor_state = {
		is_controlled_by = ROOT
	}
}

# COUNTRY IVN. The war ended with the cabinet in the capital.
IVN_cabinet_won_the_revolt = {
	has_global_flag = IVN_civil_war_won_by_cabinet
}
