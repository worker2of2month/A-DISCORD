# Event ROOT receives the offer; FROM is its original sender, not a global target.
IVN_cw_offer_valid = {
	has_global_flag = IVN_marches_revolt_active
	NOT = { has_global_flag = IVN_cw_awaiting_choice }
	exists = yes
	has_capitulated = no
	is_subject = no
	IVN_cw_minor = yes
	has_country_flag = IVN_cw_offer_pending
	OR = {
		AND = {
			has_country_flag = IVN_cw_offer_committee
			FROM = { tag = IVN }
		}
		AND = {
			has_country_flag = IVN_cw_offer_legion
			FROM = { tag = IVC }
		}
	}
	FROM = {
		exists = yes
		has_capitulated = no
		is_subject = no
		IVN_cw_principal = yes
		NOT = { has_political_power < 75 }
	}
}
