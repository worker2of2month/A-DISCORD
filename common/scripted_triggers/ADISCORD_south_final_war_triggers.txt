# Southern final war: Svetlogorye (or the Restitution Alliance that absorbed it)
# against Shahrabad. Side numbers: 1 = Shahrabad, 2 = the northern champion.

# GLOBAL. The resource war has a result, or its bounded entry path was stopped.
ADISCORD_south_resource_question_closed = {
	OR = {
		has_global_flag = ADISCORD_nam_resource_war_resolved
		AND = {
			has_global_flag = ADISCORD_nam_resource_war_retry_used
			NOT = { has_global_flag = ADISCORD_nam_resource_war_scheduled }
			NOT = { has_global_flag = ADISCORD_nam_resource_war_started }
		}
	}
}

# GLOBAL. The crisis clock runs between the resource war and the final campaign.
ADISCORD_south_crisis_open = {
	has_global_flag = ADISCORD_fresh_campaign_contract_v1
	ADISCORD_south_resource_question_closed = yes
	NOT = { has_global_flag = ADISCORD_south_final_started }
	NOT = { has_global_flag = ADISCORD_south_final_resolved }
}

ADISCORD_south_final_active = {
	has_global_flag = ADISCORD_south_final_started
	NOT = { has_global_flag = ADISCORD_south_final_resolved }
}

# COUNTRY. A sovereign country able to lead one side of the final war.
ADISCORD_south_champion_viable = {
	exists = yes
	is_subject = no
	has_capitulated = no
}

# GLOBAL. Svetlogorye leads the north while it survives as a sovereign state;
# otherwise the Restitution Alliance holds its former lands.
ADISCORD_south_north_candidate_exists = {
	OR = {
		NAM = { ADISCORD_south_champion_viable = yes }
		EFL = { ADISCORD_south_champion_viable = yes }
		AZH = { ADISCORD_south_champion_viable = yes }
	}
}

# A limited Shahrabad war keeps its own settlement; the final campaign waits for it.
ADISCORD_south_final_can_start = {
	ADISCORD_south_crisis_open = yes
	SHL = {
		ADISCORD_south_champion_viable = yes
		has_country_flag = SHL_furnaces_initialized
		NOT = { check_variable = { var = SHL_war_mzr value = 1 compare = equals } }
		NOT = { check_variable = { var = SHL_war_kyz value = 1 compare = equals } }
		NOT = { check_variable = { var = SHL_war_glp value = 1 compare = equals } }
	}
	ADISCORD_south_north_candidate_exists = yes
}

# GLOBAL. The northern leader of the current campaign is Svetlogorye itself.
ADISCORD_south_north_is_nam = {
	NAM = {
		exists = yes
		has_country_flag = ADISCORD_south_north_leader
	}
}

# COUNTRY.
ADISCORD_south_on_shl_side = {
	check_variable = { var = ADISCORD_south_side value = 1 compare = equals }
}

ADISCORD_south_on_north_side = {
	check_variable = { var = ADISCORD_south_side value = 2 compare = equals }
}

# COUNTRY. One of the desert and northern-theatre states that must take a side.
ADISCORD_south_regional_country = {
	OR = {
		tag = EFL
		tag = AZH
		tag = WEF
		tag = KDR
		tag = RHM
		tag = SDR
		tag = MZR
		tag = KYZ
		tag = GLP
	}
}

# COUNTRY. Regional faction followers enter with their leader instead of
# choosing alone; a member of an outside faction still chooses for itself.
ADISCORD_south_needs_side_choice = {
	exists = yes
	is_subject = no
	NOT = { has_variable = ADISCORD_south_side }
	OR = {
		is_in_faction = no
		is_faction_leader = yes
		faction_leader = { NOT = { ADISCORD_south_regional_country = yes } }
	}
}

# COUNTRY. Subject or faction partner of the selected northern leader.
ADISCORD_south_bound_to_north = {
	OR = {
		AND = {
			NAM = { has_country_flag = ADISCORD_south_north_leader }
			OR = {
				is_subject_of = NAM
				is_in_faction_with = NAM
			}
		}
		AND = {
			EFL = { has_country_flag = ADISCORD_south_north_leader }
			OR = {
				is_subject_of = EFL
				is_in_faction_with = EFL
			}
		}
		AND = {
			AZH = { has_country_flag = ADISCORD_south_north_leader }
			OR = {
				is_subject_of = AZH
				is_in_faction_with = AZH
			}
		}
	}
}

# GLOBAL. Either champion has lost its sovereign leadership or the two no
# longer fight each other.
ADISCORD_south_final_broken = {
	OR = {
		NOT = { SHL = { ADISCORD_south_champion_viable = yes } }
		NOT = { ADISCORD_south_north_leader_fights = yes }
	}
}

# GLOBAL. The flagged northern leader still leads its side against Shahrabad.
ADISCORD_south_north_leader_fights = {
	OR = {
		NAM = { ADISCORD_south_north_leader_viable = yes has_war_with = SHL }
		EFL = { ADISCORD_south_north_leader_viable = yes has_war_with = SHL }
		AZH = { ADISCORD_south_north_leader_viable = yes has_war_with = SHL }
	}
}

# GLOBAL.
ADISCORD_south_north_leader_exists = {
	OR = {
		NAM = { ADISCORD_south_north_leader_viable = yes }
		EFL = { ADISCORD_south_north_leader_viable = yes }
		AZH = { ADISCORD_south_north_leader_viable = yes }
	}
}

# COUNTRY.
ADISCORD_south_north_leader_viable = {
	has_country_flag = ADISCORD_south_north_leader
	ADISCORD_south_champion_viable = yes
}

# COUNTRY SHL. Shahrabad course that keeps the defeated as subjects.
ADISCORD_south_shl_keeps_subjects = {
	OR = {
		check_variable = { var = SHL_political_course value = 1 compare = equals }
		check_variable = { var = SHL_political_course value = 2 compare = equals }
	}
}

# STATE. Controlled by a participant of the given side.
ADISCORD_south_state_held_by_shl_side = {
	controller = { ADISCORD_south_on_shl_side = yes }
}

ADISCORD_south_state_held_by_north_side = {
	controller = { ADISCORD_south_on_north_side = yes }
}

# COUNTRY. Already fighting the selected northern leader in a campaign war.
ADISCORD_south_fights_north_leader = {
	OR = {
		AND = {
			NAM = { has_country_flag = ADISCORD_south_north_leader }
			has_war_with = NAM
		}
		AND = {
			EFL = { has_country_flag = ADISCORD_south_north_leader }
			has_war_with = EFL
		}
		AND = {
			AZH = { has_country_flag = ADISCORD_south_north_leader }
			has_war_with = AZH
		}
	}
}

# COUNTRY target; ROOT is the champion sending a consolidation offer. Offers stop
# once the final campaign has begun; a pending answer blocks a second offer.
ADISCORD_south_campaign_target_valid = {
	exists = yes
	is_subject = no
	has_capitulated = no
	NOT = { tag = ROOT }
	NOT = { tag = SHL }
	NOT = { tag = NAM }
	NOT = { has_variable = ADISCORD_south_offer_kind }
	NOT = { is_in_faction_with = ROOT }
	NOT = { has_war_with = ROOT }
	NOT = { has_global_flag = ADISCORD_south_final_started }
}

# COUNTRY target. Nothing is left to win from it for the champion in ROOT.
ADISCORD_south_campaign_target_settled = {
	OR = {
		exists = no
		is_subject_of = ROOT
		is_in_faction_with = ROOT
	}
}
