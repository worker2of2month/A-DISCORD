# ----- Crisis deadline -----
# The war starts no later than 1 March 2166. Each unit of
# global.ADISCORD_south_crisis_pull (0-6) moves that date 15 days earlier:
# 2166.3.1, 2166.2.14, 2166.1.30, 2166.1.15, 2165.12.31, 2165.12.16, 2165.12.1.
# The calendar has no leap years. NOT = { date < D } holds on D and later but
# never on the previous day, unlike date > D, which can miss the first hours
# of D. The offset dates below are precomputed from the same levels.

# GLOBAL. The current deadline date has come.
ADISCORD_south_deadline_reached = {
	OR = {
		AND = {
			check_variable = { var = global.ADISCORD_south_crisis_pull value = 1 compare = less_than }
			NOT = { date < 2166.3.1 }
		}
		AND = {
			check_variable = { var = global.ADISCORD_south_crisis_pull value = 1 compare = equals }
			NOT = { date < 2166.2.14 }
		}
		AND = {
			check_variable = { var = global.ADISCORD_south_crisis_pull value = 2 compare = equals }
			NOT = { date < 2166.1.30 }
		}
		AND = {
			check_variable = { var = global.ADISCORD_south_crisis_pull value = 3 compare = equals }
			NOT = { date < 2166.1.15 }
		}
		AND = {
			check_variable = { var = global.ADISCORD_south_crisis_pull value = 4 compare = equals }
			NOT = { date < 2165.12.31 }
		}
		AND = {
			check_variable = { var = global.ADISCORD_south_crisis_pull value = 5 compare = equals }
			NOT = { date < 2165.12.16 }
		}
		AND = {
			check_variable = { var = global.ADISCORD_south_crisis_pull value = 6 compare = greater_than_or_equals }
			NOT = { date < 2165.12.1 }
		}
	}
}

# COUNTRY target; ROOT is the champion sending a consolidation offer. Offers stop
# once the final campaign has begun; a pending answer blocks a second offer.
# The NAM story gate is checked at the answer and by the deferred dispatcher,
# so a demand sent while the story is pending is queued, not dropped.
ADISCORD_south_campaign_target_valid = {
	ADISCORD_south_offer_target_open = yes
	NOT = { tag = ROOT }
	NOT = { tag = SHL }
	NOT = { tag = NAM }
	NOT = { has_variable = ADISCORD_south_offer_kind }
	NOT = { is_in_faction_with = ROOT }
	NOT = { has_war_with = ROOT }
	NOT = { has_global_flag = ADISCORD_south_final_started }
	NOT = { has_global_flag = ADISCORD_south_final_resolved }
}

		check_variable = { var = ADISCORD_south_offer_kind value = 2 compare = equals }
		AND = {
			check_variable = { var = ADISCORD_south_offer_kind value = 4 compare = equals }
			NAM = { has_completed_focus = NAM_pragmatic_compact }
		}
		AND = {

			OR = {
				is_in_faction = no
				ADISCORD_south_alliance_switch_possible = yes
				NAM_eflor_can_replace_restitution_alliance = yes
			}
		}
	}
