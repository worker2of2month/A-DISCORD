# ----- Crisis deadline -----
# The deadline is a day number in global.num_days units,
# global.ADISCORD_south_deadline_day. ADISCORD_south_set_deadline writes it once
# per story campaign and only Shahrabad's pulls move it; every derived day is
# rewritten together with it, so these checks compare two variables and never
# recompute a date.

# GLOBAL. The current story campaign has its deadline.
ADISCORD_south_deadline_set = {
	has_variable = global.ADISCORD_south_deadline_day
}

# GLOBAL. The current deadline day has come.
ADISCORD_south_deadline_reached = {
	ADISCORD_south_deadline_set = yes
	check_variable = { var = global.num_days value = global.ADISCORD_south_deadline_day compare = greater_than_or_equals }
}

# GLOBAL. At most 31 days remain before the current deadline day.
ADISCORD_south_deadline_date_near = {
	ADISCORD_south_deadline_set = yes
	check_variable = { var = global.num_days value = global.ADISCORD_south_deadline_near_day compare = greater_than_or_equals }
}

# GLOBAL. At most 120 days remain before the current deadline day.
ADISCORD_south_deadline_date_in_ai_window = {
	ADISCORD_south_deadline_set = yes
	check_variable = { var = global.num_days value = global.ADISCORD_south_deadline_ai_day compare = greater_than_or_equals }
}

# GLOBAL. Before the anchor every pull is counted and subtracted when the
# anchor is set. With an anchor more than 135 days must remain, so one 15-day
# pull still leaves the 42-day declaration focus and the 56-day warning. The
# spheres agreement stops the pulls.
ADISCORD_south_deadline_can_pull = {
	NOT = { has_global_flag = ADISCORD_south_spheres_agreed }
	OR = {
		ADISCORD_south_deadline_set = no
		check_variable = { var = global.num_days value = global.ADISCORD_south_deadline_pull_guard_day compare = less_than }
	}
}
