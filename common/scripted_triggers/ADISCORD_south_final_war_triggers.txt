# Southern final war: Svetlogorye (or the Restitution Alliance that absorbed it)
# against Shahrabad. Side numbers: 1 = Shahrabad, 2 = the northern champion.

# GLOBAL. The resource war has a result, or its bounded entry path was stopped.
ADISCORD_south_resource_question_closed = {
	OR = {
		has_global_flag = ADISCORD_nam_resource_war_resolved
		AND = {
			has_global_flag = ADISCORD_nam_resource_war_retry_used
			NOT = { has_global_flag = ADISCORD_nam_resource_war_scheduled }
			NOT = { has_global_flag = ADISCORD_nam_resource_war_started }
		}
	}
}

# GLOBAL. The regional campaign uses the same lifecycle gate for every starter.
ADISCORD_south_final_war_enabled = {
	always = yes
}

ADISCORD_south_nam_story_clear = {
	NOT = { NAM = { NAM_south_story_pending = yes } }
}

# GLOBAL. The campaign calendar is shared by automatic and focus declarations.
ADISCORD_south_preparation_elapsed = {
	has_global_flag = ADISCORD_south_preparation_started
	date > 2164.12.31
}

# COUNTRY NAM. Each route has attainable preparation milestones before its war focus.
ADISCORD_south_nam_focus_ready = {
	OR = {
		AND = {
			has_global_flag = ADISCORD_nam_resource_war_retry_used
			NOT = { has_global_flag = ADISCORD_nam_resource_war_scheduled }
			NOT = { has_global_flag = ADISCORD_nam_resource_war_started }
			NOT = { has_global_flag = ADISCORD_nam_resource_war_resolved }
			has_completed_focus = NAM_seal_courts
			has_completed_focus = NAM_civilian_combine
			has_completed_focus = NAM_officer_courses
		}
		AND = {
			has_completed_focus = NAM_svetlogorye
			has_completed_focus = NAM_oil_exchange
			OR = {
				AND = {
					has_completed_focus = NAM_ladder_of_seals
					has_completed_focus = NAM_south_desert_offices
				}
				AND = {
					has_completed_focus = NAM_common_fuel_market
					has_completed_focus = NAM_south_fenn_guarantee
				}
				AND = {
					has_completed_focus = NAM_balanced_accounts
					has_completed_focus = NAM_prag_partner_supply
				}
			}
		}
		AND = {
			has_completed_focus = NAM_defeat_rump_cabinet
			has_completed_focus = NAM_defeat_harbour_economy
			has_completed_focus = NAM_defeat_road_south
		}
		AND = {
			has_completed_focus = NAM_defeat_revenge
			has_completed_focus = NAM_defeat_second_chapter
		}
		has_completed_focus = NAM_rad_great_war_plan
		has_completed_focus = NAM_hum_union_call
		has_completed_focus = NAM_board_stock_mobilisation
	}
}

# COUNTRY target; ROOT is the champion sending a consolidation offer. Offers stop
# once the final campaign has begun; a pending answer blocks a second offer.
# The NAM story gate is checked at the answer and by the deferred dispatcher,
# so a demand sent while the story is pending is queued, not dropped.
ADISCORD_south_campaign_target_valid = {
	ADISCORD_south_offer_target_open = yes
	NOT = { tag = ROOT }
	NOT = { tag = SHL }
	NOT = { tag = NAM }
	NOT = { has_variable = ADISCORD_south_offer_kind }
	NOT = { is_in_faction_with = ROOT }
	NOT = { has_war_with = ROOT }
	NOT = { has_global_flag = ADISCORD_south_final_started }
	NOT = { has_global_flag = ADISCORD_south_final_resolved }
}

# COUNTRY target. An accepted but deferred submission owns the negotiation slot.
ADISCORD_south_offer_target_open = {
	exists = yes
	is_subject = no
	has_capitulated = no
	NOT = { has_country_flag = ADISCORD_south_submits_to_SHL }
	NOT = { has_country_flag = ADISCORD_south_submits_to_NAM }
}

ADISCORD_south_offer_terms_valid = {
	ADISCORD_south_offer_target_open = yes
	has_variable = ADISCORD_south_offer_kind
	NOT = { has_global_flag = ADISCORD_south_final_started }
	NOT = { has_global_flag = ADISCORD_south_final_resolved }
	OR = {
		check_variable = { var = ADISCORD_south_offer_kind value = 1 compare = equals }
		check_variable = { var = ADISCORD_south_offer_kind value = 2 compare = equals }
		AND = {
			check_variable = { var = ADISCORD_south_offer_kind value = 4 compare = equals }
			NAM = { has_completed_focus = NAM_pragmatic_compact }
		}
		AND = {
			check_variable = { var = ADISCORD_south_offer_kind value = 3 compare = equals }
			OR = {
				is_in_faction = no
				ADISCORD_south_alliance_switch_possible = yes
			}
		}
	}
}

# COUNTRY event recipient; FROM is the original champion.
ADISCORD_south_offer_answer_valid = {
	ADISCORD_south_offer_terms_valid = yes
	FROM = { ADISCORD_south_champion_viable = yes }
	NOT = { has_war_with = FROM }
	NOT = { is_in_faction_with = FROM }
	OR = {
		NOT = { check_variable = { var = ADISCORD_south_offer_kind value = 3 compare = equals } }
		FROM = {
			OR = {
				is_in_faction = no
				is_faction_leader = yes
			}
		}
	}

	OR = {
		NOT = { check_variable = { var = ADISCORD_south_offer_kind value = 4 compare = equals } }
		FROM = {
			tag = NAM
			has_completed_focus = NAM_pragmatic_compact
		}
	}
}
