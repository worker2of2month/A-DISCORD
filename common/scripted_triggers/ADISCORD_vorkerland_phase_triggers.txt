# Normal campaigns open the nine district campaigns as soon as central
# preparation begins. A single surviving claimant may use the same controller
# in showdown/reunification only as an old-save recovery path; this prevents a
# serialized victory from stranding live district governments on the map.
ADISCORD_vorkerland_central_minor_campaign_phase_available = {
	OR = {
		has_global_flag = ADISCORD_vorkerland_phase_central_preparation
		AND = {
			OR = {
				has_global_flag = ADISCORD_vorkerland_phase_central_showdown
				has_global_flag = ADISCORD_vorkerland_phase_reunification
			}
			ADISCORD_vorkerland_has_single_surviving_claimant = yes
		}
		# Save repair for builds which announced a winner and materialised WRK
		# before the nine central administrations had been removed. This branch
		# reopens only the bounded district controller, never the claimant war.
		AND = {
			tag = WRK
			has_global_flag = ADISCORD_vorkerland_phase_postwar_integration
			has_global_flag = ADISCORD_vorkerland_reunification_verified
			OR = {
				has_country_flag = ADISCORD_vorkerland_route_worker
				has_country_flag = ADISCORD_vorkerland_route_joint
				has_country_flag = ADISCORD_vorkerland_route_utilitarian
			}
		}
	}
}
