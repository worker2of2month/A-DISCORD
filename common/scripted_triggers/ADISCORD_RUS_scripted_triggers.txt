# --- rus_last_empire_triggers ---
# RUS. The project starts once through the new-campaign imperial lifecycle.
RUS_crisis_can_begin = {
	tag = RUS
	exists = yes
	is_subject = no
	has_capitulated = no
	RUS_khan_governing = yes
	has_country_flag = ADISCORD_vorkerland_rus_last_empire_proclaimed
	NOT = { has_variable = RUS_crisis_phase }
	RUS_lab_reactor_access = yes
	NOT = { has_global_flag = RUS_crisis_world_ended }
}

# COUNTRY. Regional winners retain their own alliances and settlements.
RUS_crisis_eastern_hegemony = {
	OR = {
		AND = {
			tag = VAL
			OR = {
				has_completed_focus = VAL_Campaign_Secured
				has_completed_focus = VAL_nw_price
			}
			VAL_campaign_objectives_met = yes
		}
		AND = {
			tag = STS
			has_completed_focus = STP_pc_heg_final_kefreyt
			STP_heg_northern_final_resolved = yes
			OR = {
				has_country_flag = STP_heg_kefreyt_final_won
				VAL = { is_subject_of = STS }
				NOT = { VAL = { exists = yes } }
			}
		}
		AND = {
			tag = STP
			has_completed_focus = STP_pw_party_great_stelander
			STP_pw_party_external_order_resolved = yes
			STP_pw_party_northern_march_resolved = yes
		}
		AND = {
			tag = NOD
			check_variable = { var = STP_cw_northern_campaign_status value = 2 compare = equals }
			VAL = {
				OR = {
					exists = no
					is_subject_of = NOD
				}
			}
			STP = { RUS_crisis_northern_order_member = yes }
			STS = { RUS_crisis_northern_order_member = yes }
		}
	}
}

# COUNTRY. Winners of the southern, western and eastern final wars. Current
# wars and sovereignty belong to the caller's entry rule.

		AND = {

				AND = {

# Capitulation can fire before every playable state is occupied. The campaign
# still absorbs that neighbour instead of waiting on leftover pockets.
ADISCORD_vorkerland_rus_current_target_capitulated = {
	OR = {
		AND = {
			check_variable = { var = ADISCORD_vorkerland_rus_campaign_target value = 1 compare = equals }
			country_exists = SLA
			has_war_with = SLA
			SLA = { ADISCORD_vorkerland_rus_target_defeated = yes capital_scope = { OR = { is_controlled_by = RUS controller = { is_subject_of = RUS has_war_with = SLA } } } }
		}
		AND = {
			check_variable = { var = ADISCORD_vorkerland_rus_campaign_target value = 2 compare = equals }
			country_exists = RZA
			has_war_with = RZA
			RZA = { ADISCORD_vorkerland_rus_target_defeated = yes capital_scope = { OR = { is_controlled_by = RUS controller = { is_subject_of = RUS has_war_with = RZA } } } }
		}
		AND = {
			check_variable = { var = ADISCORD_vorkerland_rus_campaign_target value = 3 compare = equals }
			country_exists = MLR
			has_war_with = MLR
			MLR = { ADISCORD_vorkerland_rus_target_defeated = yes capital_scope = { OR = { is_controlled_by = RUS controller = { is_subject_of = RUS has_war_with = MLR } } } }
		}
		AND = {
			check_variable = { var = ADISCORD_vorkerland_rus_campaign_target value = 4 compare = equals }
			country_exists = ERT
			has_war_with = ERT
			ERT = { ADISCORD_vorkerland_rus_target_defeated = yes capital_scope = { OR = { is_controlled_by = RUS controller = { is_subject_of = RUS has_war_with = ERT } } } }
		}
		AND = {
			check_variable = { var = ADISCORD_vorkerland_rus_campaign_target value = 5 compare = equals }
			country_exists = IRT
			has_war_with = IRT
			IRT = { ADISCORD_vorkerland_rus_target_defeated = yes capital_scope = { OR = { is_controlled_by = RUS controller = { is_subject_of = RUS has_war_with = IRT } } } }

		AND = {

		}
	}
}
