	is_ai = no

# Old or pre-feature saves are unsupported and remain inert. Ordinary economy
# updates may continue only for countries already initialized on schema 15.
ADISCORD_economy_has_current_schema = {
	has_variable = ADISCORD_economy_initialized
	has_variable = ADISCORD_economy_schema_version
	check_variable = { var = ADISCORD_economy_schema_version value = 15 compare = greater_than_or_equals }
}

ADISCORD_economy_is_primary_tier_country = {
	OR = {
		ADISCORD_is_playable_development_country = yes
		is_major = yes
		is_faction_leader = yes
		has_country_flag = ADISCORD_economy_important_country
		any_enemy_country = { is_ai = no }
	}
}

ADISCORD_economy_is_secondary_tier_country = {
	is_ai = yes
	OR = {
		ADISCORD_is_secondary_development_country = yes
		has_country_flag = ADISCORD_economy_secondary_country
	}
	NOT = { ADISCORD_economy_is_primary_tier_country = yes }
}

ADISCORD_economy_is_dormant_tier_country = {
	is_ai = yes
	NOT = {
		OR = {
			ADISCORD_economy_is_primary_tier_country = yes
			ADISCORD_economy_is_secondary_tier_country = yes
		}
	}
}

ADISCORD_economy_is_dormant = {
	ADISCORD_economy_is_dormant_tier_country = yes
}

ADISCORD_economy_should_monthly_update = {
	ADISCORD_economy_has_current_schema = yes
	OR = {
		ADISCORD_economy_is_player_tier_country = yes
		ADISCORD_economy_is_primary_tier_country = yes
	}
}

ADISCORD_economy_should_weekly_update = {
	has_variable = ADISCORD_economy_initialized
	has_variable = ADISCORD_economy_weekly_source_cache_ready
	check_variable = { var = ADISCORD_economy_schema_version value = 15 compare = greater_than_or_equals }
	check_variable = { var = ADISCORD_economy_weekly_source_cache_ready value = 1 compare = greater_than_or_equals }
	OR = {
		ADISCORD_economy_is_player_tier_country = yes
		AND = {
			has_variable = ADISCORD_economy_simulation_tier
			check_variable = { var = ADISCORD_economy_simulation_tier value = 1 compare = less_than_or_equals }
		}
	}
}

# non-zero result. Temp calculations avoid fragile variable-to-variable trigger
