is_diplomatic_action_valid_lend_lease = {
	ADISCORD_diplomacy_not_dirty_zone_pair = yes
	OR = {
		FROM = {
			NOT = {
				AND = {
					tag = TVA
					is_ai = yes
					has_country_leader = { character = TVA_Dorian_Worx ruling_only = yes }
				}
			}
		}
		AND = {
			FROM = {
				tag = TVA
				is_ai = yes
				has_country_leader = { character = TVA_Dorian_Worx ruling_only = yes }
			}
			OR = {
				is_in_faction_with = FROM
				is_subject_of = FROM
			}
		}
	}
}

# VAL volunteer mandates are restricted to their current contracted fronts.
is_diplomatic_action_valid_send_volunteers = {
 ADISCORD_diplomacy_not_dirty_zone_pair = yes
 OR = {
  FROM = { NOT = { tag = VAL } }
  AND = { FROM = { has_idea = VAL_northern_volunteer_mandate } OR = { tag = VAL VAL_northern_volunteer_host = yes } }
  AND = { FROM = { has_idea = VAL_resource_volunteer_mandate } OR = { tag = VAL VAL_resource_aid_recipient = yes } }
 }
