# Kefreyt logistics, displacement and grey-market triggers.
# The feature stays country-scoped and checks only authored strategic states/tags.

VAL_trade_corridors_unlocked = {
	OR = {
		has_completed_focus = VAL_One_Ledger_One_Banner
		has_completed_focus = VAL_Reopen_Trade_Routes
	}
}

# STATE scope. A route node belongs to Kefreyt's contractual sphere only when
# both sovereignty and effective control remain inside the VAL subject network.
VAL_trade_state_in_contract_sphere = {
	OR = {
        AND = { owner = { has_country_flag = VAL_trade_transit_charter NOT = { has_war_with = VAL } } controller = { has_country_flag = VAL_trade_transit_charter NOT = { has_war_with = VAL } } }
		AND = { is_owned_by = VAL is_controlled_by = VAL }
		AND = {
			owner = { is_subject_of = VAL }
			controller = { is_subject_of = VAL }
		}
	}
}

# COUNTRY VAL. The Occidian artery links Balchansk, the Harbour and the inland
# depot. Livonn (45) is deliberately optional so an authored border settlement
# cannot permanently kill the route.
VAL_trade_route_occidia_open = {
	tag = VAL
	VAL_trade_corridors_unlocked = yes
	43 = { VAL_trade_state_in_contract_sphere = yes }
	44 = { VAL_trade_state_in_contract_sphere = yes }
	88 = { VAL_trade_state_in_contract_sphere = yes }
}

# COUNTRY VAL. The northern belt is the compact resource spine through Antolla,

VAL_trade_route_north_open = {

# COUNTRY VAL. The Stelander road uses the two border districts retained by

	29 = { VAL_trade_state_in_contract_sphere = yes }
	46 = { VAL_trade_state_in_contract_sphere = yes }
}

# COUNTRY VAL. The old Vorkerland corridor can remain a commercial route under
# a non-hostile Vorkerland successor; it does not require political domination.
VAL_trade_route_vorkerland_open = {
	tag = VAL
	VAL_trade_corridors_unlocked = yes
	33 = {
		OR = {
			VAL_trade_state_in_contract_sphere = yes
			AND = { is_owned_by = WRK is_controlled_by = WRK WRK = { has_war = no } }
			AND = { is_owned_by = WKR is_controlled_by = WKR WKR = { has_war = no } }
			AND = { is_owned_by = VAD is_controlled_by = VAD VAD = { has_war = no } }
			AND = { is_owned_by = TVA is_controlled_by = TVA TVA = { has_war = no } }
		}
	}
}

# Only a previously commissioned route counts as a disruption. Future expansion
# targets do not create fictitious shortages before Kefreyt ever opens them.
VAL_trade_network_disrupted = {
	tag = VAL
	VAL_trade_corridors_unlocked = yes
	OR = {
		AND = { has_country_flag = VAL_route_occidia_commissioned NOT = { VAL_trade_route_occidia_open = yes } }
		AND = { has_country_flag = VAL_route_north_commissioned NOT = { VAL_trade_route_north_open = yes } }
		AND = { has_country_flag = VAL_route_stelander_commissioned NOT = { VAL_trade_route_stelander_open = yes } }
		AND = { has_country_flag = VAL_route_vorkerland_commissioned NOT = { VAL_trade_route_vorkerland_open = yes } }
	}
}

# COUNTRY VAL: compare route inputs before the weekly payment, including war and autonomy changes.
VAL_trade_network_changed = {
	OR = {
		AND = { VAL_trade_route_occidia_open = yes NOT = { check_variable = { var = VAL_route_occidia_active value = 1 compare = equals } } }
		AND = { VAL_trade_route_occidia_open = no check_variable = { var = VAL_route_occidia_active value = 1 compare = equals } }
		AND = { VAL_trade_route_north_open = yes NOT = { check_variable = { var = VAL_route_north_active value = 1 compare = equals } } }
		AND = { VAL_trade_route_north_open = no check_variable = { var = VAL_route_north_active value = 1 compare = equals } }
		AND = { VAL_trade_route_stelander_open = yes NOT = { check_variable = { var = VAL_route_stelander_active value = 1 compare = equals } } }
		AND = { VAL_trade_route_stelander_open = no check_variable = { var = VAL_route_stelander_active value = 1 compare = equals } }
		AND = { VAL_trade_route_vorkerland_open = yes NOT = { check_variable = { var = VAL_route_vorkerland_active value = 1 compare = equals } } }
		AND = { VAL_trade_route_vorkerland_open = no check_variable = { var = VAL_route_vorkerland_active value = 1 compare = equals } }
		AND = { VAL_trade_corridors_unlocked = yes NOT = { OR = { has_idea = VAL_corridor_network_fragmented has_idea = VAL_corridor_network_partial has_idea = VAL_corridor_network_integrated } } }
		AND = { VAL_trade_corridors_unlocked = no OR = { has_idea = VAL_corridor_network_fragmented has_idea = VAL_corridor_network_partial has_idea = VAL_corridor_network_integrated } }
	}
}
