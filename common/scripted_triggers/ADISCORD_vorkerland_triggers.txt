# The three successor governments share one canonical claimant interface.
# District governments use ADISCORD_vorkerland_is_central_minor instead.

ADISCORD_vorkerland_is_main_claimant = {
	OR = {
		tag = WKR
		tag = VAD
		tag = TVA
	}

# Peace-conference AI evaluates this on a potential country that may not exist
# yet, so the gate cannot require exists = yes. Historical WRK stays dormant
# from the collapse until the verified postwar formation.
ADISCORD_vorkerland_wrk_release_is_forbidden = {
	OR = {
		original_tag = WRK
		tag = WRK
	}
	has_global_flag = ADISCORD_vorkerland_collapse_wars_started
	NOT = { has_global_flag = ADISCORD_vorkerland_phase_postwar_integration }
	NOT = { has_global_flag = ADISCORD_vorkerland_central_war_finished }
	NOT = { has_global_flag = ADISCORD_vorkerland_reunification_verified }
	NOT = { has_global_flag = ADISCORD_vorkerland_collapse_finished }
}

# The historical WRK tag is dormant between the initial split and the verified
# phase.6 formation. Native peace actions can otherwise release that dormant
# core holder for one tick and recreate the old country beside its claimants.
ADISCORD_vorkerland_is_premature_wrk = {
	tag = WRK
	exists = yes
	ADISCORD_vorkerland_wrk_release_is_forbidden = yes
}

ADISCORD_vorkerland_release_requires_interception = {
	OR = {
		ADISCORD_vorkerland_is_premature_wrk = yes
		AND = {
			NOT = { has_global_flag = ADISCORD_vorkerland_premature_wrk_subject_cleanup_active }
			FROM = { ADISCORD_vorkerland_is_premature_wrk = yes }
		}
	}
}

# Main claimants fight only across a live border. An unrelated current war must
# not freeze the centre: the collapse is explicitly allowed to become a
# multi-front conflict. Worker and Doctor Worx are the exception: their direct
# war is opened only by the dedicated preparation decision after one finishes
# its opening campaign.
ADISCORD_vorkerland_is_main_claimant_rival_for_ROOT = {
	is_subject = no
	NOT = { has_capitulated = yes }
	NOT = { has_war_with = ROOT }
	ADISCORD_vorkerland_is_main_claimant = yes
	ROOT = { ADISCORD_vorkerland_is_main_claimant = yes }
	NOT = {
		OR = {
			AND = { tag = TVA ROOT = { tag = WKR } }
			AND = { tag = WKR ROOT = { tag = TVA } }
		}
	}
}

# Smaller central states remain valid targets even while fighting another
# claimant. This prevents one long campaign from locking every adjacent front.
ADISCORD_vorkerland_is_central_target_for_ROOT = {
	is_subject = no
	NOT = { has_capitulated = yes }

	ROOT = { ADISCORD_vorkerland_is_main_claimant = yes }
}

# A country can be spliced into a local rivalry only while it is a real, free
# belligerent. One shared definition keeps the war builder, the bracket-cleared
# check and the coalition decision from drifting apart.

	exists = yes

	NOT = { has_capitulated = yes }

# The nine central district governments. They are the first bracket stage and

# Every tag that owns a seat in the local war graph. Used by the AI focus
# strategies, the autonomy grant and the coalition decision allowlist.

	OR = {
		ADISCORD_vorkerland_is_central_bracket_member = yes
		ADISCORD_vorkerland_is_closed_zone_bracket_member = yes
		ADISCORD_vorkerland_is_regional_combatant = yes
	}
}

# The authored rivalry table. Each edge is written exactly once, in its attacker
# direction, and is evaluated in the candidate attacker's scope with PREV as the
# candidate defender. This single list is the only definition of who fights whom
# locally; the war builder, the undirected bracket check and the validator all
# read it instead of repeating pairwise war blocks.
ADISCORD_vorkerland_is_local_attacker_of_PREV = {
	OR = {
		# Western triangle and the Ash-country contract directorate.
		AND = { tag = ZAO PREV = { tag = WPA } }
		AND = { tag = WPS PREV = { tag = ZAO } }
		AND = { tag = ZAO PREV = { tag = PSD } }
		AND = { tag = PWR PREV = { tag = ZAO } }
		AND = { tag = WPA PREV = { tag = PSD } }
		AND = { tag = WPA PREV = { tag = PWR } }
		AND = { tag = WPS PREV = { tag = PSD } }
		AND = { tag = WPS PREV = { tag = PWR } }
		AND = { tag = PWR PREV = { tag = PSD } }
		# Volnograd, Firnov-Tehlar and Macri's Ebern.
		AND = { tag = VLA PREV = { tag = EBA } }
		AND = { tag = VLA PREV = { tag = TGD } }
		AND = { tag = TGD PREV = { tag = EBA } }
		# Solyarino plain and the isolated Sloboda city board.
		AND = { tag = SOL PREV = { tag = SRA } }
		AND = { tag = SOL PREV = { tag = CSL } }
		AND = { tag = SRA PREV = { tag = CSL } }
		# Frealor mountain roads and the Golden River order.
		AND = { tag = ROM PREV = { tag = DVA } }
		AND = { tag = TRU PREV = { tag = ZTA } }
		# Eyrmi knot: the garrison staff strikes Koval's congress, and the
		# congress refuses a second city assembly in the Yorsen belt.
		AND = { tag = EGC PREV = { tag = EYR } }
		AND = { tag = EYR PREV = { tag = YOR } }
		# Vorkensberg ring: the old-city board claims the suburban councils,
		# and the Revel okrug council reclaims Ostvin as its own district.
		AND = { tag = SWB PREV = { tag = NDN } }
		AND = { tag = SWB PREV = { tag = OSV } }
		AND = { tag = REV PREV = { tag = OSV } }
		# Riven board versus the Verkhovye zemstvo assembly.
		AND = { tag = RIV PREV = { tag = VHV } }
		# Reactor perimeter: the technical council imposes its power regime.
		AND = { tag = RZA PREV = { tag = MLR } }
		AND = { tag = RZA PREV = { tag = IRT } }
		# Eastern corridor: the interior council links its land first, and the
		# eastern council forces a southern outlet of its own.
		AND = { tag = IRT PREV = { tag = ERT } }
		AND = { tag = ERT PREV = { tag = SCA } }
	}
}

# Bracket stages are launched separately because the closed-zone tags do not
# exist at the collapse. Neither stage lists a peripheral tag: those seventeen
# rivalries keep their own verified decision-driven launch path.
ADISCORD_vorkerland_is_staged_local_belligerent = {
	OR = {
		AND = {
			has_global_flag = ADISCORD_vorkerland_local_bracket_stage_central
			ADISCORD_vorkerland_is_central_bracket_member = yes
		}
		AND = {
			has_global_flag = ADISCORD_vorkerland_local_bracket_stage_closed_zone
			ADISCORD_vorkerland_is_closed_zone_bracket_member = yes
		}
	}
}

# Postcondition for a launched bracket stage. Every authored edge of the active
# stage must either exist in the diplomatic cache or be terminal because one
# participant no longer exists, capitulated, or became a subject. Checked only by
# the bounded delayed verifier, never by a recurring pulse.
ADISCORD_vorkerland_staged_local_war_graph_active_or_terminal = {
	NOT = {
		any_country = {
			ADISCORD_vorkerland_can_open_local_war = yes
			ADISCORD_vorkerland_is_staged_local_belligerent = yes
			any_other_country = {
				ADISCORD_vorkerland_can_open_local_war = yes
				ADISCORD_vorkerland_is_staged_local_belligerent = yes
				ADISCORD_vorkerland_is_local_attacker_of_PREV = yes
				NOT = { has_war_with = PREV }
			}
		}
	}
}
