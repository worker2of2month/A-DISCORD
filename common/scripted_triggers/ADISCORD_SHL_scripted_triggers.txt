# Furnace predicates run in state scope; the economic actor is always SHL.
SHL_furnace_site_controlled = {
	is_owned_by = SHL
	is_controlled_by = SHL
	SHL = {
		has_country_flag = SHL_furnaces_initialized
		has_capitulated = no
	}
}

SHL_furnace_site_idle = {
	SHL_furnace_site_controlled = yes
	NOT = { has_variable = SHL_operation_deposit }
}

SHL_country_operable = {
	tag = SHL
	has_country_flag = SHL_furnaces_initialized
	has_capitulated = no
}

SHL_project_idle = {
	SHL_country_operable = yes
	NOT = { has_variable = SHL_project_deposit }
	294 = {
		is_owned_by = SHL
		is_controlled_by = SHL
	}
}

SHL_reserve_affordable = {
	OR = {
		AND = {
			has_country_flag = SHL_supplier_premium
			check_variable = { var = ADISCORD_economy_treasury value = 70 compare = greater_than_or_equals }
		}
		AND = {
			NOT = { has_country_flag = SHL_supplier_premium }
			check_variable = { var = ADISCORD_economy_treasury value = 60 compare = greater_than_or_equals }
		}
	}
}
