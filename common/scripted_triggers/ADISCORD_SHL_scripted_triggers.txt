# Mazar tests a weak Shahrabad first and stops waiting after 24 cycles.
SHL_mazar_demand_ready = {
	check_variable = { var = SHL_mazar_state value = 0 compare = equals }
	OR = {
		check_variable = { var = SHL_cycle_number value = 24 compare = greater_than_or_equals }
		AND = {
			check_variable = { var = SHL_cycle_number value = 16 compare = greater_than_or_equals }
			OR = {
				check_variable = { var = SHL_running_count value = 6 compare = less_than_or_equals }
				has_war = yes
				has_stability < 0.35
			}
		}
	}
	SHL_mazar_can_threaten = yes
}
