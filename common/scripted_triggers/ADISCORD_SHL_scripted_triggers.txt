# Overtime doubles the packets each lit furnace burns, so it needs a reserve
# for the planned supply and is never chosen on worn furnaces. It serves a war,
# a paid arsenal obligation or a pending guard order.
SHL_ai_shift_overtime_wanted = {
	SHL_ai_furnaces_worn = no
	check_variable = { var = ADISCORD_economy_treasury value = 300 compare = greater_than_or_equals }
	OR = {
		has_war = yes
		has_variable = SHL_guard_deposit
		check_variable = { var = SHL_obligation_kind value = 3 compare = equals }
	}
}

# Careful smelting rests worn furnaces or spends a comfortable treasury on
# longer furnace life instead of income.
SHL_ai_shift_careful_wanted = {
	OR = {
		SHL_ai_furnaces_worn = yes
		AND = {
			SHL_ai_shift_overtime_wanted = no
			check_variable = { var = ADISCORD_economy_treasury value = 1200 compare = greater_than_or_equals }
		}
	}
}

# Overtime that is no longer wanted ends at once, through careful smelting when
# that is wanted and unlocked. Careful smelting is left only on rested furnaces
# once the treasury falls below 900.
SHL_ai_shift_ordinary_wanted = {
	SHL_ai_shift_overtime_wanted = no
	OR = {
		AND = {
			check_variable = { var = SHL_shift_policy value = 1 compare = equals }
			NOT = {
				AND = {
					has_completed_focus = SHL_careful_smelting
					SHL_ai_shift_careful_wanted = yes
				}
			}
		}
		AND = {
			check_variable = { var = SHL_shift_policy value = 2 compare = equals }
			SHL_ai_furnaces_rested = yes
			check_variable = { var = ADISCORD_economy_treasury value = 900 compare = less_than }
		}
	}
}
