can_lose_unity = {
	has_stability > 60
	OR = {
		has_war = no
		AND = {
			has_war = yes
			surrender_progress < 0.1
		}
	}
}

# COUNTRY allocator: one adjacent-country scan per profile enable evaluation.
# PREV is the allocator inside this iterator; FROM belongs
# to the external caller and must not be used as the neighbour's counterpart.
ADISCORD_ai_has_adjacent_war_preparation = {
	any_neighbor_country = {
		exists = yes
		has_capitulated = no
		NOT = { is_in_faction_with = PREV }
		OR = {
			has_wargoal_against = PREV
			is_justifying_wargoal_against = PREV
			PREV = { has_wargoal_against = PREV }
			PREV = { is_justifying_wargoal_against = PREV }
			check_variable = { var = ai_strategy_prepare_for_war@PREV value = 0 compare = greater_than }
			check_variable = { var = PREV.ai_strategy_prepare_for_war@THIS value = 0 compare = greater_than }
		}
	}
}

# COUNTRY target; FROM is the country allocating its front. Native preparation
# scores are read only: profiles producing them must use the underlying mission
# or route condition, otherwise preparation would keep itself alive after peace.
ADISCORD_ai_front_has_prewar_threat = {
	OR = {
		has_wargoal_against = FROM
		is_justifying_wargoal_against = FROM
		FROM = {
			has_wargoal_against = PREV
		}
		FROM = {
			is_justifying_wargoal_against = PREV
		}
		check_variable = { var = ai_strategy_prepare_for_war@FROM value = 0 compare = greater_than }
		check_variable = { var = FROM.ai_strategy_prepare_for_war@THIS value = 0 compare = greater_than }
	}
}

# COUNTRY: compare fielded equipment in thousands; empty armies need no refill.
# Temporary thresholds are local to evaluation and do not require a pulse/cache.
ADISCORD_ai_army_needs_replacements = {
	OR = {
		AND = {
			check_variable = { num_target_equipment_in_armies_k@infantry_equipment > 0 }
			set_temp_variable = { ADISCORD_ai_equipment_threshold = num_target_equipment_in_armies_k@infantry_equipment }
			multiply_temp_variable = { ADISCORD_ai_equipment_threshold = 0.85 }
			check_variable = { num_equipment_in_armies_k@infantry_equipment < ADISCORD_ai_equipment_threshold }
			stockpile_ratio = { archetype = infantry_equipment ratio < 0.05 }
		}
		AND = {
			check_variable = { num_target_equipment_in_armies_k@ADISCORD_squad_weapons_equipment > 0 }
			set_temp_variable = { ADISCORD_ai_equipment_threshold = num_target_equipment_in_armies_k@ADISCORD_squad_weapons_equipment }
			multiply_temp_variable = { ADISCORD_ai_equipment_threshold = 0.85 }
			check_variable = { num_equipment_in_armies_k@ADISCORD_squad_weapons_equipment < ADISCORD_ai_equipment_threshold }
			stockpile_ratio = { archetype = ADISCORD_squad_weapons_equipment ratio < 0.05 }
		}
		AND = {
			check_variable = { num_target_equipment_in_armies_k@support_equipment > 0 }
			set_temp_variable = { ADISCORD_ai_equipment_threshold = num_target_equipment_in_armies_k@support_equipment }
			multiply_temp_variable = { ADISCORD_ai_equipment_threshold = 0.85 }
			check_variable = { num_equipment_in_armies_k@support_equipment < ADISCORD_ai_equipment_threshold }
			stockpile_ratio = { archetype = support_equipment ratio < 0.05 }
		}
		AND = {
			check_variable = { num_target_equipment_in_armies_k@artillery_equipment > 0 }
			set_temp_variable = { ADISCORD_ai_equipment_threshold = num_target_equipment_in_armies_k@artillery_equipment }
			multiply_temp_variable = { ADISCORD_ai_equipment_threshold = 0.85 }
			check_variable = { num_equipment_in_armies_k@artillery_equipment < ADISCORD_ai_equipment_threshold }
			stockpile_ratio = { archetype = artillery_equipment ratio < 0.05 }
		}
	}
}

# A reserve can already be in transit to depleted units. Separate entry/exit
# thresholds avoid switching recruitment after each daily equipment delivery.
ADISCORD_ai_army_replacement_pressure_cleared = {
	set_temp_variable = { ADISCORD_ai_equipment_threshold = num_target_equipment_in_armies_k@infantry_equipment }
	multiply_temp_variable = { ADISCORD_ai_equipment_threshold = 0.95 }
	OR = {
		NOT = { check_variable = { num_equipment_in_armies_k@infantry_equipment < ADISCORD_ai_equipment_threshold } }
		NOT = { stockpile_ratio = { archetype = infantry_equipment ratio < 0.10 } }
	}
	set_temp_variable = { ADISCORD_ai_equipment_threshold = num_target_equipment_in_armies_k@ADISCORD_squad_weapons_equipment }
	multiply_temp_variable = { ADISCORD_ai_equipment_threshold = 0.95 }
	OR = {
		NOT = { check_variable = { num_equipment_in_armies_k@ADISCORD_squad_weapons_equipment < ADISCORD_ai_equipment_threshold } }
		NOT = { stockpile_ratio = { archetype = ADISCORD_squad_weapons_equipment ratio < 0.10 } }
	}
	set_temp_variable = { ADISCORD_ai_equipment_threshold = num_target_equipment_in_armies_k@support_equipment }
	multiply_temp_variable = { ADISCORD_ai_equipment_threshold = 0.95 }
	OR = {
		NOT = { check_variable = { num_equipment_in_armies_k@support_equipment < ADISCORD_ai_equipment_threshold } }
		NOT = { stockpile_ratio = { archetype = support_equipment ratio < 0.10 } }
	}
	set_temp_variable = { ADISCORD_ai_equipment_threshold = num_target_equipment_in_armies_k@artillery_equipment }
	multiply_temp_variable = { ADISCORD_ai_equipment_threshold = 0.95 }
	OR = {
		NOT = { check_variable = { num_equipment_in_armies_k@artillery_equipment < ADISCORD_ai_equipment_threshold } }
		NOT = { stockpile_ratio = { archetype = artillery_equipment ratio < 0.10 } }
	}
}

# Field upgrades compete with replacement deliveries and the recruitment queue.
ADISCORD_ai_can_refit_field_army = {
	ADISCORD_ai_army_replacement_pressure_cleared = yes
	NOT = { ADISCORD_economy_ai_is_crisis = yes }
	NOT = { has_manpower < 3000 }
	NOT = { has_equipment = { infantry_equipment < 3000 } }
}
