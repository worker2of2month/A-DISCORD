# Only a real central Vorkerland war edge can open the Palatinate crisis.
# The generic on_war hook may run in both belligerent scopes; the global
# scheduled flag in the companion on_action keeps the producer one-shot.
ADISCORD_rin_is_vorkerland_war_actor = {
	OR = {
		tag = WKR
		tag = VAD
		tag = TVA
	}
}

ADISCORD_rin_oath_crisis_can_schedule = {
	has_global_flag = ADISCORD_vorkerland_collapse_started
	NOT = { has_global_flag = ADISCORD_rin_oath_crisis_scheduled }
	NOT = { has_global_flag = ADISCORD_rin_oath_crisis_opened }
	NOT = { has_global_flag = ADISCORD_rin_oath_crisis_active }
	NOT = { has_global_flag = ADISCORD_rin_oath_crisis_resolved }
	RIN = { exists = yes }
	MON = { exists = yes }
}

ADISCORD_rin_oath_crisis_legacy_needs_schedule = {
	tag = RIN
	has_global_flag = ADISCORD_vorkerland_collapse_wars_started
	NOT = { has_global_flag = ADISCORD_rin_oath_crisis_scheduled }
	NOT = { has_global_flag = ADISCORD_rin_oath_crisis_opened }
	NOT = { has_global_flag = ADISCORD_rin_oath_crisis_active }
	NOT = { has_global_flag = ADISCORD_rin_oath_crisis_resolved }
	MON = { exists = yes }
}

ADISCORD_rin_is_oath_crisis_side = {
	original_tag = RIN
	OR = {
		has_country_flag = ADISCORD_rin_southern_charter_side
		has_country_flag = ADISCORD_rin_northern_crown_side
	}
}
