		subtract_from_temp_variable = { var = NAM_dir_campaign_free value = 1 }

	if = {
		limit = { has_idea = NAM_dir_campaign_dividend_spirit }
		subtract_from_temp_variable = { var = NAM_dir_campaign_free value = 1 }
	}
	if = {
		limit = { has_idea = NAM_dir_campaign_pays_on_time_spirit }
		subtract_from_temp_variable = { var = NAM_dir_campaign_free value = 1 }
	}
	if = {
		limit = { has_idea = NAM_dir_campaign_agitators_spirit }
		subtract_from_temp_variable = { var = NAM_dir_campaign_free value = 1 }
	}
	if = {
		limit = { has_idea = NAM_dir_campaign_mutineers_spirit }
		subtract_from_temp_variable = { var = NAM_dir_campaign_free value = 1 }
	}
	if = {
		limit = { has_idea = NAM_dir_campaign_return_spirit }
		subtract_from_temp_variable = { var = NAM_dir_campaign_free value = 1 }
	}
	if = {

		subtract_from_temp_variable = { var = NAM_dir_campaign_free value = 1 }
	}
	if = {

# ----- Coalition land fronts -----
# Enemy scope NAM in a coalition front country_trigger; FROM is the allocating
# member. Svetlogorye fights both principals at once, so its fighting army is
# measured against their combined strength. With r the ratio against one member,
# 1/r(EFL) + 1/r(AZH) is the coalition's strength relative to Svetlogorye. Each
# pair of bounds below lies on or inside the exact 0.8 and 1.25 limits, so a
# borderline front keeps the balanced base order.
ADISCORD_difficulty_coalition_outnumbers_nam = {
	OR = {
		fighting_army_strength_ratio = { tag = FROM ratio < 0.8 }
		AND = {
			has_war_with = EFL
			has_war_with = AZH
			OR = {
				fighting_army_strength_ratio = { tag = EFL ratio < 0.8 }
				fighting_army_strength_ratio = { tag = AZH ratio < 0.8 }
				AND = {
					fighting_army_strength_ratio = { tag = EFL ratio < 1.6 }
					fighting_army_strength_ratio = { tag = AZH ratio < 1.6 }
				}
				AND = {
					fighting_army_strength_ratio = { tag = EFL ratio < 1.2 }
					fighting_army_strength_ratio = { tag = AZH ratio < 2.4 }
				}
				AND = {
					fighting_army_strength_ratio = { tag = EFL ratio < 2.4 }
					fighting_army_strength_ratio = { tag = AZH ratio < 1.2 }
				}
				AND = {
					fighting_army_strength_ratio = { tag = EFL ratio < 1 }
					fighting_army_strength_ratio = { tag = AZH ratio < 4 }
				}
				AND = {
					fighting_army_strength_ratio = { tag = EFL ratio < 4 }
					fighting_army_strength_ratio = { tag = AZH ratio < 1 }
				}
			}
		}
	}
}

ADISCORD_difficulty_nam_outmatches_coalition = {
	OR = {
		AND = {
			OR = {
				NOT = { has_war_with = EFL }
				NOT = { has_war_with = AZH }
			}
			fighting_army_strength_ratio = { tag = FROM ratio > 1.25 }
		}

				AND = {
					fighting_army_strength_ratio = { tag = EFL ratio > 2.5 }

				}

				}
				AND = {

					fighting_army_strength_ratio = { tag = AZH ratio > 2 }
				}
				AND = {
