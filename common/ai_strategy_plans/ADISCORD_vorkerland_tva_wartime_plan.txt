ADISCORD_vorkerland_tva_experimental_core_plan = {
	name = "TVA experimental wartime core"
	desc = ""

	allowed = { original_tag = TVA }

	enable = {
		tag = TVA
		is_ai = yes
		has_global_flag = ADISCORD_vorkerland_collapse_wars_started
		NOT = { has_global_flag = ADISCORD_vorkerland_collapse_finished }
		NOT = {
			AND = {
				OR = {
					has_country_flag = ADISCORD_vorkerland_focus_tva_throughput_priority
					has_country_flag = ADISCORD_vorkerland_focus_tva_algorithmic_board
					has_country_flag = ADISCORD_vorkerland_focus_tva_specialist_priority
				}
				OR = {
					has_country_flag = ADISCORD_vorkerland_focus_tva_technical_battalions
					has_country_flag = ADISCORD_vorkerland_focus_tva_remote_fire_control
					has_country_flag = ADISCORD_vorkerland_focus_tva_adaptive_logistics
				}
			}
		}
		OR = {
			has_global_flag = ADISCORD_vorkerland_phase_collapse
			has_global_flag = ADISCORD_vorkerland_phase_regional_consolidation
			has_global_flag = ADISCORD_vorkerland_phase_central_preparation
			has_global_flag = ADISCORD_vorkerland_phase_central_showdown
		}
	}

	abort = {
		OR = {
			NOT = { tag = TVA }
			has_global_flag = ADISCORD_vorkerland_collapse_finished
			AND = {
				OR = {
					has_country_flag = ADISCORD_vorkerland_focus_tva_throughput_priority
					has_country_flag = ADISCORD_vorkerland_focus_tva_algorithmic_board
					has_country_flag = ADISCORD_vorkerland_focus_tva_specialist_priority
				}
				OR = {
					has_country_flag = ADISCORD_vorkerland_focus_tva_technical_battalions
					has_country_flag = ADISCORD_vorkerland_focus_tva_remote_fire_control
					has_country_flag = ADISCORD_vorkerland_focus_tva_adaptive_logistics
				}
			}
		}
	}

	ai_national_focuses = {
		TVA_codify_utilitarian_directorate
		TVA_publish_operational_metrics
		TVA_deploy_field_laboratories
		TVA_select_trial_protocol
	}

	weight = { factor = 3 }
}
