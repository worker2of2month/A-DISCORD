ADISCORD_vorkerland_tva_experimental_core_plan = {
	name = "TVA experimental wartime core"
	desc = ""

	allowed = { original_tag = TVA }

	enable = {
		tag = TVA
		is_ai = yes
		has_global_flag = ADISCORD_vorkerland_collapse_wars_started
		NOT = { has_global_flag = ADISCORD_vorkerland_collapse_finished }
		NOT = {
			AND = {
				OR = {
					has_country_flag = ADISCORD_vorkerland_focus_tva_throughput_priority
					has_country_flag = ADISCORD_vorkerland_focus_tva_algorithmic_board
					has_country_flag = ADISCORD_vorkerland_focus_tva_specialist_priority
				}
				OR = {
					has_country_flag = ADISCORD_vorkerland_focus_tva_technical_battalions
					has_country_flag = ADISCORD_vorkerland_focus_tva_remote_fire_control
					has_country_flag = ADISCORD_vorkerland_focus_tva_adaptive_logistics
				}
			}
		}
		OR = {
			has_global_flag = ADISCORD_vorkerland_phase_collapse
			has_global_flag = ADISCORD_vorkerland_phase_regional_consolidation
			has_global_flag = ADISCORD_vorkerland_phase_central_preparation
			has_global_flag = ADISCORD_vorkerland_phase_central_showdown
		}
	}

	abort = {
		OR = {
			NOT = { tag = TVA }
			has_global_flag = ADISCORD_vorkerland_collapse_finished
			AND = {
				OR = {
					has_country_flag = ADISCORD_vorkerland_focus_tva_throughput_priority
					has_country_flag = ADISCORD_vorkerland_focus_tva_algorithmic_board
					has_country_flag = ADISCORD_vorkerland_focus_tva_specialist_priority
				}
				OR = {
					has_country_flag = ADISCORD_vorkerland_focus_tva_technical_battalions
					has_country_flag = ADISCORD_vorkerland_focus_tva_remote_fire_control
					has_country_flag = ADISCORD_vorkerland_focus_tva_adaptive_logistics
				}
			}
		}
	}

	ai_national_focuses = {
		TVA_codify_utilitarian_directorate
		TVA_publish_operational_metrics
		TVA_deploy_field_laboratories
		TVA_select_trial_protocol
	}

	weight = { factor = 3 }
}

# Variant programmes. The directorate has no succession roll, so unlike WKR and
# VAD the AI picks its variant from the state of the war rather than from a
# collapse flag: a directorate that is already bleeding surrender progress
# converts the civil grid, a directorate that is holding preserves its
# specialists instead. The two enable blocks are complementary and each one
# latches on its own variant flag, so the choice cannot oscillate once made.

ADISCORD_vorkerland_tva_throughput_variant_plan = {
	name = "TVA continuous mobilisation programme"
	desc = ""

	allowed = {
		OR = {
			original_tag = WRK
			original_tag = TVA
		}
	}

	enable = {
		tag = TVA
		is_ai = yes
		has_global_flag = ADISCORD_vorkerland_collapse_wars_started
		NOT = { has_global_flag = ADISCORD_vorkerland_collapse_finished }
		has_government = technocracy
		has_country_leader = { character = TVA_Dorian_Worx ruling_only = yes }
		NOT = { has_country_flag = ADISCORD_vorkerland_tva_variant_preservation }
		NOT = { has_country_flag = ADISCORD_vorkerland_focus_tva_single_process }
		OR = {
			has_country_flag = ADISCORD_vorkerland_tva_variant_throughput
			surrender_progress > 0.05
		}
		OR = {
			has_global_flag = ADISCORD_vorkerland_phase_collapse
			has_global_flag = ADISCORD_vorkerland_phase_regional_consolidation
			has_global_flag = ADISCORD_vorkerland_phase_central_preparation
			has_global_flag = ADISCORD_vorkerland_phase_central_showdown
		}
	}

	abort = {
		OR = {
			NOT = { tag = TVA }
			has_global_flag = ADISCORD_vorkerland_collapse_finished
			has_country_flag = ADISCORD_vorkerland_tva_variant_preservation
			NOT = { has_country_leader = { character = TVA_Dorian_Worx ruling_only = yes } }
			has_country_flag = ADISCORD_vorkerland_focus_tva_single_process
		}
	}

	ai_national_focuses = {
		TVA_declare_continuous_mobilization
		TVA_convert_the_civil_grid
		TVA_run_the_republic_as_one_process
	}

	weight = { factor = 4 }
}
