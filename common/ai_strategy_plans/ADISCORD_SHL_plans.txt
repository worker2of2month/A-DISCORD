# ----- Preliminary course by the Conclave balance -----
# Each plan is weighted by the balance range its ratification requires; the
# first completed preliminary focus aborts the other two.

SHL_prepare_houses_plan = {
	name = "SHL preparation: tenth house"
	desc = ""
	allowed = { original_tag = SHL }
	enable = {
		tag = SHL
		is_ai = yes
		check_variable = { var = SHL_political_course value = 0 compare = equals }
		NOT = { has_completed_focus = SHL_open_conclave }
		NOT = { has_completed_focus = SHL_nur_directorate }
	}
	abort = {
		OR = {
			NOT = { tag = SHL }
			NOT = { check_variable = { var = SHL_political_course value = 0 compare = equals } }
			has_completed_focus = SHL_open_conclave
			has_completed_focus = SHL_nur_directorate
		}
	}
	ai_national_focuses = {
		SHL_new_house
		SHL_house_compromise
		SHL_house_war_mandate
		SHL_admit_tenth_house
	}
	weight = {
		factor = 30
		modifier = {
			factor = 0.1
			NOT = {
				is_power_balance_in_range = { id = SHL_conclave_balance range = SHL_balance_crests_influence }
				is_power_balance_in_range = { id = SHL_conclave_balance range = SHL_balance_crests_dominance }
			}
		}
	}
}
