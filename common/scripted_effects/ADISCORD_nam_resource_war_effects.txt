ADISCORD_nam_resource_war_schedule = {
	if = {
		limit = {
			has_global_flag = ADISCORD_fresh_campaign_contract_v1
			has_global_flag = ADISCORD_vorkerland_collapse_wars_started
			NOT = { has_global_flag = ADISCORD_nam_resource_war_scheduled }
			NOT = { has_global_flag = ADISCORD_nam_resource_war_started }
			NOT = { has_global_flag = ADISCORD_nam_resource_war_resolved }
			NOT = { has_global_flag = ADISCORD_nam_resource_war_retry_used }
			country_exists = NAM
			country_exists = EFL
			country_exists = AZH
		}
		set_global_flag = ADISCORD_nam_resource_war_scheduled
		# The Svetlogorye resource war is deliberately a later regional crisis:
		# preserve the old 90-day variance while moving the whole window +730 days.
		NAM = {
			country_event = {
				id = ADISCORD_nam_resource_war.1
				days = 850
				random_days = 90
			}
		}
	}
}

ADISCORD_nam_resource_war_start = {
	if = {
		limit = { ADISCORD_nam_resource_war_ready = yes }
		NAM = {
			if = {
				limit = { is_subject = yes }
				overlord = { set_autonomy = { target = NAM autonomy_state = autonomy_free } }
			}
			if = { limit = { is_in_faction = yes } leave_faction = yes }
		}
		clr_global_flag = ADISCORD_nam_resource_war_scheduled
		ADISCORD_nam_resource_war_begin_hostilities = yes
	}
}

ADISCORD_nam_resource_war_begin_hostilities = {
	set_global_flag = ADISCORD_nam_resource_war_started
	VAL = {
		if = {
			limit = { exists = yes has_focus_tree = VAL_focus }
			mark_focus_tree_layout_dirty = yes
			country_event = { id = ADISCORD_nam_resource_war.3 hours = 1 }
		}
	}
	clr_global_flag = ADISCORD_nam_resource_war_scheduled
	clr_global_flag = ADISCORD_nam_resource_war_EFL_defeated
	clr_global_flag = ADISCORD_nam_resource_war_AZH_defeated
	clr_global_flag = ADISCORD_nam_resource_war_SLF_defeated

	NAM = {
		add_ideas = ADISCORD_nam_extraction_defence_corps
		add_war_support = 0.20
		# Six resource-war formations arrive fully equipped. Without this
		# reserve the starting 960k core pool is spent on the first two
		# guards and the rest of the OOB sits empty.
		add_manpower = 18000
		load_oob = "NAM_resource_war"
		add_equipment_to_stockpile = { type = infantry_equipment amount = 60000 producer = NAM }
		add_equipment_to_stockpile = { type = ADISCORD_squad_weapons_equipment_0 amount = 80 producer = NAM }
		add_equipment_to_stockpile = { type = support_equipment amount = 900 producer = NAM }
		add_equipment_to_stockpile = { type = artillery_equipment amount = 700 producer = NAM }
	}
	EFL = {
		if = { limit = { is_in_faction = yes } leave_faction = yes }
		add_ideas = ADISCORD_nam_restitution_coalition
		add_war_support = 0.18
		load_oob = "EFL_nam_resource_war"
		add_equipment_to_stockpile = { type = infantry_equipment amount = 30000 producer = EFL }
		add_equipment_to_stockpile = { type = ADISCORD_squad_weapons_equipment_0 amount = 100 producer = EFL }
		add_equipment_to_stockpile = { type = support_equipment amount = 350 producer = EFL }
	}
	AZH = {
		if = { limit = { is_in_faction = yes } leave_faction = yes }
		add_ideas = ADISCORD_nam_restitution_coalition
		add_war_support = 0.22
		load_oob = "AZH_nam_resource_war"
		add_equipment_to_stockpile = { type = infantry_equipment amount = 18000 producer = AZH }
	}

	# Only states with a direct material interest mobilise expeditionary corps:
	# Rahma wants capped fuel prices; Mazar wants priority access for water convoys.
	if = {
		limit = { RHM = { exists = yes is_subject = no has_war = no } }
		RHM = { add_ideas = ADISCORD_nam_profitable_volunteer_contracts }
	}
	if = {
		limit = { MZR = { exists = yes is_subject = no has_war = no } }
		MZR = { add_ideas = ADISCORD_nam_profitable_volunteer_contracts }
	}

	EFL = {
		create_faction_from_template = {
			template = faction_template_ADISCORD_standard
			name = faction_eflor_azhar_restitution_alliance
		}
		add_to_faction = AZH
	}

	67 = { add_claim_by = EFL add_claim_by = AZH }
	# Every principal must fall before native faction peace can end the campaign.
	NAM = { ADISCORD_nam_resource_war_reserve_major = yes }
	EFL = { ADISCORD_nam_resource_war_reserve_major = yes }
	AZH = { ADISCORD_nam_resource_war_reserve_major = yes }
	EFL = { declare_war_on = { target = NAM type = annex_everything } }
	AZH = {
		add_to_war = {
			targeted_alliance = EFL
			enemy = NAM
			hostility_reason = asked_to_join
			single_target_only = yes
		}
	}

	NAM = {
		if = {
			limit = { NAM_compact_effective = yes }
			set_country_flag = NAM_compact_ratified
			set_global_flag = ADISCORD_nam_resource_war_SLF_defeated
		}
		else = {
			random_list = {
				50 = { country_event = { id = ADISCORD_nam_resource_war.2 days = 45 random_days = 30 } }
				50 = { set_global_flag = ADISCORD_nam_resource_war_SLF_defeated }
			}
		}
		activate_mission = NAM_defend_fields
		country_event = { id = ADISCORD_nam_resource_war.11 hours = 1 }
		country_event = { id = ADISCORD_nam_resource_war.50 days = 420 }
		news_event = { id = ADISCORD_nam_resource_news.1 hours = 1 }
		ADISCORD_nam_show_resource_war_superevent = yes
	}
}

	set_global_flag = ADISCORD_nam_resource_war_resolved
	VAL = { if = { limit = { exists = yes has_focus_tree = VAL_focus } mark_focus_tree_layout_dirty = yes } }
	set_global_flag = ADISCORD_nam_resource_war_peaceful_partition
	clr_global_flag = ADISCORD_nam_resource_war_started
	clr_global_flag = ADISCORD_nam_resource_war_scheduled
	clr_global_flag = ADISCORD_nam_resource_war_EFL_defeated
	clr_global_flag = ADISCORD_nam_resource_war_AZH_defeated
	clr_global_flag = ADISCORD_nam_resource_war_SLF_defeated

	EFL = { if = { limit = { is_in_faction = yes } leave_faction = yes } }
	AZH = { if = { limit = { is_in_faction = yes } leave_faction = yes } }

	EFL = {
		create_faction_from_template = {
			template = faction_template_ADISCORD_standard
			name = faction_eflor_azhar_restitution_alliance
		}
		add_to_faction = AZH
		transfer_state = 67
		transfer_state = 225
		transfer_state = 228
		transfer_state = 230
		transfer_state = 231
		transfer_state = 688
	}
	AZH = {
		transfer_state = 226
		transfer_state = 227
		transfer_state = 229
		transfer_state = 689
		transfer_state = 690
	}

	67 = { add_core_of = EFL set_state_controller_to = EFL }
	225 = { add_core_of = EFL set_state_controller_to = EFL }
	228 = { add_core_of = EFL set_state_controller_to = EFL }
	230 = { add_core_of = EFL set_state_controller_to = EFL }
	231 = { add_core_of = EFL set_state_controller_to = EFL }
	688 = { remove_core_of = SLF remove_core_of = NAM add_core_of = EFL set_state_controller_to = EFL }

	226 = { add_core_of = AZH set_state_controller_to = AZH }
	227 = { add_core_of = AZH set_state_controller_to = AZH }
	229 = { add_core_of = AZH set_state_controller_to = AZH }
	689 = { remove_core_of = NAM add_core_of = AZH set_state_controller_to = AZH }
	700 = {
		# A transfer with no previous controller can crash native province updates.
		if = {
			limit = {
				country_exists = OWNER
				country_exists = CONTROLLER
			}
			AZH = { transfer_state = 700 }
			remove_core_of = NAM
			add_core_of = AZH
			set_state_controller_to = AZH
		}
	}
	690 = { remove_core_of = NAM add_core_of = AZH set_state_controller_to = AZH }

	if = {
		limit = { NAM = { exists = yes } }
		EFL = { annex_country = { target = NAM transfer_troops = no } }
	}

	ADISCORD_nam_resource_war_clear_temporary_support = yes
	EFL = { news_event = { id = ADISCORD_nam_resource_news.7 hours = 1 } }
}

ADISCORD_nam_refresh_emergency_defence = {
	if = {
		limit = {
			has_idea = ADISCORD_nam_last_line_administration
			OR = {
				is_ai = no
				NOT = { ADISCORD_nam_resource_war_active = yes }
				surrender_progress < 0.45
			}
		}
		remove_ideas = ADISCORD_nam_last_line_administration
	}
	if = {
		limit = {
			is_ai = yes
			ADISCORD_nam_resource_war_active = yes
			surrender_progress > 0.60
			NOT = { has_idea = ADISCORD_nam_last_line_administration }
		}
		add_ideas = ADISCORD_nam_last_line_administration
	}
	if = {
		limit = {
			is_ai = yes
			ADISCORD_nam_resource_war_active = yes
			surrender_progress > 0.40
			NOT = { has_country_flag = ADISCORD_nam_emergency_reserve_called }
		}
		set_country_flag = ADISCORD_nam_emergency_reserve_called
		add_manpower = 7000
		add_equipment_to_stockpile = { type = infantry_equipment amount = 15000 producer = NAM }
		add_equipment_to_stockpile = { type = support_equipment amount = 300 producer = NAM }
		add_equipment_to_stockpile = { type = artillery_equipment amount = 150 producer = NAM }
		capital_scope = {
			create_unit = {
				division = "name = \"Last Line Guard\" division_template = \"NAM Resource Guard\" start_experience_factor = 0.20 start_equipment_factor = 1.00"
				owner = NAM
			}
			create_unit = {
				division = "name = \"Last Line Infantry\" division_template = \"NAM Resource Infantry\" start_experience_factor = 0.15 start_equipment_factor = 1.00"
				owner = NAM
			}
		}
	}
}

ADISCORD_nam_resource_war_resolve_nam_victory = {
	if = {
		limit = { ADISCORD_nam_resource_war_active = yes }
		set_global_flag = ADISCORD_nam_resource_war_resolved
		VAL = { if = { limit = { exists = yes has_focus_tree = VAL_focus } mark_focus_tree_layout_dirty = yes } }
		set_global_flag = ADISCORD_nam_resource_war_nam_victory
		clr_global_flag = ADISCORD_nam_resource_war_started
		ADISCORD_nam_resource_war_end_all_wars = yes
		NAM = {
			if = {
				limit = { SLF = { exists = yes } }
				annex_country = { target = SLF transfer_troops = no }
			}
			transfer_state = 688
		}
		688 = { remove_core_of = SLF add_core_of = NAM set_state_controller_to = NAM }

		# The Middle Loren district and its urban enclave are the annexation. State 691
		# touches NAM's mainland; state 68 is separated by Eflorian territory and
		# would leave an isolated Svetlogorsk enclave. Azhar survives while
		# recognising a Svetlogorsk claim to the Black Basin.
		NAM = {
			transfer_state = 691
		}
		691 = { remove_core_of = EFL add_core_of = NAM set_state_controller_to = NAM }
		701 = {
			# Keep ownership, cores and control under the same native validity gate.
			if = {
				limit = {
					country_exists = OWNER
					country_exists = CONTROLLER
				}
				NAM = { transfer_state = 701 }
				remove_core_of = EFL
				add_core_of = NAM
				set_state_controller_to = NAM
			}
		}
		69 = { add_claim_by = NAM }
		NAM = {
			set_country_flag = NAM_treaty_independence_recognised
			add_ideas = ADISCORD_nam_treaty_independence
			add_stability = 0.15
		}
		ADISCORD_nam_resource_war_clear_temporary_support = yes
		VAL = { VAL_deliver_nam_concession = yes VAL_settle_resource_aid = yes }
		NAM = { news_event = { id = ADISCORD_nam_resource_news.3 hours = 1 } }
	}
}

# External annexation, subjugation or peace invalidates the authored contest.
ADISCORD_nam_resource_war_close_external = {
	if = {
		limit = {
			ADISCORD_nam_resource_war_active = yes
			OR = {
				NAM = { OR = { exists = no is_subject = yes } }
				EFL = { OR = { exists = no is_subject = yes } }
				AZH = { OR = { exists = no is_subject = yes } }
				NAM = { NOT = { has_war_with = EFL } NOT = { has_war_with = AZH } }
			}
		}
		set_global_flag = ADISCORD_nam_resource_war_resolved
		clr_global_flag = ADISCORD_nam_resource_war_started
		ADISCORD_nam_resource_war_end_all_wars = yes
		ADISCORD_nam_resource_war_clear_temporary_support = yes
		VAL = { VAL_cancel_nam_concession = yes VAL_settle_resource_aid = yes }
	}
}
