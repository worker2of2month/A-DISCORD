ADISCORD_nam_resource_war_schedule = {
	if = {
		limit = {
			has_global_flag = ADISCORD_fresh_campaign_contract_v1
			has_global_flag = ADISCORD_vorkerland_collapse_wars_started
			NOT = { has_global_flag = ADISCORD_nam_resource_war_scheduled }
			NOT = { has_global_flag = ADISCORD_nam_resource_war_started }
			NOT = { has_global_flag = ADISCORD_nam_resource_war_resolved }
			NOT = { has_global_flag = ADISCORD_nam_resource_war_retry_used }
			country_exists = NAM
			country_exists = EFL
			country_exists = AZH
		}
		set_global_flag = ADISCORD_nam_resource_war_scheduled
		# The scheduled flag keeps the collapse date: the restitution clock reads
		# its age for the 300-day minimum. Event .1 at 420-480 days is the backstop.
		NAM = {
			set_variable = { var = NAM_rw_tension value = 10 }
			NAM_rw_count_entry_dispatch = yes
			country_event = {
				id = ADISCORD_nam_resource_war.1
				days = 420
				random_days = 60
			}
		}
	}
	# The same collapse edge proclaims independence and creates the ladder,
	# whether or not the neighbours can open the restitution campaign.
	NAM = { NAM_ladder_declare_independence = yes }
}

# Every queued ADISCORD_nam_resource_war.1 is counted. A dispatch that fails
# readiness may stop the entry only when no other dispatch is still queued, so
# a pending retry always owns the final decision and no war follows a refund.
NAM_rw_count_entry_dispatch = {
	add_to_variable = { var = global.NAM_rw_entry_dispatches value = 1 }
}

ADISCORD_nam_resource_war_start = {
	if = {
		limit = { ADISCORD_nam_resource_war_ready = yes }
		NAM = {
			if = {
				limit = { is_subject = yes }
				overlord = { set_autonomy = { target = NAM autonomy_state = autonomy_free } }
			}
			if = {
				limit = { is_in_faction = yes }
				leave_faction = yes
			}
		}
		clr_global_flag = ADISCORD_nam_resource_war_scheduled
		ADISCORD_nam_resource_war_begin_hostilities = yes
	}
}

			add_ideas = ADISCORD_nam_extraction_defence_corps

		# their recruits when they were placed and are delivered from escrow.

		add_equipment_to_stockpile = { type = infantry_equipment amount = 20000 producer = NAM }
		add_equipment_to_stockpile = { type = ADISCORD_squad_weapons_equipment_0 amount = 80 producer = NAM }
		add_equipment_to_stockpile = { type = support_equipment amount = 300 producer = NAM }
		add_equipment_to_stockpile = { type = artillery_equipment amount = 250 producer = NAM }
		# The templates exist before any unit refers to them: the baseline OOB
		# carries units only and every paid order spawns from the same templates.
		NAM_order_ensure_templates = yes
		load_oob = "NAM_resource_war"
		NAM_deliver_mobilisation_orders = yes
		NAM_seal_discipline_begin = yes
	}

		if = {
			limit = { is_in_faction = yes }
			leave_faction = yes
		}

				name = "Line Infantry Brigade"
				template_counter = 3
				division_names_group = ADISCORD_INF_01
				regiments = {
					infantry = { x = 0 y = 0 }
					infantry = { x = 0 y = 1 }
					infantry = { x = 0 y = 2 }
					infantry = { x = 1 y = 0 }
					infantry = { x = 1 y = 1 }
					infantry = { x = 1 y = 2 }
				}
				support = {
					engineer = { x = 0 y = 0 }
				}
			}
		}
		load_oob = "EFL_nam_resource_war"
		add_equipment_to_stockpile = { type = infantry_equipment amount = 30000 producer = EFL }
		add_equipment_to_stockpile = { type = ADISCORD_squad_weapons_equipment_0 amount = 100 producer = EFL }
		add_equipment_to_stockpile = { type = support_equipment amount = 350 producer = EFL }
	}
	AZH = {
		if = {
			limit = { is_in_faction = yes }
			leave_faction = yes
		}
		add_ideas = ADISCORD_nam_restitution_coalition
		add_war_support = 0.22
		if = {
			limit = { NOT = { has_template = "Basin Levy" } }
			division_template = {
				name = "Basin Levy"
				template_counter = 7
				division_names_group = ADISCORD_INF_01
				regiments = {
					ADISCORD_militia = { x = 0 y = 0 }
					ADISCORD_militia = { x = 0 y = 1 }
					ADISCORD_militia = { x = 1 y = 0 }
				}
			}
		}
		load_oob = "AZH_nam_resource_war"
		add_equipment_to_stockpile = { type = infantry_equipment amount = 18000 producer = AZH }
	}
	if = {
		limit = { NAM = { has_country_flag = NAM_rw_first_strike } }
		EFL = { add_war_support = 0.10 }
		AZH = { add_war_support = 0.10 }
	}

	# Only states with a direct material interest mobilise expeditionary corps:
	# Rahma wants capped fuel prices; Mazar wants priority access for water convoys.
	# A buyer already holding a Svetlogorsk fuel contract has nothing to win.
	if = {
		limit = {
			RHM = {
				exists = yes
				is_subject = no

		NAM_oil_validate_clients = yes

			set_country_flag = NAM_compact_ratified

# Timeout settlement for an oil score of 36-99. Svetlogorye survives as a rump:
# it keeps every field it still controls and its islands, cedes the fields held
# by the coalition and owes a year of fuel tribute.
ADISCORD_nam_resource_war_resolve_compromise = {
	if = {
		limit = { ADISCORD_nam_resource_war_active = yes }
		set_global_flag = ADISCORD_nam_resource_war_resolved
		NAM = { mark_focus_tree_layout_dirty = yes }
		VAL = {
			if = {
				limit = {
					exists = yes
					has_focus_tree = VAL_focus
				}
				mark_focus_tree_layout_dirty = yes
			}
		}
		set_global_flag = ADISCORD_nam_resource_war_compromise
		VAL = { VAL_cancel_nam_concession = yes }
		clr_global_flag = ADISCORD_nam_resource_war_started

		# Ownership follows wartime control before white peace returns occupations.
		67 = { NAM_rw_compromise_cede_field = yes }
		689 = { NAM_rw_compromise_cede_field = yes }
		690 = { NAM_rw_compromise_cede_field = yes }
		719 = { NAM_rw_compromise_cede_field = yes }
		688 = {
			if = {
				limit = { is_controlled_by = NAM }
				if = {
					limit = { NOT = { is_owned_by = NAM } }
					NAM = { transfer_state = 688 }
				}
				remove_core_of = SLF
			}
			else = {
				EFL = { transfer_state = 688 }
				remove_core_of = SLF
				remove_core_of = NAM
				add_core_of = EFL
				add_claim_by = NAM
				set_state_controller_to = EFL
			}
		}
		if = {
			limit = { SLF = { exists = yes } }
			688 = {
				owner = {
					if = {
						limit = { NOT = { tag = SLF } }
						annex_country = { target = SLF transfer_troops = no }
					}
				}
			}
		}
		ADISCORD_nam_resource_war_end_all_wars = yes
		NAM = {
			every_owned_state = {
				limit = { NOT = { is_controlled_by = NAM } }
				set_state_controller_to = NAM
			}
		}

		ADISCORD_nam_resource_war_clear_temporary_support = yes
		VAL = { VAL_settle_resource_aid = yes }
		# Tribute replaces any embargo on the two neighbours.
		EFL = { NAM_oil_lift_embargo = yes }
		AZH = { NAM_oil_lift_embargo = yes }
		NAM = {
			NAM_rw_start_tribute = yes
			load_focus_tree = {
				tree = NAM_defeated_focus
				keep_completed = no
			}
		}
		EFL = { news_event = { id = ADISCORD_nam_resource_news.7 hours = 1 } }
	}
}

		remove_core_of = NAM
		add_core_of = EFL
		add_claim_by = NAM
		set_state_controller_to = EFL
	}
}

# Capitulation, a lost front, a lost timeout and a lost war of return share this
# treaty. Svetlogorye loses the whole mainland; while it owns an overseas
# district it survives there in exile, otherwise it is annexed.
ADISCORD_nam_resource_war_resolve_coalition_victory = {
	if = {
		limit = {
			OR = {
				ADISCORD_nam_resource_war_active = yes
				NAM_war_of_return_active = yes
			}
		}
		if = {
			limit = { NAM_war_of_return_active = yes }
			NAM_war_of_return_close = yes
		}
		set_global_flag = ADISCORD_nam_resource_war_resolved
		NAM = { mark_focus_tree_layout_dirty = yes }
		VAL = {
			if = {
				limit = {
					exists = yes
					has_focus_tree = VAL_focus
				}
				mark_focus_tree_layout_dirty = yes
			}
		}
		set_global_flag = ADISCORD_nam_resource_war_coalition_victory
		VAL = { VAL_cancel_nam_concession = yes }
		clr_global_flag = ADISCORD_nam_resource_war_started
		# The flight to the islands is decided while the mainland is still
		# Svetlogorsk and its divisions still stand there, before the peace
		# moves any unit.
		if = {
			limit = { NAM_rw_can_go_into_exile = yes }
			NAM = { NAM_rw_exile_evacuate = yes }
		}
		ADISCORD_nam_resource_war_end_all_wars = yes

		EFL = {
			if = {
				limit = { SLF = { exists = yes } }
				annex_country = { target = SLF transfer_troops = no }
			}
		}
		# Eflor inherits the uprising district together with the uprising.
		688 = {
			remove_core_of = SLF
			if = {
				limit = { is_owned_by = EFL }
				remove_core_of = NAM
				add_core_of = EFL
				add_claim_by = NAM
				set_state_controller_to = EFL
			}
		}
		# The mainland goes to the victors in both outcomes: Eflor takes the
		# Northern Deposits and Svetlogorsk, Azhar the southern fields and the
		# harbour, Eflor any other mainland district Svetlogorye still owns.
		67 = { NAM_rw_cede_mainland_state = yes }
		688 = { NAM_rw_cede_mainland_state = yes }
		689 = { NAM_rw_cede_mainland_state = yes }
		690 = { NAM_rw_cede_mainland_state = yes }
		700 = { NAM_rw_cede_mainland_state = yes }
		719 = { NAM_rw_cede_mainland_state = yes }
		if = {
			limit = { NAM_rw_can_go_into_exile = yes }
			NAM = {
				every_owned_state = {
					limit = { NOT = { NAM_rw_island_state = yes } }
					NAM_rw_cede_mainland_state = yes
				}
			}
			NAM_rw_settle_island_exile = yes
		}
		else = {
			# Without an overseas district nothing is left to govern: the
			# original partition annexes the remainder to Eflor.
			clr_global_flag = ADISCORD_nam_resource_war_island_exile
			if = {
				limit = { NAM = { exists = yes } }
				EFL = { annex_country = { target = NAM transfer_troops = no } }
			}
			ADISCORD_nam_resource_war_clear_temporary_support = yes
			VAL = { VAL_settle_resource_aid = yes }
			NAM_oil_close_all = yes
		}
		EFL = { news_event = { id = ADISCORD_nam_resource_news.2 hours = 1 } }
	}
}

# STATE owned by NAM on the mainland. Azhar takes the southern fields and the
# harbour while it exists, Eflor everything else. Svetlogorye keeps a claim on
# the fields for a later war of return.

	if = {
		limit = {
			is_owned_by = NAM
			OR = {
				state = 689
				state = 690
				state = 700
				state = 719
			}
			country_exists = AZH
		}

		remove_core_of = NAM
		add_core_of = AZH
		add_claim_by = NAM
		set_state_controller_to = AZH
	}
	else_if = {
		limit = { is_owned_by = NAM }
		EFL = { transfer_state = PREV }
		remove_core_of = NAM
		add_core_of = EFL
		add_claim_by = NAM
		set_state_controller_to = EFL
	}
}

# ----- Flight to the islands -----
# COUNTRY NAM, inside the coalition treaty before any state changes hands.
# The transports carry the real divisions, with their own templates: 35% of
# the army that stood at the end of the war, rounded, and at least one. The
# rest is left behind division by division in a fixed order: columns cut off
# in mainland states Svetlogorye no longer controls, the front (719, 690, 688,
# 67, 689), the principals' ground (720, 721, 722, 701, 68, 70, 691, 69, 692),
# any other mainland state and the South Harbour, where the transports load,
# last. Divisions already on the islands or at sea are never left behind.
# The survivors move to the island capital, and the whole army in exile loses
# 45% strength and half its organisation for want of weapons; the stocks lose
# 65% over the side.
NAM_rw_exile_evacuate = {
	set_variable = { var = NAM_exile_divisions_before value = num_divisions }
	set_variable = { var = NAM_exile_divisions_keep value = num_divisions }
	multiply_variable = { var = NAM_exile_divisions_keep value = 0.35 }
	round_variable = NAM_exile_divisions_keep
	clamp_variable = {
		var = NAM_exile_divisions_keep
		min = 1
		max = 1000
	}
	set_temp_variable = { var = NAM_exile_to_cut value = NAM_exile_divisions_before }
	subtract_from_temp_variable = { var = NAM_exile_to_cut value = NAM_exile_divisions_keep }
	NAM_rw_exile_move_capital = yes
	every_owned_state = {
		limit = {
			NAM_rw_island_state = yes
			NOT = { is_controlled_by = NAM }
		}
		set_state_controller_to = NAM
	}
	# No division-scope delete exists: the abandoned divisions switch to a
	# temporary template, and deleting that template removes exactly them.
	if = {
		limit = { check_variable = { NAM_exile_to_cut > 0 } }
		if = {
			limit = { NOT = { has_template = "Abandoned Columns" } }
			division_template = {
				name = "Abandoned Columns"
				template_counter = 6
				division_names_group = ADISCORD_INF_01
				regiments = {
					infantry = { x = 0 y = 0 }
					infantry = { x = 0 y = 1 }
					infantry = { x = 0 y = 2 }
				}
				support = {
					artillery = { x = 0 y = 0 }
				}
			}
		}
		every_state = {
			limit = {
				check_variable = { NAM_exile_to_cut > 0 }
				NOT = { NAM_rw_island_state = yes }
				NOT = { is_controlled_by = NAM }
				any_state_division = { OWNER = { tag = NAM } }
			}
			NAM_rw_exile_abandon_state = yes
		}
		719 = { NAM_rw_exile_abandon_state = yes }
		690 = { NAM_rw_exile_abandon_state = yes }
		688 = { NAM_rw_exile_abandon_state = yes }
		67 = { NAM_rw_exile_abandon_state = yes }
		689 = { NAM_rw_exile_abandon_state = yes }
		720 = { NAM_rw_exile_abandon_state = yes }
		721 = { NAM_rw_exile_abandon_state = yes }
		722 = { NAM_rw_exile_abandon_state = yes }
		701 = { NAM_rw_exile_abandon_state = yes }
		68 = { NAM_rw_exile_abandon_state = yes }
		70 = { NAM_rw_exile_abandon_state = yes }
		691 = { NAM_rw_exile_abandon_state = yes }
		69 = { NAM_rw_exile_abandon_state = yes }
		692 = { NAM_rw_exile_abandon_state = yes }
		if = {
			limit = { check_variable = { NAM_exile_to_cut > 0 } }
			every_state = {
				limit = {
					check_variable = { NAM_exile_to_cut > 0 }
					NOT = { NAM_rw_island_state = yes }
					NOT = { state = 700 }
					any_state_division = { OWNER = { tag = NAM } }
				}
				NAM_rw_exile_abandon_state = yes
			}
		}
		700 = { NAM_rw_exile_abandon_state = yes }
		if = {
			limit = { has_template = "Abandoned Columns" }
			delete_unit = {
				division_template = "Abandoned Columns"
				disband = no
			}
			delete_unit_template_and_units = {
				division_template = "Abandoned Columns"
				disband = no
			}
		}
	}
	set_variable = { var = NAM_exile_divisions_after value = num_divisions }
	set_variable = { var = NAM_exile_divisions_lost value = NAM_exile_divisions_before }
	subtract_from_variable = { var = NAM_exile_divisions_lost value = NAM_exile_divisions_after }
	# The survivors embark before the peace moves any unit or state.
	every_state = {
		limit = {
			NOT = { NAM_rw_island_state = yes }
			any_state_division = { OWNER = { tag = NAM } }
		}
		NAM_rw_exile_embark_state = yes
	}
	NAM_rw_exile_damage_army = yes
	# Equipment that the transports cannot take goes over the side.
	set_variable = { var = NAM_exile_rifles_dumped value = num_equipment@infantry_equipment }
	set_temp_variable = { var = NAM_exile_dump value = num_equipment@infantry_equipment }
	multiply_temp_variable = { var = NAM_exile_dump value = -0.65 }
	round_temp_variable = NAM_exile_dump
	add_equipment_to_stockpile = { type = infantry_equipment amount = NAM_exile_dump producer = NAM }
	subtract_from_variable = { var = NAM_exile_rifles_dumped value = num_equipment@infantry_equipment }
	set_variable = { var = NAM_exile_heavy_dumped value = num_equipment@support_equipment }
	add_to_variable = { var = NAM_exile_heavy_dumped value = num_equipment@artillery_equipment }
	set_temp_variable = { var = NAM_exile_dump value = num_equipment@support_equipment }
	multiply_temp_variable = { var = NAM_exile_dump value = -0.65 }
	round_temp_variable = NAM_exile_dump
	add_equipment_to_stockpile = { type = support_equipment amount = NAM_exile_dump producer = NAM }
	set_temp_variable = { var = NAM_exile_dump value = num_equipment@artillery_equipment }
	multiply_temp_variable = { var = NAM_exile_dump value = -0.65 }
	round_temp_variable = NAM_exile_dump
	add_equipment_to_stockpile = { type = artillery_equipment amount = NAM_exile_dump producer = NAM }
	subtract_from_variable = { var = NAM_exile_heavy_dumped value = num_equipment@support_equipment }
	subtract_from_variable = { var = NAM_exile_heavy_dumped value = num_equipment@artillery_equipment }
	set_temp_variable = { var = NAM_exile_dump value = num_equipment@ADISCORD_squad_weapons_equipment }
	multiply_temp_variable = { var = NAM_exile_dump value = -0.65 }
	round_temp_variable = NAM_exile_dump
	add_equipment_to_stockpile = { type = ADISCORD_squad_weapons_equipment amount = NAM_exile_dump producer = NAM }
	# The remaining-mainland count is diagnostic only; it needs a separate world scan.
	if = {
		limit = { is_debug = yes }
		set_temp_variable = { var = NAM_exile_mainland_left value = 0 }
		every_state = {
			limit = {
				NOT = { NAM_rw_island_state = yes }
				any_state_division = { OWNER = { tag = NAM } }
			}
			add_to_temp_variable = { var = NAM_exile_mainland_left value = 1 }
		}
		log = "[ADISCORD][NAM][EXILE] Evacuation: divisions [?NAM_exile_divisions_before] before, [?NAM_exile_divisions_keep] carried by plan, [?NAM_exile_divisions_after] left, [?NAM_exile_divisions_lost] abandoned, [?NAM_exile_to_cut] still to abandon; mainland states still holding NAM divisions [?NAM_exile_mainland_left]; capital [NAM.Capital.GetName]; rifles dumped [?NAM_exile_rifles_dumped], heavy equipment dumped [?NAM_exile_heavy_dumped]."
	}
}

ADISCORD_nam_resource_war_resolve_nam_victory = {
	if = {
		limit = { ADISCORD_nam_resource_war_active = yes }
		set_global_flag = ADISCORD_nam_resource_war_resolved
		NAM = { mark_focus_tree_layout_dirty = yes }
		VAL = {
			if = {
				limit = {
					exists = yes
					has_focus_tree = VAL_focus
				}
				mark_focus_tree_layout_dirty = yes
			}
		}
		set_global_flag = ADISCORD_nam_resource_war_nam_victory
		clr_global_flag = ADISCORD_nam_resource_war_started
		ADISCORD_nam_resource_war_end_all_wars = yes
		NAM = {
			if = {
				limit = { SLF = { exists = yes } }
				annex_country = { target = SLF transfer_troops = no }
			}
			transfer_state = 688
		}
		688 = {
			remove_core_of = SLF
			add_core_of = NAM
			set_state_controller_to = NAM
		}

		# ----- Border award of the victory treaty -----
		# Each defeated principal cedes its border sliver: Eflor the Moren district
		# (720), the Loren borderland (721) and the Middle Loren city (701), which
		# touches Svetlogorye only through 720; Azhar cedes its seaboard (722).
		# Only a sliver still owned by that principal changes hands. Azhar keeps
		# its capital state 69, on which Svetlogorye retains a claim to the Black
		# Basin; post-war normalisation can renounce it.
		720 = { NAM_rw_award_border_state = yes }
		721 = { NAM_rw_award_border_state = yes }
		701 = { NAM_rw_award_border_state = yes }
		722 = { NAM_rw_award_border_state = yes }
		69 = { add_claim_by = NAM }
		# ----- End of the border award -----
		NAM_rw_punish_defeated_coalition = yes
		NAM = {
			set_country_flag = NAM_treaty_independence_recognised
			add_ideas = ADISCORD_nam_treaty_independence
			add_stability = 0.15
		}
		ADISCORD_nam_resource_war_clear_temporary_support = yes
		VAL = {
			VAL_deliver_nam_concession = yes
			VAL_settle_resource_aid = yes
		}
		# The post-war chapter keeps every completed focus of the main tree.
		NAM = {
			NAM_oil_refresh = yes
			load_focus_tree = {
				tree = NAM_postwar_focus
				keep_completed = yes
			}
			news_event = { id = ADISCORD_nam_resource_news.3 hours = 1 }
		}
	}
}

# Timeout settlement for a held basin without a captured border district.
# The fighting ends on the pre-war border: Svetlogorye keeps its independence
# and loads the post-war chapter, but signs no recognition treaty, wins no
# border award and imposes no punishment. ADISCORD_nam_resource_war_nam_victory
# marks every outcome in which Svetlogorye held: Kefreyt's concession and aid,
# the post-war tree and the South read it. The standoff flag distinguishes the
# texts and the missing recognition treaty.
ADISCORD_nam_resource_war_resolve_standoff = {
	if = {
		limit = { ADISCORD_nam_resource_war_active = yes }
		set_global_flag = ADISCORD_nam_resource_war_resolved
		NAM = { mark_focus_tree_layout_dirty = yes }
		VAL = {
			if = {
				limit = {
					exists = yes
					has_focus_tree = VAL_focus
				}
				mark_focus_tree_layout_dirty = yes
			}
		}
		set_global_flag = ADISCORD_nam_resource_war_nam_victory
		set_global_flag = ADISCORD_nam_resource_war_standoff
		clr_global_flag = ADISCORD_nam_resource_war_started
		ADISCORD_nam_resource_war_end_all_wars = yes
		# The uprising is an internal front: its district returns to Svetlogorye.
		NAM = {
			if = {
				limit = { SLF = { exists = yes } }
				annex_country = { target = SLF transfer_troops = no }
			}
			if = {
				limit = { 688 = { is_owned_by = SLF } }
				transfer_state = 688
			}
		}
		688 = {
			if = {
				limit = { is_owned_by = NAM }
				remove_core_of = SLF
				set_state_controller_to = NAM
			}
		}
		NAM = {
			every_owned_state = {
				limit = { NOT = { is_controlled_by = NAM } }
				set_state_controller_to = NAM
			}
		}
		ADISCORD_nam_resource_war_clear_temporary_support = yes
		VAL = {
			VAL_deliver_nam_concession = yes
			VAL_settle_resource_aid = yes
		}
		NAM = {
			NAM_oil_refresh = yes
			load_focus_tree = {
				tree = NAM_postwar_focus
				keep_completed = yes
			}
			news_event = { id = ADISCORD_nam_resource_news.3 hours = 1 }
		}
	}
}

# COUNTRY native ROOT. Keep the receipt through both capitulation callbacks:
# the final opponent may not expose has_capitulated before the treaty runs.
ADISCORD_nam_resource_war_record_capitulation = {
	if = {
		limit = {
			ADISCORD_nam_resource_war_defender_capitulation = yes
			NOT = { has_country_flag = ADISCORD_nam_capitulation_reserved }
		}
		set_country_flag = {
			flag = ADISCORD_nam_capitulation_reserved
			value = 1
			days = 2
		}
		if = {
			limit = { tag = EFL }
			NAM = { ADISCORD_nam_resource_war_mark_eflor_defeated = yes }
		}
		else_if = {
			limit = { tag = AZH }
			NAM = { ADISCORD_nam_resource_war_mark_azhar_defeated = yes }
		}
		else_if = {
			limit = { tag = SLF }
			NAM = { ADISCORD_nam_resource_war_mark_rebels_defeated = yes }
		}
	}
}

# COUNTRY native ROOT of on_uncapitulation. A liberated principal is no longer

}

# Settlement by oil score: the fields Svetlogorye still controls count
# 67 = 80, 688 = 12, 689 = 36, 690 = 24, 719 = 12. 100 or more keeps
# independence, 36-99 is the compromise peace, less than 36 is the partition.
# A held basin wins the border treaty only when Svetlogorye has also taken a
# border district of a principal; holding its own fields alone is a standoff.
ADISCORD_nam_resource_war_timeout_resolution = {
	if = {
		limit = { ADISCORD_nam_resource_war_active = yes }
		NAM_rw_compute_oil_score = yes
		if = {
			limit = {

# Svetlogorye holding every border belt and all its own fields wins the
# border treaty; the coalition holding the Northern Deposits and one more

# ----- Restitution pressure -----
# NAM_rw_tension (0-100) is the restitution clock. Every writer calls
# NAM_rw_tension_clamp, so monthly drift, focus rewards, decisions and events
# cross the same thresholds; each threshold fires once per campaign.

# COUNTRY NAM.
NAM_rw_tension_clamp = {
	clamp_variable = {
		var = NAM_rw_tension
		min = 0
		max = 100
	}
	if = {
		limit = { NAM_rw_clock_running = yes }
		if = {
			limit = {
				NOT = { has_country_flag = NAM_rw_notes_sent }
				check_variable = { var = NAM_rw_tension value = 40 compare = greater_than_or_equals }
			}
			set_country_flag = NAM_rw_notes_sent
			country_event = { id = ADISCORD_nam_resource_war.20 hours = 6 }
		}
		if = {
			limit = {
				NOT = { has_country_flag = NAM_rw_ultimatum_sent }
				check_variable = { var = NAM_rw_tension value = 70 compare = greater_than_or_equals }
			}
			set_country_flag = NAM_rw_ultimatum_sent
			country_event = { id = ADISCORD_nam_resource_war.21 days = 2 }
		}
		# Full pressure starts the war once the minimum period since the collapse
		# has passed; a later monthly pass dispatches it if the window was closed.
		if = {
			limit = {
				NOT = { has_country_flag = NAM_rw_war_dispatched }
				check_variable = { var = NAM_rw_tension value = 100 compare = greater_than_or_equals }
				NAM_rw_war_window_open = yes
			}
			set_country_flag = NAM_rw_war_dispatched
			NAM_rw_count_entry_dispatch = yes
			country_event = { id = ADISCORD_nam_resource_war.1 days = 7 random_days = 14 }
		}
	}
}

			subtract_from_temp_variable = { var = NAM_rw_tension_delta value = 2 }
		}
		if = {
			limit = { AZH = { has_country_flag = NAM_oil_client } }
			subtract_from_temp_variable = { var = NAM_rw_tension_delta value = 2 }
		}

			limit = { EFL = { has_dynamic_modifier = { modifier = NAM_fuel_embargo_modifier } } }

			limit = { AZH = { has_dynamic_modifier = { modifier = NAM_fuel_embargo_modifier } } }
			add_to_temp_variable = { var = NAM_rw_tension_delta value = 1 }
		}

# COUNTRY NAM. Input: temp NAM_order_kind (1 guard, 2 infantry, 3 motorised,
# 4 coastal). The formation takes the first fully held state of its own list:
# guard 67; infantry 690, 719, 689; motorised 67, 688; coastal 700, 689.
# Otherwise any fully held mainland state takes it, with the harbour, the port
# city and Svetlogorsk first, then any state Svetlogorye still holds, so a lost
# field or an occupied capital never swallows a paid formation.
NAM_order_spawn_one = {
	set_temp_variable = { var = NAM_order_route value = 0 }
	if = {
		limit = {
			check_variable = { NAM_order_kind = 1 }
			has_full_control_of_state = 67
		}
		67 = { NAM_create_order_by_kind = yes }
		set_temp_variable = { var = NAM_order_route value = 1 }
	}
	else_if = {
		limit = {
			check_variable = { NAM_order_kind = 2 }
			has_full_control_of_state = 690
		}
		690 = { NAM_create_order_by_kind = yes }
		set_temp_variable = { var = NAM_order_route value = 1 }
	}
	else_if = {
		limit = {
			check_variable = { NAM_order_kind = 2 }
			has_full_control_of_state = 719
		}
		719 = { NAM_create_order_by_kind = yes }
		set_temp_variable = { var = NAM_order_route value = 1 }
	}
	else_if = {
		limit = {
			check_variable = { NAM_order_kind = 2 }
			has_full_control_of_state = 689
		}
		689 = { NAM_create_order_by_kind = yes }
		set_temp_variable = { var = NAM_order_route value = 1 }
	}
	else_if = {
		limit = {
			check_variable = { NAM_order_kind = 3 }
			has_full_control_of_state = 67
		}
		67 = { NAM_create_order_by_kind = yes }
		set_temp_variable = { var = NAM_order_route value = 1 }
	}
	else_if = {
		limit = {
			check_variable = { NAM_order_kind = 3 }
			has_full_control_of_state = 688
		}
		688 = { NAM_create_order_by_kind = yes }
		set_temp_variable = { var = NAM_order_route value = 1 }
	}
	else_if = {
		limit = {
			check_variable = { NAM_order_kind = 4 }
			has_full_control_of_state = 700
		}
		700 = { NAM_create_order_by_kind = yes }
		set_temp_variable = { var = NAM_order_route value = 1 }
	}
	else_if = {
		limit = {
			check_variable = { NAM_order_kind = 4 }
			has_full_control_of_state = 689
		}
		689 = { NAM_create_order_by_kind = yes }
		set_temp_variable = { var = NAM_order_route value = 1 }
	}
	if = {
		limit = { check_variable = { NAM_order_route = 0 } }
		if = {
			limit = {
				any_owned_state = {
					is_fully_controlled_by = NAM
					NOT = { NAM_rw_island_state = yes }

		}

			random_owned_controlled_state = {
				prioritize = { 700 689 688 }
				NAM_create_order_by_kind = yes
			}
			set_temp_variable = { var = NAM_order_route value = 3 }
		}
	}
	log = "[ADISCORD][NAM][ORDERS] Formation kind [?NAM_order_kind]: placement route [?NAM_order_route] (1 own state, 2 mainland, 3 any held state, 0 none)."

		}
		clr_state_flag = NAM_line_in_progress
		set_state_flag = NAM_line_prepared
		NAM = { subtract_from_variable = { var = NAM_lines_in_progress value = 1 } }
		# Every province bordering another country gets level 1; border towns
		# with a victory point are the main approaches and get a second level.
		add_building_construction = {
			type = bunker
			level = 1
			instant_build = yes

				all_provinces = yes
				limit_to_border = yes

			level = 1
			instant_build = yes
			province = {
				all_provinces = yes
				limit_to_border = yes
				limit_to_victory_point > 0

		}
	}
	else = {
		NAM_line_release = yes
	}
}

# ----- Seal discipline -----
# NAM_seal_discipline (0-100) starts at hostilities at 70 plus any prewar focus
# bonus. NAM_seal_discipline_dynamic exists from then on and reads
# NAM_seal_org_factor = (discipline - 50) / 1000; it persists after the war.

# COUNTRY NAM. Snapshot casualties so earlier wars do not count.
NAM_seal_discipline_begin = {
	add_to_variable = { var = NAM_seal_discipline value = 70 }
	set_variable = { var = NAM_seal_casualties_seen_k value = casualties_k }
	clr_country_flag = NAM_seal_warned_50
	clr_country_flag = NAM_seal_warned_25
	if = {
		limit = { NOT = { has_dynamic_modifier = { modifier = NAM_seal_discipline_dynamic } } }
		add_dynamic_modifier = { modifier = NAM_seal_discipline_dynamic }
		ADISCORD_economy_mark_dirty = yes
	}
	NAM_seal_discipline_clamp = yes
}

# COUNTRY NAM. Clamp, refresh the modifier and open the threshold events once.
NAM_seal_discipline_clamp = {
	clamp_variable = {
		var = NAM_seal_discipline
		min = 0
		max = 100
	}
	if = {
		limit = { has_dynamic_modifier = { modifier = NAM_seal_discipline_dynamic } }
		set_temp_variable = { var = NAM_seal_org_next value = NAM_seal_discipline }
		subtract_from_temp_variable = { var = NAM_seal_org_next value = 50 }
		divide_temp_variable = { var = NAM_seal_org_next value = 1000 }
		if = {
			limit = {
				OR = {
					NOT = { has_variable = NAM_seal_org_factor }
					NOT = { check_variable = { var = NAM_seal_org_factor value = NAM_seal_org_next compare = equals } }
				}
			}
			set_variable = { var = NAM_seal_org_factor value = NAM_seal_org_next }
			force_update_dynamic_modifier = yes
		}
		# A newly reached event threshold still matters when the modifier is unchanged.
		if = {
			limit = { ADISCORD_nam_resource_war_active = yes }
			if = {
				limit = {
					NOT = { has_country_flag = NAM_seal_night }
					check_variable = { var = NAM_seal_discipline value = 0 compare = less_than_or_equals }
				}
				set_country_flag = NAM_seal_night
				country_event = { id = ADISCORD_nam_resource_war.32 hours = 1 }
			}
			else_if = {
				limit = {
					NOT = { has_country_flag = NAM_seal_warned_25 }
					check_variable = { var = NAM_seal_discipline value = 25 compare = less_than }
				}
				set_country_flag = NAM_seal_warned_25
				set_country_flag = NAM_seal_warned_50
				country_event = { id = ADISCORD_nam_resource_war.31 hours = 1 }
			}
			else_if = {
				limit = {
					NOT = { has_country_flag = NAM_seal_warned_50 }
					check_variable = { var = NAM_seal_discipline value = 50 compare = less_than }
				}
				set_country_flag = NAM_seal_warned_50
				country_event = { id = ADISCORD_nam_resource_war.30 hours = 1 }
			}
		}
	}
}

# COUNTRY NAM. on_monthly_NAM: resource-war losses cost 1 point per 5000

	if = {

# STATE field. Units = capacity x condition / 100, rounded down, plus the
# refinery and pipeline built here; zero while another country controls it.
NAM_field_update_units = {
	set_variable = { var = NAM_field_units value = 0 }
	if = {
		limit = { is_controlled_by = NAM }
		set_variable = { var = NAM_field_units value = NAM_field_capacity }
		multiply_variable = { var = NAM_field_units value = NAM_field_condition }
		divide_variable = { var = NAM_field_units value = 100 }
		# The quotient has at most two decimals, so this rounds down.
		subtract_from_variable = { var = NAM_field_units value = 0.499 }
		round_variable = NAM_field_units
		clamp_variable = {
			var = NAM_field_units
			min = 0
			max = 20
		}
		if = {
			limit = { has_state_flag = NAM_refinery_built }
			add_to_variable = { var = NAM_field_units value = 1 }
		}
		if = {
			limit = {
				has_state_flag = NAM_pipeline_built
				700 = { is_controlled_by = NAM }
			}
			add_to_variable = { var = NAM_field_units value = 1 }
		}
	}
	NAM = { add_to_variable = { var = NAM_oil_throughput value = PREV.NAM_field_units } }
}

# ----- Ladder of seals -----
# NAM_ladder_bop runs from the namestnik vertical (-1) to the districts (+1).
# The collapse edge proclaims independence once; the ladder is created only
# when it does not exist, so a repeated call never resets accumulated balance.

# COUNTRY NAM. Called by ADISCORD_nam_resource_war_schedule at the collapse.
NAM_ladder_declare_independence = {
	if = {
		limit = {
			has_global_flag = ADISCORD_fresh_campaign_contract_v1
			exists = yes
			NOT = { has_country_flag = NAM_independence_declared }
		}
		set_country_flag = NAM_independence_declared
		if = {
			limit = { NOT = { has_power_balance = { id = NAM_ladder_bop } } }
			set_power_balance = { id = NAM_ladder_bop set_value = 0 }
			# A course chosen before the ladder existed applies its shift now;
			# the course focuses push the balance themselves once it exists.
			if = {
				limit = { has_completed_focus = NAM_vertical }
				add_power_balance_value = { id = NAM_ladder_bop value = -0.25 }
			}
			else_if = {
				limit = { has_completed_focus = NAM_districts }
				add_power_balance_value = { id = NAM_ladder_bop value = 0.25 }
			}
		}
		if = {
			limit = { NOT = { has_variable = NAM_underground } }
			set_variable = { var = NAM_underground value = 20 }
		}
		# The first rival demand follows a full cycle after the declaration.
		set_country_flag = { flag = NAM_rival_cooldown value = 1 days = 240 }
		ADISCORD_economy_mark_dirty = yes
		# Event .80 announces the vanished centre and then opens the
		# declaration event ADISCORD_nam_groups.12.
		country_event = { id = ADISCORD_nam_resource_war.80 hours = 6 }
	}
}

# COUNTRY NAM. Every ladder shift outside focuses uses this guard.
# Input: temp NAM_ladder_shift (negative toward the vertical), one of the
# listed steps: add_power_balance_value rejects a variable value.
NAM_ladder_shift = {
	if = {
		limit = { has_power_balance = { id = NAM_ladder_bop } }
		if = {
			limit = { check_variable = { NAM_ladder_shift = -0.20 } }
			add_power_balance_value = { id = NAM_ladder_bop value = -0.20 }
		}
		else_if = {
			limit = { check_variable = { NAM_ladder_shift = -0.15 } }
			add_power_balance_value = { id = NAM_ladder_bop value = -0.15 }
		}
		else_if = {
			limit = { check_variable = { NAM_ladder_shift = -0.10 } }
			add_power_balance_value = { id = NAM_ladder_bop value = -0.10 }
		}
		else_if = {
			limit = { check_variable = { NAM_ladder_shift = -0.05 } }
			add_power_balance_value = { id = NAM_ladder_bop value = -0.05 }
		}
		else_if = {
			limit = { check_variable = { NAM_ladder_shift = 0.05 } }
			add_power_balance_value = { id = NAM_ladder_bop value = 0.05 }
		}
		else_if = {
			limit = { check_variable = { NAM_ladder_shift = 0.10 } }
			add_power_balance_value = { id = NAM_ladder_bop value = 0.10 }
		}
		else_if = {
			limit = { check_variable = { NAM_ladder_shift = 0.15 } }
			add_power_balance_value = { id = NAM_ladder_bop value = 0.15 }
		}
		else_if = {
			limit = { check_variable = { NAM_ladder_shift = 0.20 } }
			add_power_balance_value = { id = NAM_ladder_bop value = 0.20 }
		}
	}
}

# ----- Rivals of the ladder -----
# Rival numbers: 1 Vera Pechat (namestniks), 2 Arseny Klyuch (oil),
# 3 Taras Shtyk (garrisons), 4 Lev Gavanov (export houses). The districts have
# no rival: their discontent feeds the underground instead.

# COUNTRY NAM. Writes NAM_rival_current: the free rival of the weakest pillar,
# or 0 when every rival is gone.
NAM_rival_pick_weakest = {
	set_variable = { var = NAM_rival_current value = 0 }
	set_temp_variable = { var = NAM_rival_low value = 101 }
	if = {
		limit = {
			NAM_rival_pechat_free = yes
			check_variable = { var = NAM_grp_nam_support value = NAM_rival_low compare = less_than }
		}
		set_temp_variable = { var = NAM_rival_low value = NAM_grp_nam_support }
		set_variable = { var = NAM_rival_current value = 1 }
	}
	if = {
		limit = {
			NAM_rival_klyuch_free = yes
			check_variable = { var = NAM_grp_oil_support value = NAM_rival_low compare = less_than }
		}
		set_temp_variable = { var = NAM_rival_low value = NAM_grp_oil_support }
		set_variable = { var = NAM_rival_current value = 2 }
	}
	if = {
		limit = {
			NAM_rival_shtyk_free = yes
			check_variable = { var = NAM_grp_garr_support value = NAM_rival_low compare = less_than }
		}
		set_temp_variable = { var = NAM_rival_low value = NAM_grp_garr_support }
		set_variable = { var = NAM_rival_current value = 3 }
	}
	if = {
		limit = {
			NAM_rival_gavanov_free = yes
			check_variable = { var = NAM_grp_trade_support value = NAM_rival_low compare = less_than }
		}
		set_temp_variable = { var = NAM_rival_low value = NAM_grp_trade_support }
		set_variable = { var = NAM_rival_current value = 4 }
	}
}

# COUNTRY NAM. The 240-day demand cycle and the coup window of the Night of
# Seals. An ending of the ladder settles the order and closes both.
NAM_rivals_monthly = {
	if = {
		limit = {
			has_country_flag = NAM_coup_pending
			NOT = { has_country_flag = NAM_coup_succeeded }
			NOT = { has_country_flag = NAM_ladder_settled }
			NOT = { has_country_flag = NAM_rival_event_pending }
		}
		clr_country_flag = NAM_coup_pending
		NAM_rival_pick_weakest = yes
		if = {
			limit = { check_variable = { NAM_rival_current > 0 } }
			NAM_rival_open_coup = yes
		}
	}
	else_if = {
		limit = {
			NAM_rival_cycle_open = yes
			NOT = { has_country_flag = NAM_ladder_settled }
		}
		NAM_rival_pick_weakest = yes
		if = {
			limit = { check_variable = { NAM_rival_current > 0 } }
			set_country_flag = { flag = NAM_rival_cooldown value = 1 days = 240 }
			set_country_flag = { flag = NAM_political_petition_cooldown value = 1 days = 90 }
			set_country_flag = NAM_rival_event_pending
			country_event = { id = ADISCORD_nam_groups.20 hours = 6 }
		}
	}
}

# NAM_underground (0-100) is Yaroslav Veter's network. It works from the first
# day of the colony, strikes at the fields, the depots and the namestniks, and
# at hostilities decides the Svetlogorsk uprising; the amnesty ends it for good.
# Growth, thresholds and sabotage stop once the resource war has opened.

# COUNTRY NAM. Every writer calls this clamp; it also opens the threshold events.
NAM_underground_clamp = {
	if = {
		limit = { has_country_flag = NAM_veter_amnesty }
		set_variable = { var = NAM_underground value = 0 }
	}
	clamp_variable = {
		var = NAM_underground
		min = 0
		max = 100
	}
	NAM_underground_check_thresholds = yes
}

# COUNTRY NAM. Crossing 40, 60 or 80 opens its event once; a threshold arms
# again after the network falls 15 points below it. A jump across several
# thresholds opens only the highest one.
NAM_underground_check_thresholds = {
	if = {
		limit = { NAM_underground_active = yes }
		if = {
			limit = {
				has_country_flag = NAM_ug_alarm_40
				check_variable = { NAM_underground < 25 }
			}
			clr_country_flag = NAM_ug_alarm_40
		}
		if = {
			limit = {
				has_country_flag = NAM_ug_alarm_60
				check_variable = { NAM_underground < 45 }
			}
			clr_country_flag = NAM_ug_alarm_60
		}
		if = {
			limit = {
				has_country_flag = NAM_ug_alarm_80
				check_variable = { NAM_underground < 65 }
			}
			clr_country_flag = NAM_ug_alarm_80
		}
		if = {
			limit = {
				NAM_country_operable = yes
				NOT = { has_country_flag = NAM_ug_event_pending }
			}
			if = {
				limit = {
					NOT = { has_country_flag = NAM_ug_alarm_80 }
					check_variable = { NAM_underground > 79 }
				}
				set_country_flag = NAM_ug_alarm_80
				set_country_flag = NAM_ug_alarm_60
				set_country_flag = NAM_ug_alarm_40
				NAM_underground_open_event = yes
				country_event = { id = ADISCORD_nam_resource_war.84 hours = 6 }
			}
			else_if = {
				limit = {
					NOT = { has_country_flag = NAM_ug_alarm_60 }
					check_variable = { NAM_underground > 59 }
				}
				set_country_flag = NAM_ug_alarm_60
				set_country_flag = NAM_ug_alarm_40
				NAM_underground_open_event = yes
				country_event = { id = ADISCORD_nam_resource_war.83 hours = 6 }
			}
			else_if = {
				limit = {
					NOT = { has_country_flag = NAM_ug_alarm_40 }
					check_variable = { NAM_underground > 39 }
				}
				set_country_flag = NAM_ug_alarm_40
				NAM_underground_open_event = yes
				country_event = { id = ADISCORD_nam_resource_war.82 hours = 6 }
			}
		}
	}
}

# COUNTRY NAM. on_monthly_NAM from the first day of the colony: base +3, +2
# while the districts' support is below 40 and +1 more below 25, +2 while the
# vertical holds the ladder, -1 while the districts' support is 60 or more,
# -3 while the Svetlogorsk compact is paid. The month's change is kept for
# the category text, then the network may strike.
NAM_underground_monthly = {
	if = {
		limit = { NAM_underground_active = yes }
		set_temp_variable = { var = NAM_underground_delta value = 3 }
		if = {
			limit = { check_variable = { NAM_grp_dist_support < 40 } }
			add_to_temp_variable = { var = NAM_underground_delta value = 2 }
		}
		if = {
			limit = { check_variable = { NAM_grp_dist_support < 25 } }
			add_to_temp_variable = { var = NAM_underground_delta value = 1 }
		}
		if = {
			limit = { check_variable = { NAM_grp_dist_support > 59 } }
			add_to_temp_variable = { var = NAM_underground_delta value = -1 }
		}
		if = {
			limit = {
				NAM_ladder_formed = yes
				power_balance_value = {
					id = NAM_ladder_bop
					value < -0.10
				}
			}
			add_to_temp_variable = { var = NAM_underground_delta value = 2 }
		}
		if = {
			limit = { has_country_flag = NAM_district_compact }
			add_to_temp_variable = { var = NAM_underground_delta value = -3 }
		}
		set_variable = { var = NAM_underground_growth value = NAM_underground_delta }
		add_to_variable = { var = NAM_underground value = NAM_underground_delta }
		NAM_underground_clamp = yes
		NAM_underground_sabotage_roll = yes
	}
}

# ----- War of return -----
# The defeat tree's revenge. After the compromise peace it is one war against
# the current owners of the lost fields and the South Harbour (67, 688, 689,
# 690, 719, 700): NAM capitulation is the coalition treaty, capitulation of
# every owner returns the states, and the 365-day deadline (event .61) settles
# the contested states at the current front.
# From the island exile, once the namestnik has fallen, the same entry opens
# the exile war (global flag NAM_wor_exile_war) against Eflor and Azhar: the
# two fight in one temporary coalition and the marines land at the South
# Harbour. Their capitulation returns the homeland, applies the course's
# settlement to both (annexation and exploitation under the junta, federal
# districts under the humanists) and opens the consolidation; NAM
# capitulation annexes Svetlogorye with its islands; at the deadline the side
# holding the other's capitals wins, otherwise ownership follows the front.
# The later great war (NAM_great_war_*) reuses the participant array, the
# coalition, the capitulation receipts and the settlements of this section.
# Every terminal path runs NAM_war_of_return_close or NAM_great_war_close.

# STATE field. Registers its foreign owner once and claims the field.
NAM_wor_register_owner = {
	if = {
		limit = { NAM_wor_field_held_by_rival = yes }
		add_claim_by = NAM
		owner = {
			if = {
				limit = { NOT = { has_country_flag = NAM_wor_participant } }
				set_country_flag = NAM_wor_participant
				add_to_array = { array = global.NAM_wor_participants value = THIS }
				if = {
					limit = { NOT = { has_global_flag = NAM_wor_primary_saved } }
					set_global_flag = NAM_wor_primary_saved
					save_global_event_target_as = NAM_wor_primary
				}
			}
		}
	}
}

# COUNTRY target of the exile war or the great war. Woken from minor
# suppression before any alliance or war and registered once; the first
# independent target becomes the coalition's primary.
NAM_wor_register_target = {
	if = {
		limit = {
			NAM_wor_target_valid = yes
			NOT = { has_country_flag = NAM_wor_participant }
		}
		ADISCORD_release_non_participating_minor_optimization = yes
		set_country_flag = NAM_wor_participant
		add_to_array = { array = global.NAM_wor_participants value = THIS }
		if = {
			limit = {
				is_subject = no
				NOT = { has_global_flag = NAM_wor_primary_saved }
			}
			set_global_flag = NAM_wor_primary_saved
			save_global_event_target_as = NAM_wor_primary
		}
	}
}

# GLOBAL. Each independent target leaves its own alliance, so no outside ally
# is drawn into the war, and the primary founds a temporary coalition from the
# shared template. NAM_wor_release_participants dismantles it on every
# terminal path.
NAM_wor_form_coalition = {
	for_each_scope_loop = {
		array = global.NAM_wor_participants
		if = {
			limit = {
				is_subject = no
				is_in_faction = yes
			}
			leave_faction = yes
		}
	}
	event_target:NAM_wor_primary = {
		create_faction_from_template = {
			template = faction_template_ADISCORD_standard
			name = NAM_wor_coalition_faction
		}
		set_country_flag = NAM_wor_coalition_founder
	}
	for_each_scope_loop = {
		array = global.NAM_wor_participants
		if = {
			limit = {
				is_subject = no
				is_in_faction = no
			}
			event_target:NAM_wor_primary = { add_to_faction = PREV }
		}
	}
}

# GLOBAL. The participants not yet at war join the primary's side; each
# learns of the declaration through the given event.
NAM_wor_call_participants = {
	for_each_scope_loop = {

				targeted_alliance = event_target:NAM_wor_primary
				enemy = NAM
				hostility_reason = asked_to_join
			}
		}
	}
}

# COUNTRY NAM. Called by the focuses NAM_defeat_war_of_return (compromise),

			67 = { add_claim_by = NAM }
			688 = { add_claim_by = NAM }
			689 = { add_claim_by = NAM }
			690 = { add_claim_by = NAM }

			690 = { NAM_wor_register_owner = yes }

		}

# GLOBAL, before the cleanup. Without a decision at the deadline ownership

# ----- Consolidation of the returned south -----
# NAM_consolidation (0-100) is Svetlogorye's hold on the regions taken back in
# the exile war. It starts at 10 with the victory and grows by 2 a month; the
# junta adds terror and garrison orders and pays garrison upkeep of 3
# treasury for every conquered state, the humanists add federal treaties and
# reconstruction funds. Below 70 the regions may revolt or strike (event
# .104). From 70 the great war against the rest of the south may begin
# (NAM_great_war_can_start). NAM_consolidation_dynamic turns the hold into
# stability and compliance.

# COUNTRY NAM. A focus may already have written a value, so only a missing
# one is seeded.
NAM_consolidation_begin = {
	set_country_flag = NAM_consolidation_started
	if = {
		limit = { NOT = { has_variable = NAM_consolidation } }
		set_variable = { var = NAM_consolidation value = 10 }
	}
	NAM_consolidation_clamp = yes
	mark_focus_tree_layout_dirty = yes
}

# COUNTRY NAM. Every writer of the consolidation calls this clamp.
NAM_consolidation_clamp = {
	clamp_variable = {
		var = NAM_consolidation
		min = 0
		max = 100
	}
	if = {
		limit = { has_country_flag = NAM_consolidation_started }
		set_variable = { var = NAM_cons_mod_stability value = NAM_consolidation }
		subtract_from_variable = { var = NAM_cons_mod_stability value = 50 }
		multiply_variable = { var = NAM_cons_mod_stability value = 0.002 }
		set_variable = { var = NAM_cons_mod_compliance value = NAM_consolidation }
		multiply_variable = { var = NAM_cons_mod_compliance value = 0.0005 }
		if = {
			limit = { NOT = { has_dynamic_modifier = { modifier = NAM_consolidation_dynamic } } }
			add_dynamic_modifier = { modifier = NAM_consolidation_dynamic }
		}
		force_update_dynamic_modifier = yes
		ADISCORD_economy_mark_dirty = yes
	}
}

# COUNTRY NAM, on_monthly_NAM. Growth, garrison upkeep and the incident roll;
# the incident reads only NAM's own arrays of conquered states and districts.
NAM_consolidation_monthly = {
	if = {
		limit = {
			has_country_flag = NAM_consolidation_started
			NAM_country_operable = yes
		}
		set_temp_variable = { var = NAM_cons_delta value = 2 }
		if = {
			limit = { has_country_flag = NAM_rad_garrison_order_active }
			add_to_temp_variable = { var = NAM_cons_delta value = 3 }
			set_temp_variable = { var = NAM_treasury_amount value = 20 }
			NAM_spend_treasury = yes
		}
		if = {
			limit = { has_stability < 0.40 }
			add_to_temp_variable = { var = NAM_cons_delta value = -2 }
		}
		set_variable = { var = NAM_consolidation_growth value = NAM_cons_delta }
		add_to_variable = { var = NAM_consolidation value = NAM_cons_delta }
		NAM_consolidation_clamp = yes
		# The junta's garrisons in the conquered states.
		if = {
			limit = {
				has_country_flag = NAM_exile_radical
				check_variable = { NAM_consolidation < 100 }
			}
			set_temp_variable = { var = NAM_treasury_amount value = NAM_conquered_states^num }
			multiply_temp_variable = { var = NAM_treasury_amount value = 3 }
			if = {
				limit = { check_variable = { NAM_treasury_amount > 0 } }
				NAM_spend_treasury = yes
			}
		}
		if = {
			limit = {
				check_variable = { NAM_consolidation < 70 }
				NOT = { has_country_flag = NAM_cons_incident_cooldown }

# ----- Great war -----
# After the consolidation Svetlogorye fights the rest of the south in one war:
# Rahma, Sardan and Kadir, and Eflor or Azhar if either still stands
# independent. The focus that opens it calls NAM_great_war_start behind
# NAM_great_war_can_start. The suppressed minors wake before any alliance,
# the targets fight in one temporary coalition, every participant holds
# mandatory major status only for this war. Capitulation of all of them
# applies the course's settlement to the new conquests; NAM capitulation
# annexes Svetlogorye; at the 365-day deadline (event .106) the side holding
# the other's capitals wins, otherwise ownership follows the front. Every
# terminal path runs NAM_great_war_close.

# COUNTRY NAM. Called by the focuses NAM_rad_great_war, NAM_hum_great_war and
# NAM_board_great_war and by the renewal decision NAM_renew_great_war. A
# renewed war reopens the resolved flag; NAM_great_war_deadline_armed (360
# days) makes a .106 left over from the earlier war ignore the new one.
NAM_great_war_start = {
	if = {
		limit = { NAM_great_war_can_start = yes }
		set_global_flag = NAM_great_war_started
		clr_global_flag = NAM_great_war_resolved
		set_global_flag = {
			flag = NAM_great_war_deadline_armed
			days = 360
		}
		clear_array = global.NAM_wor_participants
		clr_global_flag = NAM_wor_primary_saved
		clear_global_event_target = NAM_wor_primary
		EFL = { NAM_wor_register_target = yes }
		AZH = { NAM_wor_register_target = yes }
		RHM = { NAM_wor_register_target = yes }
		SDR = { NAM_wor_register_target = yes }
		KDR = { NAM_wor_register_target = yes }
		if = {
			limit = { is_in_faction = yes }
			leave_faction = yes
		}
		ADISCORD_nam_resource_war_reserve_major = yes
		for_each_scope_loop = {
			array = global.NAM_wor_participants
			ADISCORD_nam_resource_war_reserve_major = yes
		}
		NAM_wor_form_coalition = yes
		declare_war_on = {
			target = event_target:NAM_wor_primary
			type = annex_everything
		}
		NAM_wor_call_participants = yes
		for_each_scope_loop = {
			array = global.NAM_wor_participants
			country_event = { id = ADISCORD_nam_resource_war.105 hours = 1 }
		}
		add_war_support = 0.10
		activate_mission = NAM_great_war_deadline
		country_event = { id = ADISCORD_nam_resource_war.105 hours = 1 }
		country_event = { id = ADISCORD_nam_resource_war.106 days = 365 }
		log = "[ADISCORD][NAM][GREAT_WAR] Svetlogorye opens the great war against the south."
	}
}

	}

# GLOBAL. Event .106 at the deadline.
NAM_great_war_resolve_front = {
	if = {
		limit = { NAM_great_war_active = yes }
		if = {
			limit = { NAM_wor_capitals_taken = yes }
			log = "[ADISCORD][NAM][GREAT_WAR] Deadline: Svetlogorye holds every remaining capital."
			NAM_great_war_resolve_victory = yes
		}
		else_if = {
			limit = { NAM_wor_capital_lost = yes }
			log = "[ADISCORD][NAM][GREAT_WAR] Deadline: the coalition holds Svetlogorye's capital."
			NAM_great_war_resolve_defeat = yes
		}
		else = {
			NAM_wor_settle_front_states = yes
			NAM_great_war_close = yes
			# A front that moved south is a victory without the course's
			# settlement of the survivors; a front that did not leaves the
			# renewal decision NAM_renew_great_war open.
			if = {
				limit = { check_variable = { NAM_wor_front_net > 0 } }
				set_global_flag = NAM_great_war_won
				log = "[ADISCORD][NAM][GREAT_WAR] Deadline: the front moves south; the great war counts as won."
			}
			NAM_wor_after_front = yes
		}
	}
}

# also costs stability and political power and the boards demand guarantees
# (event .101), from 90 they demand the seals (event .102). Three months at 90
# or more, or 100 at once, end in the corporate takeover (event .103):
# corporate boards govern the islands under the cosmetic tag
# NAM_corporate_state, while the humanist course and its federation continue.
# NAM_free_zone_slots (1-3) limits the chartered free zones on the islands.

			set_variable = { var = NAM_corp_mod_efficiency value = 0.05 }
			set_variable = { var = NAM_corp_mod_trade value = 0.20 }
			set_variable = { var = NAM_corp_mod_stability value = -0.10 }
			set_variable = { var = NAM_corp_mod_pp value = -0.20 }
		}
		else_if = {
			limit = { check_variable = { NAM_corp_influence > 69 } }
			set_variable = { var = NAM_corp_mod_research value = 0.08 }
			set_variable = { var = NAM_corp_mod_efficiency value = 0.05 }
			set_variable = { var = NAM_corp_mod_trade value = 0.15 }
			set_variable = { var = NAM_corp_mod_stability value = -0.05 }
			set_variable = { var = NAM_corp_mod_pp value = -0.10 }

			limit = { check_variable = { NAM_corp_influence > 29 } }
			set_variable = { var = NAM_corp_mod_research value = 0.05 }
			set_variable = { var = NAM_corp_mod_efficiency value = 0.03 }
			set_variable = { var = NAM_corp_mod_trade value = 0.10 }

# COUNTRY federal district inside the monthly dividend loop: adds its owner's
# dividend to the temp NAM_corp_dividend and delivers fuel or rifles. A
# disputed district pays nothing until the dispute is settled.
NAM_corp_district_dividend = {
	if = {
		limit = {
			has_variable = NAM_corp_district_owner
			NOT = { has_country_flag = NAM_corp_disputed }
		}
		if = {
			limit = { check_variable = { NAM_corp_district_owner = 1 } }
			add_to_temp_variable = { var = NAM_corp_dividend value = 8 }
			NAM = { add_fuel = 500 }
		}
		else_if = {
			limit = { check_variable = { NAM_corp_district_owner = 2 } }
			add_to_temp_variable = { var = NAM_corp_dividend value = 10 }
		}
		else_if = {
			limit = { check_variable = { NAM_corp_district_owner = 3 } }
			add_to_temp_variable = { var = NAM_corp_dividend value = 10 }
		}
		else_if = {
			limit = { check_variable = { NAM_corp_district_owner = 4 } }
			add_to_temp_variable = { var = NAM_corp_dividend value = 6 }
			NAM = { add_equipment_to_stockpile = { type = infantry_equipment amount = 300 producer = NAM } }
		}
		else_if = {
			limit = { check_variable = { NAM_corp_district_owner = 5 } }
			add_to_temp_variable = { var = NAM_corp_dividend value = 12 }
		}
		else = {
			add_to_temp_variable = { var = NAM_corp_dividend value = 15 }
		}
	}
}

# COUNTRY NAM, inside NAM_corp_monthly after the takeover. Dividends of the
# districts and the rivalry roll over NAM's own array of districts.
NAM_corp_districts_monthly = {
	set_temp_variable = { var = NAM_corp_dividend value = 0 }
	set_temp_variable = { var = NAM_corp_district_count value = 0 }
	for_each_scope_loop = {
		array = NAM_federal_districts
		if = {
			limit = {
				exists = yes
				is_subject_of = NAM
			}
			if = {
				limit = { NOT = { has_variable = NAM_corp_district_owner } }
				NAM_corp_assign_district = yes
			}
			NAM_corp_district_dividend = yes
			add_to_temp_variable = { var = NAM_corp_district_count value = 1 }
		}
	}
	set_variable = { var = NAM_corp_last_dividend value = NAM_corp_dividend }
	set_temp_variable = { var = NAM_corp_capital_change value = NAM_corp_dividend }
	NAM_corp_capital_add = yes
	if = {
		limit = {
			check_variable = { NAM_corp_district_count > 1 }
			NOT = { has_country_flag = NAM_corp_dispute_open }
			NOT = { has_country_flag = NAM_corp_dispute_cooldown }
		}
		random_list = {
			85 = { }
			15 = { NAM_corp_open_dispute = yes }
		}
	}
}

# COUNTRY NAM. Input: temp NAM_pirate_lane_input, the lane of the settling
# raid (1 oil run, 2 northern coast, 3 Luren strait, 4 Charden crossing).
# Saves the lane's owners that may still be raided as NAM_pirate_victim_a and
# NAM_pirate_victim_b and sets the temps NAM_pirate_victims (their number),
# NAM_pirate_coalition_hit (Eflor or Azhar among them) and
# NAM_pirate_lane_alerted (an owner still on guard). The owners match the
# NAM_pirate_lane_<id>_open triggers.
NAM_pirate_save_lane_victims = {
	set_temp_variable = { NAM_pirate_victims = 0 }
	set_temp_variable = { NAM_pirate_coalition_hit = 0 }
	set_temp_variable = { NAM_pirate_lane_alerted = 0 }
	if = {
		limit = { check_variable = { NAM_pirate_lane_input = 1 } }
		if = {
			limit = { EFL = { NAM_pirate_victim_valid = yes } }
			EFL = { save_event_target_as = NAM_pirate_victim_a }
			set_temp_variable = { NAM_pirate_victims = 1 }
			set_temp_variable = { NAM_pirate_coalition_hit = 1 }
		}
		if = {
			limit = { AZH = { NAM_pirate_victim_valid = yes } }
			if = {
				limit = { check_variable = { NAM_pirate_victims = 0 } }
				AZH = { save_event_target_as = NAM_pirate_victim_a }
			}
			else = {
				AZH = { save_event_target_as = NAM_pirate_victim_b }
			}
			add_to_temp_variable = { NAM_pirate_victims = 1 }
			set_temp_variable = { NAM_pirate_coalition_hit = 1 }
		}
		if = {
			limit = { NAM_pirate_lane_oil_alert = yes }
			set_temp_variable = { NAM_pirate_lane_alerted = 1 }
		}
	}
	else_if = {
		limit = { check_variable = { NAM_pirate_lane_input = 2 } }
		if = {
			limit = { RIV = { NAM_pirate_victim_valid = yes } }
			RIV = { save_event_target_as = NAM_pirate_victim_a }
			set_temp_variable = { NAM_pirate_victims = 1 }
		}
		if = {
			limit = { EFL = { NAM_pirate_victim_valid = yes } }
			if = {
				limit = { check_variable = { NAM_pirate_victims = 0 } }
				EFL = { save_event_target_as = NAM_pirate_victim_a }
			}
			else = {
				EFL = { save_event_target_as = NAM_pirate_victim_b }
			}
			add_to_temp_variable = { NAM_pirate_victims = 1 }
			set_temp_variable = { NAM_pirate_coalition_hit = 1 }
		}
		if = {
			limit = { NAM_pirate_lane_coast_alert = yes }
			set_temp_variable = { NAM_pirate_lane_alerted = 1 }
		}
	}
	else_if = {
		limit = { check_variable = { NAM_pirate_lane_input = 3 } }
		if = {
			limit = { ULV = { NAM_pirate_victim_valid = yes } }
			ULV = { save_event_target_as = NAM_pirate_victim_a }
			set_temp_variable = { NAM_pirate_victims = 1 }
		}
		if = {
			limit = { LUR = { NAM_pirate_victim_valid = yes } }
			if = {
				limit = { check_variable = { NAM_pirate_victims = 0 } }
				LUR = { save_event_target_as = NAM_pirate_victim_a }
			}
			else = {
				LUR = { save_event_target_as = NAM_pirate_victim_b }
			}
			add_to_temp_variable = { NAM_pirate_victims = 1 }
		}
		if = {
			limit = { NAM_pirate_lane_strait_alert = yes }
			set_temp_variable = { NAM_pirate_lane_alerted = 1 }
		}
	}
	else_if = {
		limit = { check_variable = { NAM_pirate_lane_input = 4 } }
		if = {
			limit = { CHD = { NAM_pirate_victim_valid = yes } }
			CHD = { save_event_target_as = NAM_pirate_victim_a }
			set_temp_variable = { NAM_pirate_victims = 1 }
		}
		if = {
			limit = { RIV = { NAM_pirate_victim_valid = yes } }
			if = {
				limit = { check_variable = { NAM_pirate_victims = 0 } }
				RIV = { save_event_target_as = NAM_pirate_victim_a }
			}
			else = {
				RIV = { save_event_target_as = NAM_pirate_victim_b }
			}
			add_to_temp_variable = { NAM_pirate_victims = 1 }
		}
		if = {
			limit = { NAM_pirate_lane_crossing_alert = yes }
			set_temp_variable = { NAM_pirate_lane_alerted = 1 }
		}
	}
}

# COUNTRY NAM, the end of a paid lane raid; the decision has cleared its
# receipt. Input: temp NAM_pirate_lane_input. The outcome is rolled once: a
# haul (50), a rich prize (15) or a failure (35). More ships, owners off guard
# and low notoriety favour the flotilla; the oil run's tankers make the richest
# prizes. A lane whose owners have all gone or become friends returns the fuel.
# The settlement only changes the owners' fuel, flags and opinion and sends an
# event to human owners alone, so a suppressed minor needs no waking.
NAM_pirate_raid_settle = {
	subtract_from_variable = { var = NAM_pirate_raids_active value = 1 }
	clamp_variable = {
		var = NAM_pirate_raids_active
		min = 0
		max = 3
	}
	NAM_pirate_save_lane_victims = yes
	if = {
		limit = { check_variable = { NAM_pirate_victims = 0 } }
		add_fuel = 1000
	}
	else = {
		set_variable = { var = NAM_pirate_last_lane value = NAM_pirate_lane_input }
		set_variable = { var = NAM_pirate_last_victims value = NAM_pirate_victims }
		set_variable = { var = NAM_pirate_last_loot value = 0 }
		set_variable = { var = NAM_pirate_last_loss value = 0 }
		random_list = {
			50 = {
				modifier = {
					factor = 1.2
					check_variable = { var = num_ships value = 2 compare = greater_than }
				}
				modifier = {
					factor = 1.2
					check_variable = { var = num_ships value = 5 compare = greater_than }
				}
				modifier = {
					factor = 0.6
					check_variable = { NAM_pirate_lane_alerted = 1 }
				}
				modifier = {
					factor = 0.7
					has_dynamic_modifier = { modifier = NAM_pirate_blockade }
				}
				modifier = {
					factor = 0.8
					check_variable = { NAM_pirate_notoriety > 59 }
				}
				set_variable = { var = NAM_pirate_last_result value = 1 }
				NAM_pirate_raid_success = yes
			}
			15 = {
				modifier = {
					factor = 1.5
					check_variable = { NAM_pirate_lane_input = 1 }
				}
				modifier = {
					factor = 0.6
					check_variable = { NAM_pirate_lane_alerted = 1 }
				}
				set_variable = { var = NAM_pirate_last_result value = 2 }
				NAM_pirate_raid_prize = yes
			}
			35 = {
				modifier = {
					factor = 0.7
					check_variable = { var = num_ships value = 5 compare = greater_than }
				}
				modifier = {
					factor = 1.3
					has_dynamic_modifier = { modifier = NAM_pirate_blockade }
				}
				set_variable = { var = NAM_pirate_last_result value = 0 }
				NAM_pirate_raid_failure = yes
			}
		}
		event_target:NAM_pirate_victim_a = { NAM_pirate_mark_victim = yes }
		if = {
			limit = { check_variable = { NAM_pirate_victims > 1 } }
			event_target:NAM_pirate_victim_b = { NAM_pirate_mark_victim = yes }
		}
		log = "[ADISCORD][NAM][PIRACY] Raid on lane [?NAM_pirate_last_lane] against [NAM_pirate_victim_a.GetTag] and [?NAM_pirate_last_victims|0] owner(s): result [?NAM_pirate_last_result], loot [?NAM_pirate_last_loot], notoriety [?NAM_pirate_notoriety]."
		country_event = { id = ADISCORD_nam_resource_war.90 hours = 1 }
	}
}

# COUNTRY NAM, NAM_war_of_return_start after the declaration, from the island
# exile only. create_unit refuses a province held by the enemy, so the landing
# first takes a beachhead: hostile armies in the South Harbour (province 16716
# is the whole of state 700) fall back to their capitals and Svetlogorye
# becomes its controller. Two marine divisions land there, a third with a
# trained order or a revenge readiness of 70. The landing never exceeds three
# divisions: further trained orders form on the island capital, and so does
# the whole landing when the harbour's controller is not at war with
# Svetlogorye. The beachhead is ordinary control: the deadline settlement
# awards the harbour only if it is still held. Every trained order is
# delivered here.
NAM_wor_revenge_landing = {
	if = {
		limit = {
			has_global_flag = ADISCORD_nam_resource_war_island_exile
			700 = { NOT = { is_owned_by = NAM } }
		}
		NAM_marine_ensure_template = yes
		set_variable = { var = NAM_marines_orders_delivered value = NAM_marine_orders_paid }
		set_variable = { var = NAM_marines_landed value = 2 }
		if = {
			limit = { NAM_landing_third_division = yes }
			add_to_variable = { var = NAM_marines_landed value = 1 }
			if = {
				limit = { check_variable = { NAM_marine_orders_paid > 0 } }
				subtract_from_variable = { var = NAM_marine_orders_paid value = 1 }
			}
		}
		set_variable = { var = NAM_marines_island value = NAM_marine_orders_paid }
		set_temp_variable = { var = NAM_landing_at_harbour value = 0 }
		# A participant invited by add_to_war in the same effect may not show the
		# war relation yet; its membership flag marks it as the enemy already.
		if = {
			limit = {
				700 = {
					controller = {
						OR = {
							has_war_with = NAM
							has_country_flag = NAM_wor_participant
						}
					}
				}
			}
			700 = {
				teleport_armies = {
					limit = {
						OR = {
							has_war_with = NAM
							has_country_flag = NAM_wor_participant
						}
					}
				}
			}
			set_province_controller = 16716
			set_temp_variable = { var = NAM_landing_at_harbour value = 1 }
		}
		else_if = {
			limit = { 700 = { is_controlled_by = NAM } }
			set_temp_variable = { var = NAM_landing_at_harbour value = 1 }
		}
		if = {
			limit = { check_variable = { NAM_landing_at_harbour = 0 } }
			add_to_variable = { var = NAM_marines_island value = NAM_marines_landed }
			set_variable = { var = NAM_marines_landed value = 0 }
		}
		set_temp_variable = { var = NAM_landing_index value = 0 }
		while_loop_effect = {
			limit = { check_variable = { var = NAM_landing_index value = NAM_marines_landed compare = less_than } }
			add_to_temp_variable = { var = NAM_landing_index value = 1 }
			700 = {
				create_unit = {
					division = "division_template = \"Island Marines\" start_experience_factor = 0.30 start_equipment_factor = 1.0 start_manpower_factor = 1.0"
					owner = NAM
					prioritize_location = 16716
					allow_spawning_on_enemy_provs = no
				}
			}
		}
