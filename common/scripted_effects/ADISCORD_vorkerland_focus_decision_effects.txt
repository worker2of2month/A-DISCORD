# The verified regional war graph opens these campaigns before the claimant
# capstones. Each launch still has a one-day faction-detachment gap, and the
# final claimant showdown remains locked behind the focus-owned preparation.

	if = {
		limit = {
			ADISCORD_vorkerland_central_minor_campaign_phase_available = yes
			has_country_flag = ADISCORD_vorkerland_focus_central_minor_launch_pending
			OR = { tag = WKR tag = VAD tag = TVA tag = WRK }
			is_subject = no
			NOT = { has_capitulated = yes }
			NOT = { OR = { has_war_with = EYR has_war_with = EGC has_war_with = RIV has_war_with = REV has_war_with = YOR has_war_with = NDN has_war_with = SWB has_war_with = VHV has_war_with = OSV } }
			any_neighbor_country = { tag = EYR }
			EYR = { exists = yes is_subject = no NOT = { has_capitulated = yes } NOT = { OR = { has_war_with = WKR has_war_with = VAD has_war_with = TVA } } }
		}
		declare_war_on = { target = EYR type = annex_everything }
		activate_mission = ADISCORD_vorkerland_focus_central_minor_launch_check
	}
	else = { ADISCORD_vorkerland_focus_cleanup_central_minor_front = yes }
}

# the verified 240-day campaign or spends the launch's single retry.

	if = {
		limit = {
			ADISCORD_vorkerland_central_minor_campaign_phase_available = yes
			OR = {
				AND = { has_country_flag = ADISCORD_vorkerland_focus_central_minor_target_eyr has_war_with = EYR }

# Exact handoff hook from the visible consolidation stage to the existing
# simultaneous WKR-VAD-TVA edge queue. No central minor may remain live.
ADISCORD_vorkerland_focus_schedule_final_showdown = {
	if = {
		limit = {
			OR = {
				AND = { tag = WKR has_country_flag = ADISCORD_vorkerland_focus_wkr_central_war_unlocked }
				AND = { tag = VAD has_country_flag = ADISCORD_vorkerland_focus_vad_central_war_unlocked }
				AND = { tag = TVA has_country_flag = ADISCORD_vorkerland_focus_tva_central_war_unlocked }
			}
			has_country_flag = ADISCORD_vorkerland_focus_central_front_prepared
