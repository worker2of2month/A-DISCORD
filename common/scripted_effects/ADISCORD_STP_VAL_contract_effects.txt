VAL_select_negotiation_posture = {
	if = {
		limit = {
			check_variable = { var = VAL_negotiation_posture value = 0 compare = equals }
		}
		random_list = {
			30 = {
				modifier = { factor = 2 check_variable = { var = VAL_contract_authority value = 75 compare = less_than } }
				modifier = { factor = 2 has_country_flag = VAL_doctrine_sustained_supply }
				set_variable = { var = VAL_negotiation_posture value = 1 }
			}
			30 = {
				modifier = { factor = 2 check_variable = { var = VAL_contract_authority value = 75 compare = greater_than_or_equals } }
				modifier = { factor = 2 has_country_flag = VAL_doctrine_resource_raider }
				set_variable = { var = VAL_negotiation_posture value = 2 }
			}
			20 = {
				modifier = { factor = 3 NOD = { NOT = { has_country_flag = NOD_crisis_posture_guardian } } }
				modifier = { factor = 2 num_equipment@infantry_equipment < 1000 }
				set_variable = { var = VAL_negotiation_posture value = 3 }
			}
			20 = {
				modifier = { factor = 2 has_country_flag = VAL_doctrine_covert_intervention }
				modifier = { factor = 2 has_country_flag = VAL_broad_resource_network }
				set_variable = { var = VAL_negotiation_posture value = 4 }
			}
		}
	}
}

VAL_save_current_stp_target = {
	if = {
		limit = {
			has_event_target = STP_crisis_resistance_side
			country_exists = event_target:STP_crisis_resistance_side
			event_target:STP_crisis_resistance_side = { has_capitulated = no }
		}
		event_target:STP_crisis_resistance_side = {
			save_global_event_target_as = VAL_STP_operation_target
		}
	}
	else_if = {
		limit = {
			has_event_target = STP_crisis_party_side
			country_exists = event_target:STP_crisis_party_side
		}
		event_target:STP_crisis_party_side = {
			save_global_event_target_as = VAL_STP_operation_target
		}
	}
	else_if = {
		limit = { country_exists = STP }
		STP = { save_global_event_target_as = VAL_STP_operation_target }
	}
}

VAL_apply_stp_family_memory = {
	if = {
		limit = { has_country_flag = $success_flag$ }
		add_political_power = -10
	}
	else = {
		set_country_flag = $success_flag$
		VAL_change_stp_leverage = { value = $first_leverage$ }
	}
	if = {
		limit = {
			check_variable = { var = $exposure_var$ value = 2 compare = greater_than_or_equals }
			has_event_target = VAL_STP_operation_target
			country_exists = event_target:VAL_STP_operation_target
		}
		event_target:VAL_STP_operation_target = {
			set_country_flag = $counter_flag$
		}
	}
	if = {
		limit = {
			check_variable = { var = $exposure_var$ value = 3 compare = greater_than_or_equals }
		}
		set_country_flag = {
			flag = $block_flag$
			days = 90
			value = 1
		}
		if = {
			limit = { country_exists = NOD }
			NOD = { set_country_flag = VAL_STP_repeated_operation_reported }
		}
	}
}

# Generic northern-family memory. Full variable and flag names are passed by each resolver.
VAL_update_northern_family_memory = {
	if = {
		limit = { check_variable = { var = $last_var$ value = $family_code$ compare = equals } }
		add_to_variable = { var = $streak_var$ value = 1 }
	}
	else = {
		set_variable = { var = $last_var$ value = $family_code$ }
		set_variable = { var = $streak_var$ value = 1 }
	}
	if = {
		limit = { check_variable = { var = $streak_var$ value = 3 compare = greater_than_or_equals } }
		set_country_flag = { flag = $block_flag$ days = 90 value = 1 }
	}
}

VAL_resolve_northern_operation = {
	VAL_update_northern_family_memory = {
		last_var = $last_var$
		streak_var = $streak_var$
		family_code = $family_code$
		block_flag = $block_flag$
	}
	if = {
		limit = { NOT = { has_country_flag = $success_flag$ } }
		set_country_flag = $success_flag$
		add_to_variable = { var = $influence_var$ value = $influence_delta$ }
		clamp_variable = { var = $influence_var$ min = 0 max = 3 }
	}
	set_country_flag = { flag = $cooldown_flag$ days = 70 value = 1 }
	VAL_clear_foreign_operation = yes
}

# Five-state mountain-deal chain.
VAL_open_mountain_contract_channel = {
	if = {
		limit = {
			NOT = { has_country_flag = VAL_foreign_operation_active }
			VAL_has_active_stp_civil_war = yes
		}
		VAL_select_negotiation_posture = yes
		VAL_save_current_stp_target = yes
		if = {
			limit = {
				has_event_target = VAL_STP_operation_target
				country_exists = event_target:VAL_STP_operation_target
			}
			event_target:VAL_STP_operation_target = {
				save_global_event_target_as = STP_val_contract_partner
			}
			set_country_flag = VAL_foreign_operation_active
			set_country_flag = VAL_STP_negotiation_state_channel
			event_target:STP_val_contract_partner = { country_event = { id = val_contract.1 } }
			set_country_flag = VAL_STP_offer_generation_1
			country_event = { id = val_contract.8 days = 35 }
		}
	}
}

VAL_advance_mountain_contract_viability = {
	clr_country_flag = VAL_STP_negotiation_state_channel
	set_country_flag = VAL_STP_negotiation_state_viability
	if = {
		limit = { VAL_STP_partner_is_viable = yes }
		clr_country_flag = VAL_STP_negotiation_state_viability
		set_country_flag = VAL_STP_negotiation_state_concessions
		country_event = { id = val_contract.2 days = 2 }
	}
	else = { VAL_abort_mountain_contract = yes }
}

VAL_choose_mountain_concessions = {
	clr_country_flag = VAL_STP_negotiation_state_concessions
	set_country_flag = VAL_STP_negotiation_state_offer
	set_variable = { var = VAL_STP_concession_count value = 1 }
	set_country_flag = VAL_STP_concession_arms_debt
	if = {
		limit = { check_variable = { var = VAL_negotiation_posture value = 1 compare = equals } }
		set_variable = { var = VAL_STP_concession_count value = 1 }
	}
	else_if = {
		limit = { check_variable = { var = VAL_negotiation_posture value = 2 compare = equals } }
		set_country_flag = VAL_STP_concession_resource_45
		set_country_flag = VAL_STP_concession_postwar_contracts
		set_variable = { var = VAL_STP_concession_count value = 3 }
	}
	else_if = {
		limit = { check_variable = { var = VAL_negotiation_posture value = 3 compare = equals } }
		set_country_flag = VAL_STP_concession_postwar_contracts
		set_variable = { var = VAL_STP_concession_count value = 1 }
	}
	else = {
		set_country_flag = VAL_STP_concession_resource_88
		set_country_flag = VAL_STP_concession_advisers
		set_variable = { var = VAL_STP_concession_count value = 3 }
		VAL_change_stp_exposure_nodrul = { value = 1 }
	}
	clr_country_flag = VAL_STP_negotiation_state_offer
	set_country_flag = VAL_STP_negotiation_state_response
	event_target:STP_val_contract_partner = { country_event = { id = val_contract.3 } }
}

VAL_accept_mountain_contract = {
	if = {
		limit = { VAL_STP_partner_is_viable = yes }
		if = {
			limit = { check_variable = { var = VAL_STP_concession_count value = 3 compare = greater_than_or_equals } }
			VAL_apply_mountain_contract = { countdown = 450 }
		}
		else_if = {
			limit = { check_variable = { var = VAL_STP_concession_count value = 2 compare = greater_than_or_equals } }
			VAL_apply_mountain_contract = { countdown = 300 }
		}
		else = { VAL_apply_mountain_contract = { countdown = 180 } }
	}

# Guarded resource-corridor attempt during the internal war.
VAL_prepare_resource_corridor = {
	if = {
		limit = {
			VAL_has_active_stp_civil_war = yes
			NOT = { has_country_flag = VAL_resource_corridor_attempted }
		}
		set_country_flag = VAL_resource_corridor_attempted
		45 = { owner = { save_global_event_target_as = VAL_corridor_owner_45 } }
		88 = { owner = { save_global_event_target_as = VAL_corridor_owner_88 } }
		if = {
			limit = { event_target:VAL_corridor_owner_45 = { tag = event_target:VAL_corridor_owner_88 } }
			set_country_flag = VAL_resource_corridor_full_mode
			event_target:VAL_corridor_owner_45 = { set_country_flag = VAL_resource_corridor_participant }
			declare_war_on = { target = event_target:VAL_corridor_owner_45 type = take_state_focus generator = { 45 88 } }
		}
		else = {
			if = { limit = { is_ai = no } country_event = { id = val_contract.40 } }
			else = {
				random_list = {
					45 = { modifier = { factor = 2 has_country_flag = VAL_supply_contracts } VAL_start_corridor_target_45 = yes }
					55 = { modifier = { factor = 2 has_country_flag = VAL_doctrine_resource_raider } VAL_start_corridor_target_88 = yes }
				}
			}
		}
	}
}
