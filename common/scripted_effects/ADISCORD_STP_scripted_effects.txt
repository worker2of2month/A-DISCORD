# Core initialization, suspicion and leader health (COUNTRY)

# STP, after successor creation but before district transfer and focus replacement.
STP_cw_preserve_council_guarantees = {
	if = {
		limit = {
			has_country_flag = STP_sided_with_Maksim_flag
			has_completed_focus = STP_cw_autonomy_guarantees
			45 = { is_owned_by = STP is_controlled_by = STP has_state_flag = STP_local_deal_asset }
			STS = { exists = yes }
			SRP = { exists = yes }
		}
		STS = { add_ideas = STP_cw_council_guarantees }
		SRP = { add_ideas = STP_cw_council_guarantees }
	}
}

# SRP: an actual external war grants emergency authority, never its announcement.
STP_cw_republics_war_command = {
	if = {
		limit = { tag = SRP has_war_with = VAL NOT = { has_idea = SRP_emergency_command } }
		add_ideas = SRP_emergency_command
		if = {
			limit = { 43 = { is_owned_by = SRP is_controlled_by = SRP } }
			43 = { add_building_construction = { type = bunker level = 1 province = 16450 instant_build = yes } }
		}
	}
}

# STS pays private suppliers; no equipment or money is taken from the recipient.
STP_cw_fund_republican_supplies = {
	if = {
		limit = { hidden_trigger = { STP_cw_republics_aid_available = yes check_variable = { var = ADISCORD_economy_treasury value = 25 compare = greater_than_or_equals } } }
		hidden_effect = {
			subtract_from_variable = { var = ADISCORD_economy_treasury value = 25 }
			add_to_variable = { var = ADISCORD_economy_current_month_action_costs value = 25 }
			ADISCORD_economy_mark_dirty = yes
		}
		SRP = {
			add_equipment_to_stockpile = { type = infantry_equipment amount = 1200 }
			add_equipment_to_stockpile = { type = support_equipment amount = 30 }
			add_timed_idea = { idea = SRP_council_supply days = 42 }
		}
	}
}

# STS: consuming both promises makes late or repeated event choices harmless.
STP_cw_ratify_republican_charter = {
	if = {
		limit = { hidden_trigger = { STP_cw_can_settle_council_guarantees = yes } }
		remove_ideas = STP_cw_council_guarantees
		add_ideas = STP_cw_republican_charter
		SRP = { remove_ideas = STP_cw_council_guarantees add_ideas = STP_cw_republican_charter }
	}
}

STP_cw_reject_republican_charter = {
	if = {
		limit = { hidden_trigger = { STP_cw_can_settle_council_guarantees = yes } }
		remove_ideas = STP_cw_council_guarantees
		add_ideas = STP_cw_central_authority
		SRP = { remove_ideas = STP_cw_council_guarantees add_ideas = SRP_distrust_stelander }
	}
}

STP_initialize_core_mechanics = {
	STP_cw_initialize_preparation = yes
	if = {
		limit = { NOT = { has_variable = STP_party_suspicion } }
		set_variable = {
			var = STP_party_suspicion
			value = 5
		}
	}

	if = {
		limit = { NOT = { has_variable = STP_leader_health_stage } }
		set_variable = {
			var = STP_leader_health_stage
			value = 1
		}
	}

	STP_refresh_party_suspicion = yes
	STP_refresh_leader_health = yes

	if = {
		limit = { has_idea = STP_hedonism_with_no_bondaries }
		ADISCORD_STP_lock_regular_army_templates = yes
	}
	else = {
		ADISCORD_STP_unlock_regular_army_templates = yes
	}

	if = {
		limit = {
			NOT = {
				has_dynamic_modifier = {
					modifier = STP_party_suspicion_dynamic_modifier
				}
			}
		}
		add_dynamic_modifier = {
			modifier = STP_party_suspicion_dynamic_modifier
		}
	}

	if = {
		limit = {
			NOT = {
				has_dynamic_modifier = {
					modifier = STP_fading_father
				}
			}
		}
		add_dynamic_modifier = {
			modifier = STP_fading_father
		}
	}
}

STP_refresh_party_suspicion = {
	clamp_variable = {
		var = STP_party_suspicion
		min = 0
		max = 100
	}

	# 0 suspicion -> +35% PP gain; 50 -> 0%; 100 -> -35%.
	set_variable = {
		var = STP_sus_political_power_factor
		value = STP_party_suspicion
	}
	multiply_variable = {
		var = STP_sus_political_power_factor
		value = -0.007
	}
	add_to_variable = {
		var = STP_sus_political_power_factor
		value = 0.35
	}
	clamp_variable = {
		var = STP_sus_political_power_factor
		min = -0.35
		max = 0.35
	}
	force_update_dynamic_modifier = yes
}

# Call with temporary variable STP_party_suspicion_change.
STP_change_party_suspicion = {
	add_to_variable = {
		var = STP_party_suspicion
		value = STP_party_suspicion_change
	}

	STP_refresh_party_suspicion = yes

	set_temp_variable = {
		var = STP_party_suspicion_change
		value = 0
	}
}

STP_refresh_leader_health = {
	clamp_variable = {
		var = STP_leader_health_stage
		min = 1
		max = 5
	}

	set_variable = {
		var = STP_fading_father_stability_factor
		value = 0
	}

	if = {
		limit = {
			check_variable = {
				var = STP_leader_health_stage
				value = 5
				compare = equals
			}
		}
		set_country_flag = STP_ivanov_dead
		if = {
			limit = { has_character = STP_Petr_Ivanov }
			set_portraits = {
				character = STP_Petr_Ivanov
				civilian = { large = GFX_portrait_STP_Petr_Ivanov_animated }
			}
		}
	}
	else = {
		clr_country_flag = STP_ivanov_dead
		if = {
			limit = { has_character = STP_Petr_Ivanov }
			set_portraits = {
				character = STP_Petr_Ivanov
				civilian = { large = GFX_portrait_STP_Petr_Ivanov }
			}
		}
	}

	if = {
		limit = {
			check_variable = {
				var = STP_leader_health_stage
				value = 2
				compare = equals
			}
		}
		set_variable = {
			var = STP_fading_father_stability_factor
			value = -0.05
		}
	}
	else_if = {
		limit = {
			check_variable = {
				var = STP_leader_health_stage
				value = 3
				compare = equals
			}
		}
		set_variable = {
			var = STP_fading_father_stability_factor
			value = -0.10
		}
	}
	else_if = {
		limit = {
			check_variable = {
				var = STP_leader_health_stage
				value = 4
				compare = equals
			}
		}
		set_variable = {
			var = STP_fading_father_stability_factor
			value = -0.20
		}
	}
	else_if = {
		limit = {
			check_variable = {
				var = STP_leader_health_stage
				value = 5
				compare = equals
			}
		}
		set_variable = {
			var = STP_fading_father_stability_factor
			value = -0.30
		}
	}
	force_update_dynamic_modifier = yes
}

# Call with temporary variable STP_requested_health_stage.
STP_set_leader_health_stage = {
	set_variable = {
		var = STP_leader_health_stage
		value = STP_requested_health_stage
	}

	STP_refresh_leader_health = yes
	if = { limit = { has_country_flag = STP_ivanov_dead } STP_cw_begin_elections = yes }

	set_temp_variable = {
		var = STP_requested_health_stage
		value = 0
	}
}

	if = {
		limit = { NOT = { has_country_flag = STP_cw_first_health_decline } }

			set_power_balance = { id = STP_shabrat_election_legitimacy set_value = -0.20 }

		if = {
			limit = { NOT = { has_country_flag = STP_battle_for_stelander_active } }
			STP_initialize_battle_for_stelander = yes
		}
	}
}

STP_cw_advance_health = {
	if = {
		limit = { NOT = { has_country_flag = STP_cw_first_health_decline } NOT = { has_country_flag = STP_cw_participant } }
		STP_cw_first_health_decline = yes
	}
	else_if = {
		limit = { NOT = { has_global_flag = STP_cw_started } check_variable = { var = STP_leader_health_stage value = 5 compare = less_than } }
		set_temp_variable = { var = STP_requested_health_stage value = STP_leader_health_stage }
		add_to_temp_variable = { var = STP_requested_health_stage value = 1 }
		STP_set_leader_health_stage = yes
		if = { limit = { check_variable = { var = STP_leader_health_stage value = 3 compare = equals } } country_event = { id = ADISCORD_STP_preparation.20 days = 7 } }
		if = { limit = { check_variable = { var = STP_leader_health_stage value = 4 compare = equals } } country_event = { id = ADISCORD_STP_preparation.21 days = 10 } }
		if = {
			limit = { check_variable = { var = STP_leader_health_stage value = 3 compare = equals } has_idea = volunteer_only }
			add_ideas = limited_conscription
			ADISCORD_economy_mark_dirty = yes
			country_event = { id = ADISCORD_STP_preparation.19 }
		}
		if = {
			limit = { has_country_flag = STP_ivanov_dead }
			STP_cw_begin_elections = yes
		}
		else = { country_event = { id = ADISCORD_STP_preparation.2 days = 1 } }
	}
}

# The opening deadline runs independently of focus completion.
# Each later health stage lasts seven weeks; elections have their own deadline.
STP_cw_schedule_health = {
	activate_mission = ivanov_health_is_getting_worse
	add_days_mission_timeout = { mission = ivanov_health_is_getting_worse days = -21 }
}

STP_cw_begin_elections = {
	if = {
		limit = {
			has_country_flag = STP_ivanov_dead
			NOT = { has_country_flag = STP_cw_elections_started }
			NOT = { has_global_flag = STP_cw_started }
		}
		STP_cw_open_preparation = yes
		set_country_flag = STP_cw_elections_started
		# Ivanov remains the public face until the election result replaces the presidency.
		remove_mission = ivanov_health_is_getting_worse
		set_variable = { var = STP_cw_completed_public_campaigns value = 0 }
		activate_mission = STP_cw_election_window
		if = { limit = { has_country_flag = STP_sided_with_Maksim_flag } activate_mission = STP_cw_credentials_review }
		country_event = { id = ADISCORD_STP_preparation.3 }
		STP_cw_launch_northern_crisis = yes
		NOD = { country_event = { id = ADISCORD_STP_cw.42 days = 112 } }
		log = "[STP CW] Elections opened: 140 days to the result"
	}
}

STP_cw_finish_elections = {
	if = {
		limit = { has_country_flag = STP_cw_elections_started NOT = { has_global_flag = STP_cw_started } NOT = { has_country_flag = STP_cw_elections_finished } }
		set_country_flag = STP_cw_elections_finished
		# Freeze the mandate before closing the BOP or distributing army reserves.
		# A disputed result favours the incumbent apparatus. Exceptional support
		# requires surviving civil and military commitments, not only the BOP threshold.
		if = {
			limit = { power_balance_value = { id = STP_shabrat_election_legitimacy value > 0.1 } }
			set_country_flag = STP_cw_shabrat_election_victory
			if = { limit = { STP_cw_limited_party_revolt_ready = yes } set_country_flag = STP_cw_limited_party_revolt }
			set_variable = { var = STP_cw_mandate_org_regain value = 0.20 }
			set_variable = { var = STP_cw_mandate_core_defence value = 0.10 }
			set_variable = { var = STP_cw_mandate_training_time value = -0.15 }
			set_variable = { var = STP_cw_mandate_mobilization value = 0.20 }
			if = {
				limit = { power_balance_value = { id = STP_shabrat_election_legitimacy value > 0.55 } }
				set_variable = { var = STP_cw_mandate_org_regain value = 0.30 }
				set_variable = { var = STP_cw_mandate_core_defence value = 0.15 }
				set_variable = { var = STP_cw_mandate_training_time value = -0.20 }
				set_variable = { var = STP_cw_mandate_mobilization value = 0.30 }
			}
			if = {
				limit = { has_country_flag = STP_cw_public_command_agreement STP_cw_public_command_available = yes }
				add_to_variable = { var = STP_cw_mandate_org_regain value = 0.10 }
				add_to_variable = { var = STP_cw_mandate_core_defence value = 0.05 }
			}
		}
		else = { set_country_flag = STP_cw_party_election_victory }
		clr_country_flag = STP_cw_public_command_agreement
		STP_cw_start = yes
		if = { limit = { NOT = { has_country_flag = STP_cw_participant } } country_event = { id = ADISCORD_STP_preparation.4 } }
		log = "[STP CW] Election mandate recorded; civil-war split requested"
	}
}

# COUNTRY scope. The first target allows 35 focus days + a 21-day response;
# subsequent operations use the normal 28-day clock.
# The fixed five-state pool avoids the separatist districts and daily scans.
STP_cw_schedule_opposition = {
	if = {
		limit = {
			tag = STP
			STP_cw_preparation_open = yes
			has_country_flag = STP_battle_for_stelander_active
			has_country_flag = STP_sided_with_the_party_flag
			NOT = { has_active_mission = STP_cw_opposition_preparation }
		}
		2 = { clr_state_flag = STP_cw_opposition_target }
		3 = { clr_state_flag = STP_cw_opposition_target }
		29 = { clr_state_flag = STP_cw_opposition_target }
		46 = { clr_state_flag = STP_cw_opposition_target }
		53 = { clr_state_flag = STP_cw_opposition_target }
		random_owned_state = {
			limit = {
				OR = { state = 2 state = 3 state = 29 state = 46 state = 53 }
				STP_region_is_operable = yes
				is_controlled_by = STP
				OR = {
					NOT = { has_state_flag = STP_resistance_administration_asset }
					NOT = { STP_region_resistance_entrenched = yes }
					AND = { OR = { state = 2 state = 29 } NOT = { has_state_flag = STP_resistance_garrison_asset } }
					AND = { OR = { state = 3 state = 46 } NOT = { has_state_flag = STP_resistance_supply_asset } }
					AND = { state = 53 NOT = { has_state_flag = STP_resistance_sabotage_asset } }
				}
			}
			set_state_flag = STP_cw_opposition_target
		}
		activate_mission = STP_cw_opposition_preparation
		if = {
			limit = { NOT = { has_country_flag = STP_cw_first_opposition_announced } }
			set_country_flag = STP_cw_first_opposition_announced
			add_days_mission_timeout = { mission = STP_cw_opposition_preparation days = 32 }
		}
	}
}

# COUNTRY scope, from the announced mission. A raid clears its target; ownership,
# operation locks and the preparation phase are checked again at the callback.
STP_cw_prepare_opposition = {
	if = {
		limit = { STP_cw_preparation_open = yes has_country_flag = STP_battle_for_stelander_active has_country_flag = STP_sided_with_the_party_flag }
		every_owned_state = {
			limit = { has_state_flag = STP_cw_opposition_target STP_region_is_operable = yes is_controlled_by = STP }
			clr_state_flag = STP_cw_opposition_target
			# Administration must have existed before this operation to prepare arms.
			if = {
				limit = { has_state_flag = STP_resistance_administration_asset }
				if = {
					limit = { OR = { state = 2 state = 29 } NOT = { has_state_flag = STP_resistance_garrison_asset } }
					set_state_flag = STP_resistance_garrison_asset
				}
				else_if = {
					limit = { OR = { state = 3 state = 46 } NOT = { has_state_flag = STP_resistance_supply_asset } STP = { ADISCORD_economy_can_spend_100 = yes } }
					STP = { ADISCORD_economy_spend_100 = yes }
					set_state_flag = STP_resistance_supply_asset
				}
				else_if = {
					limit = { state = 53 NOT = { has_state_flag = STP_resistance_sabotage_asset } STP = { NOT = { has_equipment = { infantry_equipment < 240 } } } }
					STP = {
						set_temp_variable = { var = STP_cw_rifle_cost value = 240 }
						STP_cw_pay_rifles = yes
					}
					if = {
						limit = { STP = { has_country_flag = STP_cw_rifles_paid } }
						set_variable = { var = STP_cw_sabotage_rifles value = 240 }
						set_state_flag = STP_resistance_sabotage_asset
					}
				}
			}
			STP = { add_power_balance_value = { id = STP_shabrat_election_legitimacy value = 0.04 } }
			set_temp_variable = { var = STP_region_influence_change value = 12 }
			STP_add_resistance_influence = yes
			if = {
				limit = { STP_region_resistance_entrenched = yes }
				if = { limit = { NOT = { has_state_flag = STP_resistance_administration_asset } } set_state_flag = STP_resistance_administration_asset }
			}
			log = "[STP preparation] Announced district operation completed in [THIS.GetName]"
		}
		# The election deadline alone triggers the split. A disrupted operation
		# still spends its announced time before the next target is selected.
		hidden_effect = { country_event = { id = ADISCORD_STP_preparation.5 days = 1 } }
		STP_cw_refresh_army_report = yes
	}
}

# STATE scope, before confiscation. Finding an actual network helps the party;
# an empty search discredits it. Reuse for timed inspections and player orders.
STP_cw_assess_search_evidence = {
	if = {
		limit = { STP = { STP_cw_preparation_open = yes } }
		if = {
			limit = {
				OR = {
					has_state_flag = STP_cw_opposition_target
					STP_region_has_detectable_assets = yes
					check_variable = { var = STP_cw_cached_rifles value = 0 compare = greater_than }
					check_variable = { var = STP_cw_pending_rifles value = 0 compare = greater_than }
				}
			}
			STP = {
				add_power_balance_value = { id = STP_shabrat_election_legitimacy value = -0.04 }
				set_variable = { var = STP_last_party_response value = 10 }
			}
		}
		else = {
			STP = {
				add_power_balance_value = { id = STP_shabrat_election_legitimacy value = 0.02 }
				set_variable = { var = STP_last_party_response value = 11 }
			}
		}
	}
}

# COUNTRY: forecast formations using the mobilisation conditions in STP_cw_start.
STP_cw_refresh_army_report = {
	set_variable = { var = STP_cw_report_party_brigades value = 16 }
	set_variable = { var = STP_cw_report_resistance_brigades value = 14 }
	# Early-uprising baseline; the election result transfers two brigades and 10% of reserves.
	if = { limit = { has_country_flag = STP_cw_party_reserve_ready } add_to_variable = { var = STP_cw_report_party_brigades value = 2 } }
	if = { limit = { 2 = { is_owned_by = STP has_state_flag = STP_party_reinforced_garrison_asset NOT = { STP_cw_region_goes_to_resistance = yes } NOT = { STP_cw_region_goes_to_republics = yes } } } add_to_variable = { var = STP_cw_report_party_brigades value = 2 } }
	if = { limit = { 29 = { is_owned_by = STP has_state_flag = STP_party_reinforced_garrison_asset NOT = { STP_cw_region_goes_to_resistance = yes } NOT = { STP_cw_region_goes_to_republics = yes } } } add_to_variable = { var = STP_cw_report_party_brigades value = 2 } }
	if = { limit = { 45 = { is_owned_by = STP has_state_flag = STP_party_reinforced_garrison_asset NOT = { STP_cw_region_goes_to_resistance = yes } NOT = { STP_cw_region_goes_to_republics = yes } } } add_to_variable = { var = STP_cw_report_party_brigades value = 2 } }
	if = { limit = { has_country_flag = STP_cw_abila_reserve_ready } add_to_variable = { var = STP_cw_report_resistance_brigades value = 2 } }
	if = { limit = { has_country_flag = STP_cw_local_recruits_ready } add_to_variable = { var = STP_cw_report_resistance_brigades value = 2 } }
	if = { limit = { 2 = { OR = { AND = { STP_cw_region_goes_to_resistance = yes has_state_flag = STP_resistance_garrison_asset } has_state_flag = STP_cw_garrison_evacuated } } } add_to_variable = { var = STP_cw_report_resistance_brigades value = 2 } }
	if = { limit = { 29 = { OR = { AND = { STP_cw_region_goes_to_resistance = yes has_state_flag = STP_resistance_garrison_asset } has_state_flag = STP_cw_garrison_evacuated } } } add_to_variable = { var = STP_cw_report_resistance_brigades value = 2 } }
}

# State scope. The ledger is cleared before credits, so late callbacks cannot duplicate it.
STP_cw_return_preparation_reserves = {
	STP_cw_refund_kefreyt_volunteers = yes
	# Abandoned weapons remain with the territorial owner, as local stockpiles.
	set_temp_variable = { var = STP_cw_returned_rifles value = STP_cw_cached_rifles }
	add_to_temp_variable = { var = STP_cw_returned_rifles value = STP_cw_pending_rifles }
	add_to_temp_variable = { var = STP_cw_returned_rifles value = STP_cw_sabotage_rifles }
	clear_variable = STP_cw_cached_rifles
	clear_variable = STP_cw_pending_rifles
	clear_variable = STP_cw_sabotage_rifles
	if = {
		limit = { check_variable = { var = STP_cw_returned_rifles value = 0 compare = greater_than } }
		owner = { add_equipment_to_stockpile = { type = infantry_equipment amount = PREV.STP_cw_returned_rifles } }
	}
}

# NOD owns the mobilization; a Stelander succession cannot cancel its orders.
STP_cw_start_northern_war = {
	if = {
		limit = {
			has_country_flag = STP_cw_northern_crisis_pending
			STP_cw_northern_targets_available = yes
			check_variable = { var = STP_cw_northern_campaign_status value = 1 compare = less_than }
		}
		clr_country_flag = STP_cw_northern_crisis_pending
		# Entry requires all three members to be factionless.
		YPR = {
			create_faction_from_template = { template = faction_template_ADISCORD_standard name = STP_northern_defence_alliance }
			add_to_faction = COF
			add_to_faction = TFF
			set_major = yes
			clear_variable = STP_cw_capitulation_occupier
		}
		COF = { set_major = yes clear_variable = STP_cw_capitulation_occupier }
		TFF = { set_major = yes clear_variable = STP_cw_capitulation_occupier }
		if = {
			limit = { YPR = { is_in_faction_with = COF is_in_faction_with = TFF } }
			declare_war_on = { target = YPR type = take_state_focus generator = { 20 } }
			if = {
				limit = { has_war_with = YPR }
				COF = { add_to_war = { targeted_alliance = YPR enemy = NOD hostility_reason = asked_to_join single_target_only = yes } }
				TFF = { add_to_war = { targeted_alliance = YPR enemy = NOD hostility_reason = asked_to_join single_target_only = yes } }
			}
		}
		if = {
			limit = { has_war_with = YPR has_war_with = COF has_war_with = TFF }
			set_variable = { var = STP_cw_northern_campaign_status value = 1 }
			# Northern deployment can permit an uprising before the election deadline.
			country_event = { id = ADISCORD_STP_cw.42 days = 7 }
			country_event = { id = ADISCORD_STP_preparation.12 days = 1 }
			news_event = { id = ADISCORD_STP_cw.82 }
			log = "[STP_NORTH] Northern alliance entered one war against NOD"
		}
		else = {
			# Cancel the wars opened by this incomplete coalition entry.
			if = { limit = { has_war_with = YPR } white_peace = YPR }
			if = { limit = { has_war_with = COF } white_peace = COF }
			if = { limit = { has_war_with = TFF } white_peace = TFF }
			STP_cw_dismantle_northern_alliance = yes
			YPR = { set_major = no }
			COF = { set_major = no }
			TFF = { set_major = no }
			set_variable = { var = STP_cw_northern_campaign_status value = 4 }
			STP = { country_event = { id = ADISCORD_STP_preparation.14 } }
		}
	}
	else_if = {
		limit = { has_country_flag = STP_cw_northern_crisis_pending }
		clr_country_flag = STP_cw_northern_crisis_pending
		set_variable = { var = STP_cw_northern_campaign_status value = 4 }
		STP = { country_event = { id = ADISCORD_STP_preparation.14 } }
	}
}

# Battle for Stelander regional influence. All mutators execute in state scope.
# Call a mutator after setting temporary variable STP_region_influence_change.

STP_normalize_region_influence = {
	clamp_variable = { var = STP_resistance_influence min = 0 max = 100 }
	clamp_variable = { var = STP_party_influence min = 0 max = 100 }

	# Local forces are the balancing term, so the three values always total 100.
	set_variable = { var = STP_local_forces_influence value = 100 }
	subtract_from_variable = { var = STP_local_forces_influence value = STP_resistance_influence }
	subtract_from_variable = { var = STP_local_forces_influence value = STP_party_influence }
	if = {
		limit = { check_variable = { var = STP_local_forces_influence value = 0 compare = less_than } }
		set_temp_variable = { var = STP_region_overflow value = STP_local_forces_influence }
		multiply_temp_variable = { var = STP_region_overflow value = -1 }
		set_variable = { var = STP_local_forces_influence value = 0 }
		if = {
			limit = { check_variable = { var = STP_party_influence value = STP_region_overflow compare = greater_than_or_equals } }
			subtract_from_variable = { var = STP_party_influence value = STP_region_overflow }
		}
		else = {
			subtract_from_temp_variable = { var = STP_region_overflow value = STP_party_influence }
			set_variable = { var = STP_party_influence value = 0 }
			subtract_from_variable = { var = STP_resistance_influence value = STP_region_overflow }
		}
	}
	clamp_variable = { var = STP_local_forces_influence min = 0 max = 100 }
	STP_calculate_region_alignment = yes
	STP = { if = { limit = { STP_cw_preparation_open = yes has_country_flag = STP_battle_for_stelander_active } STP_cw_refresh_army_report = yes } }
}

STP_add_resistance_influence = {
	set_temp_variable = { var = STP_region_remaining_shift value = STP_region_influence_change }
	clamp_temp_variable = { var = STP_region_remaining_shift min = 0 max = 100 }
	if = {
		limit = { check_variable = { var = STP_party_influence value = STP_region_remaining_shift compare = greater_than_or_equals } }
		subtract_from_variable = { var = STP_party_influence value = STP_region_remaining_shift }
		add_to_variable = { var = STP_resistance_influence value = STP_region_remaining_shift }
		set_temp_variable = { var = STP_region_remaining_shift value = 0 }
	}
	else = {
		add_to_variable = { var = STP_resistance_influence value = STP_party_influence }
		subtract_from_temp_variable = { var = STP_region_remaining_shift value = STP_party_influence }
		set_variable = { var = STP_party_influence value = 0 }
		if = {
			limit = { check_variable = { var = STP_local_forces_influence value = STP_region_remaining_shift compare = greater_than_or_equals } }
			subtract_from_variable = { var = STP_local_forces_influence value = STP_region_remaining_shift }
			add_to_variable = { var = STP_resistance_influence value = STP_region_remaining_shift }
		}
		else = {
			add_to_variable = { var = STP_resistance_influence value = STP_local_forces_influence }
			set_variable = { var = STP_local_forces_influence value = 0 }
		}
	}
	STP_normalize_region_influence = yes
}

STP_calculate_region_alignment = {
	set_variable = { var = STP_region_lead value = STP_resistance_influence }
	set_variable = { var = STP_region_runner_up value = STP_party_influence }
	set_variable = { var = STP_region_alignment value = 1 }
	if = {
		limit = { check_variable = { var = STP_party_influence value = STP_region_lead compare = greater_than } }
		set_variable = { var = STP_region_runner_up value = STP_region_lead }
		set_variable = { var = STP_region_lead value = STP_party_influence }
		set_variable = { var = STP_region_alignment value = 2 }
	}
	else_if = {
		limit = { check_variable = { var = STP_party_influence value = STP_region_runner_up compare = greater_than } }
		set_variable = { var = STP_region_runner_up value = STP_party_influence }
	}
	if = {
		limit = { check_variable = { var = STP_local_forces_influence value = STP_region_lead compare = greater_than } }
		set_variable = { var = STP_region_runner_up value = STP_region_lead }
		set_variable = { var = STP_region_lead value = STP_local_forces_influence }
		set_variable = { var = STP_region_alignment value = 3 }
	}
	else_if = {
		limit = { check_variable = { var = STP_local_forces_influence value = STP_region_runner_up compare = greater_than } }
		set_variable = { var = STP_region_runner_up value = STP_local_forces_influence }
	}
	set_variable = { var = STP_region_margin value = STP_region_lead }
	subtract_from_variable = { var = STP_region_margin value = STP_region_runner_up }
	set_variable = { var = STP_region_third value = 100 }
	subtract_from_variable = { var = STP_region_third value = STP_region_lead }
	subtract_from_variable = { var = STP_region_third value = STP_region_runner_up }
	set_variable = { var = STP_region_status value = 0 }
	if = {
		limit = {
			check_variable = { var = STP_region_lead value = 40 compare = greater_than_or_equals }
			check_variable = { var = STP_region_margin value = 10 compare = greater_than_or_equals }
		}
		if = { limit = { check_variable = { var = STP_region_alignment value = 1 compare = equals } } set_variable = { var = STP_region_status value = 1 } }
		else_if = { limit = { check_variable = { var = STP_region_alignment value = 2 compare = equals } } set_variable = { var = STP_region_status value = 3 } }
		else = { set_variable = { var = STP_region_status value = 5 } }
	}
	if = {
		limit = {
			OR = {
				AND = {
					check_variable = { var = STP_region_lead value = 60 compare = greater_than_or_equals }
					check_variable = { var = STP_region_margin value = 20 compare = greater_than_or_equals }
				}
				AND = {
					check_variable = { var = STP_region_lead value = 55 compare = greater_than_or_equals }
					check_variable = { var = STP_region_runner_up value = 40 compare = less_than_or_equals }
					check_variable = { var = STP_region_margin value = 15 compare = greater_than_or_equals }
					check_variable = { var = STP_region_third value = 10 compare = less_than_or_equals }
				}
			}
		}
		if = { limit = { check_variable = { var = STP_region_alignment value = 1 compare = equals } } set_variable = { var = STP_region_status value = 2 } }
		else_if = { limit = { check_variable = { var = STP_region_alignment value = 2 compare = equals } } set_variable = { var = STP_region_status value = 4 } }
		else = { set_variable = { var = STP_region_status value = 6 } }
	}
}

STP_schedule_next_party_inspection = {
	if = {
		limit = { has_country_flag = STP_battle_for_stelander_active }
		random_list = {
			10 = {
				modifier = { factor = 0 check_variable = { var = STP_last_inspection_state value = 2 compare = equals } }
				modifier = { factor = 3 2 = { STP_region_resistance_strongest = yes } }
				modifier = { factor = 2 2 = { has_state_flag = STP_region_recent_player_action } }
				modifier = { factor = 2 2 = { STP_region_has_detectable_assets = yes } }
				2 = { set_state_flag = STP_party_inspection_active }
				activate_mission = STP_party_inspection_state_2
			}
			10 = {
				modifier = { factor = 0 check_variable = { var = STP_last_inspection_state value = 3 compare = equals } }

# COUNTRY: caller sets positive temp STP_cw_rifle_cost.
# Debit each producer separately; an unspecified producer repeats the debit across mixed stocks.
STP_cw_pay_rifles = {
	clr_country_flag = STP_cw_rifles_paid
	set_temp_variable = { var = STP_cw_rifles_remaining value = STP_cw_rifle_cost }
	if = {
		limit = {
			check_variable = { var = STP_cw_rifle_cost value = 0 compare = greater_than }
			check_variable = { var = num_equipment@infantry_equipment value = STP_cw_rifle_cost compare = greater_than_or_equals }
		}
		every_possible_country = {
			limit = { PREV = { check_variable = { var = STP_cw_rifles_remaining value = 0 compare = greater_than } } }
			# PREV enters the payer; inside it PREV is this iteration's producer.
			PREV = {
				# Recheck the unpaid balance after each producer debit.
				if = {
					limit = { check_variable = { var = STP_cw_rifles_remaining value = 0 compare = greater_than } }
					set_temp_variable = { var = STP_cw_rifles_removed value = num_equipment@infantry_equipment }
					set_temp_variable = { var = STP_cw_rifle_debit value = STP_cw_rifles_remaining }
					multiply_temp_variable = { var = STP_cw_rifle_debit value = -1 }
					add_equipment_to_stockpile = { type = infantry_equipment amount = STP_cw_rifle_debit producer = PREV }
					subtract_from_temp_variable = { var = STP_cw_rifles_removed value = num_equipment@infantry_equipment }
					subtract_from_temp_variable = { var = STP_cw_rifles_remaining value = STP_cw_rifles_removed }
				}
			}
		}
		if = {
			limit = { check_variable = { var = STP_cw_rifles_remaining value = 0 compare = equals } }
			set_country_flag = STP_cw_rifles_paid
			log = "[STP_CW] [THIS.GetTag] exact producer-aware rifle payment=[?STP_cw_rifle_cost]"
		}
		else = { log = "[STP_CW] [THIS.GetTag] rifle payment incomplete, no asset created; remaining=[?STP_cw_rifles_remaining]" }
	}
}

		limit = {

			NOT = { has_manpower < 6000 }
			NOT = { has_equipment = { infantry_equipment < 600 } }
		}
		set_temp_variable = { var = STP_cw_rifle_cost value = 600 }
		STP_cw_pay_rifles = yes
		if = {
			limit = { has_country_flag = STP_cw_rifles_paid }
			add_manpower = -6000
			# Keep a capital garrison; later brigades go to the actual front.
			# The fallback also permits wartime mobilisation after losing the capital.
			if = {
				limit = { num_divisions < 3 capital_scope = { is_owned_by = PREV is_controlled_by = PREV } }
				capital_scope = { STP_cw_create_territorial_brigade = yes }
			}
			else_if = {
				limit = { any_owned_state = { is_controlled_by = PREV STP_cw_state_faces_rival = yes } }
				random_owned_controlled_state = {
					limit = { STP_cw_state_faces_rival = yes }
					STP_cw_create_territorial_brigade = yes
				}
			}
			else = {
				random_owned_controlled_state = { STP_cw_create_territorial_brigade = yes }
			}
			log = "[STP_CW] [THIS.GetTag] brigade paid: 6000 manpower, 600 rifles; remaining manpower=[?manpower] rifles=[?num_equipment@infantry_equipment]"
		}
	}
}

				check_variable = { var = STP_cw_payment_infantry_equipment value = 0 compare = greater_than }
				check_variable = { var = STP_cw_payment_ADISCORD_squad_weapons_equipment value = 0 compare = greater_than }

				check_variable = { var = STP_cw_payment_anti_air_equipment value = 0 compare = greater_than }
			} } }
			PREV = {
				if = {
					limit = { check_variable = { var = STP_cw_payment_infantry_equipment value = 0 compare = greater_than } }
					set_temp_variable = { var = STP_cw_payment_before value = num_equipment@infantry_equipment }
					set_temp_variable = { var = STP_cw_payment_debit value = STP_cw_payment_infantry_equipment }
					multiply_temp_variable = { var = STP_cw_payment_debit value = -1 }
					add_equipment_to_stockpile = { type = infantry_equipment amount = STP_cw_payment_debit producer = PREV }
					subtract_from_temp_variable = { var = STP_cw_payment_before value = num_equipment@infantry_equipment }
					subtract_from_temp_variable = { var = STP_cw_payment_infantry_equipment value = STP_cw_payment_before }
				}
				if = {
					limit = { check_variable = { var = STP_cw_payment_ADISCORD_squad_weapons_equipment value = 0 compare = greater_than } }
					set_temp_variable = { var = STP_cw_payment_before value = num_equipment@ADISCORD_squad_weapons_equipment }
					set_temp_variable = { var = STP_cw_payment_debit value = STP_cw_payment_ADISCORD_squad_weapons_equipment }
					multiply_temp_variable = { var = STP_cw_payment_debit value = -1 }
					add_equipment_to_stockpile = { type = ADISCORD_squad_weapons_equipment amount = STP_cw_payment_debit producer = PREV }
					subtract_from_temp_variable = { var = STP_cw_payment_before value = num_equipment@ADISCORD_squad_weapons_equipment }
					subtract_from_temp_variable = { var = STP_cw_payment_ADISCORD_squad_weapons_equipment value = STP_cw_payment_before }
				}
				if = {
					limit = { check_variable = { var = STP_cw_payment_support_equipment value = 0 compare = greater_than } }
					set_temp_variable = { var = STP_cw_payment_before value = num_equipment@support_equipment }
					set_temp_variable = { var = STP_cw_payment_debit value = STP_cw_payment_support_equipment }
					multiply_temp_variable = { var = STP_cw_payment_debit value = -1 }
					add_equipment_to_stockpile = { type = support_equipment amount = STP_cw_payment_debit producer = PREV }
					subtract_from_temp_variable = { var = STP_cw_payment_before value = num_equipment@support_equipment }
					subtract_from_temp_variable = { var = STP_cw_payment_support_equipment value = STP_cw_payment_before }
				}

					subtract_from_temp_variable = { var = STP_cw_payment_artillery_equipment value = STP_cw_payment_before }
				}
				if = {
					limit = { check_variable = { var = STP_cw_payment_anti_air_equipment value = 0 compare = greater_than } }
					set_temp_variable = { var = STP_cw_payment_before value = num_equipment@anti_air_equipment }

					subtract_from_temp_variable = { var = STP_cw_payment_anti_air_equipment value = STP_cw_payment_before }

# STATE: release reserved resources after transfer and before cleanup.
STP_cw_materialize_region_assets = {
	if = {
		limit = { NOT = { has_state_flag = STP_cw_assets_consumed } }
		set_state_flag = STP_cw_assets_consumed
		STP_cw_deploy_kefreyt_volunteers = yes
		# Unfinished orders already returned before the proportional allocation.
		if = {
			limit = { check_variable = { var = STP_cw_cached_rifles value = 0 compare = greater_than } }
			owner = { add_equipment_to_stockpile = { type = infantry_equipment amount = PREV.STP_cw_cached_rifles } }
			log = "[STP_CW] state [THIS.GetID] cached rifles paid=[?STP_cw_cached_rifles]"
		}
		if = {
			limit = { check_variable = { var = STP_cw_sabotage_rifles value = 0 compare = greater_than } }
			owner = { add_equipment_to_stockpile = { type = infantry_equipment amount = PREV.STP_cw_sabotage_rifles } }
			log = "[STP_CW] state [THIS.GetID] sabotage escrow paid=[?STP_cw_sabotage_rifles]"
		}
		# Each captured port or inland depot transfers up to 500 fuel from government stock.
		if = {
			limit = {
				has_state_flag = STP_resistance_supply_asset
				OR = { state = 3 state = 46 }
				is_owned_by = STS
				STP = { check_variable = { var = fuel_k value = 0.5 compare = greater_than_or_equals } }
			}
			STP = { add_fuel = -500 }
			owner = { add_fuel = 500 }
			log = "[STP_CW] state [THIS.GetID] supply asset transferred 500 existing fuel"
		}
		# The prepared railway sabotage matters only if the party keeps this district.
		# A captured district keeps its infrastructure; damage is repaired normally.
		if = {
			limit = { state = 53 is_owned_by = STP is_controlled_by = STP has_state_flag = STP_resistance_sabotage_asset infrastructure > 0 }
			damage_building = { type = infrastructure damage = 1 }
		}
		if = {
			limit = { has_state_flag = STP_cw_defensive_line_prepared }
			add_building_construction = { type = bunker level = 2 instant_build = yes province = { all_provinces = yes limit_to_border = yes } }
			clr_state_flag = STP_cw_defensive_line_prepared
		}
		clear_variable = STP_cw_cached_rifles
		clear_variable = STP_cw_sabotage_rifles
		clr_state_flag = STP_resistance_sabotage_asset
		clr_state_flag = STP_resistance_administration_asset
		clr_state_flag = STP_local_deal_asset
		clr_state_flag = STP_party_inspection_active
		# Garrison assets are consumed separately after paid mobilisation.
	}
}

STP_cw_prepare_successor = {
	set_country_flag = STP_cw_participant
	set_rule = { can_join_factions = no can_create_factions = no }
	inherit_technology = STP
	# Recruits retain the union's current legal mobilisation ceiling.
	if = { limit = { STP = { has_idea = disarmed_nation } } add_ideas = disarmed_nation }
	else_if = { limit = { STP = { has_idea = volunteer_only } } add_ideas = volunteer_only }
	else_if = { limit = { STP = { has_idea = limited_conscription } } add_ideas = limited_conscription }
	else_if = { limit = { STP = { has_idea = extensive_conscription } } add_ideas = extensive_conscription }
	else_if = { limit = { STP = { has_idea = service_by_requirement } } add_ideas = service_by_requirement }
	else_if = { limit = { STP = { has_idea = all_adults_serve } } add_ideas = all_adults_serve }
	else_if = { limit = { STP = { has_idea = scraping_the_barrel } } add_ideas = scraping_the_barrel }
	load_oob = ADISCORD_STP_civil_war_templates
	set_country_flag = STP_cw_templates_loaded
	STP_cw_apply_wartime_template_locks = yes
	set_country_flag = ADISCORD_economy_important_country
	ADISCORD_economy_initialize_country = yes
	ADISCORD_economy_set_simulation_tier = yes
	STP_cw_initialize_focus_modifiers = yes
	# Record and clear the successor's manpower pool before allocating shared reserves.
	log = "[STP_CW] newborn [THIS.GetTag] engine manpower=[?manpower]"
	# transfer_state already moved this recruitable pool out of STP. Return it,
	# then apply the mandate-dependent allocation from the original reserve.
	set_variable = { var = STP_cw_newborn_manpower value = manpower }
	STP = { add_manpower = var:PREV.STP_cw_newborn_manpower }
	multiply_variable = { var = STP_cw_newborn_manpower value = -1 }
	add_manpower = var:STP_cw_newborn_manpower
	clear_variable = STP_cw_newborn_manpower
}

# COUNTRY (STP). Human event options call this immediately before handoff;

			add_timed_idea = { idea = STP_cw_defensive_preparation days = 35 }
		}

		# Either captured depot supports infantry during mobilisation. Two depots
		# transfer their own fuel but never stack or refresh this one army-wide idea.

			limit = {
				OR = {
					3 = { is_owned_by = STS has_state_flag = STP_resistance_supply_asset }
					46 = { is_owned_by = STS has_state_flag = STP_resistance_supply_asset }
				}

			STP = { remove_mission = STP_cw_nod_warning remove_mission = STP_cw_nod_redeployment clr_country_flag = STP_cw_nod_warning_active }
		}
	}
}

# Winner country scope, original on_capitulation ROOT remains the defeated rival.
STP_cw_settle_union_victory = {
	STS = { STP_cw_remove_kefreyt_volunteers = yes }
	log = "[STP_CW] capital occupation settlement: [THIS.GetTag] defeated [ROOT.GetTag]; reported FROM=[FROM.GetTag]"
	if = { limit = { ROOT = { tag = STP } } STP_cw_end_nod_intervention = yes }
	if = { limit = { ROOT = { tag = STS } } STP_cw_close_nod_after_party_victory = yes }
	if = { limit = { has_war_with = ROOT } white_peace = ROOT }
	ROOT = { remove_ideas = { STP_cw_party_battle_spirit STP_cw_resistance_battle_spirit } }
	annex_country = { target = ROOT transfer_troops = no }
	# The resistance is a claimant to Stelander; foreign conquests stay non-core.
	if = {
		limit = { tag = STS }
		every_owned_state = {
			limit = { is_core_of = STP NOT = { is_core_of = STS } }
			add_core_of = STS
		}
	}
	if = { limit = { has_dynamic_modifier = { modifier = STP_cw_election_mandate_dynamic } } remove_dynamic_modifier = { modifier = STP_cw_election_mandate_dynamic } }
	set_country_flag = STP_cw_won_union_battle
	STP_cw_finish_mobilization = yes
	if = { limit = { tag = STP } news_event = { id = ADISCORD_STP_cw.71 hours = 1 } }
	else_if = { limit = { tag = STS } news_event = { id = ADISCORD_STP_cw.72 hours = 1 } }
	STP_cw_check_union_wars_finished = yes
}

# Only dismantle the alliance created from factionless northern members.
# Leadership may have passed from YPR to another member during the campaign.
STP_cw_dismantle_northern_alliance = {
	YPR = {

			limit = {
				is_in_faction = yes
				has_faction_template = faction_template_ADISCORD_standard
				NOT = { any_other_country = { is_in_faction_with = YPR NOT = { OR = { tag = YPR tag = COF tag = TFF } } } }

		}
	}
}

# NOD scope. Resolve only the northern campaign; every other war remains intact.
STP_cw_settle_northern_victory = {
	if = {
		limit = { OR = { STP_cw_northern_campaign_won = yes STP_cw_northern_conference_won = yes } }
		if = {
			limit = { STP_cw_northern_conference_won = yes }
			set_variable = { var = STP_cw_northern_campaign_status value = 2 }
			# These states have already been awarded to NOD's side by the conference.
			YPR = { transfer_state = 8 transfer_state = 15 transfer_state = 16 transfer_state = 19 transfer_state = 20 transfer_state = 21 transfer_state = 22 }
			TFF = { transfer_state = 83 transfer_state = 84 transfer_state = 85 transfer_state = 86 transfer_state = 87 transfer_state = 303 }
			transfer_state = 14
			if = { limit = { TFF = { is_puppet_of = NOD } } set_autonomy = { target = TFF autonomy_state = autonomy_free end_wars = no end_civil_wars = no } }
			if = { limit = { TFF = { is_in_faction_with = NOD } } remove_from_faction = TFF }
		}
		else = { set_variable = { var = STP_cw_northern_campaign_status value = 2 } }
		activate_mission = NOD_cw_northern_redeployment
		if = { limit = { has_war_with = YPR } white_peace = YPR }
		if = { limit = { has_war_with = COF } white_peace = COF }
		if = { limit = { has_war_with = TFF } white_peace = TFF }
		STP_cw_end_northern_colonial_war = yes
		STP_cw_dismantle_northern_alliance = yes
		YPR = { set_major = no }
		COF = { set_major = no }
		TFF = { set_major = no }
		if = {
			limit = { YPR = { exists = yes is_subject = no STP_cw_ypr_has_only_historical_states = yes } }
			puppet = { target = YPR end_wars = no end_civil_wars = no }
		}
		if = {
			limit = { COF = { exists = yes is_subject = no } }
			annex_country = { target = COF transfer_troops = no }
		}
		news_event = { id = ADISCORD_STP_cw.80 }
		STS = { if = { limit = { exists = yes } STP_cw_poll_nod_intervention = yes } }
		log = "[STP_NORTH] NOD victory after three accepted military defeats: YPR historical bounds checked, TFF independent"
	}
}

# Called only by the guarded immediate NOD capitulation callback, before native cap state flips.
STP_cw_settle_northern_defeat = {
	if = {
		limit = {
			check_variable = { var = STP_cw_northern_campaign_status value = 1 compare = equals }
			OR = {
				AND = { has_war_with = YPR check_variable = { var = STP_cw_capitulation_occupier value = 6 compare = equals } }
				AND = { has_war_with = COF check_variable = { var = STP_cw_capitulation_occupier value = 7 compare = equals } }
				AND = { has_war_with = TFF check_variable = { var = STP_cw_capitulation_occupier value = 8 compare = equals } }
			}
		}
		set_variable = { var = STP_cw_northern_campaign_status value = 3 }
		if = { limit = { has_war_with = YPR } white_peace = YPR }
		if = { limit = { has_war_with = COF } white_peace = COF }
		if = { limit = { has_war_with = TFF } white_peace = TFF }
		STP_cw_end_northern_colonial_war = yes
		YPR = { set_major = no }
		COF = { set_major = no }
		TFF = { set_major = no }
		if = {
			limit = { YPR = { exists = yes is_subject = no } }
			if = { limit = { 17 = { is_owned_by = NOD } } YPR = { transfer_state = 17 } }
			if = { limit = { 18 = { is_owned_by = NOD } } YPR = { transfer_state = 18 } }
		}
		news_event = { id = ADISCORD_STP_cw.81 }
		STS = { if = { limit = { exists = yes } STP_cw_poll_nod_intervention = yes } }
		log = "[STP_NORTH] NOD defeated by the north: separate peace, owned 17/18 ceded to YPR"
	}
}

STP_cw_poll_northern_campaign = {
	if = {
		limit = { check_variable = { var = STP_cw_northern_campaign_status value = 1 compare = equals } }
		if = { limit = { OR = { STP_cw_northern_campaign_won = yes STP_cw_northern_conference_won = yes } } STP_cw_settle_northern_victory = yes }
		else_if = {
			limit = {
				NOT = {
					AND = {
						has_war_with = YPR
						has_war_with = COF
						has_war_with = TFF
					}
				}
			}
			# A foreign peace or annexation cannot count as the coalition's military defeat.
			set_variable = { var = STP_cw_northern_campaign_status value = 4 }
			activate_mission = NOD_cw_northern_redeployment
			if = { limit = { has_war_with = YPR } white_peace = YPR }
			if = { limit = { has_war_with = COF } white_peace = COF }
			if = { limit = { has_war_with = TFF } white_peace = TFF }
			STP_cw_end_northern_colonial_war = yes
			YPR = { set_major = no }
			COF = { set_major = no }
			TFF = { set_major = no }
			log = "[STP_NORTH] Northern campaign closed after external peace; no scripted territorial award"
		}
	}
}

# AIN can fight only while it remains NOD's colony; an independent war is its own.
STP_cw_end_northern_colonial_war = {
	AIN = {

			if = { limit = { has_war_with = COF } white_peace = COF }
			if = { limit = { has_war_with = TFF } white_peace = TFF }
		}
	}
}
