STP_change_party_suspicion = {
	# Apply suspicion change
	add_to_variable = {
		var = STP_party_suspicion_rate
		value = STP_party_suspicion_rate_temp
	}
	clamp_variable = {
		var = STP_party_suspicion_rate
		min = 0
		max = 1
	}
	# Display value: 0–100
	set_variable = {
		var = STP_party_suspicion
		value = STP_party_suspicion_rate
	}
	multiply_variable = {
		var = STP_party_suspicion
		value = 100
	}
	# Political Power Gain:
	# +35% at 0 suspicion
	#   0% at 50 suspicion
	# -35% at 100 suspicion
	set_variable = {
		var = STP_sus_political_power_factor
		value = STP_party_suspicion_rate
	}
	multiply_variable = {
		var = STP_sus_political_power_factor
		value = -0.70
	}
	add_to_variable = {
		var = STP_sus_political_power_factor
		value = 0.35
	}
	clamp_variable = {
		var = STP_sus_political_power_factor
		min = -0.35
		max = 0.35
	}
}

STP_leader_health = {
	add_to_variable = {
		var = STP_leader_health_rate
		value = STP_leader_health_rate_temp
	}

	clamp_variable = {
		var = STP_leader_health_rate
		min = 0
		max = 1
	}
	set_variable = {
		var = STP_leader_health
		value = STP_leader_health_rate
	}
	multiply_variable = {
		var = STP_leader_health
		value = 100
	}
	#Stability
	set_temp_variable = {
		var = STP_leader_health_temp
		value = STP_leader_health_rate
	}
	add_to_temp_variable = {
		var = STP_leader_health_temp
		value = -1
	}
	set_variable = {
		var = STP_fading_father_stability_factor
		value = STP_leader_health_temp
	}
	multiply_variable = {
		var = STP_fading_father_stability_factor
		value = 0.10
	}

	#Portrait stage
	if = {
		limit = {
			check_variable = {
				var = STP_leader_health_rate
				value = 0.75
				compare = greater_than_or_equals
			}
		}
		set_variable = {
			var = STP_state_face_stage
			value = 1
		}
	}
	else_if = {
		limit = {
			check_variable = {
				var = STP_leader_health_rate
				value = 0.45
				compare = greater_than_or_equals
			}
		}
		set_variable = {
			var = STP_state_face_stage
			value = 2
		}
	}
	else_if = {
		limit = {
			check_variable = {
				var = STP_leader_health_rate
				value = 0.15
				compare = greater_than_or_equals
			}
		}
		set_variable = {
			var = STP_state_face_stage
			value = 3
		}
	}
	else_if = {
		limit = {
			check_variable = {
				var = STP_leader_health_rate
				value = 0
				compare = greater_than
			}
		}
		set_variable = {
			var = STP_state_face_stage
			value = 4
		}
	}
	else = {
		set_variable = {
			var = STP_state_face_stage
			value = 5
		}
		set_country_flag = STP_ivanov_dead
		remove_dynamic_modifier = {
			modifier = STP_fading_father
		}
	}
}
