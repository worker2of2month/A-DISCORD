STP_initialize_core_mechanics = {
	if = {
		limit = { NOT = { has_variable = STP_party_suspicion } }
		set_variable = {
			var = STP_party_suspicion
			value = 5
		}
	}

	if = {
		limit = { NOT = { has_variable = STP_leader_health_stage } }
		set_variable = {
			var = STP_leader_health_stage
			value = 1
		}
	}

	STP_refresh_party_suspicion = yes
	STP_refresh_leader_health = yes

	if = {
		limit = {
			NOT = {
				has_dynamic_modifier = {
					modifier = STP_party_suspicion_dynamic_modifier
				}
			}
		}
		add_dynamic_modifier = {
			modifier = STP_party_suspicion_dynamic_modifier
		}
	}

	if = {
		limit = {
			NOT = {
				has_dynamic_modifier = {
					modifier = STP_fading_father
				}
			}
		}
		add_dynamic_modifier = {
			modifier = STP_fading_father
		}
	}
}

STP_refresh_party_suspicion = {
	clamp_variable = {
		var = STP_party_suspicion
		min = 0
		max = 100
	}

	# 0 suspicion -> +35% PP gain; 50 -> 0%; 100 -> -35%.
	set_variable = {
		var = STP_sus_political_power_factor
		value = STP_party_suspicion
	}
	multiply_variable = {
		var = STP_sus_political_power_factor
		value = -0.007
	}
	add_to_variable = {
		var = STP_sus_political_power_factor
		value = 0.35
	}
	clamp_variable = {
		var = STP_sus_political_power_factor
		min = -0.35
		max = 0.35
	}
}

# Call with temporary variable STP_party_suspicion_change.
STP_change_party_suspicion = {
	add_to_variable = {
		var = STP_party_suspicion
		value = STP_party_suspicion_change
	}

	STP_refresh_party_suspicion = yes

	set_temp_variable = {
		var = STP_party_suspicion_change
		value = 0
	}
}

STP_refresh_leader_health = {
	clamp_variable = {
		var = STP_leader_health_stage
		min = 1
		max = 5
	}

	set_variable = {
		var = STP_fading_father_stability_factor
		value = 0
	}

	if = {
		limit = {
			check_variable = {
				var = STP_leader_health_stage
				value = 5
				compare = equals
			}
		}
		set_country_flag = STP_ivanov_dead
	}
	else = {
		clr_country_flag = STP_ivanov_dead
	}

	if = {
		limit = {
			check_variable = {
				var = STP_leader_health_stage
				value = 2
				compare = equals
			}
		}
		set_variable = {
			var = STP_fading_father_stability_factor
			value = -0.05
		}
	}
	else_if = {
		limit = {
			check_variable = {
				var = STP_leader_health_stage
				value = 3
				compare = equals
			}
		}
		set_variable = {
			var = STP_fading_father_stability_factor
			value = -0.10
		}
	}
	else_if = {
		limit = {
			check_variable = {
				var = STP_leader_health_stage
				value = 4
				compare = equals
			}
		}
		set_variable = {
			var = STP_fading_father_stability_factor
			value = -0.20
		}
	}
	else_if = {
		limit = {
			check_variable = {
				var = STP_leader_health_stage
				value = 5
				compare = equals
			}
		}
		set_variable = {
			var = STP_fading_father_stability_factor
			value = -0.30
		}
	}
}

# Call with temporary variable STP_requested_health_stage.
STP_set_leader_health_stage = {
	set_variable = {
		var = STP_leader_health_stage
		value = STP_requested_health_stage
	}

	STP_refresh_leader_health = yes

	set_temp_variable = {
		var = STP_requested_health_stage
		value = 0
	}
}
