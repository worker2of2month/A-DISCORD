# COUNTRY IVN. The committee suspends elections. A retreat from the north may
# already have installed the same etatist role, so the role is added once.
IVN_adopt_emergency_order = {
	if = {
		limit = { NOT = { has_government = etatism } }
		set_politics = {
			ruling_party = etatism
			elections_allowed = no
		}
	}
	if = {
		limit = { NOT = { has_country_flag = ADISCORD_vorkerland_vadim_etatist_role_added } }
		add_country_leader_role = {
			character = IVN_Vadim_Ivanchik
			promote_leader = yes
			country_leader = {
				ideology = etatism_ideology
				expire = "2200.1.1.1"
			}
		}
		set_country_flag = ADISCORD_vorkerland_vadim_etatist_role_added
	}
	IVN_party_representatives_emergency = yes
}

# ----- Committee and staff, frontier trust -----
# IVN_command_bop measures who commands the army of the marches: the civil
# Committee (negative side) or the Staff of the marches (positive side). The
# state variable IVN_march_trust (0-100) on the eight frontier marches records
# how far each march still trusts the capital. The revolt reads both: the
# balance sets the legion's share of the army, trust decides which marches
# follow Vedrina.

# COUNTRY IVN. One-time opening from the root focus. The revolt calls it as
# well, so its consumers exist even if the root focus was never completed.
IVN_marches_initialize = {
	if = {
		limit = { NOT = { has_country_flag = IVN_marches_initialized } }
		set_country_flag = IVN_marches_initialized
		set_power_balance = {
			id = IVN_command_bop
			set_value = 0.10
		}
		92 = { set_variable = { var = IVN_march_trust value = 45 } }
		95 = { set_variable = { var = IVN_march_trust value = 35 } }
		96 = { set_variable = { var = IVN_march_trust value = 45 } }
		97 = { set_variable = { var = IVN_march_trust value = 50 } }
		98 = { set_variable = { var = IVN_march_trust value = 60 } }
		99 = { set_variable = { var = IVN_march_trust value = 60 } }
		100 = { set_variable = { var = IVN_march_trust value = 65 } }
		101 = { set_variable = { var = IVN_march_trust value = 70 } }
		IVN_frontier_marches_refresh = yes
	}
}

# STATE. Adds the temporary IVN_trust_delta to the march's trust.
IVN_march_trust_change = {
	add_to_variable = { var = IVN_march_trust value = IVN_trust_delta }
	clamp_variable = {
		var = IVN_march_trust
		min = 0
		max = 100
	}
	IVN_march_trust_refresh = yes
}

# STATE. Swaps the visible modifier only when trust crosses 70 or 40.
IVN_march_trust_refresh = {
	if = {
		limit = { check_variable = { var = IVN_march_trust value = 70 compare = greater_than_or_equals } }
		if = {
			limit = { has_dynamic_modifier = { modifier = IVN_march_disloyal } }
			remove_dynamic_modifier = { modifier = IVN_march_disloyal }
		}
		if = {
			limit = { NOT = { has_dynamic_modifier = { modifier = IVN_march_loyal } } }
			add_dynamic_modifier = { modifier = IVN_march_loyal }
		}
	}
	else_if = {
		limit = { check_variable = { var = IVN_march_trust value = 40 compare = less_than } }
		if = {
			limit = { has_dynamic_modifier = { modifier = IVN_march_loyal } }
			remove_dynamic_modifier = { modifier = IVN_march_loyal }
		}
		if = {
			limit = { NOT = { has_dynamic_modifier = { modifier = IVN_march_disloyal } } }
			add_dynamic_modifier = { modifier = IVN_march_disloyal }
		}
	}
	else = {
		if = {
			limit = { has_dynamic_modifier = { modifier = IVN_march_loyal } }
			remove_dynamic_modifier = { modifier = IVN_march_loyal }
		}
		if = {
			limit = { has_dynamic_modifier = { modifier = IVN_march_disloyal } }
			remove_dynamic_modifier = { modifier = IVN_march_disloyal }
		}
	}
}

# ----- Defection of a march during the revolt -----

# STATE, ROOT = the side that wins the march. The legion's local garrison
# deserts with the state; the loser's report names the march.
IVN_march_defect_to_root = {
	save_event_target_as = IVN_defected_march
	ROOT = {
		transfer_state = PREV
		add_manpower = 3000
	}
	set_state_controller_to = ROOT
	set_temp_variable = { var = IVN_trust_delta value = 10 }
	if = {
		limit = { ROOT = { tag = IVC } }
		set_temp_variable = { var = IVN_trust_delta value = -10 }
	}
	IVN_march_trust_change = yes
	if = {
		limit = { ROOT = { tag = IVN } }
		IVC = {
			add_manpower = -3000
			country_event = { id = ADISCORD_ivn.7 }
		}
	}
	else = {
		IVN = {
			add_manpower = -3000
			country_event = { id = ADISCORD_ivn.7 }
		}
	}
	ROOT = { ADISCORD_economy_mark_dirty = yes }
	IVN = { ADISCORD_economy_mark_dirty = yes }
	IVC = { ADISCORD_economy_mark_dirty = yes }
}

# ----- Chauvinist revolt of the marches -----
# A failed northern expedition hands power to the emergency cabinet, and the
# frontier army that came back from the north rises as the Marches Legion
# (IVC). Both outcomes end with IVN annexing IVC, so the Itoran tag, its focus
# progress and western influence survive whichever side wins.

# COUNTRY IVN. The four northern marches of the expedition. Vedrina is the
# legion's capital and always rises; 92, 96 and 97 stay with the cabinet when
# their trust is at least 70. All three border Vedrina, so the legion's land
# stays contiguous whichever of them it keeps.
IVN_marches_revolt_states = {
	95 = { IVN_marches_revolt_take_state = yes }
	92 = { IVN_marches_revolt_take_state = yes }
	96 = { IVN_marches_revolt_take_state = yes }
	97 = { IVN_marches_revolt_take_state = yes }
}

# STATE. Only land Itora still owns and controls passes to the legion.
IVN_marches_revolt_take_state = {
	if = {
		limit = {
			is_owned_by = IVN
			is_controlled_by = IVN
			OR = {
				state = 95
				check_variable = { var = IVN_march_trust value = 70 compare = less_than }
			}
		}
		IVC = { transfer_state = PREV }
		set_state_controller_to = IVC
	}
}

# COUNTRY IVN. The balance of the committee and the staff decides how much of
# the army follows the generals: half under the staff's control, a quarter
# under the committee's, otherwise about a third.
IVN_marches_revolt_transfer_units = {
	if = {
		limit = {
			power_balance_value = {
				id = IVN_command_bop
				value > 0.35
			}
		}
		transfer_units_fraction = {
			target = IVC
			size = 0.5
			stockpile_ratio = 0.35
			navy_ratio = 0
			air_ratio = 0.4
		}
	}
	else_if = {
		limit = {
			power_balance_value = {
				id = IVN_command_bop
				value < -0.35
			}
		}
		transfer_units_fraction = {
			target = IVC
			size = 0.25
			stockpile_ratio = 0.2
			navy_ratio = 0
			air_ratio = 0.2
		}
	}
	else = {
		transfer_units_fraction = {
			target = IVC
			size = 0.35
			stockpile_ratio = 0.3
			navy_ratio = 0
			air_ratio = 0.3
		}
	}
}

# COUNTRY IVN. The emergency cabinet takes power and the legion goes to war.
# Karn's appointment before the expedition gives him his own reserve; a
# general sent into retirement rises with fewer men and leaves the cabinet
# a loyal officer corps.
IVN_start_marches_revolt = {
	set_global_flag = IVN_marches_revolt_active
	IVN_marches_initialize = yes
	IVN_adopt_emergency_order = yes
	IVN_marches_revolt_states = yes
	IVN_marches_revolt_defections = yes
	IVC = {
		set_capital = { state = 95 }
		inherit_technology = IVN
		add_ideas = minister_IVC_Viktor_Laren
		ADISCORD_economy_initialize_country = yes
		promote_character = {
			character = IVN_Rostislav_Karn
			ideology = chauvinism_ideology
		}
		set_stability = 0.45
		set_war_support = 0.70
		add_manpower = 20000
		IVC_legion_initialize = yes
	}
	if = {
		limit = { has_completed_focus = IVN_command_promote_karn }
		IVC = {
			add_manpower = 10000
			army_experience = 25
		}
	}
	else_if = {
		limit = { has_completed_focus = IVN_command_sideline_karn }
		IVC = { add_manpower = -10000 }
		add_timed_idea = {
			idea = IVN_cw_loyal_officers
			days = 180
		}
	}
	# Parliament's answer on the Patriotic Union: an outlawed union has already
	# lost its branches, a tolerated one brings them to the legion.
	if = {
		limit = { has_completed_focus = IVN_parliament_ban_the_union }
		IVC = { add_manpower = -5000 }
	}
	else_if = {
		limit = { has_completed_focus = IVN_parliament_tolerate_the_union }
		IVC = { add_manpower = 5000 }
	}
	IVN_marches_revolt_transfer_units = yes
	add_timed_idea = {
		idea = IVN_cw_divided_army
		days = 180
	}
	IVC = {
		declare_war_on = {
			target = IVN
			type = annex_everything
		}
	}
	mark_focus_tree_layout_dirty = yes
	ADISCORD_economy_mark_dirty = yes
	news_event = { id = ADISCORD_ivn.3 hours = 1 }
	IVN_show_civil_war_superevent = yes
}

# COUNTRY IVN after its own capitulation. The legion's government, leader and
# player move into IVN before the annexation removes IVC. The legion's
# political course survives as the government's spirit; its flag is cleared
# once converted.
IVN_hand_power_to_legion = {
	if = {
		limit = { IVC = { is_ai = no } }
		change_tag_from = IVC
	}
	IVC = {
		if = {
			limit = { has_character = IVN_Rostislav_Karn }
			set_nationality = {
				character = IVN_Rostislav_Karn
				target_country = IVN
			}
		}
	}
	set_politics = {
		ruling_party = chauvinism
		elections_allowed = no
	}
	promote_character = {
		character = IVN_Rostislav_Karn
		ideology = chauvinism_ideology
	}
	IVN_party_representatives_legion = yes
	if = {
		limit = { has_global_flag = IVN_legion_course_council }
		add_ideas = IVN_rule_of_the_legion_council
	}
	else_if = {
		limit = { has_global_flag = IVN_legion_course_union }
		add_ideas = IVN_rule_of_the_patriotic_union
	}
	clr_global_flag = IVN_legion_course_council
	clr_global_flag = IVN_legion_course_union
}

# COUNTRY IVN. Every terminal path reunites the marches under IVN. Timed
# measures still running on a frontier march were paid by a side of the
# revolt; their flags close with it.
IVN_resolve_marches_revolt = {
	if = {
		limit = { has_war_with = IVC }
		white_peace = IVC
	}
	if = {
		limit = { country_exists = IVC }
		annex_country = {
			target = IVC
			transfer_troops = yes
		}
	}
	clr_global_flag = IVN_marches_revolt_active
	set_global_flag = IVN_marches_revolt_resolved
	if = {
		limit = { has_government = chauvinism }
		set_global_flag = IVN_civil_war_won_by_legion
	}
	else = {
		set_global_flag = IVN_civil_war_won_by_cabinet
	}
	clr_global_flag = IVN_legion_course_council
	clr_global_flag = IVN_legion_course_union
	if = {
		limit = { has_idea = IVN_cw_divided_army }
		remove_ideas = IVN_cw_divided_army
	}
	if = {
		limit = { has_idea = IVN_cw_loyal_officers }
		remove_ideas = IVN_cw_loyal_officers
	}
	if = {
		limit = { has_idea = IVN_cw_war_bonds }
		remove_ideas = IVN_cw_war_bonds
	}
	mark_focus_tree_layout_dirty = yes
	ADISCORD_economy_mark_dirty = yes
	if = {
		limit = { has_government = chauvinism }
		news_event = { id = ADISCORD_ivn.5 hours = 2 }
	}
	else = {
		news_event = { id = ADISCORD_ivn.4 hours = 2 }
	}
}
