# COUNTRY IVN. Refreshes the focus aggregate after a variable change.
IVN_refresh_order = {
	if = {
		limit = {
			tag = IVN
			has_global_flag = IVN_civil_war_won_by_legion
		}
		IVN_refresh_imperium = yes
	}
	else = {
		if = {
			limit = { NOT = { has_dynamic_modifier = { modifier = IVN_order_dynamic } } }
			add_dynamic_modifier = { modifier = IVN_order_dynamic }
		}
	}
	force_update_dynamic_modifier = yes
	ADISCORD_economy_mark_dirty = yes
}

# COUNTRY IVN. The committee suspends elections. A retreat from the north may
# already have installed the same etatist role, so the role is added once.
IVN_adopt_emergency_order = {
	if = {
		limit = {
			NOT = {
				OR = {
					has_global_flag = IVN_civil_war_won_by_directory
					has_global_flag = IVN_civil_war_won_by_communes
					has_global_flag = IVN_civil_war_won_by_marches
				}
			}
		}
		if = {
			limit = { NOT = { has_government = etatism } }
			set_politics = {
				ruling_party = etatism
				elections_allowed = no
			}
		}
		if = {
			limit = { NOT = { has_country_flag = ADISCORD_vorkerland_vadim_etatist_role_added } }
			add_country_leader_role = {
				character = IVN_Vadim_Ivanchik
				promote_leader = yes
				country_leader = {
					ideology = etatism_ideology
					expire = "2200.1.1.1"
				}
			}
			set_country_flag = ADISCORD_vorkerland_vadim_etatist_role_added
		}
		IVN_party_representatives_emergency = yes
	}
}

# ----- Committee and staff, frontier trust -----
# IVN_command_bop measures who commands the army of the marches: the civil
# Committee (negative side) or the Staff of the marches (positive side). The
# state variable IVN_march_trust (0-100) on the eight frontier marches records
# how far each march still trusts the capital. The revolt reads both: the
# balance sets the Directorate's army share, while trusted marches
# support local mobilisation.

# COUNTRY IVN. One-time opening from the root focus. The revolt calls it as
# well, so its consumers exist even if the root focus was never completed.
IVN_marches_initialize = {
	if = {
		limit = { NOT = { has_country_flag = IVN_marches_initialized } }
		set_country_flag = IVN_marches_initialized
		set_power_balance = {
			id = IVN_command_bop
			set_value = 0.10
		}
		92 = { set_variable = { var = IVN_march_trust value = 45 } }
		95 = { set_variable = { var = IVN_march_trust value = 35 } }
		96 = { set_variable = { var = IVN_march_trust value = 45 } }
		97 = { set_variable = { var = IVN_march_trust value = 50 } }
		98 = { set_variable = { var = IVN_march_trust value = 60 } }
		99 = { set_variable = { var = IVN_march_trust value = 60 } }
		100 = { set_variable = { var = IVN_march_trust value = 65 } }
		101 = { set_variable = { var = IVN_march_trust value = 70 } }
		IVN_frontier_marches_refresh = yes
	}
}

# STATE. Adds the temporary IVN_trust_delta to the march's trust.
IVN_march_trust_change = {
	add_to_variable = { var = IVN_march_trust value = IVN_trust_delta }
	clamp_variable = {
		var = IVN_march_trust
		min = 0
		max = 100
	}
	IVN_march_trust_refresh = yes
}

# STATE. Swaps the visible modifier only when trust crosses 70 or 40.
IVN_march_trust_refresh = {
	if = {
		limit = { check_variable = { var = IVN_march_trust value = 70 compare = greater_than_or_equals } }
		if = {
			limit = { has_dynamic_modifier = { modifier = IVN_march_disloyal } }
			remove_dynamic_modifier = { modifier = IVN_march_disloyal }
		}
		if = {
			limit = { NOT = { has_dynamic_modifier = { modifier = IVN_march_loyal } } }
			add_dynamic_modifier = { modifier = IVN_march_loyal }
		}
	}
	else_if = {
		limit = { check_variable = { var = IVN_march_trust value = 40 compare = less_than } }
		if = {
			limit = { has_dynamic_modifier = { modifier = IVN_march_loyal } }
			remove_dynamic_modifier = { modifier = IVN_march_loyal }
		}
		if = {
			limit = { NOT = { has_dynamic_modifier = { modifier = IVN_march_disloyal } } }
			add_dynamic_modifier = { modifier = IVN_march_disloyal }
		}
	}
	else = {
		if = {
			limit = { has_dynamic_modifier = { modifier = IVN_march_loyal } }
			remove_dynamic_modifier = { modifier = IVN_march_loyal }
		}
		if = {
			limit = { has_dynamic_modifier = { modifier = IVN_march_disloyal } }
			remove_dynamic_modifier = { modifier = IVN_march_disloyal }
		}
	}
}

# deserts with the state; the loser's report names the march.

		transfer_state = PREV
		add_manpower = 3000
	}
	set_state_controller_to = ROOT
	set_temp_variable = { var = IVN_trust_delta value = 10 }

		limit = { ROOT = { tag = IVC } }
		set_temp_variable = { var = IVN_trust_delta value = -10 }
	}
	IVN_march_trust_change = yes
	if = {
		limit = { ROOT = { tag = IVN } }
		IVC = {
			add_manpower = -3000
			country_event = { id = ADISCORD_ivn.7 }
		}
	}
	else = {
		IVN = {
			add_manpower = -3000
			country_event = { id = ADISCORD_ivn.7 }
		}
	}
	ROOT = { ADISCORD_economy_mark_dirty = yes }
	IVN = { ADISCORD_economy_mark_dirty = yes }
	IVC = { ADISCORD_economy_mark_dirty = yes }
}

# COUNTRY donor. Recipient is a local event target set by the caller. Refund
# unfinished work before moving its pools; zero them to prevent a revived tag
# receiving the same resources for a second time.
IVN_cw_transfer_reserves = {
	IVN_cw_close_programmes = yes
	set_variable = { var = IVN_cw_transfer_manpower value = manpower }
	set_variable = { var = IVN_cw_transfer_power value = political_power }
	event_target:IVN_cw_recipient = {
		add_manpower = var:PREV.IVN_cw_transfer_manpower
		add_political_power = var:PREV.IVN_cw_transfer_power
		add_to_variable = { var = ADISCORD_economy_treasury value = PREV.ADISCORD_economy_treasury }
		add_to_variable = { var = ADISCORD_economy_debt value = PREV.ADISCORD_economy_debt }
	}
	multiply_variable = { var = IVN_cw_transfer_manpower value = -1 }
	multiply_variable = { var = IVN_cw_transfer_power value = -1 }
	add_manpower = var:IVN_cw_transfer_manpower
	add_political_power = var:IVN_cw_transfer_power
	clear_variable = IVN_cw_transfer_manpower
	clear_variable = IVN_cw_transfer_power
	set_variable = { var = ADISCORD_economy_treasury value = 0 }
	set_variable = { var = ADISCORD_economy_debt value = 0 }
	transfer_units_fraction = {
		target = event_target:IVN_cw_recipient
		size = 0
		stockpile_ratio = 1
		navy_ratio = 0
		air_ratio = 0
	}
	if = {
		limit = { has_idea = IVN_cw_military_guarantees }
		event_target:IVN_cw_recipient = { add_ideas = IVN_cw_military_guarantees }
		remove_ideas = IVN_cw_military_guarantees
	}
	if = {
		limit = { has_idea = IVN_cw_communal_guarantees }
		event_target:IVN_cw_recipient = { add_ideas = IVN_cw_communal_guarantees }
		remove_ideas = IVN_cw_communal_guarantees
	}
	if = {
		limit = { has_idea = IVN_cw_regional_guarantees }
		event_target:IVN_cw_recipient = { add_ideas = IVN_cw_regional_guarantees }
		remove_ideas = IVN_cw_regional_guarantees
	}
	# The existing island administration remains a subject through government changes.
	if = {
		limit = { IIA = { is_subject_of = PREV } }
		event_target:IVN_cw_recipient = {
			set_autonomy = { target = IIA autonomy_state = autonomy_island_administration freedom_level = 0 }
		}
	}
	ADISCORD_economy_mark_dirty = yes
	event_target:IVN_cw_recipient = { ADISCORD_economy_mark_dirty = yes }
}

	add_to_variable = { var = IVN_imp_training value = IVC_leg_training }

	add_to_variable = { var = IVN_imp_stability value = IVC_leg_stability }

	add_to_variable = { var = IVN_imp_political_power value = IVC_leg_political_power }
	set_variable = { var = IVN_imp_research value = IVN_ord_research }
	set_variable = { var = IVN_imp_factory_output value = IVN_ord_factory_output }
	add_to_variable = { var = IVN_imp_factory_output value = IVC_leg_factory_output }
	set_variable = { var = IVN_imp_construction value = IVN_ord_construction }
	add_to_variable = { var = IVN_imp_construction value = IVC_leg_construction }
	set_variable = { var = IVN_imp_consumer_goods value = IVN_ord_consumer_goods }

		limit = { NOT = { has_dynamic_modifier = { modifier = IVN_exercitus_imperialis } } }

		add_dynamic_modifier = { modifier = IVN_oeconomia_solis_nigri }
	}

	ADISCORD_economy_mark_dirty = yes
}
