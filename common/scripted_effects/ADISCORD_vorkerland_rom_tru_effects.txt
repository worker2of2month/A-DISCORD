ADISCORD_vorkerland_rom_initialize_war_content = {
	if = {
		limit = { NOT = { has_country_flag = ADISCORD_vorkerland_rom_war_content_initialized } }
		set_country_flag = ADISCORD_vorkerland_rom_war_content_initialized
		if = {
			limit = { NOT = { has_idea = ADISCORD_vorkerland_rom_desert_eagle_staff } }
			add_ideas = ADISCORD_vorkerland_rom_desert_eagle_staff
		}
		set_popularities = {
			etatism = 45
			pragmatism = 30
			humanism = 15
			chauvinism = 5
			utilitarism = 5
		}
		# The shared collapse setup already grants 6,000 men, 1,200 rifles and
		# 100 support sets. These finite deltas bring ROM to the approved total.
		add_manpower = 3000
		add_equipment_to_stockpile = { type = infantry_equipment_0 amount = 400 producer = ROM }
		add_equipment_to_stockpile = { type = support_equipment_1 amount = 20 producer = ROM }
		add_equipment_to_stockpile = { type = artillery_equipment_1 amount = 48 producer = ROM }
	}
}
