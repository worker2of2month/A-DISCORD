ADISCORD_vorkerland_apply_dirty_state_modifier = {
    if = {
        limit = {
            NOT = { has_dynamic_modifier = { modifier = ADISCORD_vorkerland_dirty_state } }
        }
        add_dynamic_modifier = { modifier = ADISCORD_vorkerland_dirty_state }
	}
}

ADISCORD_vorkerland_setup_sla = {
	log = "[ADISCORD Vorkerland] SLA ownerless-state setup started"
	# Dirty-zone states are ownerless before opening. transfer_state assumes a
	# previous country owner and crashes the 1.19.2 relation cache on null.
	49 = { add_core_of = SLA set_state_owner_to = SLA }
	51 = { add_core_of = SLA set_state_owner_to = SLA }
	155 = { add_core_of = SLA set_state_owner_to = SLA }
	176 = { add_core_of = SLA set_state_owner_to = SLA }
	187 = { add_core_of = SLA set_state_owner_to = SLA }
	191 = { add_core_of = SLA set_state_owner_to = SLA }
	log = "[ADISCORD Vorkerland] SLA state ownership assigned"
	SLA = {
		set_capital = { state = 49 }
		ADISCORD_grant_2150_technology_baseline = yes
		ADISCORD_economy_initialize_country = yes
		ADISCORD_vorkerland_prepare_conflict_country = yes
		load_oob = "SLA_vorkerland_collapse"
	}
	log = "[ADISCORD Vorkerland] SLA setup completed"
}
