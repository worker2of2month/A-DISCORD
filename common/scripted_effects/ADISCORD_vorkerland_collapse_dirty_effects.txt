			country_exists = SLA
			country_exists = MLR
			country_exists = RZA
			country_exists = SCA
			country_exists = ERT
			country_exists = IRT
		}

		clr_global_flag = ADISCORD_vorkerland_dirty_cascade_recovery_scheduled

}

ADISCORD_vorkerland_start_khan_border_war = {
	if = {
		limit = {
			country_exists = RUS
			country_exists = SLA
			NOT = { has_global_flag = ADISCORD_vorkerland_khan_border_war_started }
			NOT = { has_global_flag = ADISCORD_vorkerland_khan_border_war_resolved }
			RUS = { NOT = { has_war_with = SLA } }
		}
		set_global_flag = ADISCORD_vorkerland_khan_border_war_started
		49 = { add_claim_by = RUS }
		176 = { add_claim_by = RUS }
		RUS = {
			add_equipment_to_stockpile = { type = ADISCORD_squad_weapons_equipment_0 amount = 350 producer = RUS }
			add_equipment_to_stockpile = { type = support_equipment_1 amount = 100 producer = RUS }
			declare_war_on = { target = SLA type = annex_everything }
		}
	}
}

ADISCORD_vorkerland_check_khan_border_war = {
	if = {
		limit = {
			has_global_flag = ADISCORD_vorkerland_khan_border_war_started
			country_exists = SLA
			has_war_with = SLA
			controls_state = 49
			controls_state = 176
		}
		clr_global_flag = ADISCORD_vorkerland_khan_border_war_started
		set_global_flag = ADISCORD_vorkerland_khan_border_war_resolved
		white_peace = SLA
		49 = { set_state_owner_to = RUS set_state_controller_to = RUS }
		176 = { set_state_owner_to = RUS set_state_controller_to = RUS }
		SLA = { set_capital = { state = 51 } }
	}
}
