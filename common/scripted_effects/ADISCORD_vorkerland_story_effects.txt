# Compact story layer for the background Vorkerland civil war. These effects
# may change only narrative flags and small country resources. They do not own
# wars, peace, state transfer, annexation, subjects, or phase progression.

ADISCORD_vorkerland_story_announce_verified_showdown = {
	if = {
		limit = {
			has_global_flag = ADISCORD_vorkerland_central_showdown_started
			ADISCORD_vorkerland_central_showdown_edges_verified = yes
			ADISCORD_vorkerland_central_showdown_required = yes
			NOT = { has_global_flag = ADISCORD_vorkerland_central_showdown_launch_failed }
			NOT = { has_global_flag = ADISCORD_vorkerland_story_showdown_announced }
		}
		set_global_flag = ADISCORD_vorkerland_story_showdown_announced
		news_event = { id = ADISCORD_vorkerland_story.1 }
	}
}

# The collapse roll decides what happened to Nikita Worker and records it in one
# of four global flags. That roll lives in the collapse layer, which owns no
# presentation, so the world would otherwise never learn the single most
# consequential fact of the whole scenario. Call this once from the collapse
# opening, after the fate flags are set.
#
# The guard is released again when no fate flag is present so that a call placed
# too early cannot permanently silence the report.
ADISCORD_vorkerland_story_report_worker_fate = {
	if = {
		limit = { NOT = { has_global_flag = ADISCORD_vorkerland_story_worker_fate_reported } }
		set_global_flag = ADISCORD_vorkerland_story_worker_fate_reported
		if = {
			limit = { has_global_flag = ADISCORD_vorkerland_worker_safe_with_loyalists }
			set_global_flag = ADISCORD_vorkerland_story_worker_survival_reported
			news_event = { id = ADISCORD_vorkerland_story.10 }
		}
		else_if = {
			limit = { has_global_flag = ADISCORD_vorkerland_worker_rescued_by_vlad }
			set_global_flag = ADISCORD_vorkerland_story_worker_rescue_reported
			news_event = { id = ADISCORD_vorkerland_story.11 }
		}
		else_if = {
			limit = { has_global_flag = ADISCORD_vorkerland_worker_missing }
			set_global_flag = ADISCORD_vorkerland_story_worker_disappearance_reported
			news_event = { id = ADISCORD_vorkerland_story.12 }
		}
		else_if = {
			limit = { has_global_flag = ADISCORD_vorkerland_worker_killed }
			set_global_flag = ADISCORD_vorkerland_story_worker_death_reported
			news_event = { id = ADISCORD_vorkerland_story.13 }
		}
		else = {
			clr_global_flag = ADISCORD_vorkerland_story_worker_fate_reported
		}
	}
}

# Presentation for the twelve iconic objectives. The phase layer resolves the
# capture and writes global.ADISCORD_vorkerland_objective_last; this reads that
# id, records it so the same centre can never be reported twice, and fires the
# single objective news event.
#
# Called from ADISCORD_vorkerland_story_check_first_claimant_capital_fall, which
# on_state_control_changed already invokes after the twelve objective checks. It
# therefore observes the completed result of the whole state change rather than
# a half-applied one, and one control change can never queue several objective
# reports that would all render with the last id.
ADISCORD_vorkerland_story_report_iconic_objective = {
	if = {
		limit = {
			check_variable = { var = global.ADISCORD_vorkerland_objective_last value = 1 compare = greater_than_or_equals }
			check_variable = { var = global.ADISCORD_vorkerland_objective_last value = 12 compare = less_than_or_equals }
			NOT = {
				is_in_array = {
					array = global.ADISCORD_vorkerland_story_objectives_reported
					value = global.ADISCORD_vorkerland_objective_last
				}
			}
			# The three claimant capitals already carry a dedicated first-fall
			# report. Without these the same city would be announced twice on the
			# same day by two different events.
			NOT = {
				AND = {
					check_variable = { var = global.ADISCORD_vorkerland_objective_last value = 1 compare = equals }
					has_global_flag = ADISCORD_vorkerland_story_wkr_capital_fell_first
				}
			}
			NOT = {
				AND = {
					check_variable = { var = global.ADISCORD_vorkerland_objective_last value = 2 compare = equals }
					has_global_flag = ADISCORD_vorkerland_story_vad_capital_fell_first
				}
			}
			NOT = {
				AND = {
					check_variable = { var = global.ADISCORD_vorkerland_objective_last value = 3 compare = equals }
					has_global_flag = ADISCORD_vorkerland_story_tva_capital_fell_first
				}
			}
		}
		# A global array is the ordinary array argument with a global. prefix.
		# The script API defines no separate global-array effect or trigger, and
		# a mistyped one is accepted by the parser and then does nothing.
		add_to_array = {
			array = global.ADISCORD_vorkerland_story_objectives_reported
			value = global.ADISCORD_vorkerland_objective_last
		}
		if = {
			limit = { check_variable = { var = global.ADISCORD_vorkerland_objective_last value = 1 compare = equals } }
			news_event = { id = ADISCORD_vorkerland_story.50 }
		}
		else_if = {
			limit = { check_variable = { var = global.ADISCORD_vorkerland_objective_last value = 2 compare = equals } }
			news_event = { id = ADISCORD_vorkerland_story.51 }
		}
		else_if = {
			limit = { check_variable = { var = global.ADISCORD_vorkerland_objective_last value = 3 compare = equals } }
			news_event = { id = ADISCORD_vorkerland_story.52 }
		}
		else_if = {
			limit = { check_variable = { var = global.ADISCORD_vorkerland_objective_last value = 4 compare = equals } }
			news_event = { id = ADISCORD_vorkerland_story.53 }
		}
		else_if = {
			limit = { check_variable = { var = global.ADISCORD_vorkerland_objective_last value = 5 compare = equals } }
			news_event = { id = ADISCORD_vorkerland_story.54 }
		}
		else_if = {
			limit = { check_variable = { var = global.ADISCORD_vorkerland_objective_last value = 6 compare = equals } }
			news_event = { id = ADISCORD_vorkerland_story.55 }
		}
		else_if = {
			limit = { check_variable = { var = global.ADISCORD_vorkerland_objective_last value = 7 compare = equals } }
			news_event = { id = ADISCORD_vorkerland_story.56 }
		}
		else_if = {
			limit = { check_variable = { var = global.ADISCORD_vorkerland_objective_last value = 8 compare = equals } }
			news_event = { id = ADISCORD_vorkerland_story.57 }
		}
		else_if = {
			limit = { check_variable = { var = global.ADISCORD_vorkerland_objective_last value = 9 compare = equals } }
			news_event = { id = ADISCORD_vorkerland_story.58 }
		}
		else_if = {
			limit = { check_variable = { var = global.ADISCORD_vorkerland_objective_last value = 10 compare = equals } }
			news_event = { id = ADISCORD_vorkerland_story.59 }
		}
		else_if = {
			limit = { check_variable = { var = global.ADISCORD_vorkerland_objective_last value = 11 compare = equals } }
			news_event = { id = ADISCORD_vorkerland_story.60 }
		}
		else_if = {
			limit = { check_variable = { var = global.ADISCORD_vorkerland_objective_last value = 12 compare = equals } }
			news_event = { id = ADISCORD_vorkerland_story.61 }
		}
	}
}

# ROOT is the new controller and FROM is the old controller in
# on_state_control_changed. Requiring the changed controller to hold the exact
# capital prevents an unrelated state change from naming the wrong victor.
ADISCORD_vorkerland_story_check_first_claimant_capital_fall = {
	if = {
		limit = {
			has_global_flag = ADISCORD_vorkerland_central_showdown_started
			ADISCORD_vorkerland_central_showdown_edges_verified = yes
			NOT = { has_global_flag = ADISCORD_vorkerland_central_showdown_launch_failed }
			NOT = { has_global_flag = ADISCORD_vorkerland_story_first_claimant_capital_fell }
			FROM = { tag = WKR }
			32 = { is_controlled_by = ROOT }
			NOT = { tag = WKR }
		}
		set_global_flag = ADISCORD_vorkerland_story_first_claimant_capital_fell
		set_global_flag = ADISCORD_vorkerland_story_wkr_capital_fell_first
		WKR = {
			if = {
				limit = { exists = yes }
				add_war_support = -0.02
				add_command_power = -5
			}
		}
		ADISCORD_vorkerland_story_dispatch_wkr_capital_fall = yes
	}
	else_if = {
		limit = {
			has_global_flag = ADISCORD_vorkerland_central_showdown_started
			ADISCORD_vorkerland_central_showdown_edges_verified = yes
			NOT = { has_global_flag = ADISCORD_vorkerland_central_showdown_launch_failed }
			NOT = { has_global_flag = ADISCORD_vorkerland_story_first_claimant_capital_fell }
			FROM = { tag = VAD }
			75 = { is_controlled_by = ROOT }
			NOT = { tag = VAD }
		}
		set_global_flag = ADISCORD_vorkerland_story_first_claimant_capital_fell
		set_global_flag = ADISCORD_vorkerland_story_vad_capital_fell_first
		VAD = {
			if = {
				limit = { exists = yes }
				add_war_support = -0.02
				add_command_power = -5
			}
		}
		ADISCORD_vorkerland_story_dispatch_vad_capital_fall = yes
	}
	else_if = {
		limit = {
			has_global_flag = ADISCORD_vorkerland_central_showdown_started
			ADISCORD_vorkerland_central_showdown_edges_verified = yes
			NOT = { has_global_flag = ADISCORD_vorkerland_central_showdown_launch_failed }
			NOT = { has_global_flag = ADISCORD_vorkerland_story_first_claimant_capital_fell }
			FROM = { tag = TVA }
			36 = { is_controlled_by = ROOT }
			NOT = { tag = TVA }
		}
		set_global_flag = ADISCORD_vorkerland_story_first_claimant_capital_fell
		set_global_flag = ADISCORD_vorkerland_story_tva_capital_fell_first
		TVA = {
			if = {
				limit = { exists = yes }
				add_war_support = -0.02
				add_command_power = -5
			}
		}
		ADISCORD_vorkerland_story_dispatch_tva_capital_fall = yes
	}
	# Runs after the capital chain above so the iconic-objective report can see
	# which capital, if any, has just been announced separately. This effect is
	# the story layer's single on_state_control_changed entry point, so the
	# objective news needs no additional hook outside the story files.
	ADISCORD_vorkerland_story_report_iconic_objective = yes
}

ADISCORD_vorkerland_story_offer_first_claimant_command_choice = {
	# on_capitulation calls this inside FROM, the winner, so the report has to be
	# scoped back to ROOT. Otherwise the news would name the side that won as the
	# side that fell.
	ROOT = { ADISCORD_vorkerland_story_report_claimant_capitulation = yes }
	if = {
		limit = {
			has_global_flag = ADISCORD_vorkerland_central_showdown_started
			NOT = { has_global_flag = ADISCORD_vorkerland_central_war_finished }
			NOT = { has_global_flag = ADISCORD_vorkerland_reunification_verified }
			NOT = { has_global_flag = ADISCORD_vorkerland_story_first_claimant_capitulated }
			NOT = { tag = WRK }
			ROOT = { ADISCORD_vorkerland_is_main_claimant = yes }
		}
		set_global_flag = ADISCORD_vorkerland_story_first_claimant_capitulated
		if = {
			limit = { tag = WTD TVA = { exists = yes } }
			TVA = { country_event = { id = ADISCORD_vorkerland_story.5 } }
		}
		else_if = {
			limit = { tag = VLA WKR = { exists = yes } }
			WKR = { country_event = { id = ADISCORD_vorkerland_story.5 } }
		}
		else_if = {
			limit = { tag = SOL VAD = { exists = yes } }
			VAD = { country_event = { id = ADISCORD_vorkerland_story.5 } }
		}
		else_if = {
			limit = { is_subject = yes }
			OVERLORD = { country_event = { id = ADISCORD_vorkerland_story.5 } }
		}
		else = {
			country_event = { id = ADISCORD_vorkerland_story.5 }
		}
	}
}
