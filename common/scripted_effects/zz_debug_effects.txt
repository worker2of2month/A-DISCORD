d_collaboration = {
	if = {
		limit = { tag = FROM }
		log = "Error:"
		log = "  right click a country to select it for adding collaboration"
		log = "  you can specify collaboration amount to add or just use 'collaboration' to add 10%"
	}
	else = {
		FROM = {
			add_collaboration = {
				target = PREV
				value = args^0?0.1
			}
			log = "Added [?args^0?0.1|H%] collaboration for [PREV.GetName]"
		}
	}
}

d_retire_country_leader = {
	random_character = {
		limit = { is_country_leader = yes }
		log = "Character [THIS.GetName] retired from [ROOT.GetNameDef]"
	}
	retire_country_leader = yes
}

d_clear_tech = {
	set_technology = { ADISCORD_tech_postwar_weapon_standardization = 0 popup = no }
	log = "tech reset: ADISCORD_tech_postwar_weapon_standardization"
}

d_add_tech = {
	set_technology = { ADISCORD_tech_postwar_weapon_standardization = 1 popup = no }
	log = "tech researched: ADISCORD_tech_postwar_weapon_standardization"
}

# COUNTRY: invoked only by a human's debug decision. Never placed on a pulse.
ADISCORD_debug_war_snapshot = {
    if = {
        limit = { is_debug = yes is_ai = no }
        log = "[ADISCORD_PEACE] SNAPSHOT BEGIN country=[THIS.GetTag]"
        ADISCORD_debug_war_country_dump = yes
        every_enemy_country = {
            log = "[ADISCORD_PEACE] ENEMY of [ROOT.GetTag]"
            ADISCORD_debug_war_country_dump = yes
        }
        VAL = { log = "[ADISCORD_PEACE] VAL stage=[?VAL_frontier_stage] target=[?VAL_frontier_target]" }
        168 = { ADISCORD_debug_war_state_dump = yes }
        461 = { ADISCORD_debug_war_state_dump = yes }
        log = "[ADISCORD_PEACE] SNAPSHOT END"
    }
}

ADISCORD_debug_war_country_dump = {
    log = "[ADISCORD_PEACE] COUNTRY [THIS.GetTag] owned_states=[?num_owned_states] surrender=[?surrender_progress]"
    if = { limit = { has_capitulated = yes } log = "[ADISCORD_PEACE] capitulated=yes" }
    else = { log = "[ADISCORD_PEACE] capitulated=no" }
    if = { limit = { is_major = yes } log = "[ADISCORD_PEACE] major=yes" }
    if = { limit = { has_country_flag = VAL_frontier_member } log = "[ADISCORD_PEACE] frontier_member=yes" }
    if = { limit = { has_country_flag = VAL_frontier_guarantor } log = "[ADISCORD_PEACE] frontier_guarantor=yes" }
    if = { limit = { has_country_flag = VAL_frontier_defeated } log = "[ADISCORD_PEACE] frontier_defeated_receipt=yes" }
    if = { limit = { is_in_faction = yes } faction_leader = { log = "[ADISCORD_PEACE] faction_leader=[THIS.GetTag]" } }
    capital_scope = { log = "[ADISCORD_PEACE] capital_state=[THIS.GetID]" }
    every_owned_state = { ADISCORD_debug_war_state_dump = yes }
}

# STATE: separate native scopes keep owner and controller readable even in partitioned wars.
ADISCORD_debug_war_state_dump = {
    log = "[ADISCORD_PEACE] STATE [THIS.GetID]"
    owner = { log = "[ADISCORD_PEACE] owner=[THIS.GetTag]" }
    controller = { log = "[ADISCORD_PEACE] controller=[THIS.GetTag]" }
}
