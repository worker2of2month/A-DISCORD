# Kefreyt logistics, displacement and grey-market runtime.
# All recurring work is country-scoped to VAL and uses a fixed set of states.

# COUNTRY VAL. Retention applies after ordinary income modifiers and before payment.
VAL_black_market_calculate_retention = {
    set_temp_variable = { var = VAL_black_market_loss_rate_temp value = 0 }
    if = {
        limit = { check_variable = { var = VAL_black_market_pressure value = 75 compare = greater_than_or_equals } }
        set_temp_variable = { var = VAL_black_market_loss_rate_temp value = 0.50 }
    }
    else_if = {
        limit = { check_variable = { var = VAL_black_market_pressure value = 50 compare = greater_than_or_equals } }
        set_temp_variable = { var = VAL_black_market_loss_rate_temp value = 0.25 }
    }
    else_if = {
        limit = { check_variable = { var = VAL_black_market_pressure value = 25 compare = greater_than_or_equals } }
        set_temp_variable = { var = VAL_black_market_loss_rate_temp value = 0.10 }
    }
    set_temp_variable = { var = VAL_black_market_retention_temp value = 1 }
    subtract_from_temp_variable = { var = VAL_black_market_retention_temp value = VAL_black_market_loss_rate_temp }
}

# These forecast hooks receive fresh gross values; they never debit cash or statistics.
VAL_black_market_reduce_monthly_income = {
    VAL_black_market_calculate_retention = yes
    multiply_variable = { var = ADISCORD_economy_monthly_income value = VAL_black_market_retention_temp }
}

VAL_black_market_reduce_weekly_bonus = {
    VAL_black_market_calculate_retention = yes
    set_variable = { var = VAL_black_market_bonus_loss value = 0 }
    if = {
        limit = { check_variable = { var = ADISCORD_economy_final_weekly_income_bonus value = 0 compare = greater_than } }
        set_variable = { var = VAL_black_market_bonus_loss value = ADISCORD_economy_final_weekly_income_bonus }
        multiply_variable = { var = VAL_black_market_bonus_loss value = VAL_black_market_loss_rate_temp }
        subtract_from_variable = { var = ADISCORD_economy_final_weekly_income_bonus value = VAL_black_market_bonus_loss }
    }
}

# Called only by the native weekly budget settlement. Corridor receipts accumulate
# between settlements, independent of the ordering of the two weekly on_actions.
VAL_black_market_record_budget_loss = {
    VAL_black_market_calculate_retention = yes
    set_temp_variable = { var = VAL_black_market_budget_loss_temp value = ADISCORD_economy_monthly_income }
    divide_temp_variable = { var = VAL_black_market_budget_loss_temp value = VAL_black_market_retention_temp }
    multiply_temp_variable = { var = VAL_black_market_budget_loss_temp value = VAL_black_market_loss_rate_temp }
    multiply_temp_variable = { var = VAL_black_market_budget_loss_temp value = 3 }
    divide_temp_variable = { var = VAL_black_market_budget_loss_temp value = 13 }
    add_to_temp_variable = { var = VAL_black_market_budget_loss_temp value = VAL_black_market_bonus_loss }
    set_variable = { var = VAL_black_market_loss_last_week value = VAL_black_market_budget_loss_temp }
    add_to_variable = { var = VAL_black_market_loss_last_week value = VAL_black_market_route_loss_pending }
    set_variable = { var = VAL_black_market_route_loss_pending value = 0 }
    add_to_variable = { var = VAL_black_market_loss_total value = VAL_black_market_budget_loss_temp }
}

			limit = { has_country_flag = VAL_route_north_commissioned }
			set_country_flag = VAL_route_west_commissioned
			clr_country_flag = VAL_route_north_commissioned
		}
		if = {
			limit = { has_country_flag = VAL_route_north_upgraded }
			set_country_flag = VAL_route_west_upgraded
			clr_country_flag = VAL_route_north_upgraded
		}
		if = {
			limit = { has_variable = VAL_route_north_active }
			set_variable = { var = VAL_route_west_active value = VAL_route_north_active }
			set_variable = { var = VAL_route_north_active value = 0 }
		}
		set_country_flag = VAL_trade_route_west_migration_v1
	}
	ADISCORD_economy_mark_dirty = yes
	if = { limit = { NOT = { has_variable = VAL_black_market_loss_last_week } } set_variable = { var = VAL_black_market_loss_last_week value = 0 } }
	if = { limit = { NOT = { has_variable = VAL_black_market_loss_total } } set_variable = { var = VAL_black_market_loss_total value = 0 } }
	if = { limit = { NOT = { has_variable = VAL_black_market_route_loss_pending } } set_variable = { var = VAL_black_market_route_loss_pending value = 0 } }
	if = {
		limit = { NOT = { has_variable = VAL_trade_corridors_active } }
		set_variable = { var = VAL_trade_corridors_active value = 0 }
	}
	if = {
		limit = { NOT = { has_variable = VAL_trade_corridors_capacity } }
		set_variable = { var = VAL_trade_corridors_capacity value = 0 }
	}
	if = {
		limit = { NOT = { has_variable = VAL_route_occidia_active } }
		set_variable = { var = VAL_route_occidia_active value = 0 }
	}
	if = {
		limit = { NOT = { has_variable = VAL_route_west_active } }
		set_variable = { var = VAL_route_west_active value = 0 }
	}
	if = {
		limit = { NOT = { has_variable = VAL_route_north_active } }
		set_variable = { var = VAL_route_north_active value = 0 }
	}
	if = {
		limit = { NOT = { has_variable = VAL_route_stelander_active } }
		set_variable = { var = VAL_route_stelander_active value = 0 }
	}
	if = {
		limit = { NOT = { has_variable = VAL_route_vorkerland_active } }
		set_variable = { var = VAL_route_vorkerland_active value = 0 }
	}
	if = {
		limit = { NOT = { has_variable = VAL_route_south_active } }
		set_variable = { var = VAL_route_south_active value = 0 }
	}
	if = {
		limit = { NOT = { has_variable = VAL_corridor_security } }
		set_variable = { var = VAL_corridor_security value = 0 }
	}
	if = {
		limit = { NOT = { has_variable = VAL_black_market_pressure } }
		set_variable = { var = VAL_black_market_pressure value = 8 }
	}
	if = {
		limit = { NOT = { has_variable = VAL_displaced_population } }
		set_variable = { var = VAL_displaced_population value = 0 }
	}
	clamp_variable = { var = VAL_corridor_security min = 0 max = 3 }
	clamp_variable = { var = VAL_black_market_pressure min = 0 max = 100 }
	clamp_variable = { var = VAL_displaced_population min = 0 max = 100 }
	VAL_refresh_trade_network = yes
	VAL_refresh_black_market_state = yes
	if = {
		limit = { NOT = { has_variable = VAL_refugee_housing } }
		set_variable = { var = VAL_refugee_housing value = 20 }
		set_variable = { var = VAL_local_volunteer_pool value = 0 }
		add_to_variable = { var = VAL_local_volunteer_pool value = 24.state_population_k }
		add_to_variable = { var = VAL_local_volunteer_pool value = 42.state_population_k }
		add_to_variable = { var = VAL_local_volunteer_pool value = 48.state_population_k }
		add_to_variable = { var = VAL_local_volunteer_pool value = 54.state_population_k }
		add_to_variable = { var = VAL_local_volunteer_pool value = 55.state_population_k }
		add_to_variable = { var = VAL_local_volunteer_pool value = 56.state_population_k }
		add_to_variable = { var = VAL_local_volunteer_pool value = 57.state_population_k }
		# Population is in thousands; the pool is in ten-thousands: an additional 2%.
		multiply_variable = { var = VAL_local_volunteer_pool value = 0.002 }
	}
	VAL_refresh_refugee_state = yes
	VAL_open_refugee_waves = yes
}

VAL_refresh_trade_network = {
	# First opening permanently commissions a route. Losing it later creates a
	# real commercial disruption rather than a penalty for unconquered land.
	if = {
		limit = { VAL_trade_route_occidia_open = yes }
		set_country_flag = VAL_route_occidia_commissioned
	}
	if = {
		limit = { VAL_trade_route_west_open = yes }
		set_country_flag = VAL_route_west_commissioned
	}
	if = {
		limit = { VAL_trade_route_north_open = yes }
		set_country_flag = VAL_route_north_commissioned
	}
	if = {
		limit = { VAL_trade_route_stelander_open = yes }
		set_country_flag = VAL_route_stelander_commissioned
	}
	if = {
		limit = { VAL_trade_route_vorkerland_open = yes }
		set_country_flag = VAL_route_vorkerland_commissioned
	}
	if = {
		limit = { VAL_trade_route_south_open = yes }
		set_country_flag = VAL_route_south_commissioned
	}

	set_variable = { var = VAL_trade_corridors_active value = 0 }
	set_variable = { var = VAL_trade_corridors_capacity value = 0 }
	set_variable = { var = VAL_route_occidia_active value = 0 }
	set_variable = { var = VAL_route_west_active value = 0 }
	set_variable = { var = VAL_route_north_active value = 0 }
	set_variable = { var = VAL_route_stelander_active value = 0 }
	set_variable = { var = VAL_route_vorkerland_active value = 0 }
	set_variable = { var = VAL_route_south_active value = 0 }

	if = {
		limit = { has_country_flag = VAL_route_occidia_commissioned }
		add_to_variable = { var = VAL_trade_corridors_capacity value = 1 }
	}
	if = {
		limit = { has_country_flag = VAL_route_west_commissioned }
		add_to_variable = { var = VAL_trade_corridors_capacity value = 1 }
	}
	if = {
		limit = { has_country_flag = VAL_route_north_commissioned }
		add_to_variable = { var = VAL_trade_corridors_capacity value = 1 }
	}
	if = {
		limit = { has_country_flag = VAL_route_stelander_commissioned }
		add_to_variable = { var = VAL_trade_corridors_capacity value = 1 }
	}
	if = {
		limit = { has_country_flag = VAL_route_vorkerland_commissioned }
		add_to_variable = { var = VAL_trade_corridors_capacity value = 1 }
	}
	if = {
		limit = { has_country_flag = VAL_route_south_commissioned }
		add_to_variable = { var = VAL_trade_corridors_capacity value = 1 }
	}

	if = {
		limit = { VAL_trade_route_occidia_open = yes }
		set_variable = { var = VAL_route_occidia_active value = 1 }
		add_to_variable = { var = VAL_trade_corridors_active value = 1 }
	}
	if = {
		limit = { VAL_trade_route_west_open = yes }
		set_variable = { var = VAL_route_west_active value = 1 }
		add_to_variable = { var = VAL_trade_corridors_active value = 1 }
	}
	if = {
		limit = { VAL_trade_route_north_open = yes }
		set_variable = { var = VAL_route_north_active value = 1 }
		add_to_variable = { var = VAL_trade_corridors_active value = 1 }
	}
	if = {
		limit = { VAL_trade_route_stelander_open = yes }
		set_variable = { var = VAL_route_stelander_active value = 1 }
		add_to_variable = { var = VAL_trade_corridors_active value = 1 }
	}
	if = {
		limit = { VAL_trade_route_vorkerland_open = yes }
		set_variable = { var = VAL_route_vorkerland_active value = 1 }
		add_to_variable = { var = VAL_trade_corridors_active value = 1 }
	}
	if = {
		limit = { VAL_trade_route_south_open = yes }
		set_variable = { var = VAL_route_south_active value = 1 }
		add_to_variable = { var = VAL_trade_corridors_active value = 1 }
	}

	if = {
		limit = { VAL_trade_corridors_unlocked = yes }
		if = {
			limit = { check_variable = { var = VAL_trade_corridors_active value = 6 compare = greater_than_or_equals } }
			if = {
				limit = { has_idea = VAL_corridor_network_fragmented }
				remove_ideas = VAL_corridor_network_fragmented
			}
			if = {
				limit = { has_idea = VAL_corridor_network_partial }
				remove_ideas = VAL_corridor_network_partial
			}
			if = {
				limit = { NOT = { has_idea = VAL_corridor_network_integrated } }
				add_ideas = VAL_corridor_network_integrated
			}
		}
		else_if = {
			limit = {
				OR = {
					check_variable = { var = VAL_trade_corridors_active value = 2 compare = greater_than_or_equals }
					AND = {
						check_variable = { var = VAL_trade_corridors_capacity value = 1 compare = equals }
						check_variable = { var = VAL_trade_corridors_active value = 1 compare = equals }
					}
				}
			}
			if = {
				limit = { has_idea = VAL_corridor_network_fragmented }
				remove_ideas = VAL_corridor_network_fragmented
			}
			if = {
				limit = { has_idea = VAL_corridor_network_integrated }
				remove_ideas = VAL_corridor_network_integrated
			}
			if = {
				limit = { NOT = { has_idea = VAL_corridor_network_partial } }
				add_ideas = VAL_corridor_network_partial
			}
		}
		else = {
			if = {
				limit = { has_idea = VAL_corridor_network_partial }
				remove_ideas = VAL_corridor_network_partial
			}
			if = {
				limit = { has_idea = VAL_corridor_network_integrated }
				remove_ideas = VAL_corridor_network_integrated
			}
			if = {
				limit = { NOT = { has_idea = VAL_corridor_network_fragmented } }
				add_ideas = VAL_corridor_network_fragmented
			}
		}
	}
	else = {
		if = {
			limit = { has_idea = VAL_corridor_network_fragmented }
			remove_ideas = VAL_corridor_network_fragmented
		}
		if = {
			limit = { has_idea = VAL_corridor_network_partial }
			remove_ideas = VAL_corridor_network_partial
		}
		if = {
			limit = { has_idea = VAL_corridor_network_integrated }
			remove_ideas = VAL_corridor_network_integrated
		}
	}
}

VAL_change_black_market_pressure = {
	# Internal state reconciliation must never be expanded into player-facing
	# decision tooltips. Callers provide concise custom_effect_tooltip text.
	hidden_effect = {
		if = {
			limit = { VAL_trade_corridors_unlocked = yes }
				add_to_variable = { var = VAL_black_market_pressure value = VAL_black_market_pressure_change }
				clamp_variable = { var = VAL_black_market_pressure min = 0 max = 100 }
				VAL_refresh_black_market_state = yes
		}
		else = {
			set_variable = { var = VAL_black_market_pressure value = 0 }
			set_variable = { var = VAL_black_market_last_delta value = 0 }
			VAL_refresh_black_market_state = yes
		}
	}
}

VAL_refresh_black_market_state = {
	if = {
		limit = { NOT = { VAL_trade_corridors_unlocked = yes } }
		set_variable = { var = VAL_black_market_pressure value = 0 }
		set_variable = { var = VAL_black_market_last_delta value = 0 }
		remove_ideas = VAL_black_market_low
		remove_ideas = VAL_black_market_high
		remove_ideas = VAL_black_market_crisis
	}
	else = {
			if = {
				limit = { check_variable = { var = VAL_black_market_pressure value = 75 compare = greater_than_or_equals } }
				if = {
					limit = { has_idea = VAL_black_market_low }
					remove_ideas = VAL_black_market_low
				}
				if = {
					limit = { has_idea = VAL_black_market_high }
					remove_ideas = VAL_black_market_high
				}
				if = {
					limit = { NOT = { has_idea = VAL_black_market_crisis } }
					add_ideas = VAL_black_market_crisis
				}
			}
			else_if = {
				limit = { check_variable = { var = VAL_black_market_pressure value = 50 compare = greater_than_or_equals } }
				if = {
					limit = { has_idea = VAL_black_market_low }
					remove_ideas = VAL_black_market_low
				}
				if = {
					limit = { has_idea = VAL_black_market_crisis }
					remove_ideas = VAL_black_market_crisis
				}
				if = {
					limit = { NOT = { has_idea = VAL_black_market_high } }
					add_ideas = VAL_black_market_high
				}
			}
			else_if = {
				limit = { check_variable = { var = VAL_black_market_pressure value = 25 compare = greater_than_or_equals } }
				if = {
					limit = { has_idea = VAL_black_market_high }
					remove_ideas = VAL_black_market_high
				}
				if = {
					limit = { has_idea = VAL_black_market_crisis }
					remove_ideas = VAL_black_market_crisis
				}
				if = {
					limit = { NOT = { has_idea = VAL_black_market_low } }
					add_ideas = VAL_black_market_low
				}
			}
			else = {
				if = {
					limit = { has_idea = VAL_black_market_low }
					remove_ideas = VAL_black_market_low
				}
				if = {
					limit = { has_idea = VAL_black_market_high }
					remove_ideas = VAL_black_market_high
				}
				if = {
					limit = { has_idea = VAL_black_market_crisis }
					remove_ideas = VAL_black_market_crisis
				}
			}
	}
}

VAL_update_black_market_weekly = {
	if = {
		limit = { VAL_trade_corridors_unlocked = yes }
			set_temp_variable = { var = VAL_black_market_weekly_delta value = 0 }

			if = {
				limit = {
					has_country_flag = VAL_route_occidia_commissioned
					NOT = { check_variable = { var = VAL_route_occidia_active value = 1 compare = equals } }
				}
				add_to_temp_variable = { var = VAL_black_market_weekly_delta value = 3 }
			}
			if = {
				limit = {
					has_country_flag = VAL_route_west_commissioned
					NOT = { check_variable = { var = VAL_route_west_active value = 1 compare = equals } }
				}
				add_to_temp_variable = { var = VAL_black_market_weekly_delta value = 3 }
			}
			if = {
				limit = {
					has_country_flag = VAL_route_north_commissioned
					NOT = { check_variable = { var = VAL_route_north_active value = 1 compare = equals } }
				}
				add_to_temp_variable = { var = VAL_black_market_weekly_delta value = 3 }
			}
			if = {
				limit = {
					has_country_flag = VAL_route_stelander_commissioned
					NOT = { check_variable = { var = VAL_route_stelander_active value = 1 compare = equals } }
				}
				add_to_temp_variable = { var = VAL_black_market_weekly_delta value = 3 }
			}
			if = {
				limit = {
					has_country_flag = VAL_route_vorkerland_commissioned
					NOT = { check_variable = { var = VAL_route_vorkerland_active value = 1 compare = equals } }
				}
				add_to_temp_variable = { var = VAL_black_market_weekly_delta value = 3 }
			}

			if = {
				limit = { has_war = yes }
				add_to_temp_variable = { var = VAL_black_market_weekly_delta value = 2 }
			}
			if = {
				limit = { check_variable = { var = VAL_population_present value = VAL_refugee_housing compare = greater_than } }
				add_to_temp_variable = { var = VAL_black_market_weekly_delta value = 3 }
			}
			if = {
				limit = {
					VAL_trade_corridors_unlocked = yes
					check_variable = { var = VAL_trade_corridors_active value = VAL_trade_corridors_capacity compare = greater_than_or_equals }
					check_variable = { var = VAL_trade_corridors_capacity value = 0 compare = greater_than }
				}
				add_to_temp_variable = { var = VAL_black_market_weekly_delta value = -2 }
			}

			subtract_from_temp_variable = { var = VAL_black_market_weekly_delta value = VAL_corridor_security }
			add_to_temp_variable = { var = VAL_black_market_weekly_delta value = VAL_broker_dependence }
			set_variable = { var = VAL_black_market_last_delta value = VAL_black_market_weekly_delta }
			add_to_variable = { var = VAL_black_market_pressure value = VAL_black_market_weekly_delta }
			clamp_variable = { var = VAL_black_market_pressure min = 0 max = 100 }
			VAL_refresh_black_market_state = yes
	}
	else = {
		set_temp_variable = { var = VAL_black_market_weekly_delta value = 0 }
		set_variable = { var = VAL_black_market_last_delta value = 0 }
		set_variable = { var = VAL_black_market_pressure value = 0 }
		VAL_refresh_black_market_state = yes
	}
}

VAL_refresh_refugee_state = {
	set_variable = { var = VAL_population_present value = VAL_displaced_population }
	if = { limit = { has_variable = VAL_refugee_training_escrow } add_to_variable = { var = VAL_population_present value = VAL_refugee_training_escrow } }
	if = { limit = { has_variable = VAL_refugee_labor_escrow } add_to_variable = { var = VAL_population_present value = VAL_refugee_labor_escrow } }
	if = {
		limit = { check_variable = { var = VAL_population_present value = VAL_refugee_housing compare = greater_than } }
		if = {
			limit = { NOT = { has_idea = VAL_refugee_strain } }
			add_ideas = VAL_refugee_strain
		}
	}
	else = {
		if = {
			limit = { has_idea = VAL_refugee_strain }
			remove_ideas = VAL_refugee_strain
		}
	}
}
