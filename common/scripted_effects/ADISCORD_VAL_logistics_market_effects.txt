# Kefreyt logistics, displacement and grey-market runtime.
# All recurring work is country-scoped to VAL and uses a fixed set of states.

VAL_corridor_income_10 = {
	ADISCORD_economy_initialize_country = yes
	add_to_variable = { var = ADISCORD_economy_treasury value = 10 }
	add_to_variable = { var = ADISCORD_economy_current_month_action_income value = 10 }
	ADISCORD_economy_mark_dirty = yes
}

VAL_corridor_income_20 = {
	ADISCORD_economy_initialize_country = yes
	add_to_variable = { var = ADISCORD_economy_treasury value = 20 }
	add_to_variable = { var = ADISCORD_economy_current_month_action_income value = 20 }
	ADISCORD_economy_mark_dirty = yes
}

VAL_initialize_logistics_market = {
	if = { limit = { NOT = { has_variable = VAL_trade_corridors_active } } set_variable = { var = VAL_trade_corridors_active value = 0 } }
	if = { limit = { NOT = { has_variable = VAL_trade_corridors_capacity } } set_variable = { var = VAL_trade_corridors_capacity value = 0 } }
	if = { limit = { NOT = { has_variable = VAL_route_occidia_active } } set_variable = { var = VAL_route_occidia_active value = 0 } }
	if = { limit = { NOT = { has_variable = VAL_route_north_active } } set_variable = { var = VAL_route_north_active value = 0 } }
	if = { limit = { NOT = { has_variable = VAL_route_stelander_active } } set_variable = { var = VAL_route_stelander_active value = 0 } }
	if = { limit = { NOT = { has_variable = VAL_route_vorkerland_active } } set_variable = { var = VAL_route_vorkerland_active value = 0 } }
	if = { limit = { NOT = { has_variable = VAL_corridor_security } } set_variable = { var = VAL_corridor_security value = 0 } }
	if = { limit = { NOT = { has_variable = VAL_black_market_pressure } } set_variable = { var = VAL_black_market_pressure value = 8 } }
	if = { limit = { NOT = { has_variable = VAL_displaced_population } } set_variable = { var = VAL_displaced_population value = 0 } }
	clamp_variable = { var = VAL_corridor_security min = 0 max = 3 }
	clamp_variable = { var = VAL_black_market_pressure min = 0 max = 100 }
	clamp_variable = { var = VAL_displaced_population min = 0 max = 100 }
	VAL_refresh_trade_network = yes
	VAL_refresh_black_market_state = yes
	VAL_refresh_refugee_state = yes
}

VAL_refresh_trade_network = {
	# First opening permanently commissions a route. Losing it later creates a
	# real commercial disruption rather than a penalty for unconquered land.
	if = { limit = { VAL_trade_route_occidia_open = yes } set_country_flag = VAL_route_occidia_commissioned }
	if = { limit = { VAL_trade_route_north_open = yes } set_country_flag = VAL_route_north_commissioned }
	if = { limit = { VAL_trade_route_stelander_open = yes } set_country_flag = VAL_route_stelander_commissioned }
	if = { limit = { VAL_trade_route_vorkerland_open = yes } set_country_flag = VAL_route_vorkerland_commissioned }

	set_variable = { var = VAL_trade_corridors_active value = 0 }
	set_variable = { var = VAL_trade_corridors_capacity value = 0 }
	set_variable = { var = VAL_route_occidia_active value = 0 }
	set_variable = { var = VAL_route_north_active value = 0 }
	set_variable = { var = VAL_route_stelander_active value = 0 }
	set_variable = { var = VAL_route_vorkerland_active value = 0 }

	if = { limit = { has_country_flag = VAL_route_occidia_commissioned } add_to_variable = { var = VAL_trade_corridors_capacity value = 1 } }
	if = { limit = { has_country_flag = VAL_route_north_commissioned } add_to_variable = { var = VAL_trade_corridors_capacity value = 1 } }
	if = { limit = { has_country_flag = VAL_route_stelander_commissioned } add_to_variable = { var = VAL_trade_corridors_capacity value = 1 } }
	if = { limit = { has_country_flag = VAL_route_vorkerland_commissioned } add_to_variable = { var = VAL_trade_corridors_capacity value = 1 } }

	if = {
		limit = { VAL_trade_route_occidia_open = yes }
		set_variable = { var = VAL_route_occidia_active value = 1 }
		add_to_variable = { var = VAL_trade_corridors_active value = 1 }
	}
	if = {
		limit = { VAL_trade_route_north_open = yes }
		set_variable = { var = VAL_route_north_active value = 1 }
		add_to_variable = { var = VAL_trade_corridors_active value = 1 }
	}
	if = {
		limit = { VAL_trade_route_stelander_open = yes }
		set_variable = { var = VAL_route_stelander_active value = 1 }
		add_to_variable = { var = VAL_trade_corridors_active value = 1 }
	}
	if = {
		limit = { VAL_trade_route_vorkerland_open = yes }
		set_variable = { var = VAL_route_vorkerland_active value = 1 }
		add_to_variable = { var = VAL_trade_corridors_active value = 1 }
	}

	if = {
		limit = { VAL_trade_corridors_unlocked = yes }
		if = {
			limit = { check_variable = { var = VAL_trade_corridors_active value = 4 compare = greater_than_or_equals } }
			if = { limit = { has_idea = VAL_corridor_network_fragmented } remove_ideas = VAL_corridor_network_fragmented }
			if = { limit = { has_idea = VAL_corridor_network_partial } remove_ideas = VAL_corridor_network_partial }
			if = { limit = { NOT = { has_idea = VAL_corridor_network_integrated } } add_ideas = VAL_corridor_network_integrated }
		}
		else_if = {
			limit = {
				OR = {
					check_variable = { var = VAL_trade_corridors_active value = 2 compare = greater_than_or_equals }
					AND = {
						check_variable = { var = VAL_trade_corridors_capacity value = 1 compare = equals }
						check_variable = { var = VAL_trade_corridors_active value = 1 compare = equals }
					}
				}
			}
			if = { limit = { has_idea = VAL_corridor_network_fragmented } remove_ideas = VAL_corridor_network_fragmented }
			if = { limit = { has_idea = VAL_corridor_network_integrated } remove_ideas = VAL_corridor_network_integrated }
			if = { limit = { NOT = { has_idea = VAL_corridor_network_partial } } add_ideas = VAL_corridor_network_partial }
		}
		else = {
			if = { limit = { has_idea = VAL_corridor_network_partial } remove_ideas = VAL_corridor_network_partial }
			if = { limit = { has_idea = VAL_corridor_network_integrated } remove_ideas = VAL_corridor_network_integrated }
			if = { limit = { NOT = { has_idea = VAL_corridor_network_fragmented } } add_ideas = VAL_corridor_network_fragmented }
		}
	}
	else = {
		if = { limit = { has_idea = VAL_corridor_network_fragmented } remove_ideas = VAL_corridor_network_fragmented }
		if = { limit = { has_idea = VAL_corridor_network_partial } remove_ideas = VAL_corridor_network_partial }
		if = { limit = { has_idea = VAL_corridor_network_integrated } remove_ideas = VAL_corridor_network_integrated }
	}
}

VAL_pay_trade_corridors = {
	if = {
		limit = { VAL_trade_route_occidia_open = yes }
		VAL_corridor_income_20 = yes
		if = { limit = { has_country_flag = VAL_route_occidia_upgraded } VAL_corridor_income_10 = yes }
	}
	if = {
		limit = { VAL_trade_route_north_open = yes }
		VAL_corridor_income_20 = yes
		if = { limit = { has_country_flag = VAL_route_north_upgraded } VAL_corridor_income_10 = yes }
	}
	if = {
		limit = { VAL_trade_route_stelander_open = yes }
		VAL_corridor_income_20 = yes
		if = { limit = { has_country_flag = VAL_route_stelander_upgraded } VAL_corridor_income_10 = yes }
	}
	if = {
		limit = { VAL_trade_route_vorkerland_open = yes }
		VAL_corridor_income_20 = yes
		if = { limit = { has_country_flag = VAL_route_vorkerland_upgraded } VAL_corridor_income_10 = yes }
	}
	if = {
		limit = { has_idea = VAL_emergency_bypass VAL_trade_network_disrupted = yes }
		VAL_corridor_income_20 = yes
	}
}

VAL_change_black_market_pressure = {
	add_to_variable = { var = VAL_black_market_pressure value = VAL_black_market_pressure_change }
	clamp_variable = { var = VAL_black_market_pressure min = 0 max = 100 }
	VAL_refresh_black_market_state = yes
}

VAL_refresh_black_market_state = {
	if = {
		limit = { check_variable = { var = VAL_black_market_pressure value = 75 compare = greater_than_or_equals } }
		if = { limit = { has_idea = VAL_black_market_low } remove_ideas = VAL_black_market_low }
		if = { limit = { has_idea = VAL_black_market_high } remove_ideas = VAL_black_market_high }
		if = { limit = { NOT = { has_idea = VAL_black_market_crisis } } add_ideas = VAL_black_market_crisis }
	}
	else_if = {
		limit = { check_variable = { var = VAL_black_market_pressure value = 50 compare = greater_than_or_equals } }
		if = { limit = { has_idea = VAL_black_market_low } remove_ideas = VAL_black_market_low }
		if = { limit = { has_idea = VAL_black_market_crisis } remove_ideas = VAL_black_market_crisis }
		if = { limit = { NOT = { has_idea = VAL_black_market_high } } add_ideas = VAL_black_market_high }
	}
	else_if = {
		limit = { check_variable = { var = VAL_black_market_pressure value = 25 compare = greater_than_or_equals } }
		if = { limit = { has_idea = VAL_black_market_high } remove_ideas = VAL_black_market_high }
		if = { limit = { has_idea = VAL_black_market_crisis } remove_ideas = VAL_black_market_crisis }
		if = { limit = { NOT = { has_idea = VAL_black_market_low } } add_ideas = VAL_black_market_low }
	}
	else = {
		if = { limit = { has_idea = VAL_black_market_low } remove_ideas = VAL_black_market_low }
		if = { limit = { has_idea = VAL_black_market_high } remove_ideas = VAL_black_market_high }
		if = { limit = { has_idea = VAL_black_market_crisis } remove_ideas = VAL_black_market_crisis }
	}
}

VAL_update_black_market_weekly = {
	set_temp_variable = { var = VAL_black_market_weekly_delta value = 0 }

	if = { limit = { has_country_flag = VAL_route_occidia_commissioned NOT = { VAL_trade_route_occidia_open = yes } } add_to_temp_variable = { var = VAL_black_market_weekly_delta value = 3 } }
	if = { limit = { has_country_flag = VAL_route_north_commissioned NOT = { VAL_trade_route_north_open = yes } } add_to_temp_variable = { var = VAL_black_market_weekly_delta value = 3 } }
	if = { limit = { has_country_flag = VAL_route_stelander_commissioned NOT = { VAL_trade_route_stelander_open = yes } } add_to_temp_variable = { var = VAL_black_market_weekly_delta value = 3 } }
	if = { limit = { has_country_flag = VAL_route_vorkerland_commissioned NOT = { VAL_trade_route_vorkerland_open = yes } } add_to_temp_variable = { var = VAL_black_market_weekly_delta value = 3 } }

	if = { limit = { has_war = yes } add_to_temp_variable = { var = VAL_black_market_weekly_delta value = 2 } }
	if = { limit = { check_variable = { var = VAL_displaced_population value = 20 compare = greater_than_or_equals } } add_to_temp_variable = { var = VAL_black_market_weekly_delta value = 1 } }
	if = {
		limit = {
			VAL_trade_corridors_unlocked = yes
			NOT = { VAL_trade_network_disrupted = yes }
			check_variable = { var = VAL_trade_corridors_capacity value = 0 compare = greater_than }
		}
		add_to_temp_variable = { var = VAL_black_market_weekly_delta value = -2 }
	}

	subtract_from_temp_variable = { var = VAL_black_market_weekly_delta value = VAL_corridor_security }
	add_to_variable = { var = VAL_black_market_pressure value = VAL_black_market_weekly_delta }
	clamp_variable = { var = VAL_black_market_pressure min = 0 max = 100 }
	VAL_refresh_black_market_state = yes
}

VAL_refresh_refugee_state = {
	if = {
		limit = { check_variable = { var = VAL_displaced_population value = 20 compare = greater_than_or_equals } }
		if = { limit = { NOT = { has_idea = VAL_refugee_strain } } add_ideas = VAL_refugee_strain }
	}
	else = {
		if = { limit = { has_idea = VAL_refugee_strain } remove_ideas = VAL_refugee_strain }
	}
}

VAL_update_refugees_weekly = {
	if = {
		limit = { VAL_regional_displacement_crisis = yes }
		if = {
			limit = { has_country_flag = VAL_refugee_border_closed }
			add_to_variable = { var = VAL_displaced_population value = 2 }
		}
		else = { add_to_variable = { var = VAL_displaced_population value = 5 } }
	}
	else_if = {
		limit = { check_variable = { var = VAL_displaced_population value = 0 compare = greater_than } }
		add_to_variable = { var = VAL_displaced_population value = -1 }
	}
	clamp_variable = { var = VAL_displaced_population min = 0 max = 100 }
	VAL_refresh_refugee_state = yes
}

VAL_logistics_market_weekly = {
	VAL_refresh_trade_network = yes
	if = { limit = { VAL_trade_corridors_unlocked = yes } VAL_pay_trade_corridors = yes }
	VAL_update_refugees_weekly = yes
	VAL_update_black_market_weekly = yes
}
