# --- rus_crisis_effects ---
# RUS owns the phase: 1 mobilization, 2 coalition war, 3 coalition defeated,
# 4 empire dissolved, 5 project closed, 6 epilogue. Timers never restart.
RUS_crisis_check_start = {
	if = {
		limit = { RUS_crisis_can_begin = yes }
		RUS_crisis_begin = yes
	}
}

RUS_crisis_begin = {
	if = {
		limit = { RUS_crisis_can_begin = yes }
		set_variable = { var = RUS_crisis_phase value = 1 }
		activate_mission = RUS_crisis_invasion_countdown
		activate_mission = RUS_crisis_laser_countdown
		country_event = { id = ADISCORD_rus_crisis.2 }
		# 90 days before the 455-day strike deadline.
		country_event = { id = ADISCORD_rus_crisis.14 days = 365 }
		country_event = { id = ADISCORD_rus_crisis.21 days = 365 }
		country_event = { id = ADISCORD_rus_crisis.22 days = 395 }
		country_event = { id = ADISCORD_rus_crisis.23 days = 425 }
		every_country = {
			limit = {
				OR = {
					RUS_crisis_coalition_candidate = yes
					AND = {
						RUS_crisis_player_challenger = yes
						has_war_with = RUS
					}
				}
			}
			ADISCORD_release_non_participating_minor_optimization = yes
			set_country_flag = RUS_crisis_warned
			activate_mission = RUS_crisis_defence_countdown
			country_event = { id = ADISCORD_rus_crisis.1 }
		}
	}
}

# The terminal epilogue cannot open a new war after its global settlement.
RUS_crisis_enforce_truce = {
	if = {
		limit = {
			has_global_flag = RUS_crisis_world_ended
			NOT = { has_global_flag = RUS_crisis_truce_in_progress }
		}
		set_global_flag = RUS_crisis_truce_in_progress
		ROOT = { white_peace = FROM }
		clr_global_flag = RUS_crisis_truce_in_progress
	}
}

# COUNTRY defender. The roster is captured before declaring war because native
# war callbacks may immediately call allies. Preserve other campaigns' ownership.
RUS_crisis_register_defender = {
	if = {
		limit = { NOT = { has_country_flag = RUS_crisis_defender } }
		ADISCORD_release_non_participating_minor_optimization = yes
		set_country_flag = RUS_crisis_defender
		add_to_array = { array = global.RUS_crisis_defenders value = THIS }
		if = {
			limit = {
				tag = NAM
				NOT = { has_country_flag = NAM_rus_oil_offered }
			}
			country_event = { id = ADISCORD_nam_oil.10 hours = 6 }
			set_country_flag = NAM_rus_oil_offered
		}
		if = {
			limit = {
				OR = {
					is_major = no
					RUS_crisis_other_major_required = yes
				}
			}
			set_country_flag = RUS_crisis_added_major
			set_major = yes
		}
	}
}

RUS_crisis_launch = {
	if = {
		limit = { check_variable = { var = RUS_crisis_phase value = 1 compare = equals } }
		if = {
			limit = {
				RUS_crisis_project_active = yes
				RUS_lab_reactor_access = yes
			}
			set_variable = { var = RUS_crisis_phase value = 2 }
			clear_global_event_target = RUS_crisis_war_anchor
			# An existing player offensive supplies the native war for intervention.
			every_enemy_country = {
				limit = {
					RUS_crisis_player_challenger = yes
					is_subject = no
					has_capitulated = no
					RUS_crisis_side_compatible = yes
					NOT = { has_event_target = RUS_crisis_war_anchor }
				}
				RUS_crisis_register_defender = yes
				save_global_event_target_as = RUS_crisis_war_anchor
				RUS = {
					every_enemy_country = {
						limit = { RUS_crisis_defending_bloc = yes }
						RUS_crisis_accept_native_entry = yes
					}
				}
				RUS_crisis_confirm_intervention_entry = yes
			}
			MON = {
				if = {
					limit = { RUS_crisis_coalition_candidate = yes }
					RUS_crisis_start_intervention = yes
				}
			}
			every_country = {
				limit = { RUS_crisis_coalition_candidate = yes }
				RUS_crisis_start_intervention = yes
			}
			for_each_scope_loop = {
				array = global.RUS_crisis_defenders
				country_event = { id = ADISCORD_rus_crisis.3 }
			}
			country_event = { id = ADISCORD_rus_crisis.3 }
			country_event = { id = ADISCORD_rus_crisis.7 days = 1 }
		}
		else = { RUS_crisis_close = yes }
	}
}

# COUNTRY challenger. An active intervention already owns the war and settlement.
RUS_crisis_player_attack = {
	if = {
		limit = { RUS_crisis_player_attack_available = yes }
		if = {
			limit = { RUS = { check_variable = { var = RUS_crisis_phase value = 2 compare = equals } } }
			RUS_crisis_start_intervention = yes
		}
		else = {
			# A pre-emptive war does not start or reset the Khan's orbital programme.
			declare_war_on = {
				target = RUS
				type = annex_everything
			}
		}
	}
}

# COUNTRY sovereign intervener. Keep one native war without merging factions.
# Register before callbacks; enemies of the existing war side cannot join it.
RUS_crisis_start_intervention = {
	RUS_crisis_refresh_war_anchor = yes
	if = {
		limit = {
			RUS_crisis_can_enter_common_war = yes
			NOT = { has_country_flag = RUS_crisis_defender }
		}
		RUS_crisis_register_defender = yes
		set_country_flag = RUS_crisis_entry_pending
		if = {
			limit = { NOT = { has_event_target = RUS_crisis_war_anchor } }
			save_global_event_target_as = RUS_crisis_war_anchor
			declare_war_on = { target = RUS type = annex_everything }
		}
		else_if = {
			limit = {
				NOT = { has_war_with = RUS }
				event_target:RUS_crisis_war_anchor = { has_war_with = RUS }
			}
			RUS_crisis_request_intervention_entry = yes
		}
		if = {
			limit = { has_war_with = RUS }
			RUS_crisis_confirm_intervention_entry = yes
		}
		else = {
			country_event = { id = ADISCORD_rus_crisis.9 days = 1 }
		}
	}
}

# RUS. Regional winners that were not free at launch receive one offer; the
# answer revalidates entry because the event may stay open.
RUS_crisis_call_second_wave = {
	if = {
		limit = {
			RUS_crisis_project_active = yes
			OR = {
				check_variable = { var = RUS_crisis_phase value = 2 compare = equals }
				check_variable = { var = RUS_crisis_phase value = 3 compare = equals }
			}
		}
		every_country = {
			limit = { RUS_crisis_second_wave_candidate = yes }
			country_event = { id = ADISCORD_rus_crisis.15 }
		}
	}
}

# COUNTRY regional winner. After a defeated first wave the same programme
# reopens the common war; its deadline is never restarted.
RUS_crisis_enter_second_wave = {
	if = {
		limit = {
			RUS_crisis_second_wave_candidate = yes
			RUS_crisis_can_enter_common_war = yes
			RUS = {
				RUS_crisis_project_active = yes
				OR = {
					check_variable = { var = RUS_crisis_phase value = 2 compare = equals }
					check_variable = { var = RUS_crisis_phase value = 3 compare = equals }
				}
			}
		}
		RUS_crisis_remove_volunteer_mandate = yes
		RUS = {
			if = {
				limit = { check_variable = { var = RUS_crisis_phase value = 3 compare = equals } }
				set_variable = { var = RUS_crisis_phase value = 2 }
			}
		}
		RUS_crisis_start_intervention = yes
		RUS = { country_event = { id = ADISCORD_rus_crisis.16 } }
	}
}

# RUS monthly. Only the fixed regional winners can hold the mandate.
RUS_crisis_update_volunteers = {
	if = {
		limit = { check_variable = { var = RUS_crisis_phase value = 2 compare = equals } }
		VAL = { RUS_crisis_update_volunteer_mandate = yes }
		STS = { RUS_crisis_update_volunteer_mandate = yes }
		STP = { RUS_crisis_update_volunteer_mandate = yes }
		NOD = { RUS_crisis_update_volunteer_mandate = yes }
		SHL = { RUS_crisis_update_volunteer_mandate = yes }
		NAM = { RUS_crisis_update_volunteer_mandate = yes }
		WRK = { RUS_crisis_update_volunteer_mandate = yes }
		IVN = { RUS_crisis_update_volunteer_mandate = yes }
	}
}

# COUNTRY. The idea only relaxes native volunteer limits; sending stays a
# native diplomatic action, so a human government decides for itself.
RUS_crisis_update_volunteer_mandate = {
	if = {
		limit = { RUS_crisis_volunteer_sender = yes }
		if = {
			limit = { NOT = { has_idea = RUS_crisis_volunteer_mandate } }
			add_ideas = RUS_crisis_volunteer_mandate
		}
	}
	else = { RUS_crisis_remove_volunteer_mandate = yes }
}

RUS_crisis_clear_volunteers = {
	VAL = { RUS_crisis_remove_volunteer_mandate = yes }
	STS = { RUS_crisis_remove_volunteer_mandate = yes }
	STP = { RUS_crisis_remove_volunteer_mandate = yes }
	NOD = { RUS_crisis_remove_volunteer_mandate = yes }
	SHL = { RUS_crisis_remove_volunteer_mandate = yes }
	NAM = { RUS_crisis_remove_volunteer_mandate = yes }
	WRK = { RUS_crisis_remove_volunteer_mandate = yes }
	IVN = { RUS_crisis_remove_volunteer_mandate = yes }
}

RUS_crisis_remove_volunteer_mandate = {
	if = {
		limit = { has_idea = RUS_crisis_volunteer_mandate }
		remove_ideas = RUS_crisis_volunteer_mandate
	}
}

# COUNTRY entrant. Resolve the saved anchor as a scope before the native war
# effect. Returning through PREV puts the entrant in THIS and the anchor in
# PREV, independently of the launch or delayed event's ROOT and FROM.
RUS_crisis_request_intervention_entry = {
	if = {
		limit = {
			has_event_target = RUS_crisis_war_anchor
			NOT = { has_war_with = RUS }
		}
		event_target:RUS_crisis_war_anchor = {
			if = {
				limit = { has_war_with = RUS }
				PREV = {
					add_to_war = {
						targeted_alliance = PREV
						enemy = RUS
						hostility_reason = asked_to_join
						single_target_only = yes
					}
					log = "[RUS_CRISIS] [This.GetTag]: coalition entry requested."
				}
			}
		}
	}
}

# COUNTRY ally, PREV is already on the intended side of the war against RUS.
RUS_crisis_join_intervention = {
	if = {
		limit = {
			exists = yes
			has_capitulated = no
			NOT = { has_country_flag = RUS_crisis_defender }
			NOT = { tag = RUS }
			NOT = { is_subject_of = RUS }
			NOT = { is_in_faction_with = RUS }
			NOT = { has_war_with = PREV }
			RUS_crisis_can_enter_common_war = yes
			PREV = { has_war_with = RUS }
		}
		RUS_crisis_register_defender = yes
		set_country_flag = RUS_crisis_entry_pending
		if = {
			limit = { NOT = { has_war_with = RUS } }
			add_to_war = {
				targeted_alliance = PREV
				enemy = RUS
				hostility_reason = asked_to_join
				single_target_only = yes
			}
		}
		if = {
			limit = { has_war_with = RUS }
			RUS_crisis_confirm_intervention_entry = yes
		}
		else = {
			country_event = { id = ADISCORD_rus_crisis.9 days = 1 }
		}
	}
}

# Native war entry can settle after hourly events. Allow a full daily tick
# before the bounded retry and again before checking its result.
RUS_crisis_retry_intervention_entry = {
	if = {
		limit = { has_country_flag = RUS_crisis_entry_pending }
		if = {
			limit = { RUS_crisis_entry_still_valid = yes }
			RUS_crisis_refresh_war_anchor = yes
			if = {
				limit = { has_war_with = RUS }
				RUS_crisis_confirm_intervention_entry = yes
			}
			else = {
				RUS_crisis_request_intervention_entry = yes
				country_event = { id = ADISCORD_rus_crisis.11 days = 1 }
			}
		}
		else = { RUS_crisis_remove_defender = yes }
	}
}

RUS_crisis_settle_intervention_entry = {
	if = {
		limit = { has_country_flag = RUS_crisis_entry_pending }
		if = {
			limit = {
				RUS_crisis_entry_still_valid = yes
				has_war_with = RUS
			}
			RUS_crisis_confirm_intervention_entry = yes
		}
		else = {
			log = "[RUS_CRISIS] [This.GetTag]: coalition entry not confirmed; pending membership withdrawn."
			RUS_crisis_remove_defender = yes
		}
		RUS = { RUS_crisis_check_external_end = yes }
	}
}

# A native ally entry can precede its leader's scripted invitations.
# Register and clear pending before expansion to terminate allied cycles.
RUS_crisis_accept_native_entry = {
	if = {
		limit = {
			OR = {
				NOT = { has_country_flag = RUS_crisis_defender }
				has_country_flag = RUS_crisis_entry_pending
			}
		}
		RUS_crisis_register_defender = yes
		RUS_crisis_confirm_intervention_entry = yes
	}
}

# Form the alliance only after native war entry is confirmed. Existing blocs
# remain intact; their members still participate through the common war.
RUS_crisis_form_reactor_alliance = {
	if = {
		limit = {
			RUS = { check_variable = { var = RUS_crisis_phase value = 2 compare = equals } }
			MON = {
				exists = yes
				is_subject = no
				has_capitulated = no
				has_country_flag = RUS_crisis_defender
				NOT = { has_country_flag = RUS_crisis_entry_pending }
				has_war_with = RUS
				OR = {
					is_in_faction = no
					is_faction_leader = yes
				}
			}
			BTL = {
				exists = yes
				is_subject = no
				has_capitulated = no
				has_country_flag = RUS_crisis_defender
				NOT = { has_country_flag = RUS_crisis_entry_pending }
				has_war_with = RUS
				has_war_together_with = MON
				RUS_crisis_side_compatible = yes
				OR = {
					is_in_faction = no
					is_in_faction_with = MON
				}
			}
		}
		MON = {
			if = {
				limit = { is_in_faction = no }
				create_faction_from_template = {
					template = faction_template_ADISCORD_standard
					name = RUS_crisis_reactor_alliance
				}
			}
			if = {
				limit = { BTL = { is_in_faction = no } }
				add_to_faction = BTL
			}
			if = {
				limit = {
					TMR = {
						exists = yes
						is_subject = no
						has_capitulated = no
						is_in_faction = no
						has_country_flag = RUS_crisis_defender
						NOT = { has_country_flag = RUS_crisis_entry_pending }
						has_war_with = RUS
						has_war_together_with = MON
						RUS_crisis_side_compatible = yes
					}
				}
				add_to_faction = TMR
			}
		}
	}
}

RUS_crisis_confirm_intervention_entry = {
	if = {
		limit = { has_country_flag = RUS_crisis_entry_pending }
		log = "[RUS_CRISIS] [This.GetTag]: coalition entry confirmed."
	}
	clr_country_flag = RUS_crisis_entry_pending
	RUS_crisis_form_reactor_alliance = yes
	every_allied_country = {
		limit = { NOT = { tag = PREV } }
		RUS_crisis_join_intervention = yes
	}
	RUS_crisis_join_subjects = yes
}

# Dependencies are acyclic. Each subject follows its own overlord's war side.
RUS_crisis_join_subjects = {
	every_subject_country = { RUS_crisis_join_intervention = yes }
}

RUS_crisis_clear_preparation = {
	remove_ideas = RUS_crisis_reactor_guard
}

RUS_crisis_clear_roster = {
	if = {
		limit = { RUS_crisis_project_active = no }
		RUS_crisis_clear_preparation = yes
	}
	for_each_scope_loop = {
		array = global.RUS_crisis_defenders
		RUS_crisis_clear_defender = yes
	}
	clear_array = global.RUS_crisis_defenders
	clear_global_event_target = RUS_crisis_war_anchor
	RUS_crisis_clear_volunteers = yes
	every_country = {
		limit = { has_country_flag = RUS_crisis_warned }
		clr_country_flag = RUS_crisis_warned
		remove_mission = RUS_crisis_defence_countdown
	}
}

# COUNTRY. Removal happens outside an active iteration over the same array.
RUS_crisis_remove_defender = {
	RUS_crisis_clear_defender = yes
	remove_from_array = { array = global.RUS_crisis_defenders value = THIS }
	RUS_crisis_refresh_war_anchor = yes
}

# The native war can outlive its first declarer. Rebind only within its roster.
RUS_crisis_refresh_war_anchor = {
	if = {
		limit = { has_event_target = RUS_crisis_war_anchor }
		if = {
			limit = {
				NOT = {
					event_target:RUS_crisis_war_anchor = {
						exists = yes
						has_country_flag = RUS_crisis_defender
						OR = {
							has_war_with = RUS
							AND = {
								has_capitulated = no
								has_country_flag = RUS_crisis_entry_pending
							}
						}
					}
				}
			}
			clear_global_event_target = RUS_crisis_war_anchor
		}
	}
	if = {
		limit = { NOT = { has_event_target = RUS_crisis_war_anchor } }
		for_each_scope_loop = {
			array = global.RUS_crisis_defenders
			if = {
				limit = {
					NOT = { has_event_target = RUS_crisis_war_anchor }
					exists = yes
					has_country_flag = RUS_crisis_defender
					has_war_with = RUS
				}
				save_global_event_target_as = RUS_crisis_war_anchor
			}
		}
	}
}

RUS_crisis_clear_defender = {
	if = {
		limit = {
			has_country_flag = RUS_crisis_entry_pending
			RUS = { check_variable = { var = RUS_crisis_phase value = 2 compare = greater_than } }
		}
		# The native request may still enter after its campaign receipt is cleared.
		set_country_flag = { flag = RUS_crisis_entry_cancelled days = 2 }
		country_event = { id = ADISCORD_rus_crisis.13 hours = 1 }
	}
	clr_country_flag = RUS_crisis_defender
	clr_country_flag = RUS_crisis_entry_pending
	if = {
		limit = { has_country_flag = RUS_crisis_added_major }
		if = {
			limit = { RUS_crisis_other_major_required = no }
			set_major = no
		}
		clr_country_flag = RUS_crisis_added_major
	}
}

# COUNTRY. This receipt can only come from this campaign's pending entry.
RUS_crisis_cancel_pending_intervention = {
	if = {
		limit = { has_country_flag = RUS_crisis_entry_cancelled }
		if = {
			limit = {
				has_war_with = RUS
				RUS = { check_variable = { var = RUS_crisis_phase value = 2 compare = greater_than } }
			}
			white_peace = RUS
		}
		clr_country_flag = RUS_crisis_entry_cancelled
	}
}

# COUNTRY. Called after the ending campaign clears its own major receipt.
RUS_crisis_release_other_major = {
	if = {
		limit = { has_country_flag = RUS_crisis_defender }
		set_country_flag = RUS_crisis_added_major
	}
	else_if = {
		limit = { RUS_crisis_other_major_required = no }
		set_major = no
	}
}

RUS_crisis_close = {
	if = {
		limit = {
			check_variable = { var = RUS_crisis_phase value = 0 compare = greater_than }
			check_variable = { var = RUS_crisis_phase value = 3 compare = less_than }
		}
		set_country_flag = RUS_crisis_laser_disabled
		RUS_last_sky_clear_offers = yes
		set_variable = { var = RUS_crisis_phase value = 5 }
		every_country = {
			limit = {
				OR = {
					has_country_flag = RUS_crisis_warned
					has_country_flag = RUS_crisis_defender
				}
			}
			country_event = { id = ADISCORD_rus_crisis.6 }
		}
		RUS_crisis_clear_roster = yes
	}
}

# Run after peace/annex/puppet callbacks have settled. This delayed check does
# not recreate a war ended by another settlement.
RUS_crisis_check_external_end = {
	if = {
		limit = {
			OR = {
				check_variable = { var = RUS_crisis_phase value = 1 compare = equals }
				check_variable = { var = RUS_crisis_phase value = 2 compare = equals }
			}
		}
		if = {
			limit = {
				OR = {
					exists = no
					is_subject = yes
				}
			}
			RUS_crisis_close = yes
		}
		else = {
			clear_temp_array = RUS_crisis_departures
			for_each_scope_loop = {
				array = global.RUS_crisis_defenders
				if = {
					limit = {
						OR = {
							exists = no
							AND = {
								NOT = { has_country_flag = RUS_crisis_entry_pending }
								NOT = { has_war_with = RUS }
							}
						}
					}
					RUS_crisis_clear_defender = yes
					add_to_temp_array = { array = RUS_crisis_departures value = THIS }
				}
			}
			for_each_scope_loop = {
				array = RUS_crisis_departures
				remove_from_array = { array = global.RUS_crisis_defenders value = THIS }
			}
			clear_temp_array = RUS_crisis_departures
			RUS_crisis_refresh_war_anchor = yes
			RUS_crisis_resolve_capitulation = yes
		}
	}
}

# Native capitulation checks include the current callback before the engine's
# has_capitulated update. A liberated member blocks victory again.
RUS_crisis_record_capitulation = {
	if = {
		limit = {
			ROOT = {
				tag = RUS
				exists = yes
				has_country_flag = ADISCORD_vorkerland_rus_last_empire_proclaimed
				NOT = { check_variable = { var = RUS_crisis_phase value = 4 compare = equals } }
			}
			FROM = { exists = yes }
		}
		RUS = {
			if = {
				limit = { NOT = { check_variable = { var = RUS_crisis_phase value = 2 compare = equals } } }
				RUS_crisis_clear_roster = yes
				FROM = {
					save_global_event_target_as = RUS_crisis_hegemon
					RUS_crisis_register_defender = yes
				}
				set_variable = { var = RUS_crisis_phase value = 2 }
			}
			FROM = {
				if = {
					limit = { is_subject = yes }
					OVERLORD = { save_global_event_target_as = RUS_crisis_hegemon }
				}
				else = { save_global_event_target_as = RUS_crisis_hegemon }
			}
			set_country_flag = RUS_crisis_capitulation_current
			RUS_crisis_resolve_capitulation = yes
			clr_country_flag = RUS_crisis_capitulation_current
			set_country_flag = { flag = RUS_crisis_capitulation_reserved days = 2 }
		}
	}
	else_if = {
		limit = {
			RUS = { check_variable = { var = RUS_crisis_phase value = 2 compare = equals } }
			ROOT = { has_country_flag = RUS_crisis_defender }
			FROM = {
				OR = {
					tag = RUS
					is_subject_of = RUS
				}
			}
		}
		ROOT = { set_country_flag = RUS_crisis_capitulation_current }
		RUS = { RUS_crisis_resolve_capitulation = yes }
		ROOT = {
			clr_country_flag = RUS_crisis_capitulation_current
			set_country_flag = { flag = RUS_crisis_capitulation_reserved days = 2 }
		}
	}
	else_if = {
		limit = {
			RUS = { check_variable = { var = RUS_crisis_phase value = 2 compare = equals } }
			ROOT = { has_country_flag = RUS_crisis_defender }
			FROM = {
				NOT = {
					OR = {
						tag = RUS
						is_subject_of = RUS
					}
				}
			}
		}
		# A foreign victory leaves this country to its own war settlement.
		ROOT = { RUS_crisis_remove_defender = yes }
		RUS = { RUS_crisis_resolve_capitulation = yes }
	}
}

RUS_crisis_resolve_capitulation = {
	if = {
		limit = { check_variable = { var = RUS_crisis_phase value = 2 compare = equals } }
		if = {
			limit = {
				OR = {
					has_capitulated = yes
					has_country_flag = RUS_crisis_capitulation_current
				}
			}
			RUS_crisis_dissolve_empire = yes
		}
		else_if = {
			limit = { RUS_crisis_all_defenders_defeated = yes }
			RUS_crisis_annex_hegemony = yes
		}
	}
}

RUS_crisis_annex_hegemony = {
	set_variable = { var = RUS_crisis_phase value = 3 }
	# Preserve the full roster and territory before native peace merges war outcomes.
	clear_temp_array = RUS_crisis_conquests
	for_each_scope_loop = {
		array = global.RUS_crisis_defenders
		if = {
			limit = { NOT = { is_in_array = { array = global.RUS_crisis_conquered_countries value = THIS } } }
			add_to_array = { array = global.RUS_crisis_conquered_countries value = THIS }
		}
		every_owned_state = {
			add_to_temp_array = { array = RUS_crisis_conquests value = THIS }
			PREV = { add_to_array = { array = RUS_crisis_annexed_states value = PREV } }
		}
	}
	for_each_scope_loop = {
		array = global.RUS_crisis_defenders
		if = {
			limit = { has_war_with = RUS }
			RUS = { white_peace = PREV }
		}
		RUS = { annex_country = { target = PREV transfer_troops = no } }
	}
	for_each_scope_loop = {
		array = RUS_crisis_conquests
		set_state_flag = RUS_crisis_conquered_state
		set_state_owner_to = RUS
		set_state_controller_to = RUS
	}
	clear_temp_array = RUS_crisis_conquests
	RUS_crisis_clear_roster = yes
	ADISCORD_economy_mark_dirty = yes
	country_event = { id = ADISCORD_rus_crisis.5 }
}

# RUS. Earlier conquests retain their former owners within this campaign.
# Restore only land still held by the empire; foreign acquisitions stay outside.
RUS_crisis_restore_conquests = {
	for_each_scope_loop = {
		array = global.RUS_crisis_conquered_countries
		for_each_scope_loop = {
			array = RUS_crisis_annexed_states
			if = {
				limit = { is_owned_by = RUS }
				PREV = {
					if = {
						limit = { exists = no }
						set_country_flag = RUS_crisis_fragment
					}
				}
				set_state_owner_to = PREV
				set_state_controller_to = PREV
				remove_core_of = RUS
				remove_claim_by = RUS
				clr_state_flag = RUS_crisis_conquered_state
			}
		}
		clear_array = RUS_crisis_annexed_states
		RUS_crisis_prepare_restored_fragment = yes
	}
	clear_array = global.RUS_crisis_conquered_countries
}

# COUNTRY fragment. Existing independent neighbours do not enter this cleanup.
RUS_crisis_prepare_restored_fragment = {
	if = {
		limit = {
			exists = yes
			has_country_flag = RUS_crisis_fragment
		}
		if = {
			limit = { is_subject = yes }
			overlord = {
				set_autonomy = {
					target = PREV
					autonomy_state = autonomy_free
					end_wars = no
					end_civil_wars = no
				}
			}
		}
		if = {
			limit = { is_in_faction = yes }
			leave_faction = yes
		}
		ADISCORD_economy_mark_dirty = yes
	}
}

RUS_crisis_dissolve_empire = {
	set_country_flag = RUS_crisis_laser_disabled
	set_variable = { var = RUS_crisis_phase value = 4 }
	RUS_campaign_shutdown = yes
	# The credited victor receives the choice after the Khan's empire dissolves.
	event_target:RUS_crisis_hegemon = {
		set_country_flag = RUS_crisis_settlement_rights
		# The proclaimed imperial form selects the settlement narrative.
		set_variable = { var = RUS_crisis_defeated_course value = 1 }
		if = {
			limit = { RUS = { has_cosmetic_tag = RUS_black_banner_empire } }
			set_variable = { var = RUS_crisis_defeated_course value = 2 }
		}
		else_if = {
			limit = { RUS = { has_cosmetic_tag = RUS_restoration_state } }
			set_variable = { var = RUS_crisis_defeated_course value = 3 }
		}
		country_event = { id = ADISCORD_rus_crisis.4 }
		if = {
			limit = { has_global_flag = RUS_khan_died }
			RUS_show_khan_died_superevent = yes
		}
		else = {
			ADISCORD_vorkerland_show_khan_defeated_superevent = yes
		}
	}
	for_each_scope_loop = {
		array = global.RUS_crisis_defenders
		if = {
			limit = { has_war_with = RUS }
			white_peace = RUS
		}
	}
	every_enemy_country = { white_peace = RUS }
	every_subject_country = {
		if = {
			limit = { is_in_faction_with = RUS }
			leave_faction = yes
		}
		if = {
			limit = {
				OR = {
					tag = SLA
					tag = RZA
					tag = MLR
					tag = ERT
					tag = IRT
					tag = SCA
				}
			}
			set_country_flag = RUS_crisis_fragment
		}
		every_enemy_country = {
			limit = { has_country_flag = RUS_crisis_defender }
			white_peace = PREV
		}
		RUS = {
			set_autonomy = {
				target = PREV
				autonomy_state = autonomy_free
				end_wars = no
				end_civil_wars = no
			}
		}
	}
	RUS_retire_current_ruler = yes
	remove_ideas = RUS_last_empire_spirit
	remove_ideas = RUS_aimaq_compact
	clr_country_flag = ADISCORD_vorkerland_rus_sla_campaign_active
	ADISCORD_vorkerland_rus_abort_dirty_campaign = yes
	RUS_crisis_restore_conquests = yes
	# Snapshot owned territory before the last transfer removes the RUS country.
	clear_temp_array = RUS_crisis_imperial_states
	every_owned_state = {
		add_to_temp_array = { array = RUS_crisis_imperial_states value = THIS }
	}
	# Consume the defeated army before returning the captured state snapshot.
	# The temporary occupier gains no land outside that snapshot.
	event_target:RUS_crisis_hegemon = {
		annex_country = { target = RUS transfer_troops = no }
	}
	for_each_scope_loop = {
		array = RUS_crisis_imperial_states
		RUS_crisis_restore_fragment_state = yes
	}
	clear_temp_array = RUS_crisis_imperial_states

	SLA = { RUS_crisis_prepare_restored_fragment = yes }
	RZA = { RUS_crisis_prepare_restored_fragment = yes }
	MLR = { RUS_crisis_prepare_restored_fragment = yes }
	ERT = { RUS_crisis_prepare_restored_fragment = yes }
	IRT = { RUS_crisis_prepare_restored_fragment = yes }
	SCA = { RUS_crisis_prepare_restored_fragment = yes }
	TMR = { RUS_crisis_prepare_restored_fragment = yes }
	VEL = { RUS_crisis_prepare_restored_fragment = yes }
	RLY = { RUS_crisis_prepare_restored_fragment = yes }
	RUS_crisis_clear_roster = yes
}

# The empty authority tag carries ruined land, not a surviving government.
# Capture each queue once. Player handoff precedes all losses of territory;
# reading the ending never owns or cancels the scheduled destruction waves.
RUS_crisis_reduce_world_to_ruins = {
	125 = { RUS_crisis_ruin_state = yes }
	66 = {
		set_state_owner_to = RUS
		set_state_controller_to = RUS
		# The protected underground complex does not preserve a surface airfield.
		if = {
			limit = { air_base > 0 }
			set_building_level = { type = air_base level = 0 instant_build = yes }
		}
	}
	clear_array = global.RUS_world_country_queue
	clear_array = global.RUS_world_state_queue
	clear_array = global.RUS_world_frontier_queue
	clear_array = global.RUS_world_cleanup_queue
	every_country = {
		add_to_array = { array = global.RUS_world_country_queue value = THIS }
		if = {
			limit = {
				NOT = { tag = RUS }
				NOT = { tag = EXZ }
			}
			add_to_array = { array = global.RUS_world_cleanup_queue value = THIS }
			if = {
				limit = { is_ai = no }
				EXZ = { change_tag_from = PREV }
			}
		}
	}
	every_state = {
		limit = {
			NOT = { state = 66 }
			NOT = { state = 125 }
		}
		add_to_array = { array = global.RUS_world_state_queue value = THIS }
	}
	125 = { RUS_crisis_enqueue_ruin_neighbors = yes }
	EXZ = {
		set_cosmetic_tag = RUS_world_anarchy
		set_capital = { state = 125 }
		every_enemy_country = { white_peace = PREV }
		set_stability = 0
		set_war_support = 0
		set_political_power = 0
		add_ideas = RUS_world_ended_isolation
	}
	RUS = {
		set_cosmetic_tag = RUS_last_bunker
		set_capital = { state = 66 }
		every_enemy_country = { white_peace = PREV }
	}
}

# RUS. Each native daily callback drains fixed-size portions of persistent queues.
# The last queue removes countries only after all state transfers are complete,
# so annexing one large country cannot bypass the territorial wave budget.
RUS_crisis_ruin_next_wave = {
	if = {
		limit = {
			has_global_flag = RUS_crisis_world_ended
			NOT = { has_global_flag = RUS_crisis_world_ruins_complete }
		}
		if = {
			limit = { check_variable = { var = global.RUS_world_country_queue^num value = 0 compare = greater_than } }
			set_temp_variable = { var = RUS_world_next_country value = global.RUS_world_country_queue^0 }
			remove_from_array = { array = global.RUS_world_country_queue index = 0 }
			var:RUS_world_next_country = { RUS_crisis_destroy_country_assets = yes }
		}
		RUS_crisis_ruin_next_state = yes
		RUS_crisis_ruin_next_state = yes
		RUS_crisis_ruin_next_state = yes
		if = {
			limit = {
				check_variable = { var = global.RUS_world_state_queue^num value = 0 compare = equals }
				check_variable = { var = global.RUS_world_country_queue^num value = 0 compare = equals }
				check_variable = { var = global.RUS_world_cleanup_queue^num value = 0 compare = greater_than }
			}
			set_temp_variable = { var = RUS_world_next_country value = global.RUS_world_cleanup_queue^0 }
			remove_from_array = { array = global.RUS_world_cleanup_queue index = 0 }
			var:RUS_world_next_country = {
				if = {
					limit = {
						exists = yes
						NOT = { any_owned_state = { always = yes } }
					}
					EXZ = { annex_country = { target = PREV transfer_troops = no } }
				}
			}
		}
		if = {
			limit = {
				check_variable = { var = global.RUS_world_country_queue^num value = 0 compare = equals }
				check_variable = { var = global.RUS_world_state_queue^num value = 0 compare = equals }
				check_variable = { var = global.RUS_world_cleanup_queue^num value = 0 compare = equals }
			}
			set_global_flag = RUS_crisis_world_ruins_complete
			clear_array = global.RUS_world_country_queue
			clear_array = global.RUS_world_state_queue
			clear_array = global.RUS_world_frontier_queue
			clear_array = global.RUS_world_cleanup_queue
		}
	}
}

# FIFO neighbours spread outward from the reactor. A disconnected landmass
# starts only when the current frontier is exhausted. The bunker is never queued.
RUS_crisis_enqueue_ruin_neighbors = {
	every_neighbor_state = {
		limit = {
			is_in_array = { array = global.RUS_world_state_queue value = THIS }
			NOT = { is_in_array = { array = global.RUS_world_frontier_queue value = THIS } }
		}
		add_to_array = { array = global.RUS_world_frontier_queue value = THIS }
	}
}

# Extract before transferring, since native callbacks can change country existence.
RUS_crisis_ruin_next_state = {
	if = {
		limit = { check_variable = { var = global.RUS_world_state_queue^num value = 0 compare = greater_than } }
		if = {
			limit = { check_variable = { var = global.RUS_world_frontier_queue^num value = 0 compare = greater_than } }
			set_temp_variable = { var = RUS_world_next_state value = global.RUS_world_frontier_queue^0 }
			remove_from_array = { array = global.RUS_world_frontier_queue index = 0 }
		}
		else = {
			set_temp_variable = { var = RUS_world_next_state value = global.RUS_world_state_queue^0 }
		}
		var:RUS_world_next_state = {
			remove_from_array = { array = global.RUS_world_state_queue value = THIS }
			RUS_crisis_ruin_state = yes
			RUS_crisis_enqueue_ruin_neighbors = yes
		}
	}
}

# RUS. Each warning tier fires once. A full gauge waits for peace and the
# imperial territory, then replaces the 90-day mobilisation with an immediate war.
RUS_suspicion_check = {
	if = {
		limit = { RUS_suspicion_exposure_ready = yes }
		RUS_suspicion_expose = yes
	}
	else_if = {
		limit = {
			check_variable = { var = RUS_suspicion value = 75 compare = greater_than_or_equals }
			NOT = { check_variable = { var = RUS_suspicion_warning value = 2 compare = greater_than_or_equals } }
		}
		set_variable = { var = RUS_suspicion_warning value = 2 }
		country_event = { id = ADISCORD_rus_crisis.17 }
	}
	else_if = {
		limit = {
			check_variable = { var = RUS_suspicion value = 50 compare = greater_than_or_equals }
			NOT = { check_variable = { var = RUS_suspicion_warning value = 1 compare = greater_than_or_equals } }
		}
		set_variable = { var = RUS_suspicion_warning value = 1 }
		country_event = { id = ADISCORD_rus_crisis.17 }
	}
}

# RUS monthly. Keep these terms in sync with GetRUSMoraleStatus. The economy
# casualty delta is not reused: AI countries may refresh it only yearly.
RUS_morale_monthly = {
	if = {
		limit = { RUS_morale_active = yes }
		# Native casualties count only current wars; a smaller value after peace is no gain.
		set_temp_variable = { var = RUS_morale_losses_k value = casualties_k }
		subtract_from_temp_variable = { var = RUS_morale_losses_k value = RUS_morale_casualties_seen_k }
		set_variable = { var = RUS_morale_casualties_seen_k value = casualties_k }
		clamp_temp_variable = { var = RUS_morale_losses_k min = 0 max = 50 }
		set_temp_variable = { var = RUS_morale_delta value = RUS_morale_losses_k }
		if = {
			limit = { has_completed_focus = RUS_black_army_tribunals }
			divide_temp_variable = { var = RUS_morale_delta value = -10 }
		}
		else = {
			divide_temp_variable = { var = RUS_morale_delta value = -5 }
		}
		if = {
			limit = { has_war = no }
			add_to_temp_variable = { var = RUS_morale_delta value = 3 }
		}
		if = {
			limit = { has_completed_focus = RUS_imperial_field_kitchens }
			add_to_temp_variable = { var = RUS_morale_delta value = 2 }
		}
		if = {
			limit = { power_balance_value = { id = RUS_state_balance value > 0.60 } }
			add_to_temp_variable = { var = RUS_morale_delta value = 2 }
		}
		else_if = {
			limit = { power_balance_value = { id = RUS_state_balance value > 0.10 } }
			add_to_temp_variable = { var = RUS_morale_delta value = 1 }
		}
		RUS_morale_add = yes
	}
}

# RUS. A worse band announces its crisis once; recovery re-arms it silently.
# Death is settled by a delayed event outside the calling native callback.
RUS_morale_check = {
	set_temp_variable = { var = RUS_morale_band value = 0 }
	if = {
		limit = { check_variable = { var = RUS_army_morale value = 15 compare = less_than } }
		set_temp_variable = { var = RUS_morale_band value = 3 }
	}
	else_if = {
		limit = { check_variable = { var = RUS_army_morale value = 35 compare = less_than } }
		set_temp_variable = { var = RUS_morale_band value = 2 }
	}
	else_if = {
		limit = { check_variable = { var = RUS_army_morale value = 60 compare = less_than } }
		set_temp_variable = { var = RUS_morale_band value = 1 }
	}
	if = {
		limit = {
			check_variable = { var = RUS_army_morale value = 0 compare = less_than_or_equals }
			has_war = yes
		}
		country_event = { id = ADISCORD_rus_crisis.19 hours = 1 }
	}
	else_if = {
		limit = { check_variable = { var = RUS_morale_band value = RUS_morale_stage compare = greater_than } }
		set_variable = { var = RUS_morale_stage value = RUS_morale_band }
		country_event = { id = ADISCORD_rus_crisis.18 }
	}
	else = {
		# A band re-arms only once morale clears its threshold by ten points,
		# so a front swinging across one threshold does not repeat the crisis.
		set_temp_variable = { var = RUS_morale_rearm value = 0 }
		if = {
			limit = { check_variable = { var = RUS_army_morale value = 25 compare = less_than } }
			set_temp_variable = { var = RUS_morale_rearm value = 3 }
		}
		else_if = {
			limit = { check_variable = { var = RUS_army_morale value = 45 compare = less_than } }
			set_temp_variable = { var = RUS_morale_rearm value = 2 }
		}
		else_if = {
			limit = { check_variable = { var = RUS_army_morale value = 70 compare = less_than } }
			set_temp_variable = { var = RUS_morale_rearm value = 1 }
		}
		if = {
			limit = { check_variable = { var = RUS_morale_rearm value = RUS_morale_stage compare = less_than } }
			set_variable = { var = RUS_morale_stage value = RUS_morale_rearm }
		}
	}
}

# RUS. Credit the independent enemy holding the most imperial land, including
# its subjects' occupations; otherwise the Montar Empire or any enemy.
RUS_morale_select_victor = {
	clear_global_event_target = RUS_crisis_hegemon
	set_temp_variable = { var = RUS_morale_best_states value = 0 }
	every_enemy_country = {
		limit = {
			is_subject = no
			has_capitulated = no
		}
		set_temp_variable = { var = RUS_morale_held_states value = 0 }
		every_controlled_state = {
			limit = { is_owned_by = RUS }
			add_to_temp_variable = { var = RUS_morale_held_states value = 1 }
		}
		every_subject_country = {
			every_controlled_state = {
				limit = { is_owned_by = RUS }
				add_to_temp_variable = { var = RUS_morale_held_states value = 1 }
			}
		}
		if = {
			limit = { check_variable = { var = RUS_morale_held_states value = RUS_morale_best_states compare = greater_than } }
			set_temp_variable = { var = RUS_morale_best_states value = RUS_morale_held_states }
			save_global_event_target_as = RUS_crisis_hegemon
		}
	}
	if = {
		limit = { NOT = { has_event_target = RUS_crisis_hegemon } }
		if = {
			limit = {
				MON = {
					has_war_with = RUS
					is_subject = no
					has_capitulated = no
				}
			}
			MON = { save_global_event_target_as = RUS_crisis_hegemon }
		}
		else = {
			random_enemy_country = {
				limit = {
					is_subject = no
					has_capitulated = no
				}
				save_global_event_target_as = RUS_crisis_hegemon
			}
		}
	}
}

	if = {
		limit = {
			tag = RUS
			NOT = { has_capitulated = yes }
			NOT = { has_country_flag = ADISCORD_vorkerland_rus_last_empire_proclaimed }

			limit = { has_war_with = SLA }
			set_variable = { var = ADISCORD_vorkerland_rus_campaign_target value = 1 }
			if = {
				limit = { ADISCORD_vorkerland_rus_current_target_ready = yes }
				ADISCORD_vorkerland_rus_settle_dirty_target = yes
			}
		}
		if = {
			limit = { has_war_with = RZA }
			set_variable = { var = ADISCORD_vorkerland_rus_campaign_target value = 2 }
			if = {
				limit = { ADISCORD_vorkerland_rus_current_target_ready = yes }
				ADISCORD_vorkerland_rus_settle_dirty_target = yes
			}
		}
		if = {
			limit = { has_war_with = MLR }
			set_variable = { var = ADISCORD_vorkerland_rus_campaign_target value = 3 }
			if = {
				limit = { ADISCORD_vorkerland_rus_current_target_ready = yes }
				ADISCORD_vorkerland_rus_settle_dirty_target = yes
			}
		}
		if = {
			limit = { has_war_with = ERT }
			set_variable = { var = ADISCORD_vorkerland_rus_campaign_target value = 4 }
			if = {
				limit = { ADISCORD_vorkerland_rus_current_target_ready = yes }
				ADISCORD_vorkerland_rus_settle_dirty_target = yes
			}
		}
		if = {
			limit = { has_war_with = IRT }
			set_variable = { var = ADISCORD_vorkerland_rus_campaign_target value = 5 }
			if = {
				limit = { ADISCORD_vorkerland_rus_current_target_ready = yes }
				ADISCORD_vorkerland_rus_settle_dirty_target = yes
			}
		}

			set_variable = { var = ADISCORD_vorkerland_rus_campaign_target value = 6 }
			if = {
				limit = { ADISCORD_vorkerland_rus_current_target_ready = yes }

ADISCORD_vorkerland_rus_proclaim_last_empire = {
	if = {
		limit = {
			tag = RUS
			NOT = { has_country_flag = ADISCORD_vorkerland_rus_last_empire_proclaimed }
			ADISCORD_vorkerland_rus_has_empire_territory = yes
			RUS_khan_governing = yes
		}
		set_country_flag = ADISCORD_vorkerland_rus_last_empire_proclaimed
		clr_country_flag = ADISCORD_vorkerland_rus_sla_campaign_active
		clear_variable = ADISCORD_vorkerland_rus_campaign_target
		if = {
			limit = { has_idea = RUS_aimaq_compact }
			swap_ideas = { remove_idea = RUS_aimaq_compact add_idea = RUS_last_empire_spirit }
		}
		else_if = {
			limit = { NOT = { has_idea = RUS_last_empire_spirit } }
			add_ideas = RUS_last_empire_spirit
		}
		# The faction that outweighs the compromise range names the empire.
		if = {
			limit = { power_balance_value = { id = RUS_state_balance value > 0.10 } }
			set_cosmetic_tag = RUS_black_banner_empire
		}
		else_if = {
			limit = { power_balance_value = { id = RUS_state_balance value < -0.10 } }
			set_cosmetic_tag = RUS_restoration_state
		}
		else = {
			set_cosmetic_tag = RUS_last_empire
		}
		set_politics = {
			ruling_party = etatism
			elections_allowed = no
		}
		set_portraits = {
			character = RUS_Mark_Rustan
			civilian = { large = GFX_portrait_RUS_Mark_Rustan_dictator }
		}
		set_country_leader_portrait = {
			ideology = etatism
			portrait = GFX_portrait_RUS_Mark_Rustan_dictator
		}
		RUS_black_army_empire_reform = yes
		# Relyn must be able to prepare as soon as imperial frontier wars unlock.
		RLY = { ADISCORD_release_non_participating_minor_optimization = yes }
		ADISCORD_economy_mark_dirty = yes
		ADISCORD_vorkerland_show_last_empire_superevent = yes
		# Imperial focus branches use allow_branch on the proclamation flag.
		mark_focus_tree_layout_dirty = yes
		RUS_morale_begin = yes
		RUS_crisis_check_start = yes
	}
}

			RUS_bunker_excavate_affordable = yes
		}
		set_variable = { var = RUS_bunker_project value = 1 }
		set_variable = { var = RUS_bunker_project_depth value = RUS_bunker_depth }
		set_variable = { var = RUS_bunker_escrow_pp value = 20 }
		set_variable = { var = RUS_bunker_escrow_cash value = 50 }
		set_variable = { var = RUS_bunker_escrow_equipment value = 20 }
		add_political_power = -20
		subtract_from_variable = { var = ADISCORD_economy_treasury value = 50 }
		add_to_variable = { var = ADISCORD_economy_current_month_action_costs value = 50 }
		add_equipment_to_stockpile = { type = support_equipment amount = -20 }
		ADISCORD_economy_mark_dirty = yes
	}
}

# that faction; support below 50% costs political power without a pull.

# STATE scope. Ownership is checked again at delivery.
RUS_reactor_works_finish_state = {
	if = {
		limit = { has_state_flag = RUS_reactor_works_in_progress }
		if = {
			limit = {
				RUS_reactor_works_target_valid = yes
				RUS = { RUS_khan_governing = yes }
			}
			clr_state_flag = RUS_reactor_works_in_progress
			RUS = {
				clear_variable = RUS_reactor_works_deposit
				ADISCORD_economy_mark_dirty = yes
			}
			set_state_flag = RUS_reactor_works_restored
			RUS_reactor_works_restore_resources = yes
			add_extra_state_shared_building_slots = 1
			add_building_construction = {
				type = industrial_complex
				level = 1
				instant_build = yes
			}
			if = {
				limit = { RUS = { has_completed_focus = RUS_reactor_foundries } }
				RUS_reactor_foundry_expand = yes
			}
		}
		else = {
			RUS = { RUS_reactor_works_refund = yes }
		}
	}
}

		set_temp_variable = { var = RUS_lab_refund_cash value = RUS_lab_escrow_cash }

RUS_bunker_weekly_supply = {
	if = {
		limit = {
			has_country_flag = RUS_bunker_initialized
		}
		if = {
			limit = { RUS_bunker_operational = yes }
			# A returning country needs current production before this week's delivery.
			if = {
				limit = { NOT = { has_dynamic_modifier = { modifier = RUS_bunker_complex } } }
				RUS_bunker_refresh = yes
			}
			RUS_bunker_food_pressure = yes
			add_to_variable = { var = RUS_bunker_food value = RUS_bunker_food_balance }
			clamp_variable = { var = RUS_bunker_food min = 0 max = 100000 }
			if = {
				limit = { check_variable = { var = RUS_bunker_food value = RUS_bunker_food_capacity compare = greater_than } }
				set_variable = { var = RUS_bunker_food value = RUS_bunker_food_capacity }
			}
			set_variable = { var = RUS_bunker_food_space value = RUS_bunker_food_capacity }
			subtract_from_variable = { var = RUS_bunker_food_space value = RUS_bunker_food }
			# Stock changes affect national bonuses only when the ration state changes.
			RUS_bunker_classify_food = yes
			if = {
				limit = { NOT = { check_variable = { var = RUS_bunker_food_state value = RUS_bunker_food_state_now compare = equals } } }
				RUS_bunker_refresh = yes
			}
			if = {
				limit = { check_variable = { var = RUS_bunker_layer_3 value = 1 compare = equals } }
				add_equipment_to_stockpile = { type = infantry_equipment_0 amount = 36 producer = RUS }
				add_equipment_to_stockpile = { type = support_equipment amount = 1 producer = RUS }
			}
			if = {
				limit = { check_variable = { var = RUS_bunker_layer_3 value = 2 compare = equals } }
				add_equipment_to_stockpile = { type = infantry_equipment_0 amount = 90 producer = RUS }
				add_equipment_to_stockpile = { type = support_equipment amount = 2 producer = RUS }
			}
			if = {
				limit = { has_completed_focus = RUS_anti_air_posts }
				add_equipment_to_stockpile = { type = ADISCORD_anti_air_equipment_2163 amount = 2 producer = RUS }
			}
		}
		else_if = {
			# Ownership and ruler changes need bounded cleanup even without a control callback.
			limit = {
				OR = {
					has_dynamic_modifier = { modifier = RUS_bunker_complex }
					has_variable = RUS_bunker_project
					check_variable = { var = RUS_bunker_food_output value = 0 compare = greater_than }
					check_variable = { var = RUS_bunker_food_use value = 0 compare = greater_than }
				}
			}
			RUS_bunker_refresh = yes
		}
	}
}

	}

RUS_work_order_clear = {
	clear_variable = RUS_work_order
	clear_variable = RUS_work_order_pp
	clear_variable = RUS_work_order_cash
	clear_variable = RUS_work_order_equipment
}

RUS_work_order_refund = {
	if = {
		limit = { has_variable = RUS_work_order }
		set_temp_variable = { var = RUS_order_refund_pp value = RUS_work_order_pp }
		set_temp_variable = { var = RUS_order_refund_cash value = RUS_work_order_cash }
		set_temp_variable = { var = RUS_order_refund_equipment value = RUS_work_order_equipment }
		RUS_work_order_clear = yes
		remove_decision = RUS_civil_factory_order
		remove_decision = RUS_artillery_order
		add_political_power = RUS_order_refund_pp
		add_to_variable = { var = ADISCORD_economy_treasury value = RUS_order_refund_cash }
		add_to_variable = { var = ADISCORD_economy_current_month_action_income value = RUS_order_refund_cash }
		add_equipment_to_stockpile = { type = support_equipment amount = RUS_order_refund_equipment producer = RUS }
		ADISCORD_economy_mark_dirty = yes
	}
}

# Offer state: 1 = pending, 2 = paid, 3 = refused. The ending reads settled
# RUS outcomes; all other shutdowns discard them and pending offers always close.
RUS_last_sky_clear_offers = {
	if = {
		limit = {
			OR = {
				NOT = { tag = RUS }
				NOT = { has_global_flag = RUS_crisis_world_ended }
				check_variable = { var = RUS_last_sky_science value = 2 compare = less_than }
			}
		}
		clear_variable = RUS_last_sky_science
	}
	if = {
		limit = {
			OR = {
				NOT = { tag = RUS }
				NOT = { has_global_flag = RUS_crisis_world_ended }
				check_variable = { var = RUS_last_sky_army value = 2 compare = less_than }
			}
		}
		clear_variable = RUS_last_sky_army
	}
	if = {
		limit = {
			OR = {
				NOT = { tag = RUS }
				NOT = { has_global_flag = RUS_crisis_world_ended }
				check_variable = { var = RUS_last_sky_districts value = 2 compare = less_than }
			}
		}
		clear_variable = RUS_last_sky_districts
	}
}
