ADISCORD_vorkerland_prepare_conflict_country = {
	# Do not run an unscoped spirit scrub here. Most old autonomies had only the
	# shared WRK crisis spirit, so the former blanket cleanup left their national
	# spirit row empty and made a local transition look like a world-wide wipe.
	# Every removal below is tied to a tag that actually starts with that idea.
	if = {
		limit = { tag = WRK }
		remove_ideas = WRK_ashes_of_the_crown
		remove_ideas = WRK_hourglass_of_discord
		remove_ideas = WRK_constitution_of_the_republic
		remove_ideas = ADISCORD_vorkerland_erased_nations
		if = {
			limit = {
				OR = {
					has_country_leader_ideology = vorkerism
					has_country_leader_ideology = neo_vorkerism
				}
				NOT = {
					OR = {
						has_idea = WRK_birthplace_of_the_first_revolution
						has_idea = WRK_birthplace_of_the_first_revolution_front_republic
						has_idea = WRK_birthplace_of_the_first_revolution_renewed_mandate
					}
				}
			}
			add_ideas = WRK_birthplace_of_the_first_revolution
		}
	}
	else_if = {
		limit = { tag = VAD }
		remove_ideas = WRK_ashes_of_the_crown
		remove_ideas = WRK_hourglass_of_discord
		remove_ideas = WRK_constitution_of_the_republic
	}
	else_if = {
		limit = { tag = VLA }
		remove_ideas = WRK_hourglass_of_discord
		remove_ideas = VLA_national_spirit
		add_ideas = ADISCORD_vorkerland_republics_from_the_ruins
	}
	else_if = {
		limit = { OR = { tag = ZAO tag = PWR tag = ROM tag = SOL tag = TRU } }
		remove_ideas = WRK_hourglass_of_discord
		add_ideas = ADISCORD_vorkerland_republics_from_the_ruins
	}
	else_if = {
		limit = { tag = TVA }
		add_ideas = ADISCORD_vorkerland_mobilized_periphery
		add_ideas = ADISCORD_vorkerland_tva_field_directorate
		add_ideas = ADISCORD_vorkerland_tva_ideological_fanaticism
	}
	else_if = {
		limit = { tag = WTD }
		add_ideas = ADISCORD_vorkerland_mobilized_periphery
		add_ideas = ADISCORD_vorkerland_wtd_worx_protocols
	}
	else_if = { limit = { tag = WPA } add_ideas = ADISCORD_vorkerland_wpa_municipal_compact }
	else_if = { limit = { tag = WPS } add_ideas = ADISCORD_vorkerland_wps_factory_councils }
	else_if = { limit = { tag = EBA } add_ideas = ADISCORD_vorkerland_eba_free_quays }
	else_if = { limit = { tag = ZTA } add_ideas = ADISCORD_vorkerland_zta_golden_river_order }
	else_if = { limit = { tag = TGD } add_ideas = ADISCORD_vorkerland_tgd_grid_collapse }
	else_if = {
		limit = { tag = CSL }
		add_ideas = ADISCORD_vorkerland_mobilized_periphery
		add_ideas = ADISCORD_vorkerland_csl_isolated_city
	}
	else_if = {
		limit = { OR = { tag = EYR tag = EGC tag = RIV tag = REV tag = YOR tag = NDN tag = SWB tag = VHV tag = OSV tag = PSD tag = DVA tag = SRA tag = IBL tag = IBA } }
		add_ideas = ADISCORD_vorkerland_mobilized_periphery
	}
	# Like TFR's common Volunteer Militia, this thin locked template exists so a
	# minor can turn one political mobilisation into frontage without acquiring
	# a duplicate long-term force-design track.
	if = {
		limit = { ADISCORD_vorkerland_is_minor_combatant = yes }
		division_template = {
			name = "Emergency Militia"
			is_locked = yes
			force_allow_recruiting = yes
			regiments = {
				ADISCORD_militia = { x = 0 y = 0 }
				ADISCORD_militia = { x = 0 y = 1 }
				ADISCORD_militia = { x = 0 y = 2 }
			}
		}
	}
	# Every collapse participant receives cores on the territory assigned during
	# its one-time setup. This also covers legacy tags whose old history files do
	# not know about the post-collapse partition.
	every_owned_state = { add_core_of = ROOT }
}

# 12-15 formation main claimant. PREV is the country owning the capital scope.

ADISCORD_vorkerland_wrk_activate_vla_auxiliaries = {
	if = {
		limit = { tag = WRK VLA = { exists = yes is_subject_of = ROOT } }
		# A finite contract delivery represents Volnograd opening its arsenals.
		if = {
			limit = { VLA = { NOT = { has_country_flag = ADISCORD_vorkerland_auxiliary_contract_activated } } }
			add_equipment_to_stockpile = { type = infantry_equipment_0 amount = 300 producer = VLA }
			VLA = { set_country_flag = ADISCORD_vorkerland_auxiliary_contract_activated }
		}
		VLA = {
			ADISCORD_vorkerland_sync_republics_from_ruins = yes
			if = {
				limit = { WRK = { has_war_with = VAD } NOT = { has_war_with = VAD } }

			}
			if = {
				limit = { WRK = { has_war_with = TVA } NOT = { has_war_with = TVA } }
				add_to_war = { targeted_alliance = WRK enemy = TVA hostility_reason = asked_to_join }
			}
		}
	}
}

# A released country may keep the old confederation faction in the diplomatic
# cache even after its autonomy changed. Civil-war declarations against a
# country in that faction are rejected by the engine, so each war launcher uses
# these focused detach effects immediately before it opens its fronts.
ADISCORD_vorkerland_leave_inherited_faction = {
	if = {
		limit = { is_subject = no is_faction_leader = yes }
		dismantle_faction = yes
	}
	else_if = {
		limit = { is_subject = no is_in_faction = yes }
		leave_faction = yes

}

	set_global_flag = ADISCORD_vorkerland_northern_wars_began
	clr_global_flag = ADISCORD_vorkerland_regional_war_launch_scheduled

		limit = {
			RUS = { exists = yes }

			NOT = { has_global_flag = ADISCORD_vorkerland_ivanland_intervention_available }
		}
		RUS = { country_event = { id = ADISCORD_vorkerland_collapse.30 days = 30 random_days = 60 } }
	}
}

	clr_global_flag = ADISCORD_vorkerland_restore_zao_wpa
	clr_global_flag = ADISCORD_vorkerland_restore_zao_wps
	clr_global_flag = ADISCORD_vorkerland_restore_zao_psd
	clr_global_flag = ADISCORD_vorkerland_restore_zao_pwr
	clr_global_flag = ADISCORD_vorkerland_restore_wpa_psd
	clr_global_flag = ADISCORD_vorkerland_restore_wpa_pwr
	clr_global_flag = ADISCORD_vorkerland_restore_wps_psd
	clr_global_flag = ADISCORD_vorkerland_restore_wps_pwr
	clr_global_flag = ADISCORD_vorkerland_northern_wars_restored
	if = { limit = { ZAO = { has_war_with = WPA } } set_global_flag = ADISCORD_vorkerland_restore_zao_wpa }
	if = { limit = { ZAO = { has_war_with = WPS } } set_global_flag = ADISCORD_vorkerland_restore_zao_wps }
	if = { limit = { ZAO = { has_war_with = PSD } } set_global_flag = ADISCORD_vorkerland_restore_zao_psd }
	if = { limit = { ZAO = { has_war_with = PWR } } set_global_flag = ADISCORD_vorkerland_restore_zao_pwr }
	if = { limit = { WPA = { has_war_with = PSD } } set_global_flag = ADISCORD_vorkerland_restore_wpa_psd }
	if = { limit = { WPA = { has_war_with = PWR } } set_global_flag = ADISCORD_vorkerland_restore_wpa_pwr }
	if = { limit = { WPS = { has_war_with = PSD } } set_global_flag = ADISCORD_vorkerland_restore_wps_psd }
	if = { limit = { WPS = { has_war_with = PWR } } set_global_flag = ADISCORD_vorkerland_restore_wps_pwr }

	ZAO = {
		if = { limit = { country_exists = WPA has_war_with = WPA } white_peace = WPA }
		if = { limit = { country_exists = WPS has_war_with = WPS } white_peace = WPS }
		if = { limit = { country_exists = PSD has_war_with = PSD } white_peace = PSD }
		if = { limit = { country_exists = PWR has_war_with = PWR } white_peace = PWR }
	}
	WPA = {
		if = { limit = { country_exists = PSD has_war_with = PSD } white_peace = PSD }
		if = { limit = { country_exists = PWR has_war_with = PWR } white_peace = PWR }
	}
	WPS = {
		if = { limit = { country_exists = PSD has_war_with = PSD } white_peace = PSD }
		if = { limit = { country_exists = PWR has_war_with = PWR } white_peace = PWR }
	}

	# Krait accepts protection at the start of the operation. It is not part of
	# the northern defence front and therefore never fights its future overlord.
	IVN = {
		puppet = IBL
		set_autonomy = { target = IBL autonomy_state = autonomy_puppet freedom_level = 0.10 }
		if = {
			limit = { NOT = { has_country_flag = ADISCORD_vorkerland_ivanland_expedition_supplied } }
			add_manpower = 8000
			add_equipment_to_stockpile = { type = infantry_equipment_0 amount = 1600 producer = IVN }
			add_equipment_to_stockpile = { type = support_equipment amount = 120 producer = IVN }
			add_equipment_to_stockpile = { type = artillery_equipment amount = 48 producer = IVN }
			add_fuel = 5000
			set_country_flag = ADISCORD_vorkerland_ivanland_expedition_supplied
		}
	}
	IBL = {
		set_country_flag = ADISCORD_vorkerland_ivanland_client
		ADISCORD_vorkerland_appoint_selevyostrov = yes
		country_event = { id = ADISCORD_vorkerland_collapse.42 days = 1 }
	}

	PWR = {
		create_faction_from_template = {
			template = faction_template_ADISCORD_standard
			name = faction_vorkerland_northern_defense_front
		}
		add_to_faction = ZAO
		add_to_faction = WPA
		add_to_faction = WPS
		add_to_faction = PSD
	}

	ZAO = { add_manpower = 1000 add_equipment_to_stockpile = { type = infantry_equipment_0 amount = 250 producer = ZAO } add_timed_idea = { idea = ADISCORD_vorkerland_northern_defense_front days = 260 } }
	WPA = { add_manpower = 1000 add_equipment_to_stockpile = { type = infantry_equipment_0 amount = 250 producer = WPA } add_timed_idea = { idea = ADISCORD_vorkerland_northern_defense_front days = 260 } }
	WPS = { add_manpower = 1000 add_equipment_to_stockpile = { type = infantry_equipment_0 amount = 250 producer = WPS } add_timed_idea = { idea = ADISCORD_vorkerland_northern_defense_front days = 260 } }
	PWR = { add_manpower = 1000 add_equipment_to_stockpile = { type = infantry_equipment_0 amount = 250 producer = PWR } add_timed_idea = { idea = ADISCORD_vorkerland_northern_defense_front days = 260 } }
	PSD = { add_manpower = 1000 add_equipment_to_stockpile = { type = infantry_equipment_0 amount = 250 producer = PSD } add_timed_idea = { idea = ADISCORD_vorkerland_northern_defense_front days = 260 } }

	IVN = { declare_war_on = { target = PWR type = take_state_focus generator = { 90 } } }
	ZAO = { if = { limit = { NOT = { has_war_with = IVN } } add_to_war = { targeted_alliance = PWR enemy = IVN hostility_reason = asked_to_join } } }
	WPA = { if = { limit = { NOT = { has_war_with = IVN } } add_to_war = { targeted_alliance = PWR enemy = IVN hostility_reason = asked_to_join } } }
	WPS = { if = { limit = { NOT = { has_war_with = IVN } } add_to_war = { targeted_alliance = PWR enemy = IVN hostility_reason = asked_to_join } } }
	PSD = { if = { limit = { NOT = { has_war_with = IVN } } add_to_war = { targeted_alliance = PWR enemy = IVN hostility_reason = asked_to_join } } }
}

ADISCORD_vorkerland_appoint_mateusk = {
	set_politics = { ruling_party = pragmatism elections_allowed = yes }
	promote_character = {
		character = IBA_Matvey_Mateusk
		ideology = pragmatism_ideology
	}
	set_country_leader_portrait = {
		ideology = pragmatism
		portrait = GFX_portrait_IBA_Matvey_Mateusk
	}
	set_country_flag = ADISCORD_vorkerland_mateusk_character_repair_v2
}

# Krait must receive its new land before state 91 changes hands. Otherwise the
# tag becomes landless and may disappear before the second Ivanland client is
# created.
ADISCORD_vorkerland_expand_krait_client = {
	if = {
		limit = { country_exists = IBL }
		IBL = {
			transfer_state = 93
			transfer_state = 94
			set_capital = { state = 93 }
		}
		93 = { add_core_of = IBL add_claim_by = IBL set_state_controller_to = IBL }
		94 = { add_core_of = IBL add_claim_by = IBL set_state_controller_to = IBL }
		IBL = {
			ADISCORD_vorkerland_appoint_selevyostrov = yes
			set_country_flag = ADISCORD_vorkerland_ivanland_client
			add_manpower = 1500
			add_equipment_to_stockpile = { type = infantry_equipment_0 amount = 350 producer = IVN }
		}
		IVN = {
			puppet = IBL
			set_autonomy = { target = IBL autonomy_state = autonomy_puppet freedom_level = 0.10 }
		}
		IBL = {
			ADISCORD_vorkerland_appoint_selevyostrov = yes
			country_event = { id = ADISCORD_vorkerland_collapse.42 days = 1 }
		}
		set_global_flag = ADISCORD_vorkerland_ivanland_krait_frontier_secured
	}
}

# The final settlement leaves Ivanland with two separate clients: Krait in
# states 93-94 and Mateusk's Norvane in states 90-91.
ADISCORD_vorkerland_setup_ivanland_mandate = {
	ADISCORD_vorkerland_expand_krait_client = yes
	IBA = {
		transfer_state = 90
		transfer_state = 91
	}
	90 = { add_core_of = IBA add_claim_by = IBA set_state_controller_to = IBA }
	91 = { remove_core_of = IBL add_core_of = IBA add_claim_by = IBA set_state_controller_to = IBA }
	71 = { add_claim_by = IBA }
	IBA = {
		set_capital = { state = 90 }
		if = {
			limit = { NOT = { has_country_flag = ADISCORD_vorkerland_iba_initialized } }
			ADISCORD_grant_2150_technology_baseline = yes
			ADISCORD_grant_technology_profile_fragment_low_tech = yes
			ADISCORD_economy_initialize_country = yes
			ADISCORD_vorkerland_prepare_conflict_country = yes
			load_oob = "IBA_vorkerland_collapse"
			add_manpower = 3500
			add_equipment_to_stockpile = { type = infantry_equipment_0 amount = 700 producer = IVN }
			set_country_flag = ADISCORD_vorkerland_iba_initialized
		}
	}
	IVN = {
		if = {
			limit = { IBA = { NOT = { is_subject_of = IVN } } }
			puppet = IBA
			set_autonomy = { target = IBA autonomy_state = autonomy_puppet freedom_level = 0.10 }
		}
	}
	# Puppeting may rebuild the new country's politics and replace an already
	# promoted leader with a generated one. Appoint Mateusk only after that
	# rebuild, then make one deferred assertion on the following day.
	IBA = {
		ADISCORD_vorkerland_appoint_mateusk = yes
		country_event = { id = ADISCORD_vorkerland_collapse.40 days = 1 }
	}
}

# A successful Ivanland operation also secures Zaozersk. When the republic has
# already been eliminated, state 72 is detached from its current holder and the
# historical tag is released there before the subject relationship is applied.
ADISCORD_vorkerland_ivanland_secure_zaozersk = {
	if = {
		limit = { NOT = { has_global_flag = ADISCORD_vorkerland_ivanland_zaozersk_secured } }
		if = {
			limit = { NOT = { country_exists = ZAO } }
			IVN = { transfer_state = 72 }
			72 = { add_core_of = ZAO set_state_controller_to = IVN }
			IVN = {
				release_autonomy = {
					target = ZAO
					autonomy_state = autonomy_puppet
					freedom_level = 0.10
				}
			}
		}
		else = {
			ZAO = {
				if = { limit = { country_exists = WPA has_war_with = WPA } white_peace = WPA }
				if = { limit = { country_exists = WPS has_war_with = WPS } white_peace = WPS }
				if = { limit = { country_exists = PSD has_war_with = PSD } white_peace = PSD }
				if = { limit = { country_exists = PWR has_war_with = PWR } white_peace = PWR }
				if = { limit = { country_exists = ROM has_war_with = ROM } white_peace = ROM }
				if = {
					limit = { is_subject = yes }
					overlord = { set_autonomy = { target = ZAO autonomy_state = autonomy_free } }
				}
				if = { limit = { is_in_faction = yes } leave_faction = yes }
			}
			IVN = {
				puppet = ZAO
				set_autonomy = { target = ZAO autonomy_state = autonomy_puppet freedom_level = 0.10 }
			}
		}
		ZAO = {
			ADISCORD_vorkerland_sync_independence_cosmetic = yes
			ADISCORD_vorkerland_sync_republics_from_ruins = yes
			set_country_flag = ADISCORD_vorkerland_ivanland_protectorate
		}
		set_global_flag = ADISCORD_vorkerland_ivanland_zaozersk_secured
	}
}

# Only Workerland's own victory can restore its former republic autonomies.
# The two rolls are independent. Ivanland's timely guarantee suppresses this
# entire settlement and leaves both republics sovereign.
ADISCORD_vorkerland_resolve_unguaranteed_free_republics = {
	if = {
		limit = { NOT = { has_global_flag = ADISCORD_vorkerland_free_republics_resolution_done } }
		if = {
			limit = { NOT = { has_global_flag = ADISCORD_vorkerland_free_republics_guaranteed_by_ivn } }
			WRK = {
				if = {
					limit = { ROM = { exists = yes is_subject = no } }
					random_list = {
						50 = {
							puppet = ROM
							set_autonomy = { target = ROM autonomy_state = autonomy_republic_in_Vorkerland freedom_level = 0.10 }
							ROM = { ADISCORD_vorkerland_sync_republics_from_ruins = yes }
						}
						50 = {
							if = { limit = { NOT = { has_war_with = ROM } } declare_war_on = { target = ROM type = annex_everything } }
						}
					}
				}
				if = {
					limit = { TRU = { exists = yes is_subject = no } }
					random_list = {
						50 = {
							puppet = TRU
							set_autonomy = { target = TRU autonomy_state = autonomy_republic_in_Vorkerland freedom_level = 0.10 }
							TRU = { ADISCORD_vorkerland_sync_republics_from_ruins = yes }
						}
						50 = {
							if = { limit = { NOT = { has_war_with = TRU } } declare_war_on = { target = TRU type = annex_everything } }
						}
					}
				}
			}
		}
		set_global_flag = ADISCORD_vorkerland_free_republics_resolution_done
	}
}

ADISCORD_vorkerland_ivanland_intervention_success = {
	if = {
		limit = { NOT = { has_global_flag = ADISCORD_vorkerland_ivanland_intervention_resolved } }
		set_global_flag = ADISCORD_vorkerland_ivanland_intervention_resolved
		set_global_flag = ADISCORD_vorkerland_ivanland_intervened
		set_global_flag = ADISCORD_vorkerland_ivanland_intervention_succeeded
		set_global_flag = ADISCORD_vorkerland_ivanland_mandate_proclaimed
		clr_global_flag = ADISCORD_vorkerland_ivanland_intervention_failed
		clr_global_flag = ADISCORD_vorkerland_ivanland_intervention_active
		log = "[ADISCORD][VORKERLAND] Ivanland intervention resolved: SUCCESS; IBL receives states 93-94 and IBA receives states 90-91."
		ADISCORD_vorkerland_end_ivanland_intervention_wars = yes
		ADISCORD_vorkerland_setup_ivanland_mandate = yes
		ADISCORD_vorkerland_ivanland_secure_zaozersk = yes
		ADISCORD_vorkerland_end_ivanland_intervention_wars = yes
		country_event = { id = ADISCORD_vorkerland_collapse.43 days = 2 }
		if = {
			limit = { NOT = { has_global_flag = ADISCORD_vorkerland_ivanland_success_news_shown } }
			set_global_flag = ADISCORD_vorkerland_ivanland_success_news_shown
			news_event = { id = ADISCORD_vorkerland_news.1 }
		}
	}
	else_if = {
		limit = { has_global_flag = ADISCORD_vorkerland_ivanland_intervention_succeeded }
		ADISCORD_vorkerland_end_ivanland_intervention_wars = yes
	}
}

ADISCORD_vorkerland_ivanland_intervention_failure = {
	if = {
		limit = { NOT = { has_global_flag = ADISCORD_vorkerland_ivanland_intervention_resolved } }
		set_global_flag = ADISCORD_vorkerland_ivanland_intervention_resolved
		set_global_flag = ADISCORD_vorkerland_ivanland_intervention_failed
		clr_global_flag = ADISCORD_vorkerland_ivanland_intervention_succeeded
		clr_global_flag = ADISCORD_vorkerland_ivanland_intervention_active
		clr_global_flag = ADISCORD_vorkerland_ivanland_mandate_proclaimed
		log = "[ADISCORD][VORKERLAND] Ivanland intervention resolved: FAILURE; IBA is dissolved and the pre-settlement borders are restored."
		ADISCORD_vorkerland_end_ivanland_intervention_wars = yes
		if = {
			limit = { country_exists = IBA }
			# Annex first, then restore both clients' pre-settlement land. This is
			# primarily a safeguard for a save made during the proclamation tick.
			IVN = { annex_country = { target = IBA transfer_troops = yes } }
			90 = { remove_core_of = IBA }
			91 = { remove_core_of = IBA }
			if = {
				limit = { country_exists = PWR }
				PWR = { transfer_state = 90 }
				90 = { set_state_controller_to = PWR }
			}
			if = {
				limit = { country_exists = IBL }
				IBL = {
					transfer_state = 91
					set_capital = { state = 91 }
				}
				91 = { add_core_of = IBL set_state_controller_to = IBL }
			}
			if = {
				limit = {
					has_global_flag = ADISCORD_vorkerland_ivanland_krait_frontier_secured
					country_exists = PSD
				}
				PSD = {
					transfer_state = 93
					transfer_state = 94
				}
				93 = { remove_core_of = IBL set_state_controller_to = PSD }
				94 = { remove_core_of = IBL set_state_controller_to = PSD }
			}
		}
		clr_global_flag = ADISCORD_vorkerland_ivanland_krait_frontier_secured
		ADISCORD_vorkerland_end_ivanland_intervention_wars = yes
		country_event = { id = ADISCORD_vorkerland_collapse.43 days = 2 }
		IVN = {
			set_politics = {
				ruling_party = etatism
				elections_allowed = no
			}
			if = {
				limit = { NOT = { has_country_flag = ADISCORD_vorkerland_vadim_etatist_role_added } }
				add_country_leader_role = {
					character = IVN_Vadim_Ivanchik
					promote_leader = yes
					country_leader = {
						ideology = etatism_ideology
						expire = "2200.1.1.1"
					}
				}
				set_country_flag = ADISCORD_vorkerland_vadim_etatist_role_added
			}
			set_country_leader_portrait = {
				ideology = etatism
				portrait = GFX_portrait_IVN_Vadim_Ivanchik_after_retreat
			}
		}
		if = {
			limit = { NOT = { has_global_flag = ADISCORD_vorkerland_ivanland_failure_news_shown } }
			set_global_flag = ADISCORD_vorkerland_ivanland_failure_news_shown
			news_event = { id = ADISCORD_vorkerland_news.2 }
		}
	}
}

# Independent republic cosmetics must never survive a later dependency. This
# shared synchronizer is called at collapse setup and by autonomy on-actions.
ADISCORD_vorkerland_sync_independence_cosmetic = {
	if = {
		limit = {
			has_global_flag = ADISCORD_vorkerland_collapse_started
			OR = { tag = ROM tag = TRU tag = ZAO }
		}
		if = {
			limit = { is_subject = yes }
			drop_cosmetic_tag = yes
		}
		else_if = {
			limit = { tag = ROM }
			set_cosmetic_tag = ROM_frealor_republic
		}
		else_if = {
			limit = { tag = TRU }
			set_cosmetic_tag = TRU_zolotorevsk_republic
		}
		else = { set_cosmetic_tag = ZAO_zaozersk_republic }
	}
}

# Country-scoped monthly exhaustion. WRK, VAD and TVA call this separately, so the
# shared variable names remain independent and each side pays only its losses.
ADISCORD_vorkerland_initialize_civil_war_exhaustion = {
	set_variable = { var = ADISCORD_vorkerland_civil_war_exhaustion value = 1 }
	set_variable = { var = ADISCORD_vorkerland_civil_war_casualties_delta_k value = 0 }
	set_variable = { var = ADISCORD_vorkerland_civil_war_casualties_snapshot_k value = casualties_k }
	ADISCORD_vorkerland_refresh_civil_war_exhaustion = yes
}

ADISCORD_vorkerland_update_civil_war_exhaustion = {
	if = {
		limit = {
			NOT = { has_variable = ADISCORD_vorkerland_civil_war_casualties_snapshot_k }
		}
		set_variable = {
			var = ADISCORD_vorkerland_civil_war_casualties_snapshot_k
			value = casualties_k
		}
	}

	set_variable = {
		var = ADISCORD_vorkerland_civil_war_casualties_delta_k
		value = casualties_k
	}
	subtract_from_variable = {
		var = ADISCORD_vorkerland_civil_war_casualties_delta_k
		value = ADISCORD_vorkerland_civil_war_casualties_snapshot_k
	}
	clamp_variable = {
		var = ADISCORD_vorkerland_civil_war_casualties_delta_k
		min = 0
		max = 10000
	}
	set_variable = {
		var = ADISCORD_vorkerland_civil_war_casualties_snapshot_k
		value = casualties_k
	}

	if = {
		limit = {
			NOT = { has_global_flag = ADISCORD_vorkerland_central_war_finished }
			has_war = yes
		}
		add_to_variable = {
			var = ADISCORD_vorkerland_civil_war_exhaustion
			value = 2
		}
		if = {
			limit = {
				check_variable = {
					var = ADISCORD_vorkerland_civil_war_casualties_delta_k
					value = 100
					compare = greater_than_or_equals
				}
			}
			add_to_variable = {
				var = ADISCORD_vorkerland_civil_war_exhaustion
				value = 6
			}
		}
		else_if = {
			limit = {
				check_variable = {
					var = ADISCORD_vorkerland_civil_war_casualties_delta_k
					value = 25
					compare = greater_than_or_equals
				}
			}
			add_to_variable = {
				var = ADISCORD_vorkerland_civil_war_exhaustion
				value = 3
			}
		}
		else_if = {
			limit = {
				check_variable = {
					var = ADISCORD_vorkerland_civil_war_casualties_delta_k
					value = 5
					compare = greater_than_or_equals
				}
			}
			add_to_variable = {
				var = ADISCORD_vorkerland_civil_war_exhaustion
				value = 1
			}
		}
	}
	clamp_variable = {
		var = ADISCORD_vorkerland_civil_war_exhaustion
		min = 1
		max = 100
	}
	ADISCORD_vorkerland_refresh_civil_war_exhaustion = yes
}

ADISCORD_vorkerland_refresh_civil_war_exhaustion = {
	set_variable = { var = ADISCORD_vorkerland_civil_war_exhaustion_level value = 0 }
	if = {
		limit = {
			check_variable = {
				var = ADISCORD_vorkerland_civil_war_exhaustion
				value = 80
				compare = greater_than_or_equals
			}
		}
		set_variable = { var = ADISCORD_vorkerland_civil_war_exhaustion_level value = 4 }
	}
	else_if = {
		limit = {
			check_variable = {
				var = ADISCORD_vorkerland_civil_war_exhaustion
				value = 60
				compare = greater_than_or_equals
			}
		}
		set_variable = { var = ADISCORD_vorkerland_civil_war_exhaustion_level value = 3 }
	}
	else_if = {
		limit = {
			check_variable = {
				var = ADISCORD_vorkerland_civil_war_exhaustion
				value = 40
				compare = greater_than_or_equals
			}
		}
		set_variable = { var = ADISCORD_vorkerland_civil_war_exhaustion_level value = 2 }
	}
	else_if = {
		limit = {
			check_variable = {
				var = ADISCORD_vorkerland_civil_war_exhaustion
				value = 20
				compare = greater_than_or_equals
			}
		}
		set_variable = { var = ADISCORD_vorkerland_civil_war_exhaustion_level value = 1 }
	}

	set_variable = {
		var = ADISCORD_vorkerland_civil_war_exhaustion_war_support_factor
		value = ADISCORD_vorkerland_civil_war_exhaustion
	}
	multiply_variable = {
		var = ADISCORD_vorkerland_civil_war_exhaustion_war_support_factor
		value = -0.002
	}
	set_variable = {
		var = ADISCORD_vorkerland_civil_war_exhaustion_stability_factor
		value = ADISCORD_vorkerland_civil_war_exhaustion
	}
	multiply_variable = {
		var = ADISCORD_vorkerland_civil_war_exhaustion_stability_factor
		value = -0.001
	}
	set_variable = {
		var = ADISCORD_vorkerland_civil_war_exhaustion_industrial_capacity_factory
		value = ADISCORD_vorkerland_civil_war_exhaustion
	}
	multiply_variable = {
		var = ADISCORD_vorkerland_civil_war_exhaustion_industrial_capacity_factory
		value = -0.001
	}
	set_variable = {
		var = ADISCORD_vorkerland_civil_war_exhaustion_army_morale_factor
		value = ADISCORD_vorkerland_civil_war_exhaustion
	}
	multiply_variable = {
		var = ADISCORD_vorkerland_civil_war_exhaustion_army_morale_factor
		value = -0.0005
	}
	set_variable = {
		var = ADISCORD_vorkerland_civil_war_exhaustion_surrender_limit
		value = ADISCORD_vorkerland_civil_war_exhaustion
	}
	multiply_variable = {
		var = ADISCORD_vorkerland_civil_war_exhaustion_surrender_limit
		value = -0.001
	}

	if = {
		limit = {
			check_variable = {
				var = ADISCORD_vorkerland_civil_war_exhaustion
				value = 1
				compare = greater_than_or_equals
			}
		}
		if = {
			limit = {
				NOT = {
					has_dynamic_modifier = {
						modifier = ADISCORD_vorkerland_civil_war_exhaustion
					}
				}
			}
			add_dynamic_modifier = { modifier = ADISCORD_vorkerland_civil_war_exhaustion }
		}
		force_update_dynamic_modifier = yes
	}
	else = {
		if = {
			limit = {
				has_dynamic_modifier = {
					modifier = ADISCORD_vorkerland_civil_war_exhaustion
				}
			}
			remove_dynamic_modifier = { modifier = ADISCORD_vorkerland_civil_war_exhaustion }
		}
	}
}

ADISCORD_vorkerland_reset_civil_war_exhaustion = {
	set_variable = { var = ADISCORD_vorkerland_civil_war_exhaustion value = 0 }
	set_variable = { var = ADISCORD_vorkerland_civil_war_casualties_delta_k value = 0 }
	set_variable = {
		var = ADISCORD_vorkerland_civil_war_casualties_snapshot_k
		value = casualties_k
	}
	ADISCORD_vorkerland_refresh_civil_war_exhaustion = yes
}

# Capitulation and state-control hooks can fire before a peace conference has
# finished transferring occupations. This guarded checker is safe to call from
# all of them and from the monthly fallback without scheduling duplicate events.
ADISCORD_vorkerland_check_central_outcome = {
	if = {
		limit = {
			has_global_flag = ADISCORD_vorkerland_collapse_wars_started
			NOT = { has_global_flag = ADISCORD_vorkerland_central_war_finished }
			NOT = { has_global_flag = ADISCORD_vorkerland_collapse_finished }
			NOT = { has_global_flag = ADISCORD_vorkerland_central_outcome_pending }
		}
		if = {
			limit = { WRK = { exists = yes ADISCORD_vorkerland_worker_victory_candidate = yes } }
			set_global_flag = ADISCORD_vorkerland_central_outcome_pending
			WRK = { country_event = { id = ADISCORD_vorkerland_collapse.20 hours = 1 } }
		}
		else_if = {
			limit = { VAD = { exists = yes ADISCORD_vorkerland_vlad_victory_candidate = yes } }
			set_global_flag = ADISCORD_vorkerland_central_outcome_pending
			VAD = { country_event = { id = ADISCORD_vorkerland_collapse.21 hours = 1 } }
		}
		else_if = {
			limit = { TVA = { exists = yes ADISCORD_vorkerland_dorian_victory_candidate = yes } }
			set_global_flag = ADISCORD_vorkerland_central_outcome_pending
			TVA = { country_event = { id = ADISCORD_vorkerland_collapse.22 hours = 1 } }
		}
	}
}
