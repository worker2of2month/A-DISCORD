ADISCORD_vorkerland_refresh_war_weariness = {
	set_variable = {
		var = ADISCORD_vorkerland_war_weariness_war_support_factor
		value = ADISCORD_vorkerland_war_weariness_level
	}
	multiply_variable = { var = ADISCORD_vorkerland_war_weariness_war_support_factor value = -0.10 }
	set_variable = {
		var = ADISCORD_vorkerland_war_weariness_stability_factor
		value = ADISCORD_vorkerland_war_weariness_level
	}
	multiply_variable = { var = ADISCORD_vorkerland_war_weariness_stability_factor value = -0.05 }
	set_variable = {
		var = ADISCORD_vorkerland_war_weariness_consumer_goods_factor
		value = ADISCORD_vorkerland_war_weariness_level
	}
	multiply_variable = { var = ADISCORD_vorkerland_war_weariness_consumer_goods_factor value = 0.03 }
	if = {
		limit = {
			has_dynamic_modifier = { modifier = ADISCORD_vorkerland_war_weariness }
		}
		force_update_dynamic_modifier = yes
	}
	else = {
		add_dynamic_modifier = { modifier = ADISCORD_vorkerland_war_weariness }
	}
}

ADISCORD_vorkerland_increase_war_weariness = {
	# Removing the compatibility idea is harmless in a fresh game and migrates
	# an old save before the stronger dynamic values are applied.
	remove_ideas = ADISCORD_vorkerland_war_weariness
	add_to_variable = { var = ADISCORD_vorkerland_war_weariness_level value = 1 }
	clamp_variable = { var = ADISCORD_vorkerland_war_weariness_level min = 1 max = 5 }
	add_to_variable = { var = ADISCORD_vorkerland_surrender_limit_offset value = -0.05 }
	clamp_variable = { var = ADISCORD_vorkerland_surrender_limit_offset min = -1 max = 0 }
	ADISCORD_vorkerland_refresh_war_weariness = yes
}

ADISCORD_vorkerland_clear_war_weariness = {
	remove_ideas = ADISCORD_vorkerland_war_weariness
	if = {
		limit = {
			has_dynamic_modifier = { modifier = ADISCORD_vorkerland_war_weariness }
		}
		remove_dynamic_modifier = { modifier = ADISCORD_vorkerland_war_weariness }
	}
	clear_variable = ADISCORD_vorkerland_war_weariness_level
	clear_variable = ADISCORD_vorkerland_war_weariness_war_support_factor
	clear_variable = ADISCORD_vorkerland_war_weariness_stability_factor
	clear_variable = ADISCORD_vorkerland_war_weariness_consumer_goods_factor
	clear_variable = ADISCORD_vorkerland_surrender_limit_offset
}

ADISCORD_vorkerland_migrate_war_weariness = {
	if = {
		limit = { has_idea = ADISCORD_vorkerland_war_weariness }
		remove_ideas = ADISCORD_vorkerland_war_weariness
		set_variable = { var = ADISCORD_vorkerland_war_weariness_level value = 1 }
		set_variable = { var = ADISCORD_vorkerland_surrender_limit_offset value = -0.05 }
		ADISCORD_vorkerland_refresh_war_weariness = yes
		if = {
			limit = {
				NOT = { has_capitulated = yes }
				has_war = yes
			}
			activate_mission = ADISCORD_vorkerland_stalemate_deadline
		}
	}
	# Saves made with the first dynamic-weariness implementation already have
	# a level but no capitulation decay variable. Reconstruct the missed decay.
	if = {
		limit = {
			has_dynamic_modifier = { modifier = ADISCORD_vorkerland_war_weariness }
			NOT = { has_variable = ADISCORD_vorkerland_surrender_limit_offset }
		}
		set_variable = {
			var = ADISCORD_vorkerland_surrender_limit_offset
			value = ADISCORD_vorkerland_war_weariness_level
		}
		multiply_variable = { var = ADISCORD_vorkerland_surrender_limit_offset value = -0.05 }
		clamp_variable = { var = ADISCORD_vorkerland_surrender_limit_offset min = -1 max = 0 }
		ADISCORD_vorkerland_refresh_war_weariness = yes
	}
}
