# Dirty Zone opening presentation helpers.
# The reveal lifecycle itself remains owned by ADISCORD_vorkerland_collapse.85.

ADISCORD_vorkerland_clear_dirty_zone_smoke = {
	if = {
		limit = { has_global_flag = ADISCORD_vorkerland_dirty_opened }

		# The main Dirty Zone haze is created globally at startup as actors 1 and 2.
		# Recreate both ids with the same inert mesh but no particle emitter. Using
		# the existing ids replaces the live particle actors and also makes the
		# cleanup deterministic after save-load.
		create_entity = {
			entity = ADISCORD_global_clear_entity
			id = 1
			x = 42
			y = 21
			z = -100.000
			province = 16531
			state = 42
			rotation = 1.2
			scale = 0.03
			min_zoom = 5555.0
		}
		create_entity = {
			entity = ADISCORD_global_clear_entity
			id = 2
			x = 42
			y = 21
			z = -100.000
			province = 12863
			state = 42
			rotation = 1.2
			scale = 0.02
			min_zoom = 5555.0
		}

		# The Unity Tower landmark carries its own independent smog/fire particles.
		# Recreate that actor too so opening the Zone cannot leave a local smoke
		# source behind. Preserve the correct intact/ruined visual state.
		if = {
			limit = { has_global_flag = ADISCORD_vorkerland_unity_tower_destruction_resolved }
			40 = {
				create_entity = {
					entity = ADISCORD_unity_tower_ruins_entity
					id = 610040
					x = 3159.92992
					y = 725.68
					z = 2.83464
					rotation = 5.410520681
					scale = 0.15
					min_zoom = 2000
				}
			}
		}
		else = {
			40 = {
				create_entity = {
					entity = ADISCORD_vorkerland_pyramid_clear_entity
					id = 610040
					x = 3159.92992
					y = 725.68
					z = 2.83464
					rotation = 5.410520681
					scale = 0.15
					min_zoom = 2000
				}
			}
		}

		set_global_flag = ADISCORD_vorkerland_dirty_zone_smoke_cleared
	}
}

ADISCORD_vorkerland_finalize_dirty_zone_opening_presentation = {
	if = {
		limit = {
			has_global_flag = ADISCORD_vorkerland_dirty_opened
			NOT = { has_global_flag = ADISCORD_vorkerland_dirty_zone_presentation_finalized }
		}
		set_global_flag = ADISCORD_vorkerland_dirty_zone_presentation_finalized
		ADISCORD_vorkerland_clear_dirty_zone_smoke = yes

		if = {
			limit = {
				NOT = { has_global_flag = ADISCORD_vorkerland_dirty_zone_news_published }
				RUS = { exists = yes }
			}
			set_global_flag = ADISCORD_vorkerland_dirty_zone_news_published
			RUS = { news_event = { id = ADISCORD_vorkerland_dirty_zone.1 } }
		}
	}
}

# Save-load repair: startup recreates the two global pollution actors and the
# Vorkerland landmark. If the Zone was already opened, this later on_startup
# hook overwrites all three with their durable particle-free variants.
ADISCORD_vorkerland_restore_dirty_zone_opening_visual = {
	if = {
		limit = { has_global_flag = ADISCORD_vorkerland_dirty_opened }
		ADISCORD_vorkerland_clear_dirty_zone_smoke = yes
	}
}
