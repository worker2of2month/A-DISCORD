# Stelander-Kefreyt crisis: canonical schema, mutation API, and dynamic-spirit refresh.

ADISCORD_STP_VAL_initialize_schema = {
	if = {
		limit = {
			OR = {
				NOT = { has_variable = ADISCORD_STP_VAL_crisis_schema_version }
				check_variable = { var = ADISCORD_STP_VAL_crisis_schema_version value = 2 compare = less_than }
			}
		}

		if = {
			limit = { tag = STP }

			# Legacy suspicion used a 0-1 scale. Migrate it before applying defaults.
			if = {
				limit = {
					has_variable = STP_party_suspicion
					check_variable = { var = STP_party_suspicion value = 0 compare = greater_than_or_equals }
					check_variable = { var = STP_party_suspicion value = 1 compare = less_than_or_equals }
				}
				multiply_variable = { var = STP_party_suspicion value = 100 }
			}
			if = {
				limit = {
					has_variable = STP_party_suspicion
					check_variable = { var = STP_party_suspicion value = 0 compare = less_than }
				}
				set_variable = { var = STP_party_suspicion value = 0 }
			}
			if = {
				limit = {
					has_variable = STP_party_suspicion
					check_variable = { var = STP_party_suspicion value = 100 compare = greater_than }
				}
				set_variable = { var = STP_party_suspicion value = 100 }
			}

			# Legacy face 5 represented the terminal portrait and normalizes to live health 4.
			if = {
				limit = {
					NOT = { has_variable = STP_leader_health_stage }
					has_variable = STP_state_face_stage
					check_variable = { var = STP_state_face_stage value = 5 }
				}
				set_variable = { var = STP_leader_health_stage value = 4 }
			}
			else_if = {
				limit = {
					NOT = { has_variable = STP_leader_health_stage }
					has_variable = STP_state_face_stage
				}
				set_variable = { var = STP_leader_health_stage value = STP_state_face_stage }
				clamp_variable = { var = STP_leader_health_stage min = 1 max = 4 }
			}

			if = { limit = { NOT = { has_variable = STP_leader_health_stage } } set_variable = { var = STP_leader_health_stage value = 1 } }
			if = { limit = { NOT = { has_variable = STP_resistance_readiness } } set_variable = { var = STP_resistance_readiness value = 10 } }
			if = { limit = { NOT = { has_variable = STP_party_suspicion } } set_variable = { var = STP_party_suspicion value = 5 } }
			if = { limit = { NOT = { has_variable = STP_crisis_phase } } set_variable = { var = STP_crisis_phase value = 1 } }
			if = { limit = { NOT = { has_variable = STP_side_commitment } } set_variable = { var = STP_side_commitment value = 0 } }
			if = { limit = { NOT = { has_variable = STP_node_palace } } set_variable = { var = STP_node_palace value = 0 } }
			if = { limit = { NOT = { has_variable = STP_node_officers } } set_variable = { var = STP_node_officers value = 0 } }
			if = { limit = { NOT = { has_variable = STP_node_mountains } } set_variable = { var = STP_node_mountains value = 0 } }
			if = { limit = { NOT = { has_variable = STP_node_market } } set_variable = { var = STP_node_market value = 0 } }
			if = { limit = { NOT = { has_variable = STP_node_street } } set_variable = { var = STP_node_street value = 0 } }
			if = { limit = { NOT = { has_variable = STP_node_val_channel } } set_variable = { var = STP_node_val_channel value = 0 } }
			if = { limit = { NOT = { has_variable = STP_security_posture } } set_variable = { var = STP_security_posture value = 0 } }
			if = { limit = { NOT = { has_variable = STP_resistance_posture } } set_variable = { var = STP_resistance_posture value = 0 } }
			if = { limit = { NOT = { has_variable = STP_security_adaptation_palace } } set_variable = { var = STP_security_adaptation_palace value = 0 } }
			if = { limit = { NOT = { has_variable = STP_security_adaptation_officers } } set_variable = { var = STP_security_adaptation_officers value = 0 } }
			if = { limit = { NOT = { has_variable = STP_security_adaptation_mountains } } set_variable = { var = STP_security_adaptation_mountains value = 0 } }
			if = { limit = { NOT = { has_variable = STP_security_adaptation_market } } set_variable = { var = STP_security_adaptation_market value = 0 } }
			if = { limit = { NOT = { has_variable = STP_security_adaptation_street } } set_variable = { var = STP_security_adaptation_street value = 0 } }
			if = { limit = { NOT = { has_variable = STP_security_adaptation_foreign } } set_variable = { var = STP_security_adaptation_foreign value = 0 } }

			# Restore a legacy literal STP only while it is still the pre-outcome country.
			if = {
				limit = {
					NOT = { has_country_flag = STP_main_campaign_side }
					NOT = { has_country_flag = STP_internal_war_started }
					NOT = { has_country_flag = STP_crisis_party_side }
					NOT = { has_country_flag = STP_crisis_resistance_side }
					NOT = { has_country_flag = STP_postwar_campaign_side }
					NOT = { has_country_flag = STP_internal_outcome_finalized }
				}
				set_country_flag = STP_main_campaign_side
			}
			if = {
				limit = {
					has_country_flag = STP_main_campaign_side
					NOT = { has_country_flag = STP_internal_war_started }
					NOT = { has_country_flag = STP_internal_outcome_finalized }
					NOT = { has_country_flag = STP_shabrat_available }
					NOT = { has_country_flag = STP_shabrat_lost }
					has_character = STP_maksim_shabrat
				}
				set_country_flag = STP_shabrat_available
			}

			# Compatibility mirror. New logic reads STP_leader_health_stage.
			set_variable = { var = STP_state_face_stage value = STP_leader_health_stage }
		}
		else_if = {
			limit = { tag = VAL }

			# Reconstruct a pre-schema VAL exactly once from the historical base and completed rewards.
			if = {
				limit = { NOT = { has_variable = VAL_contract_authority } }
				set_variable = { var = VAL_contract_authority value = 35 }
				if = {
					limit = { has_completed_focus = VAL_The_Contract_State }
					add_to_variable = { var = VAL_contract_authority value = 5 }
				}
				if = {
					limit = { has_completed_focus = VAL_The_Weaponry_Baron }
					add_to_variable = { var = VAL_contract_authority value = 10 }
				}
				if = {
					limit = { has_completed_focus = VAL_Price_Of_Loyalty }
					add_to_variable = { var = VAL_contract_authority value = 5 }
				}
				if = {
					limit = { has_completed_focus = VAL_Count_The_Captains }
					add_to_variable = { var = VAL_contract_authority value = 5 }
				}
				if = {
					limit = { has_completed_focus = VAL_One_Ledger_One_Banner }
					add_to_variable = { var = VAL_contract_authority value = 10 }
				}
				if = {
					limit = { has_completed_focus = VAL_Export_Rifles_Not_Promises }
					add_to_variable = { var = VAL_contract_authority value = 5 }
				}
				if = {
					limit = { has_completed_focus = VAL_Morns_Supply_Trains }
					add_to_variable = { var = VAL_contract_authority value = 5 }
				}
				if = {
					limit = { has_completed_focus = VAL_Dead_Villages_Still_Count }
					add_to_variable = { var = VAL_contract_authority value = 5 }
				}
				if = {
					limit = { has_completed_focus = VAL_Different_Views_On_Freedom }
					add_to_variable = { var = VAL_contract_authority value = 5 }
				}
				if = {
					limit = { has_completed_focus = VAL_Contracts_Outlive_Kings }
					add_to_variable = { var = VAL_contract_authority value = 5 }
				}
			}
			if = {
				limit = { check_variable = { var = VAL_contract_authority value = 0 compare = less_than } }
				set_variable = { var = VAL_contract_authority value = 0 }
			}
			if = {
				limit = { check_variable = { var = VAL_contract_authority value = 100 compare = greater_than } }
				set_variable = { var = VAL_contract_authority value = 100 }
			}

			if = { limit = { NOT = { has_variable = VAL_STP_leverage } } set_variable = { var = VAL_STP_leverage value = 0 } }
			if = { limit = { NOT = { has_variable = VAL_contract_band } } set_variable = { var = VAL_contract_band value = -1 } }
			if = { limit = { NOT = { has_variable = VAL_STP_final_war_preparation } } set_variable = { var = VAL_STP_final_war_preparation value = 0 } }
			if = { limit = { NOT = { has_variable = VAL_negotiation_posture } } set_variable = { var = VAL_negotiation_posture value = 0 } }
			if = { limit = { NOT = { has_variable = VAL_CIN_influence } } set_variable = { var = VAL_CIN_influence value = 0 } }
			if = { limit = { NOT = { has_variable = VAL_OSF_influence } } set_variable = { var = VAL_OSF_influence value = 0 } }
			if = { limit = { NOT = { has_variable = VAL_APH_influence } } set_variable = { var = VAL_APH_influence value = 0 } }
			if = { limit = { NOT = { has_variable = VAL_CIN_contract_posture } } set_variable = { var = VAL_CIN_contract_posture value = 0 } }
			if = { limit = { NOT = { has_variable = VAL_OSF_contract_posture } } set_variable = { var = VAL_OSF_contract_posture value = 0 } }
			if = { limit = { NOT = { has_variable = VAL_APH_contract_posture } } set_variable = { var = VAL_APH_contract_posture value = 0 } }
			if = { limit = { NOT = { has_variable = VAL_STP_exposure_intel } } set_variable = { var = VAL_STP_exposure_intel value = 0 } }
			if = { limit = { NOT = { has_variable = VAL_STP_exposure_supply } } set_variable = { var = VAL_STP_exposure_supply value = 0 } }
			if = { limit = { NOT = { has_variable = VAL_STP_exposure_concession } } set_variable = { var = VAL_STP_exposure_concession value = 0 } }
			if = { limit = { NOT = { has_variable = VAL_STP_exposure_garrison } } set_variable = { var = VAL_STP_exposure_garrison value = 0 } }
			if = { limit = { NOT = { has_variable = VAL_STP_exposure_nodrul } } set_variable = { var = VAL_STP_exposure_nodrul value = 0 } }
			if = { limit = { NOT = { has_variable = VAL_CIN_last_family } } set_variable = { var = VAL_CIN_last_family value = 0 } }
			if = { limit = { NOT = { has_variable = VAL_CIN_repeat_streak } } set_variable = { var = VAL_CIN_repeat_streak value = 0 } }
			if = { limit = { NOT = { has_variable = VAL_OSF_last_family } } set_variable = { var = VAL_OSF_last_family value = 0 } }
			if = { limit = { NOT = { has_variable = VAL_OSF_repeat_streak } } set_variable = { var = VAL_OSF_repeat_streak value = 0 } }
			if = { limit = { NOT = { has_variable = VAL_APH_last_family } } set_variable = { var = VAL_APH_last_family value = 0 } }
			if = { limit = { NOT = { has_variable = VAL_APH_repeat_streak } } set_variable = { var = VAL_APH_repeat_streak value = 0 } }
			if = { limit = { NOT = { has_variable = VAL_STP_missed_debt_payments } } set_variable = { var = VAL_STP_missed_debt_payments value = 0 } }
		}

		# Version 1 saves already own an authority variable, so grant only the
		# central-spine rewards introduced with the focus-specialisation pass.
		if = {
			limit = {
				tag = VAL
				has_variable = ADISCORD_STP_VAL_crisis_schema_version
				check_variable = { var = ADISCORD_STP_VAL_crisis_schema_version value = 2 compare = less_than }
			}
			if = { limit = { has_completed_focus = VAL_Price_Of_Loyalty } add_to_variable = { var = VAL_contract_authority value = 5 } }
			if = { limit = { has_completed_focus = VAL_Count_The_Captains } add_to_variable = { var = VAL_contract_authority value = 5 } }
			if = { limit = { has_completed_focus = VAL_One_Ledger_One_Banner } add_to_variable = { var = VAL_contract_authority value = 10 } }
			if = { limit = { has_completed_focus = VAL_Contracts_Outlive_Kings } add_to_variable = { var = VAL_contract_authority value = 5 } }
			clamp_variable = { var = VAL_contract_authority min = 0 max = 100 }
		}

		# Schema assignment is deliberately last: all legacy reconstruction precedes it.
		set_variable = { var = ADISCORD_STP_VAL_crisis_schema_version value = 2 }
	}
}

ADISCORD_STP_VAL_refresh_presentation_mirrors = {
	# Lightweight compatibility mirrors are rebuilt from authoritative state on
	# startup.  No daily pulse is required for either presentation value.
	if = {
		limit = { tag = STP }
		if = { limit = { check_variable = { var = STP_leader_health_stage value = 1 compare = equals } } set_variable = { var = STP_leader_health value = 100 } }
		else_if = { limit = { check_variable = { var = STP_leader_health_stage value = 2 compare = equals } } set_variable = { var = STP_leader_health value = 75 } }
		else_if = { limit = { check_variable = { var = STP_leader_health_stage value = 3 compare = equals } } set_variable = { var = STP_leader_health value = 45 } }
		else_if = { limit = { check_variable = { var = STP_leader_health_stage value = 4 compare = equals } } set_variable = { var = STP_leader_health value = 15 } }
		else = { set_variable = { var = STP_leader_health value = 0 } }
	}
	else_if = {
		limit = { tag = VAL }
		if = { limit = { NOT = { has_variable = VAL_STP_campaign_readiness } } set_variable = { var = VAL_STP_campaign_readiness value = 0 } }
		VAL_recalculate_stp_campaign_readiness = yes
	}
}

STP_set_health_stage = {
	set_temp_variable = { var = STP_requested_health_stage value = ADISCORD_STP_VAL_effect_value }
	if = {
		limit = {
			check_variable = { var = STP_requested_health_stage value = 1 compare = greater_than_or_equals }
			check_variable = { var = STP_requested_health_stage value = 5 compare = less_than_or_equals }
			NOT = { check_variable = { var = STP_leader_health_stage value = STP_requested_health_stage } }
		}
		set_variable = { var = STP_leader_health_stage value = STP_requested_health_stage }
		set_variable = { var = STP_state_face_stage value = STP_leader_health_stage }
		if = { limit = { check_variable = { var = STP_leader_health_stage value = 1 compare = equals } } set_variable = { var = STP_leader_health value = 100 } }
		else_if = { limit = { check_variable = { var = STP_leader_health_stage value = 2 compare = equals } } set_variable = { var = STP_leader_health value = 75 } }
		else_if = { limit = { check_variable = { var = STP_leader_health_stage value = 3 compare = equals } } set_variable = { var = STP_leader_health value = 45 } }
		else_if = { limit = { check_variable = { var = STP_leader_health_stage value = 4 compare = equals } } set_variable = { var = STP_leader_health value = 15 } }
		else = { set_variable = { var = STP_leader_health value = 0 } }
		STP_refresh_crisis_modifier = yes
	}
}

	set_temp_variable = { var = STP_requested_crisis_phase value = ADISCORD_STP_VAL_effect_value }

		limit = {
			check_variable = { var = STP_requested_crisis_phase value = 0 compare = greater_than_or_equals }

# Readiness is a compact operational score, not another click currency.  It is
# rebuilt only after relevant state changes and directly powers the temporary
# war-plan spirit once the canonical countdown expires.
VAL_recalculate_stp_campaign_readiness = {
	set_variable = { var = VAL_STP_campaign_readiness value = VAL_STP_final_war_preparation }
	set_temp_variable = { var = VAL_STP_leverage_component value = VAL_STP_leverage }
	multiply_temp_variable = { var = VAL_STP_leverage_component value = 0.35 }
	add_to_variable = { var = VAL_STP_campaign_readiness value = VAL_STP_leverage_component }

	if = { limit = { check_variable = { var = VAL_contract_authority value = 50 compare = greater_than_or_equals } } add_to_variable = { var = VAL_STP_campaign_readiness value = 5 } }
	if = { limit = { check_variable = { var = VAL_contract_authority value = 75 compare = greater_than_or_equals } } add_to_variable = { var = VAL_STP_campaign_readiness value = 5 } }
	if = { limit = { has_country_flag = VAL_STP_intel_43 has_country_flag = VAL_STP_intel_44 has_country_flag = VAL_STP_intel_45 has_country_flag = VAL_STP_intel_88 } add_to_variable = { var = VAL_STP_campaign_readiness value = 10 } }
	if = { limit = { has_country_flag = VAL_STP_client_garrison } add_to_variable = { var = VAL_STP_campaign_readiness value = 10 } }
	if = { limit = { has_country_flag = VAL_STP_resource_rights_45 } add_to_variable = { var = VAL_STP_campaign_readiness value = 5 } }
	if = { limit = { has_country_flag = VAL_STP_resource_rights_88 } add_to_variable = { var = VAL_STP_campaign_readiness value = 5 } }
	if = { limit = { check_variable = { var = VAL_CIN_influence value = 2 compare = greater_than_or_equals } } add_to_variable = { var = VAL_STP_campaign_readiness value = 5 } }
	if = { limit = { check_variable = { var = VAL_OSF_influence value = 2 compare = greater_than_or_equals } } add_to_variable = { var = VAL_STP_campaign_readiness value = 5 } }
	if = {
		limit = {
			country_exists = NOD
			NOD = { OR = { has_war = yes has_country_flag = NOD_limited_war_active has_country_flag = NOD_crisis_posture_ypr has_country_flag = NOD_crisis_posture_cof has_country_flag = NOD_crisis_posture_beshay } }
		}
		add_to_variable = { var = VAL_STP_campaign_readiness value = 10 }
	}
	if = {
		limit = {
			country_exists = NOD
			NOD = { has_country_flag = NOD_crisis_posture_guardian has_war = no }
		}
		subtract_from_variable = { var = VAL_STP_campaign_readiness value = 15 }
	}
	clamp_variable = { var = VAL_STP_campaign_readiness min = 0 max = 100 }
	VAL_refresh_stp_campaign_plan_modifier = yes
}

	set_variable = { var = VAL_STP_campaign_planning_factor value = 0 }

		limit = { check_variable = { var = VAL_STP_campaign_readiness value = 50 compare = less_than } }

		set_variable = { var = VAL_STP_campaign_planning_factor value = -0.05 }
		set_variable = { var = VAL_STP_campaign_supply_factor value = 0.05 }

		set_variable = { var = VAL_STP_campaign_defence_factor value = 0.03 }

		set_variable = { var = VAL_STP_campaign_attack_factor value = 0.10 }
		set_variable = { var = VAL_STP_campaign_defence_factor value = 0.05 }
		set_variable = { var = VAL_STP_campaign_planning_factor value = 0.10 }
		set_variable = { var = VAL_STP_campaign_supply_factor value = -0.05 }

		limit = {
			has_country_flag = VAL_STP_final_war_active

	}
}
