VAL_initialize_contract_authority = {
	if = { limit = { NOT = { has_variable = VAL_CIN_influence } } set_variable = { var = VAL_CIN_influence value = 0 } }
	if = { limit = { NOT = { has_variable = VAL_OSF_influence } } set_variable = { var = VAL_OSF_influence value = 0 } }
	if = { limit = { NOT = { has_variable = VAL_APH_influence } } set_variable = { var = VAL_APH_influence value = 0 } }
	if = { limit = { NOT = { has_variable = VAL_contract_authority } } set_variable = { var = VAL_contract_authority value = 35 } }
	clamp_variable = { var = VAL_contract_authority min = 0 max = 100 }
	VAL_refresh_contract_modifier = yes
}

VAL_change_contract_authority = {
	VAL_initialize_contract_authority = yes
	add_to_variable = { var = VAL_contract_authority value = ADISCORD_VAL_effect_value }
	clamp_variable = { var = VAL_contract_authority min = 0 max = 100 }
	VAL_refresh_contract_modifier = yes
}

# --- foreign_operation_effects ---
# CIN, OSF and APH contracts.

# VAL export escrow uses one ledger for the shared foreign-operation slot.
# Store measured debits on each producer tag, including absent tags.
# Cross-scope debit measurement lasts until the loop ends.
# Infantry stock preserves producer and quantity; named variants are not tracked.
VAL_reserve_export_rifles = {
	clr_country_flag = VAL_export_rifles_reserved
	set_temp_variable = { var = VAL_export_rifles_remaining value = 2500 }
	if = { limit = { has_completed_focus = VAL_Foreign_Broker_Licences } set_temp_variable = { var = VAL_export_rifles_remaining value = 5000 } }
	if = {
		limit = { OR = {
			AND = { has_completed_focus = VAL_Foreign_Broker_Licences NOT = { has_equipment = { infantry_equipment < 5000 } } }
			AND = { NOT = { has_completed_focus = VAL_Foreign_Broker_Licences } NOT = { has_equipment = { infantry_equipment < 2500 } } }
		} }
		every_possible_country = {
			limit = { VAL = { check_variable = { var = VAL_export_rifles_remaining value = 0 compare = greater_than } } }
			# Recheck because the matching producer list can be collected before execution.
			if = {
				limit = { VAL = { check_variable = { var = VAL_export_rifles_remaining value = 0 compare = greater_than } } }
				VAL = {
					set_variable = { var = VAL_export_rifles_removed value = num_equipment@infantry_equipment }
					set_temp_variable = { var = VAL_export_rifle_debit value = VAL_export_rifles_remaining }
					multiply_temp_variable = { var = VAL_export_rifle_debit value = -1 }
					add_equipment_to_stockpile = { type = infantry_equipment amount = VAL_export_rifle_debit producer = PREV }
					subtract_from_variable = { var = VAL_export_rifles_removed value = num_equipment@infantry_equipment }
					subtract_from_temp_variable = { var = VAL_export_rifles_remaining value = VAL_export_rifles_removed }
				}
				if = {
					limit = { VAL = { check_variable = { var = VAL_export_rifles_removed value = 0 compare = greater_than } } }
					set_variable = { var = VAL_export_rifles value = VAL.VAL_export_rifles_removed }
				}
			}
		}
		clear_variable = VAL_export_rifles_removed
		if = {
			limit = { check_variable = { var = VAL_export_rifles_remaining value = 0 compare = equals } }
			# Freeze the quoted income and paid volume until settlement, including across focus completion.
			if = {
				limit = { has_completed_focus = VAL_Foreign_Broker_Licences }
				if = { limit = { has_completed_focus = VAL_Northern_Clearing_House } set_country_flag = { flag = VAL_export_rifles_reserved value = 40 } }
				else = { set_country_flag = { flag = VAL_export_rifles_reserved value = 30 } }
			}
			else_if = { limit = { has_completed_focus = VAL_Northern_Clearing_House } set_country_flag = { flag = VAL_export_rifles_reserved value = 20 } }
			else = { set_country_flag = { flag = VAL_export_rifles_reserved value = 15 } }
		}
		else = { VAL_refund_export_rifles = yes }
	}
}

# COUNTRY (CIN/OSF): settle the one reserved northern order to its actual buyer.
# PREV within the producer loop restores this recipient; its inner PREV is the producer.
VAL_deliver_export_rifles = {
	if = {
		limit = {
			OR = { tag = CIN tag = OSF }
			exists = yes has_capitulated = no NOT = { has_war_with = VAL }
			VAL = { has_country_flag = VAL_export_rifles_reserved has_capitulated = no }
		}
		every_possible_country = {
			limit = { has_variable = VAL_export_rifles }
			PREV = { add_equipment_to_stockpile = { type = infantry_equipment amount = PREV.VAL_export_rifles producer = PREV } }
			clear_variable = VAL_export_rifles
		}
		VAL = {
			ADISCORD_economy_initialize_country = yes

# The saved leader identifies the existing alliance; only new memberships carry receipts.
VAL_frontier_join_coalition = {

        limit = { has_event_target = VAL_frontier_coalition_anchor }
        if = {
            limit = {
                is_in_faction = no

                OR = { is_subject = no overlord = { is_in_faction_with = event_target:VAL_frontier_coalition_anchor } }
            }
            save_event_target_as = VAL_frontier_joining_country
            event_target:VAL_frontier_coalition_anchor = { faction_leader = { add_to_faction = event_target:VAL_frontier_joining_country } }

            }
        }
    }
}

                    NOT = {
                        any_other_country = {
                            is_in_faction_with = event_target:VAL_frontier_coalition_anchor

            }
            clr_country_flag = VAL_frontier_created_coalition
        }
        CIN = { VAL_frontier_leave_coalition = yes }
        OSF = { VAL_frontier_leave_coalition = yes }
        APH = { VAL_frontier_leave_coalition = yes }

}

# COUNTRY VAL. Native relation callbacks can recurse when a subject joins;
# one synchronous guard keeps each enemy/subject pair within this dispatch.
VAL_call_subjects_to_wars = {
	if = {
		limit = {
			tag = VAL
			exists = yes
			has_war = yes
			is_subject = no
			NOT = { has_country_flag = VAL_subject_war_dispatch_active }
		}
		set_country_flag = VAL_subject_war_dispatch_active
		every_enemy_country = {
			save_event_target_as = VAL_subject_war_enemy
			VAL = {
				every_subject_country = {
					limit = {
						has_capitulated = no
						NOT = { has_war_with = VAL }
						NOT = { has_war_with = event_target:VAL_subject_war_enemy }
						NOT = { is_in_faction_with = event_target:VAL_subject_war_enemy }
					}
					add_to_war = {
						targeted_alliance = VAL
						enemy = event_target:VAL_subject_war_enemy
						hostility_reason = asked_to_join
						single_target_only = yes
					}
				}
			}
		}
		clr_country_flag = VAL_subject_war_dispatch_active
	}
}

# Final campaign: 1 preparation, 2 war, 3 victory, 4 closed. Only VAL's weekly
# hook advances escalation; startup and war callbacks never advance the clock.
VAL_final_crisis_initialize = {
	if = {
		limit = { NOT = { has_variable = VAL_final_crisis_phase } VAL_final_crisis_available = yes VAL_frontier_idle = yes }
		set_variable = { var = VAL_final_crisis_phase value = 1 }
		set_variable = { var = VAL_final_escalation value = 0 }
		set_variable = { var = VAL_final_supply_bonus value = 0 }
		set_variable = { var = VAL_final_planning_bonus value = 0 }
		set_variable = { var = VAL_final_reserve_share value = 0 }
		add_dynamic_modifier = { modifier = VAL_final_war_preparation }
		if = {
			limit = { check_variable = { var = VAL_refugee_nodrul_admitted value = 3 compare = less_than } }
			set_country_flag = VAL_refugee_nodrul_seen
			set_country_flag = { flag = VAL_refugee_nodrul_window days = 180 }
		}
		unlock_decision_tooltip = VAL_final_register_reserves
		unlock_decision_tooltip = VAL_final_stockpile_supplies
		unlock_decision_tooltip = VAL_final_reconnaissance
	}
}

# COUNTRY participant. Preserve major status which belonged to another system.
VAL_final_crisis_register_member = {
	set_country_flag = VAL_final_war_member
	if = { limit = { is_major = no } set_country_flag = VAL_final_war_added_major set_major = yes }
}

VAL_final_crisis_join_war = {
	if = {
		limit = { has_war_with = NOD NOT = { check_variable = { var = VAL_final_crisis_phase value = 2 compare = equals } } }
		set_variable = { var = VAL_final_crisis_phase value = 2 }
		NOD = { VAL_final_crisis_register_member = yes }
		STP = {
			if = {
				limit = { exists = yes has_capitulated = no OR = { is_in_faction_with = NOD is_subject_of = NOD NOD = { is_subject_of = STP } } NOT = { is_in_faction_with = VAL is_subject_of = VAL } }
				if = { limit = { NOT = { has_war_with = VAL } } add_to_war = { targeted_alliance = NOD enemy = VAL hostility_reason = asked_to_join single_target_only = yes } }
				if = { limit = { has_war_with = VAL } VAL_final_crisis_register_member = yes }
			}
		}
		STS = {
			if = {
				limit = { exists = yes has_capitulated = no OR = { is_in_faction_with = NOD is_subject_of = NOD } NOT = { is_in_faction_with = VAL is_subject_of = VAL } }
				if = { limit = { NOT = { has_war_with = VAL } } add_to_war = { targeted_alliance = NOD enemy = VAL hostility_reason = asked_to_join single_target_only = yes } }
				if = { limit = { has_war_with = VAL } VAL_final_crisis_register_member = yes }
			}
		}
		if = { limit = { NOT = { has_dynamic_modifier = { modifier = VAL_final_war_preparation } } } add_dynamic_modifier = { modifier = VAL_final_war_preparation } }
		VAL_call_frontier_partner = yes
		VAL_call_subjects_to_wars = yes
	}
}

VAL_final_crisis_launch = {
	if = {
		limit = { VAL_final_crisis_preparing = yes }
		NOD = { VAL_final_crisis_register_member = yes }
		if = {
			limit = { NOD = { is_subject_of = STP } }
			STP = { VAL_final_crisis_register_member = yes }
			declare_war_on = { target = STP type = annex_everything }
			NOD = { if = { limit = { NOT = { has_war_with = VAL } } add_to_war = { targeted_alliance = STP enemy = VAL hostility_reason = asked_to_join single_target_only = yes } } }
		}
		else = { declare_war_on = { target = NOD type = annex_everything } }
		VAL_final_crisis_join_war = yes
	}
}

VAL_final_refund_supply_order = {
	if = {
		limit = { has_variable = VAL_final_supply_deposit }
		add_to_variable = { var = ADISCORD_economy_treasury value = VAL_final_supply_deposit }
		add_to_variable = { var = ADISCORD_economy_current_month_action_income value = VAL_final_supply_deposit }
		clear_variable = VAL_final_supply_deposit
		ADISCORD_economy_mark_dirty = yes
	}
}

VAL_final_crisis_close = {
	set_variable = { var = VAL_final_crisis_phase value = 4 }
	if = { limit = { NOD = { OR = { is_subject_of = VAL has_country_flag = VAL_final_settlement_commit has_country_flag = VAL_final_capitulation_immediate } } has_capitulated = no is_subject = no } set_variable = { var = VAL_final_crisis_phase value = 3 } }
	if = { limit = { has_dynamic_modifier = { modifier = VAL_final_war_preparation } } remove_dynamic_modifier = { modifier = VAL_final_war_preparation } }
	VAL_final_refund_supply_order = yes

	NOD = { clr_country_flag = VAL_final_war_member if = { limit = { has_country_flag = VAL_final_war_added_major } set_major = no clr_country_flag = VAL_final_war_added_major } }
	STP = { clr_country_flag = VAL_final_war_member if = { limit = { has_country_flag = VAL_final_war_added_major } set_major = no clr_country_flag = VAL_final_war_added_major } }
	STS = { clr_country_flag = VAL_final_war_member if = { limit = { has_country_flag = VAL_final_war_added_major } set_major = no clr_country_flag = VAL_final_war_added_major } }
}

VAL_final_crisis_reconcile = {
	if = {
		limit = { check_variable = { var = VAL_final_crisis_phase value = 0 compare = greater_than } check_variable = { var = VAL_final_crisis_phase value = 3 compare = less_than } }
		if = { limit = { OR = { has_capitulated = yes is_subject = yes has_country_flag = VAL_stelander_defeated } } VAL_final_crisis_close = yes }
		else_if = {
			limit = { check_variable = { var = VAL_final_crisis_phase value = 1 compare = equals } }
			if = { limit = { has_war_with = NOD VAL_final_campaign_separate_from_frontier = yes } VAL_final_crisis_join_war = yes }
			else_if = { limit = { VAL_final_crisis_available = no } VAL_final_crisis_close = yes }
		}
		else_if = { limit = { NOT = { OR = { has_war_with = NOD has_war_with = STP has_war_with = STS } } } VAL_final_crisis_close = yes }
	}

	# Postwar repair is intentionally retried by the weekly reconciliation pass.
	# This covers both fresh settlements and development saves loaded after a
	# previously completed final war, without relaxing any territorial requirement.
	if = {
		limit = {
			check_variable = { var = VAL_final_crisis_phase value = 3 compare = greater_than_or_equals }
			has_capitulated = no
			is_subject = no
			NOT = { has_cosmetic_tag = VAL_commonwealth }
		}
		if = { limit = { VAL_can_form_northern_administration = yes } VAL_form_northern_administration = yes }
		STP = {
			if = {
				limit = { exists = yes is_subject_of = VAL has_cosmetic_tag = STL_VAL_administration }
				VAL_cede_stelander_border = yes
			}
		}
		STS = {
			if = {
				limit = { exists = yes is_subject_of = VAL has_cosmetic_tag = STL_VAL_administration }
				VAL_cede_stelander_border = yes
			}
		}
	}
}

VAL_final_crisis_weekly = {
	VAL_final_crisis_initialize = yes
	VAL_final_crisis_reconcile = yes
	if = {
		limit = { VAL_final_crisis_preparing = yes }
		add_to_variable = { var = VAL_final_escalation value = 5 }
		clamp_variable = { var = VAL_final_escalation min = 0 max = 100 }
		if = { limit = { check_variable = { var = VAL_final_escalation value = 100 compare = greater_than_or_equals } } VAL_final_crisis_launch = yes }
	}
}
