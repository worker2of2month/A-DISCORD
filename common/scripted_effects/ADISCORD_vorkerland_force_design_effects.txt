# Save-safe repair for AI countries created by the collapse.  It runs once per
# claimant from a country-scoped weekly pulse, so old saves receive the same
# unlocks without reloading ground OOB files or duplicating divisions.

# New campaigns deploy these wings during the collapse setup.  This guarded
# fallback gives pre-fix saves the same visible air force once the relevant
# claimant again controls its designated airfield.
ADISCORD_vorkerland_deploy_missing_claimant_air_wings = {
	if = {
		limit = {
			has_global_flag = ADISCORD_vorkerland_collapse_wars_started
			NOT = { has_global_flag = ADISCORD_vorkerland_collapse_finished }
			NOT = { has_country_flag = ADISCORD_vorkerland_claimant_air_wings_deployed }
			OR = { tag = WRK tag = VAD tag = TVA }
		}
		if = {
			limit = { tag = WRK controls_state = 32 }
			add_fuel = 7500
			load_oob = "WRK_vorkerland_collapse_air"
			set_country_flag = ADISCORD_vorkerland_claimant_air_wings_deployed
		}
		else_if = {
			limit = { tag = VAD controls_state = 75 }
			add_fuel = 7500
			load_oob = "VAD_vorkerland_collapse_air"
			set_country_flag = ADISCORD_vorkerland_claimant_air_wings_deployed
		}
		else_if = {
			limit = { tag = TVA controls_state = 38 }
			add_fuel = 7500
			load_oob = "TVA_vorkerland_collapse_air"
			set_country_flag = ADISCORD_vorkerland_claimant_air_wings_deployed
		}
	}
}

ADISCORD_vorkerland_bootstrap_ai_force_designs = {
	if = {
		limit = {
			is_ai = yes
			has_global_flag = ADISCORD_vorkerland_collapse_wars_started
			NOT = { has_global_flag = ADISCORD_vorkerland_collapse_finished }
			NOT = { has_country_flag = ADISCORD_vorkerland_force_designs_bootstrapped }
			OR = { tag = WRK tag = VAD tag = TVA }
		}
		set_country_flag = ADISCORD_vorkerland_force_designs_bootstrapped
		set_technology = {
			ADISCORD_tech_semi_autonomous_combat_modules = 1
			ADISCORD_tech_reclaimed_jet_platforms = 1
		}
		# Old saves still contain the obsolete 2+2 mobile group.  One bounded
		# delivery lets the AI rebuild it into the current 4+6 armoured group and
		# leaves a small reinforcement reserve while production ramps up.
		if = {
			limit = { has_equipment = { ADISCORD_combat_platform_archetype < 120 } }
			add_equipment_to_stockpile = {
				type = ADISCORD_combat_platform_2170
				amount = 160
				producer = ROOT
			}
		}
		if = {
			limit = { has_equipment = { infantry_equipment < 500 } }
			add_equipment_to_stockpile = {
				type = infantry_equipment_0
				amount = 500
				producer = ROOT
			}
		}
		if = {
			limit = { has_equipment = { ADISCORD_squad_weapons_equipment < 40 } }
			add_equipment_to_stockpile = {
				type = ADISCORD_squad_weapons_equipment_0
				amount = 40
				producer = ROOT
			}
		}
		if = {
			limit = { has_equipment = { support_equipment < 80 } }
			add_equipment_to_stockpile = {
				type = support_equipment_1
				amount = 80
				producer = ROOT
			}
		}
		if = {
			limit = { has_equipment = { artillery_equipment < 12 } }
			add_equipment_to_stockpile = {
				type = artillery_equipment_1
				amount = 12
				producer = ROOT
			}
		}
		if = {
			limit = { has_equipment = { ADISCORD_fighter_archetype < 12 } }
			add_equipment_to_stockpile = {
				type = ADISCORD_fighter_airframe_2163
				amount = 12
				producer = ROOT
			}
		}
	}
}
