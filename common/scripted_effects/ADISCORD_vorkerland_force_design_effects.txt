# Fresh-campaign AI bootstrap. collapse.2 calls this once for each main
# claimant immediately after the verified partition enters the wartime phase.
ADISCORD_vorkerland_bootstrap_ai_force_designs = {
	if = {
		limit = {
			is_ai = yes
			has_global_flag = ADISCORD_vorkerland_collapse_wars_started
			NOT = { has_global_flag = ADISCORD_vorkerland_collapse_finished }
			NOT = { has_country_flag = ADISCORD_vorkerland_force_designs_bootstrapped }
			OR = { tag = WKR tag = VAD tag = TVA }
		}
		set_country_flag = ADISCORD_vorkerland_force_designs_bootstrapped
		set_technology = {
			ADISCORD_tech_semi_autonomous_combat_modules = 1
			ADISCORD_tech_reclaimed_jet_platforms = 1
			ADISCORD_tech_battlefield_attack_aircraft = 1
		}
		# One bounded delivery lets the AI build the authored 4+6 armoured group
		# and leaves a small reinforcement reserve while production ramps up.
		if = {
			limit = { has_equipment = { ADISCORD_combat_platform_archetype < 120 } }
			add_equipment_to_stockpile = {
				type = ADISCORD_combat_platform_2170
				amount = 160
				producer = ROOT
			}
		}
		if = {
			limit = { has_equipment = { infantry_equipment < 500 } }
			add_equipment_to_stockpile = {
				type = infantry_equipment_0
				amount = 500
				producer = ROOT
			}
		}
		if = {
			limit = { has_equipment = { ADISCORD_squad_weapons_equipment < 40 } }
			add_equipment_to_stockpile = {
				type = ADISCORD_squad_weapons_equipment_0
				amount = 40
				producer = ROOT
			}
		}
		if = {
			limit = { has_equipment = { support_equipment < 80 } }
			add_equipment_to_stockpile = {
				type = support_equipment_1
				amount = 80
				producer = ROOT
			}
		}
		if = {
			limit = { has_equipment = { artillery_equipment < 12 } }
			add_equipment_to_stockpile = {
				type = artillery_equipment_1
				amount = 12
				producer = ROOT
			}
		}
		if = {
			limit = { has_equipment = { ADISCORD_fighter_archetype < 12 } }
			add_equipment_to_stockpile = {
				type = ADISCORD_fighter_airframe_2163
				amount = 12
				producer = ROOT
			}
		}
		if = {
			limit = { has_equipment = { ADISCORD_cas_archetype < 6 } }
			add_equipment_to_stockpile = {
				type = ADISCORD_cas_airframe_2170
				amount = 6
				producer = ROOT
			}
		}
	}
}
