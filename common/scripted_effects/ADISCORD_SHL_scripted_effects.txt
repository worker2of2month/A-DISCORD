SHL_schedule_cycle = {
	if = {
		limit = {
			SHL_country_operable = yes
			NOT = { has_country_flag = SHL_cycle_pending }
		}
		set_country_flag = SHL_cycle_pending
		country_event = { id = ADISCORD_SHL.1 days = 28 }
	}
}

SHL_run_cycle = {
	if = {
		limit = { has_country_flag = { flag = SHL_cycle_pending days > 27 } }
		clr_country_flag = SHL_cycle_pending
		if = {
			limit = { SHL_country_operable = yes }
			SHL_produce_period = yes
			SHL_refresh_furnaces = yes
			SHL_update_tensions = yes
			SHL_check_obligation_deadline = yes
			SHL_check_crises = yes
			ADISCORD_south_crisis_cycle = yes
			SHL_crisis_draw = yes
			SHL_life_draw = yes
		}
		# A capitulated country keeps its clock; stale callbacks cannot advance it.
		if = {
			limit = { exists = yes }
			set_country_flag = SHL_cycle_pending
			country_event = { id = ADISCORD_SHL.1 days = 28 }
		}
	}
}

SHL_produce_period = {
	SHL_refresh_delivery_terms = yes
	SHL_ai_allocate_deliveries = yes
	SHL_reconcile_obligation = yes
	clear_variable = SHL_tenth_crest_delivered
	set_variable = { var = SHL_last_cycle_income value = 0 }
	set_variable = { var = SHL_cycle_idle_sites value = 0 }
	add_to_variable = { var = SHL_cycle_number value = 1 }
	# The metal exchange quotes one price per cycle; the cartel keeps 60.
	if = {
		limit = { check_variable = { var = SHL_export_mode value = 1 compare = equals } }
		SHL_roll_exchange_price = yes
	}
	else = { set_variable = { var = SHL_exchange_price value = 60 } }
	287 = { SHL_produce_site = yes }
	288 = { SHL_produce_site = yes }
	289 = { SHL_produce_site = yes }
	290 = { SHL_produce_site = yes }
	291 = { SHL_produce_site = yes }
	292 = { SHL_produce_site = yes }
	293 = { SHL_produce_site = yes }
	294 = { SHL_produce_site = yes }
	295 = { SHL_produce_site = yes }
	if = {
		limit = {
			check_variable = { var = SHL_project_stage value = 4 compare = greater_than_or_equals }
			294 = { SHL_furnace_site_controlled = yes }
			check_variable = { var = SHL_tenth_running value = 1 compare = equals }
		}
		set_variable = { var = SHL_tenth_period_cost value = 1 }
		set_variable = { var = SHL_tenth_period_wear value = 6 }
		set_variable = { var = SHL_tenth_period_income value = 60 }
		if = {
			limit = { check_variable = { var = SHL_shift_policy value = 1 compare = equals } }
			set_variable = { var = SHL_tenth_period_cost value = 2 }
			set_variable = { var = SHL_tenth_period_wear value = 10 }
			set_variable = { var = SHL_tenth_period_income value = 90 }
		}
		else_if = {
			limit = { check_variable = { var = SHL_shift_policy value = 2 compare = equals } }
			set_variable = { var = SHL_tenth_period_wear value = 3 }
			set_variable = { var = SHL_tenth_period_income value = 25 }
		}
		subtract_from_variable = { var = SHL_tenth_period_income value = SHL_income_deduction }
		clamp_variable = { var = SHL_tenth_period_income min = 0 max = 90 }
		if = {
			limit = {
				check_variable = { var = SHL_regular_supply value = 1 compare = equals }
				check_variable = { var = SHL_tenth_stock value = SHL_tenth_period_cost compare = less_than }
			}
			set_variable = { var = SHL_tenth_planned_packets value = 1 }
			if = {
				limit = {
					check_variable = { var = SHL_tenth_period_cost value = 2 compare = equals }
					check_variable = { var = SHL_tenth_stock value = 1 compare = less_than }
				}
				set_variable = { var = SHL_tenth_planned_packets value = 2 }
			}
			set_variable = { var = SHL_tenth_planned_price value = SHL_tenth_planned_packets }
			multiply_variable = { var = SHL_tenth_planned_price value = SHL_packet_price }
			if = {
				limit = {
					check_variable = { var = ADISCORD_economy_treasury value = SHL_tenth_planned_price compare = greater_than_or_equals }
				}
				subtract_from_variable = { var = ADISCORD_economy_treasury value = SHL_tenth_planned_price }
				add_to_variable = { var = SHL_tenth_stock value = SHL_tenth_planned_packets }
			}
		}
		if = {
			limit = { check_variable = { var = SHL_tenth_stock value = SHL_tenth_period_cost compare = greater_than_or_equals } }
			subtract_from_variable = { var = SHL_tenth_stock value = SHL_tenth_period_cost }
			add_to_variable = { var = SHL_tenth_wear value = SHL_tenth_period_wear }
			SHL_deliver_tenth_heat = yes
			add_to_variable = { var = ADISCORD_economy_treasury value = SHL_tenth_period_income }
			add_to_variable = { var = SHL_last_cycle_income value = SHL_tenth_period_income }
			set_variable = { var = SHL_tenth_shortages value = 0 }
		}
		else = {
			add_to_variable = { var = SHL_tenth_shortages value = 1 }
			add_to_variable = { var = SHL_tenth_wear value = 12 }
		}
		if = {
			limit = {
				OR = {
					check_variable = { var = SHL_tenth_wear value = 100 compare = greater_than_or_equals }
					check_variable = { var = SHL_tenth_shortages value = 2 compare = greater_than_or_equals }
				}
			}
			set_variable = { var = SHL_tenth_running value = 0 }
			custom_effect_tooltip = SHL_tenth_stopped_tt
		}
		clamp_variable = { var = SHL_tenth_wear min = 0 max = 100 }
	}
	# Deliver relief after all nine sites, so recipient order cannot grant a free cycle.
	if = {
		limit = { check_variable = { var = SHL_obligation_progress value = 2 compare = greater_than_or_equals } }
		SHL_complete_obligation = yes
	}
	# House credits repay 25 per cycle for six cycles each.
	if = {
		limit = { check_variable = { var = SHL_credit_cycles value = 0 compare = greater_than } }
		add_to_variable = { var = ADISCORD_economy_treasury value = 25 }
		add_to_variable = { var = SHL_last_cycle_income value = 25 }
		subtract_from_variable = { var = SHL_credit_cycles value = 1 }
	}
	# Metal contracts settle after the furnaces, inside the same cycle report.
	if = {
		limit = { check_variable = { var = SHL_metal_clients value = 0 compare = greater_than } }
		SHL_metal_settle_clients = yes
	}
	ADISCORD_economy_clamp_treasury = yes
}

SHL_produce_site = {
	clear_variable = SHL_crest_delivered
	if = {
		limit = {
			SHL_furnace_site_controlled = yes
			check_variable = { var = SHL_furnace_running value = 1 compare = equals }
		}
		# A mobilisation quota overrides any recipient chosen in the meantime.
		if = {
			limit = { has_state_flag = SHL_quota_site }
			set_variable = { var = SHL_furnace_recipient value = 1 }
		}
		set_variable = { var = SHL_period_cost value = 1 }
		set_variable = { var = SHL_period_wear value = 6 }
		set_variable = { var = SHL_period_income value = 60 }
		if = {
			limit = { SHL = { check_variable = { var = SHL_shift_policy value = 1 compare = equals } } }
			set_variable = { var = SHL_period_cost value = 2 }
			set_variable = { var = SHL_period_wear value = 10 }
			set_variable = { var = SHL_period_income value = 90 }
		}
		else_if = {
			limit = { SHL = { check_variable = { var = SHL_shift_policy value = 2 compare = equals } } }
			set_variable = { var = SHL_period_wear value = 3 }
			set_variable = { var = SHL_period_income value = 25 }
		}
		subtract_from_variable = { var = SHL_period_wear value = SHL_furnace_training }
		# PREV is SHL: every producing site pays the same active deductions.
		subtract_from_variable = { var = SHL_period_income value = PREV.SHL_income_deduction }
		clamp_variable = { var = SHL_period_income min = 0 max = 90 }
		# Exported metal follows the exchange quote instead of the cartel's 60.
		if = {
			limit = {
				check_variable = { var = SHL_furnace_recipient value = 0 compare = equals }
				SHL = { check_variable = { var = SHL_export_mode value = 1 compare = equals } }
			}
			add_to_variable = { var = SHL_period_income value = PREV.SHL_exchange_price }
			subtract_from_variable = { var = SHL_period_income value = 60 }
			clamp_variable = { var = SHL_period_income min = 0 max = 90 }
		}
		if = {
			limit = {
				SHL_site_on_qanat_border = yes
				SHL = { check_variable = { var = SHL_water_crisis value = 1 compare = equals } }
			}
			add_to_variable = { var = SHL_period_wear value = 4 }
		}
		if = {
			limit = {
				SHL = { check_variable = { var = SHL_regular_supply value = 1 compare = equals } }
				check_variable = { var = SHL_furnace_stock value = SHL_period_cost compare = less_than }
			}
			set_variable = { var = SHL_planned_packets value = 1 }
			if = {
				limit = {
					check_variable = { var = SHL_period_cost value = 2 compare = equals }
					check_variable = { var = SHL_furnace_stock value = 1 compare = less_than }
				}
				set_variable = { var = SHL_planned_packets value = 2 }
			}
			set_variable = { var = SHL_planned_price value = SHL_planned_packets }
			multiply_variable = { var = SHL_planned_price value = PREV.SHL_packet_price }
			if = {
				limit = {
					SHL = {
						check_variable = { var = ADISCORD_economy_treasury value = PREV.SHL_planned_price compare = greater_than_or_equals }
					}
				}
				SHL = { subtract_from_variable = { var = ADISCORD_economy_treasury value = PREV.SHL_planned_price } }
				add_to_variable = { var = SHL_furnace_stock value = SHL_planned_packets }
			}
		}
		if = {
			limit = { check_variable = { var = SHL_furnace_stock value = SHL_period_cost compare = greater_than_or_equals } }
			subtract_from_variable = { var = SHL_furnace_stock value = SHL_period_cost }
			add_to_variable = { var = SHL_furnace_wear value = SHL_period_wear }
			set_variable = { var = SHL_furnace_shortages value = 0 }
			# Explicit country input avoids reading the state after the scope switch.
			set_variable = { var = SHL_period_cash value = SHL_period_income }
			if = {
				limit = { SHL_obligation_here = yes }
				SHL_apply_site_obligation = yes
			}
			else = { SHL_deliver_site_heat = yes }
			SHL = { SHL_collect_site_income = yes }
		}
		else = {
			add_to_variable = { var = SHL_furnace_shortages value = 1 }
			add_to_variable = { var = SHL_furnace_wear value = 12 }
			SHL = {
				add_to_variable = { var = SHL_cycle_idle_sites value = 1 }
				if = {
					limit = { NOT = { has_country_flag = SHL_shortage_notice_seen } }
					set_country_flag = SHL_shortage_notice_seen
					country_event = { id = ADISCORD_SHL.3 }
				}
			}
		}
		clamp_variable = { var = SHL_furnace_wear min = 0 max = 100 }
		if = {
			limit = {
				check_variable = { var = SHL_furnace_wear value = 90 compare = greater_than_or_equals }
				check_variable = { var = SHL_furnace_wear value = 100 compare = less_than }
				SHL = { SHL_accident_ready = yes }
			}
			set_state_flag = SHL_accident_site
			SHL = {
				set_country_flag = SHL_accident_pending
				country_event = { id = ADISCORD_SHL.40 }
			}
		}
		if = {
			limit = {
				OR = {
					check_variable = { var = SHL_furnace_wear value = 100 compare = greater_than_or_equals }
					check_variable = { var = SHL_furnace_shortages value = 2 compare = greater_than_or_equals }
				}
			}
			SHL_extinguish_site = yes
		}
	}
}

	add_to_variable = { var = SHL_last_cycle_income value = PREV.SHL_period_cash }
}

SHL_extinguish_site = {
	if = {
		limit = { check_variable = { var = SHL_furnace_running value = 1 compare = equals } }
		set_variable = { var = SHL_furnace_running value = 0 }
		clear_variable = SHL_restart_settled
		if = {
			limit = { check_variable = { var = SHL_furnace_owner value = 0 compare = equals } }
			set_variable = { var = SHL_furnace_owner value = 1 }
			SHL = { add_to_variable = { var = SHL_city_support value = 5 } }
		}
		SHL = {
			add_to_variable = { var = SHL_unrest value = 5 }
			SHL_clamp_tensions = yes
		}
		SHL = {
			if = {
				limit = { NOT = { has_country_flag = SHL_first_extinction_seen } }
				set_country_flag = SHL_first_extinction_seen
				country_event = { id = ADISCORD_SHL.4 }
			}
		}
	}
}

# The packet quote is re-read before the debit, so a Svetlogorye contract or
# embargo signed since the last refresh prices this purchase.
SHL_begin_supply = {
	SHL_refresh_packet_price = yes
	if = {
		limit = {
			FROM = {
				SHL_furnace_site_idle = yes
				check_variable = { var = SHL_furnace_stock value = 2 compare = less_than_or_equals }
			}
			check_variable = { var = ADISCORD_economy_treasury value = SHL_packet_price compare = greater_than_or_equals }
		}
		subtract_from_variable = { var = ADISCORD_economy_treasury value = SHL_packet_price }
		FROM = {
			set_variable = { var = SHL_operation value = 1 }
			set_variable = { var = SHL_operation_deposit value = PREV.SHL_packet_price }
			set_variable = { var = SHL_operation_pp value = 0 }
		}
		SHL_refresh_furnaces = yes
	}
}

SHL_finish_supply = {
	if = {
		limit = { FROM = { check_variable = { var = SHL_operation value = 1 compare = equals } } }
		if = {
			limit = {
				FROM = {
					SHL_furnace_site_controlled = yes
					check_variable = { var = SHL_furnace_stock value = 2 compare = less_than_or_equals }
				}
			}
			FROM = {
				add_to_variable = { var = SHL_furnace_stock value = 1 }
				clamp_variable = { var = SHL_furnace_stock min = 0 max = 3 }
				clear_variable = SHL_operation_deposit
				clear_variable = SHL_operation_pp
				clear_variable = SHL_operation
			}
		}
		else = { SHL_cancel_operation = yes }
		SHL_refresh_furnaces = yes
	}
}

# Input: temp SHL_balance_shift (negative toward the crests), one of the
# literals below. Native add_power_balance_value takes no variable, so the
# chain maps every caller's step to a fixed value.
SHL_balance_shift = {
	if = {
		limit = { has_power_balance = { id = SHL_conclave_balance } }
		if = {
			limit = { check_variable = { var = SHL_balance_shift value = -0.30 compare = equals } }
			add_power_balance_value = { id = SHL_conclave_balance value = -0.30 }
		}
		else_if = {
			limit = { check_variable = { var = SHL_balance_shift value = -0.25 compare = equals } }
			add_power_balance_value = { id = SHL_conclave_balance value = -0.25 }
		}
		else_if = {
			limit = { check_variable = { var = SHL_balance_shift value = -0.15 compare = equals } }
			add_power_balance_value = { id = SHL_conclave_balance value = -0.15 }
		}
		else_if = {
			limit = { check_variable = { var = SHL_balance_shift value = -0.10 compare = equals } }
			add_power_balance_value = { id = SHL_conclave_balance value = -0.10 }
		}
		else_if = {
			limit = { check_variable = { var = SHL_balance_shift value = -0.05 compare = equals } }
			add_power_balance_value = { id = SHL_conclave_balance value = -0.05 }
		}
		else_if = {
			limit = { check_variable = { var = SHL_balance_shift value = -0.03 compare = equals } }
			add_power_balance_value = { id = SHL_conclave_balance value = -0.03 }
		}
		else_if = {
			limit = { check_variable = { var = SHL_balance_shift value = 0.02 compare = equals } }
			add_power_balance_value = { id = SHL_conclave_balance value = 0.02 }
		}
		else_if = {
			limit = { check_variable = { var = SHL_balance_shift value = 0.03 compare = equals } }
			add_power_balance_value = { id = SHL_conclave_balance value = 0.03 }
		}
		else_if = {
			limit = { check_variable = { var = SHL_balance_shift value = 0.05 compare = equals } }
			add_power_balance_value = { id = SHL_conclave_balance value = 0.05 }
		}
		else_if = {
			limit = { check_variable = { var = SHL_balance_shift value = 0.10 compare = equals } }
			add_power_balance_value = { id = SHL_conclave_balance value = 0.10 }
		}
		else_if = {
			limit = { check_variable = { var = SHL_balance_shift value = 0.15 compare = equals } }
			add_power_balance_value = { id = SHL_conclave_balance value = 0.15 }
		}
		else_if = {
			limit = { check_variable = { var = SHL_balance_shift value = 0.25 compare = equals } }
			add_power_balance_value = { id = SHL_conclave_balance value = 0.25 }
		}
		else_if = {
			limit = { check_variable = { var = SHL_balance_shift value = 0.30 compare = equals } }
			add_power_balance_value = { id = SHL_conclave_balance value = 0.30 }
		}
	}
}

# Runs once after each settled production period.
SHL_update_tensions = {
	if = {
		limit = { has_idea = SHL_unregistered_quarters }
		add_to_variable = { var = SHL_unrest value = 5 }
	}
	if = {
		limit = {
			check_variable = { var = SHL_project_stage value = 4 compare = less_than }
			check_variable = { var = SHL_house_votes value = 0 compare = greater_than }
			check_variable = { var = SHL_political_course value = 5 compare = less_than }
			NOT = { has_country_flag = SHL_no_crests }
		}
		add_to_variable = { var = SHL_house_pressure value = 5 }
	}
	if = {
		limit = {
			check_variable = { var = SHL_shift_policy value = 1 compare = equals }
			NOT = { has_completed_focus = SHL_common_smelting }
		}
		add_to_variable = { var = SHL_unrest value = 4 }
	}
	else_if = {
		limit = { check_variable = { var = SHL_shift_policy value = 2 compare = equals } }
		subtract_from_variable = { var = SHL_unrest value = 6 }
	}
	add_to_variable = { var = SHL_unrest value = SHL_cycle_idle_sites }
	if = {
		limit = { check_variable = { var = SHL_city_support value = 40 compare = greater_than_or_equals } }
		subtract_from_variable = { var = SHL_unrest value = 2 }
	}
	if = {
		limit = { has_completed_focus = SHL_shift_arbitration }
		subtract_from_variable = { var = SHL_unrest value = 1 }
	}
	if = {
		limit = { SHL_house_mandate_current = yes }
		subtract_from_variable = { var = SHL_house_pressure value = 2 }
	}
	else_if = {
		limit = { SHL_city_mandate_current = yes }
		subtract_from_variable = { var = SHL_unrest value = 2 }
		add_to_variable = { var = SHL_house_pressure value = 3 }
	}
	else_if = {
		limit = { SHL_directorate_mandate_current = yes }
		add_to_variable = { var = SHL_house_pressure value = 3 }
	}
	else_if = {
		limit = { check_variable = { var = SHL_political_course value = 4 compare = equals } }
		add_to_variable = { var = SHL_unrest value = 3 }
		subtract_from_variable = { var = SHL_house_pressure value = 2 }
	}
	else_if = {
		limit = { check_variable = { var = SHL_political_course value = 5 compare = equals } }
		subtract_from_variable = { var = SHL_unrest value = 3 }
	}
	else_if = {
		limit = { check_variable = { var = SHL_house_votes value = 7 compare = less_than } }
		add_to_variable = { var = SHL_house_pressure value = 2 }
	}
	# Before ratification a conclave short of seven house votes lets the quarters
	# gain every cycle; a ratified course no longer counts hereditary votes.
	if = {
		limit = {
			check_variable = { var = SHL_political_course value = 0 compare = equals }
			check_variable = { var = SHL_house_votes value = 7 compare = less_than }
		}
		set_temp_variable = { var = SHL_balance_shift value = 0.02 }
		SHL_balance_shift = yes
	}
	# A commune without crests keeps the pressure scale at zero.
	if = {
		limit = { has_country_flag = SHL_no_crests }
		set_variable = { var = SHL_house_pressure value = 0 }
	}
	SHL_house_demand_tick = yes
	SHL_clamp_tensions = yes
}

			}

			limit = { has_country_flag = SHL_mzr_offensive }
			declare_war_on = {
				target = MZR
				type = take_state_focus
				generator = { 269 273 }
			}
		}
		else = {
			MZR = {
				declare_war_on = {
					target = SHL
					type = take_state_focus
					generator = { 707 }
				}
			}
		}
		KYZ = { SHL_join_mzr_side = yes }
		GLP = { SHL_join_mzr_side = yes }
		KYZ = {
			if = {
				limit = {
					exists = yes
					is_in_faction_with = SHL
					NOT = { has_war_with = MZR }
				}
				add_to_war = {
					targeted_alliance = SHL
					enemy = MZR

				}
				set_country_flag = SHL_mzr_shl_partner

SHL_close_mzr_war = {
	# Close the dispatcher guard before white peace can invoke on_peace.
	set_variable = { var = SHL_war_mzr value = 0 }
	SHL_peace_mzr_side = yes
	KYZ = {
		if = {
			limit = { has_country_flag = SHL_mzr_shl_partner }
			SHL_peace_mzr_side = yes
		}
	}
	GLP = {
		if = {
			limit = { has_country_flag = SHL_mzr_shl_partner }
			SHL_peace_mzr_side = yes
		}
	}
	# Annexation can move leadership to the remaining signatory before on_annex.
	KYZ = { SHL_dissolve_inherited_mzr_pact = yes }
	GLP = { SHL_dissolve_inherited_mzr_pact = yes }
	MZR = {
		if = {
			limit = { has_country_flag = SHL_mzr_temporary_faction }
			if = {
				limit = { is_faction_leader = yes }
				dismantle_faction = yes
			}
			clr_country_flag = SHL_mzr_temporary_faction
		}
	}
	KYZ = {
		clr_country_flag = SHL_mzr_shl_partner
		clr_country_flag = SHL_mzr_war_partner
	}
	GLP = {
		clr_country_flag = SHL_mzr_shl_partner
		clr_country_flag = SHL_mzr_war_partner
	}
	SHL_clear_mzr_diplomacy = yes
	set_variable = { var = SHL_mazar_state value = 3 }
	clr_country_flag = SHL_war_mzr_clock
	clr_country_flag = SHL_mzr_offensive
	SHL_clean_campaign = yes
	SHL = { SHL_release_mzr_principal = yes }
	MZR = { SHL_release_mzr_principal = yes }
}

# Cancellation settles the remaining receipt once, including money already spent.
# An unpaid advance becomes ordinary public debt instead of negative cash.
SHL_cancel_obligation = {
	if = {
		limit = { has_variable = SHL_obligation_kind }
		add_to_variable = { var = ADISCORD_economy_treasury value = SHL_obligation_deposit }
		set_variable = { var = SHL_obligation_payment value = ADISCORD_economy_treasury }
		clamp_variable = { var = SHL_obligation_payment min = 0 max = SHL_obligation_liability }
		subtract_from_variable = { var = ADISCORD_economy_treasury value = SHL_obligation_payment }
		subtract_from_variable = { var = SHL_obligation_liability value = SHL_obligation_payment }
		add_to_variable = { var = ADISCORD_economy_debt value = SHL_obligation_liability }
		SHL_clear_obligation = yes
		clear_variable = SHL_obligation_payment
		ADISCORD_economy_mark_dirty = yes
		ADISCORD_economy_clamp_treasury = yes
	}
}

# PREV is the producing site. A pledge diverts cash, other mandates divert metal.
SHL_progress_obligation = {
	if = {
		limit = {
			NOT = { check_variable = { var = SHL_obligation_kind value = 4 compare = equals } }
			check_variable = { var = SHL_obligation_kind value = 6 compare = less_than }
		}
		PREV = { set_variable = { var = SHL_period_cash value = 0 } }
	}
	if = {
		limit = { check_variable = { var = SHL_obligation_kind value = 4 compare = equals } }
		set_variable = { var = SHL_obligation_payment value = PREV.SHL_period_cash }
		clamp_variable = { var = SHL_obligation_payment min = 0 max = SHL_obligation_liability }
		subtract_from_variable = { var = SHL_obligation_liability value = SHL_obligation_payment }
		PREV = { subtract_from_variable = { var = SHL_period_cash value = SHL.SHL_obligation_payment } }
		if = {
			limit = { check_variable = { var = SHL_obligation_liability value = 0 compare = less_than_or_equals } }
			SHL_clear_obligation = yes
		}
		clear_variable = SHL_obligation_payment
	}
	else_if = {
		limit = {
			NOT = { has_variable = SHL_obligation_overdue }
			check_variable = { var = SHL_obligation_progress value = 2 compare = less_than }
		}
		SHL_refresh_relief_readiness = yes
		if = {
			limit = {
				OR = {
					NOT = { check_variable = { var = SHL_obligation_kind value = 5 compare = equals } }
					check_variable = { var = SHL_obligation_recipient_ready value = 1 compare = equals }
				}
			}
			PREV = { set_variable = { var = SHL_period_cash value = 0 } }
			set_variable = { var = SHL_obligation_work value = 1 }
			if = {
				limit = { check_variable = { var = SHL_shift_policy value = 1 compare = equals } }
				set_variable = { var = SHL_obligation_work value = 2 }
			}
			set_variable = { var = SHL_obligation_remaining value = 2 }
			subtract_from_variable = { var = SHL_obligation_remaining value = SHL_obligation_progress }
			clamp_variable = { var = SHL_obligation_work min = 0 max = SHL_obligation_remaining }
			add_to_variable = { var = SHL_obligation_progress value = SHL_obligation_work }
			if = {
				limit = { check_variable = { var = SHL_obligation_kind value = 3 compare = equals } }
				SHL_deliver_arsenal_batch = yes
			}
		}
		clear_variable = SHL_obligation_recipient_ready
		clear_variable = SHL_obligation_work
		clear_variable = SHL_obligation_remaining
	}
}

# FROM is the receiving furnace. The donor's packet leaves at once and
# arrives seven days later; one transfer at a time.
SHL_begin_brotherly_supply = {
	if = {
		limit = {
			SHL_country_operable = yes
			NOT = { has_variable = SHL_brotherly_donor }
			FROM = {
				SHL_furnace_site_controlled = yes
				check_variable = { var = SHL_furnace_owner value = 1 compare = equals }
				check_variable = { var = SHL_furnace_stock value = 1 compare = less_than_or_equals }
			}
		}
		set_variable = { var = SHL_brotherly_donor value = 0 }
		287 = { SHL_consider_brotherly_donor = yes }
		288 = { SHL_consider_brotherly_donor = yes }
		289 = { SHL_consider_brotherly_donor = yes }
		290 = { SHL_consider_brotherly_donor = yes }
		291 = { SHL_consider_brotherly_donor = yes }
		292 = { SHL_consider_brotherly_donor = yes }
		293 = { SHL_consider_brotherly_donor = yes }
		294 = { SHL_consider_brotherly_donor = yes }
		295 = { SHL_consider_brotherly_donor = yes }
		if = {
			limit = { check_variable = { var = SHL_brotherly_donor value = 0 compare = greater_than } }
			set_variable = { var = SHL_brotherly_delta value = -1 }
			SHL_adjust_brotherly_donor = yes
			clear_variable = SHL_brotherly_delta
			FROM = { set_state_flag = SHL_brotherly_recipient }
			SHL_refresh_furnaces = yes
		}
		else = { clear_variable = SHL_brotherly_donor }
	}
}

# FROM is the supported country (GLP or MZR). The receipt lives on the target
# so both neighbours can be supported at once.
SHL_begin_strike_support = {
	if = {
		limit = {
			SHL_country_operable = yes
			check_variable = { var = ADISCORD_economy_treasury value = 50 compare = greater_than_or_equals }
			NOT = { has_political_power < 20 }
			FROM = {
				exists = yes
				is_subject = no
				NOT = { has_war_with = SHL }
				NOT = { has_variable = SHL_strike_support_deposit }
			}
		}
		subtract_from_variable = { var = ADISCORD_economy_treasury value = 50 }
		add_political_power = -20
		FROM = { set_variable = { var = SHL_strike_support_deposit value = 50 } }
	}
}

	if = {
		limit = { FROM = { has_variable = SHL_strike_support_deposit } }
		add_to_variable = { var = ADISCORD_economy_treasury value = FROM.SHL_strike_support_deposit }
		add_political_power = 20
		FROM = { clear_variable = SHL_strike_support_deposit }
	}
}

				SHL_country_operable = yes
				FROM = {
					exists = yes
					NOT = { has_war_with = SHL }
				}
			}

				add_to_variable = { var = SHL_cons_councils_support value = 25 }
				clamp_variable = { var = SHL_cons_councils_support min = 0 max = 100 }

					limit = {

					}
					set_country_flag = SHL_cons_councils_backed
					ADISCORD_south_offer_alliance = yes
				}
			}
		}
		else = { SHL_cancel_strike_support = yes }
	}
}

	}

# ----- Integration of satellites (postwar epilogue) -----
# COUNTRY satellite. Each course pays SHL_integration_price per annexed
# satellite; the regional tags are the only possible subjects.

SHL_integrate_satellite = {
	if = {
		limit = {
			exists = yes
			is_subject_of = SHL
			SHL = { check_variable = { var = ADISCORD_economy_treasury value = SHL_integration_price compare = greater_than_or_equals } }
			# Junior partners chosen by a postwar focus keep their satellite status.
			NOT = {
				AND = {
					tag = KDR
					SHL = { has_completed_focus = SHL_r_kadir_junior }
				}
			}
			NOT = {
				AND = {
					tag = GLP
					SHL = { has_completed_focus = SHL_h_veyr_junior_partner }
				}
			}
		}
		SHL = {
			subtract_from_variable = { var = ADISCORD_economy_treasury value = SHL_integration_price }
			add_to_variable = { var = SHL_integrated_count value = 1 }
			annex_country = {
				target = PREV
				transfer_troops = yes
			}
		}
	}
}

# ---------------------------------------------------------------------------
# Metal contracts of the Golden Gates. Buyers are the six fixed tags of the
# SHL_metal_contract_offer decision, so no settlement scans the world. The
# buyer carries SHL_metal_client and SHL_metal_arrears; Shahrabad counts
# SHL_metal_clients. The 28-day cycle moves 60 per paid contract; war between
# the parties or two unpaid cycles end it.
# ---------------------------------------------------------------------------

# COUNTRY buyer, ROOT of ADISCORD_SHL.130. Both sides are rechecked at the answer.
SHL_metal_sign_contract = {
	clr_country_flag = SHL_metal_offer_pending
	if = {
		limit = {
			NOT = { has_country_flag = SHL_metal_client }
			NOT = { has_war_with = SHL }
			SHL = {
				SHL_country_operable = yes
				SHL_metal_export_furnace_free = yes
				check_variable = { var = SHL_metal_clients value = 3 compare = less_than }
			}
		}
		set_country_flag = SHL_metal_client
		set_variable = { var = SHL_metal_arrears value = 0 }
		add_dynamic_modifier = { modifier = SHL_metal_supply }
		SHL = {
			add_to_variable = { var = SHL_metal_clients value = 1 }
			add_to_variable = { var = SHL_gui_dirty value = 1 }
			country_event = { id = ADISCORD_SHL.131 hours = 6 }
		}
	}
	else = {
		set_country_flag = {
			flag = SHL_metal_offer_lapsed
			value = 1
			days = 7
		}
		SHL = {
			if = {
				limit = { SHL_country_operable = yes }
				country_event = { id = ADISCORD_SHL.132 hours = 6 }
			}
		}
	}
}

# COUNTRY buyer, ROOT of ADISCORD_SHL.130.
SHL_metal_decline_contract = {
	clr_country_flag = SHL_metal_offer_pending
	SHL = {
		if = {
			limit = { SHL_country_operable = yes }
			country_event = { id = ADISCORD_SHL.132 hours = 6 }
		}
	}
}

# COUNTRY client. Ends the contract on both sides without a report; the
# SHL_metal_contract_end decision and the annexation cleanup use it directly.
SHL_metal_drop_client = {
	if = {
		limit = { has_country_flag = SHL_metal_client }
		clr_country_flag = SHL_metal_client
		clear_variable = SHL_metal_arrears
		if = {
			limit = { has_dynamic_modifier = { modifier = SHL_metal_supply } }
			remove_dynamic_modifier = { modifier = SHL_metal_supply }
		}
		SHL = {
			subtract_from_variable = { var = SHL_metal_clients value = 1 }
			clamp_variable = { var = SHL_metal_clients min = 0 max = 6 }
			add_to_variable = { var = SHL_gui_dirty value = 1 }
		}
	}
}

# COUNTRY client. Involuntary end: the Conclave is told in ADISCORD_SHL.133.
SHL_metal_end_contract = {
	if = {
		limit = { has_country_flag = SHL_metal_client }
		SHL_metal_drop_client = yes
		SHL = {
			if = {
				limit = { SHL_country_operable = yes }
				country_event = { id = ADISCORD_SHL.133 hours = 6 }
			}
		}
	}
}

# COUNTRY client, called from the SHL cycle. A paid cycle moves 60 from the
# buyer's treasury to Shahrabad's and books both monthly ledgers; an unpaid
# cycle is tolerated once.
SHL_metal_settle_client = {
	if = {
		limit = { has_country_flag = SHL_metal_client }
		if = {
			limit = {
				OR = {
					exists = no
					has_war_with = SHL
				}
			}
			SHL_metal_end_contract = yes
		}
		else_if = {
			limit = { check_variable = { var = ADISCORD_economy_treasury value = 60 compare = greater_than_or_equals } }
			subtract_from_variable = { var = ADISCORD_economy_treasury value = 60 }
			add_to_variable = { var = ADISCORD_economy_current_month_action_costs value = 60 }
			set_variable = { var = SHL_metal_arrears value = 0 }
			ADISCORD_economy_mark_dirty = yes
			SHL = {
				add_to_variable = { var = ADISCORD_economy_treasury value = 60 }
				add_to_variable = { var = ADISCORD_economy_current_month_action_income value = 60 }
				add_to_variable = { var = SHL_last_cycle_income value = 60 }
				ADISCORD_economy_mark_dirty = yes
			}
		}
		else_if = {
			limit = { check_variable = { var = SHL_metal_arrears value = 1 compare = less_than } }
			add_to_variable = { var = SHL_metal_arrears value = 1 }
		}
		else = { SHL_metal_end_contract = yes }
	}
}

SHL_metal_clear_clients = {
	if = {
		limit = { check_variable = { var = SHL_metal_clients value = 0 compare = greater_than } }
		VAL = { SHL_metal_drop_client = yes }
		STP = { SHL_metal_drop_client = yes }
		STS = { SHL_metal_drop_client = yes }
		RUS = { SHL_metal_drop_client = yes }
		GLP = { SHL_metal_drop_client = yes }
		NAM = { SHL_metal_drop_client = yes }
	}
	set_variable = { var = SHL_metal_clients value = 0 }
}
