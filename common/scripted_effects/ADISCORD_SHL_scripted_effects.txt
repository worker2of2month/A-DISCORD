SHL_schedule_cycle = {
	if = {
		limit = {
			SHL_country_operable = yes
			NOT = { has_country_flag = SHL_cycle_pending }
		}
		set_country_flag = SHL_cycle_pending
		country_event = { id = ADISCORD_SHL.1 days = 28 }
	}
}

SHL_run_cycle = {
	if = {
		limit = { has_country_flag = { flag = SHL_cycle_pending days > 27 } }
		clr_country_flag = SHL_cycle_pending
		if = {
			limit = { SHL_country_operable = yes }
			SHL_produce_period = yes
			SHL_refresh_furnaces = yes
			SHL_update_tensions = yes
			SHL_check_obligation_deadline = yes
			SHL_check_crises = yes
			ADISCORD_south_crisis_cycle = yes
			SHL_fire_cycle = yes
			SHL_crisis_draw = yes
		}
		# A capitulated country keeps its clock; stale callbacks cannot advance it.
		if = {
			limit = { exists = yes }
			set_country_flag = SHL_cycle_pending
			country_event = { id = ADISCORD_SHL.1 days = 28 }
		}
	}
}

SHL_produce_period = {
	SHL_reconcile_obligation = yes
	set_variable = { var = SHL_last_cycle_income value = 0 }
	set_variable = { var = SHL_cycle_idle_sites value = 0 }
	add_to_variable = { var = SHL_cycle_number value = 1 }
	287 = { SHL_produce_site = yes }
	288 = { SHL_produce_site = yes }
	289 = { SHL_produce_site = yes }
	290 = { SHL_produce_site = yes }
	291 = { SHL_produce_site = yes }
	292 = { SHL_produce_site = yes }
	293 = { SHL_produce_site = yes }
	294 = { SHL_produce_site = yes }
	295 = { SHL_produce_site = yes }
	if = {
		limit = {
			check_variable = { var = SHL_project_stage value = 4 compare = greater_than_or_equals }
			294 = { SHL_furnace_site_controlled = yes }
			check_variable = { var = SHL_tenth_running value = 1 compare = equals }
		}
		set_variable = { var = SHL_tenth_period_cost value = 1 }
		set_variable = { var = SHL_tenth_period_wear value = 6 }
		set_variable = { var = SHL_tenth_period_income value = 35 }
		if = {
			limit = { check_variable = { var = SHL_shift_policy value = 1 compare = equals } }
			set_variable = { var = SHL_tenth_period_cost value = 2 }
			set_variable = { var = SHL_tenth_period_wear value = 10 }
			set_variable = { var = SHL_tenth_period_income value = 50 }
		}
		else_if = {
			limit = { check_variable = { var = SHL_shift_policy value = 2 compare = equals } }
			set_variable = { var = SHL_tenth_period_wear value = 3 }
			set_variable = { var = SHL_tenth_period_income value = 25 }
		}
		subtract_from_variable = { var = SHL_tenth_period_income value = SHL_income_deduction }
		clamp_variable = { var = SHL_tenth_period_income min = 0 max = 50 }
		if = {
			limit = {
				check_variable = { var = SHL_regular_supply value = 1 compare = equals }
				check_variable = { var = SHL_tenth_stock value = SHL_tenth_period_cost compare = less_than }
			}
			set_variable = { var = SHL_tenth_planned_packets value = 1 }
			if = {
				limit = {
					check_variable = { var = SHL_tenth_period_cost value = 2 compare = equals }
					check_variable = { var = SHL_tenth_stock value = 1 compare = less_than }
				}
				set_variable = { var = SHL_tenth_planned_packets value = 2 }
			}
			set_variable = { var = SHL_tenth_planned_price value = SHL_tenth_planned_packets }
			multiply_variable = { var = SHL_tenth_planned_price value = 25 }
			if = {
				limit = { check_variable = { var = ADISCORD_economy_treasury value = SHL_tenth_planned_price compare = greater_than_or_equals } }
				subtract_from_variable = { var = ADISCORD_economy_treasury value = SHL_tenth_planned_price }
				add_to_variable = { var = SHL_tenth_stock value = SHL_tenth_planned_packets }
			}
		}
		if = {
			limit = { check_variable = { var = SHL_tenth_stock value = SHL_tenth_period_cost compare = greater_than_or_equals } }
			subtract_from_variable = { var = SHL_tenth_stock value = SHL_tenth_period_cost }
			add_to_variable = { var = SHL_tenth_wear value = SHL_tenth_period_wear }
			add_to_variable = { var = ADISCORD_economy_treasury value = SHL_tenth_period_income }
			add_to_variable = { var = SHL_last_cycle_income value = SHL_tenth_period_income }
			set_variable = { var = SHL_tenth_shortages value = 0 }
		}
		else = {
			add_to_variable = { var = SHL_tenth_shortages value = 1 }
			add_to_variable = { var = SHL_tenth_wear value = 12 }
		}
		if = {
			limit = {
				OR = {
					check_variable = { var = SHL_tenth_wear value = 100 compare = greater_than_or_equals }
					check_variable = { var = SHL_tenth_shortages value = 2 compare = greater_than_or_equals }
				}
			}
			set_variable = { var = SHL_tenth_running value = 0 }
			custom_effect_tooltip = SHL_tenth_stopped_tt
		}
		clamp_variable = { var = SHL_tenth_wear min = 0 max = 100 }
	}
	# Deliver relief after all nine sites, so recipient order cannot grant a free cycle.
	if = {
		limit = { check_variable = { var = SHL_obligation_progress value = 2 compare = greater_than_or_equals } }
		SHL_complete_obligation = yes
	}
	ADISCORD_economy_clamp_treasury = yes
}

SHL_produce_site = {
	if = {
		limit = {
			SHL_furnace_site_controlled = yes
			check_variable = { var = SHL_furnace_running value = 1 compare = equals }
		}
		set_variable = { var = SHL_period_cost value = 1 }
		set_variable = { var = SHL_period_wear value = 6 }
		set_variable = { var = SHL_period_income value = 35 }
		if = {
			limit = { SHL = { check_variable = { var = SHL_shift_policy value = 1 compare = equals } } }
			set_variable = { var = SHL_period_cost value = 2 }
			set_variable = { var = SHL_period_wear value = 10 }
			set_variable = { var = SHL_period_income value = 50 }
		}
		else_if = {
			limit = { SHL = { check_variable = { var = SHL_shift_policy value = 2 compare = equals } } }
			set_variable = { var = SHL_period_wear value = 3 }
			set_variable = { var = SHL_period_income value = 25 }
		}
		subtract_from_variable = { var = SHL_period_wear value = SHL_furnace_training }
		# PREV is SHL: every producing site pays the same active deductions.
		subtract_from_variable = { var = SHL_period_income value = PREV.SHL_income_deduction }
		clamp_variable = { var = SHL_period_income min = 0 max = 50 }
		if = {
			limit = {
				SHL_site_on_qanat_border = yes
				SHL = { check_variable = { var = SHL_water_crisis value = 1 compare = equals } }
			}
			add_to_variable = { var = SHL_period_wear value = 4 }
		}
		if = {
			limit = {
				SHL = { check_variable = { var = SHL_regular_supply value = 1 compare = equals } }
				check_variable = { var = SHL_furnace_stock value = SHL_period_cost compare = less_than }
			}
			set_variable = { var = SHL_planned_packets value = 1 }
			if = {
				limit = {
					check_variable = { var = SHL_period_cost value = 2 compare = equals }
					check_variable = { var = SHL_furnace_stock value = 1 compare = less_than }
				}
				set_variable = { var = SHL_planned_packets value = 2 }
			}
			set_variable = { var = SHL_planned_price value = SHL_planned_packets }
			multiply_variable = { var = SHL_planned_price value = 25 }
			if = {
				limit = { SHL = { check_variable = { var = ADISCORD_economy_treasury value = PREV.SHL_planned_price compare = greater_than_or_equals } } }
				SHL = { subtract_from_variable = { var = ADISCORD_economy_treasury value = PREV.SHL_planned_price } }
				add_to_variable = { var = SHL_furnace_stock value = SHL_planned_packets }
			}
		}
		if = {
			limit = { check_variable = { var = SHL_furnace_stock value = SHL_period_cost compare = greater_than_or_equals } }
			subtract_from_variable = { var = SHL_furnace_stock value = SHL_period_cost }
			add_to_variable = { var = SHL_furnace_wear value = SHL_period_wear }
			set_variable = { var = SHL_furnace_shortages value = 0 }
			# Explicit country input avoids reading the state after the scope switch.
			set_variable = { var = SHL_period_cash value = SHL_period_income }
			SHL_apply_site_obligation = yes
			SHL = { SHL_collect_site_income = yes }
		}
		else = {
			add_to_variable = { var = SHL_furnace_shortages value = 1 }
			add_to_variable = { var = SHL_furnace_wear value = 12 }
			SHL = {
				add_to_variable = { var = SHL_cycle_idle_sites value = 1 }
				if = {
					limit = { NOT = { has_country_flag = SHL_shortage_notice_seen } }
					set_country_flag = SHL_shortage_notice_seen
					country_event = { id = ADISCORD_SHL.3 }
				}
			}
		}
		clamp_variable = { var = SHL_furnace_wear min = 0 max = 100 }
		if = {
			limit = {
				check_variable = { var = SHL_furnace_wear value = 90 compare = greater_than_or_equals }
				check_variable = { var = SHL_furnace_wear value = 100 compare = less_than }
				SHL = { SHL_accident_ready = yes }
			}
			set_state_flag = SHL_accident_site
			SHL = {
				set_country_flag = SHL_accident_pending
				country_event = { id = ADISCORD_SHL.40 }
			}
		}
		if = {
			limit = {
				OR = {
					check_variable = { var = SHL_furnace_wear value = 100 compare = greater_than_or_equals }
					check_variable = { var = SHL_furnace_shortages value = 2 compare = greater_than_or_equals }
				}
			}
			SHL_extinguish_site = yes
		}
	}
}

# PREV is the paying production site; this effect has no other callers.
SHL_collect_site_income = {
	add_to_variable = { var = ADISCORD_economy_treasury value = PREV.SHL_period_cash }
	add_to_variable = { var = SHL_last_cycle_income value = PREV.SHL_period_cash }
}

SHL_extinguish_site = {
	if = {
		limit = { check_variable = { var = SHL_furnace_running value = 1 compare = equals } }
		set_variable = { var = SHL_furnace_running value = 0 }
		clear_variable = SHL_restart_settled
		if = {
			limit = { check_variable = { var = SHL_furnace_owner value = 0 compare = equals } }
			set_variable = { var = SHL_furnace_owner value = 1 }
			SHL = { add_to_variable = { var = SHL_city_support value = 5 } }
		}
		SHL = {
			add_to_variable = { var = SHL_unrest value = 5 }
			SHL_clamp_tensions = yes
		}
		SHL = {
			if = {
				limit = { NOT = { has_country_flag = SHL_first_extinction_seen } }
				set_country_flag = SHL_first_extinction_seen
				country_event = { id = ADISCORD_SHL.4 }
			}
		}
	}
}

SHL_cancel_operation = {

SHL_finish_project = {
	if = {
		limit = { has_variable = SHL_project_deposit }
		if = {
			limit = {
				SHL_country_operable = yes
				294 = {
					is_owned_by = SHL
					is_controlled_by = SHL
				}
			}
			if = {
				limit = { check_variable = { var = SHL_project_operation value = 1 compare = equals } }
				set_variable = { var = SHL_project_stage value = 1 }
				SHL_clear_project_receipt = yes
				country_event = { id = ADISCORD_SHL.7 days = 1 }
			}
			else_if = {
				limit = { check_variable = { var = SHL_project_operation value = 2 compare = equals } }
				if = {
					limit = { 294 = { infrastructure < 5 } }
					294 = { add_building_construction = { type = infrastructure level = 1 instant_build = yes } }
				}
				else = { add_to_variable = { var = ADISCORD_economy_treasury value = SHL_project_deposit } }
				set_variable = { var = SHL_project_stage value = 2 }
				SHL_clear_project_receipt = yes
			}
			else_if = {
				limit = { check_variable = { var = SHL_project_operation value = 3 compare = equals } }
				if = {
					limit = { 294 = { free_building_slots = { building = industrial_complex size = 1 include_locked = no } } }
					294 = { add_building_construction = { type = industrial_complex level = 1 instant_build = yes } }
					set_variable = { var = SHL_project_stage value = 3 }
					SHL_clear_project_receipt = yes
				}
				else = { SHL_cancel_project = yes }
			}
			else_if = {
				limit = { check_variable = { var = SHL_project_operation value = 4 compare = equals } }
				set_variable = { var = SHL_project_stage value = 4 }
				set_variable = { var = SHL_tenth_running value = 1 }
				set_variable = { var = SHL_tenth_wear value = 10 }
				set_variable = { var = SHL_tenth_stock value = 2 }
				set_variable = { var = SHL_tenth_shortages value = 0 }
				SHL_clear_project_receipt = yes
				country_event = { id = ADISCORD_SHL.5 }
			}
			else_if = {
				limit = { check_variable = { var = SHL_project_operation value = 5 compare = equals } }
				subtract_from_variable = { var = SHL_tenth_wear value = 30 }
				clamp_variable = { var = SHL_tenth_wear min = 0 max = 100 }
				SHL_clear_project_receipt = yes
			}
			else_if = {
				limit = { check_variable = { var = SHL_project_operation value = 6 compare = equals } }
				if = {
					limit = { check_variable = { var = SHL_tenth_stock value = 1 compare = greater_than_or_equals } }
					set_variable = { var = SHL_tenth_running value = 1 }
					set_variable = { var = SHL_tenth_shortages value = 0 }
					set_variable = { var = SHL_tenth_wear value = 20 }
					SHL_clear_project_receipt = yes
				}
				else = { SHL_cancel_project = yes }
			}
		}
		else = { SHL_cancel_project = yes }
		SHL_refresh_furnaces = yes
	}
}

# Runs once after each settled production period.
SHL_update_tensions = {
	if = {
		limit = {
			check_variable = { var = SHL_shift_policy value = 1 compare = equals }
			NOT = { has_completed_focus = SHL_common_smelting }
		}
		add_to_variable = { var = SHL_unrest value = 4 }
	}
	else_if = {
		limit = { check_variable = { var = SHL_shift_policy value = 2 compare = equals } }
		subtract_from_variable = { var = SHL_unrest value = 2 }
	}
	add_to_variable = { var = SHL_unrest value = SHL_cycle_idle_sites }
	if = {
		limit = { check_variable = { var = SHL_city_support value = 40 compare = greater_than_or_equals } }
		subtract_from_variable = { var = SHL_unrest value = 2 }
	}
	if = {
		limit = { has_completed_focus = SHL_shift_arbitration }
		subtract_from_variable = { var = SHL_unrest value = 1 }
	}
	if = {
		limit = { check_variable = { var = SHL_political_course value = 1 compare = equals } }
		subtract_from_variable = { var = SHL_house_pressure value = 2 }
	}
	else_if = {
		limit = { check_variable = { var = SHL_political_course value = 2 compare = equals } }
		subtract_from_variable = { var = SHL_unrest value = 2 }
		add_to_variable = { var = SHL_house_pressure value = 3 }
	}
	else_if = {
		limit = { check_variable = { var = SHL_political_course value = 3 compare = equals } }
		add_to_variable = { var = SHL_house_pressure value = 3 }
	}
	else_if = {
		limit = { check_variable = { var = SHL_political_course value = 4 compare = equals } }
		add_to_variable = { var = SHL_unrest value = 3 }
		subtract_from_variable = { var = SHL_house_pressure value = 2 }
	}
	else_if = {
		limit = { check_variable = { var = SHL_political_course value = 5 compare = equals } }
		subtract_from_variable = { var = SHL_unrest value = 3 }
	}
	else_if = {
		limit = { check_variable = { var = SHL_house_votes value = 7 compare = less_than } }
		add_to_variable = { var = SHL_house_pressure value = 2 }
	}
	SHL_clamp_tensions = yes
}

			SHL_country_operable = yes

		add_to_variable = { var = ADISCORD_economy_treasury value = SHL_obligation_deposit }

		clamp_variable = { var = SHL_obligation_payment min = 0 max = SHL_obligation_liability }
		subtract_from_variable = { var = ADISCORD_economy_treasury value = SHL_obligation_payment }
		subtract_from_variable = { var = SHL_obligation_liability value = SHL_obligation_payment }
		add_to_variable = { var = ADISCORD_economy_debt value = SHL_obligation_liability }
		SHL_clear_obligation = yes
		clear_variable = SHL_obligation_payment
		ADISCORD_economy_mark_dirty = yes
		ADISCORD_economy_clamp_treasury = yes
	}
}

# Ownership loss or a new constitution settles the old institution's receipt.
# Occupation alone pauses work; liberation resumes its stocks and progress.
SHL_reconcile_obligation = {
	if = {
		limit = { has_variable = SHL_obligation_kind }
		if = {
			limit = { check_variable = { var = SHL_obligation_kind value = 6 compare = equals } }
			set_variable = { var = SHL_obligation_course value = SHL_political_course }
		}
		# Ratification preserves the provisional programme and its paid work.
		if = {
			limit = {
				check_variable = { var = SHL_obligation_course value = 0 compare = equals }
				check_variable = { var = SHL_political_course value = SHL_obligation_kind compare = equals }
				check_variable = { var = SHL_obligation_kind value = 3 compare = less_than_or_equals }
			}
			set_variable = { var = SHL_obligation_course value = SHL_political_course }
			set_variable = { var = SHL_obligation_owner value = SHL_political_course }
			subtract_from_variable = { var = SHL_obligation_owner value = 1 }
		}
		if = {
			limit = { NOT = { check_variable = { var = SHL_political_course value = SHL_obligation_course compare = equals } } }
			SHL_cancel_obligation = yes
		}
		if = {
			limit = { check_variable = { var = SHL_obligation_site value = 287 compare = equals } }
			287 = { SHL_check_obligation_owner = yes }
		}
		if = {
			limit = { check_variable = { var = SHL_obligation_site value = 288 compare = equals } }
			288 = { SHL_check_obligation_owner = yes }
		}
		if = {
			limit = { check_variable = { var = SHL_obligation_site value = 289 compare = equals } }
			289 = { SHL_check_obligation_owner = yes }
		}
		if = {
			limit = { check_variable = { var = SHL_obligation_site value = 290 compare = equals } }
			290 = { SHL_check_obligation_owner = yes }
		}
		if = {

# PREV is the producing site. A pledge diverts cash, other mandates divert metal.
SHL_progress_obligation = {
	if = {
		limit = { check_variable = { var = SHL_obligation_kind value = 4 compare = equals } }
		set_variable = { var = SHL_obligation_payment value = PREV.SHL_period_cash }
		clamp_variable = { var = SHL_obligation_payment min = 0 max = SHL_obligation_liability }
		subtract_from_variable = { var = SHL_obligation_liability value = SHL_obligation_payment }
		PREV = { subtract_from_variable = { var = SHL_period_cash value = SHL.SHL_obligation_payment } }
		if = {
			limit = { check_variable = { var = SHL_obligation_liability value = 0 compare = less_than_or_equals } }
			SHL_clear_obligation = yes
		}
		clear_variable = SHL_obligation_payment
	}
	else_if = {
		limit = {
			NOT = { has_variable = SHL_obligation_overdue }
			check_variable = { var = SHL_obligation_progress value = 2 compare = less_than }
		}
		SHL_refresh_relief_readiness = yes
		if = {
			limit = {
				OR = {
					NOT = { check_variable = { var = SHL_obligation_kind value = 5 compare = equals } }
					check_variable = { var = SHL_obligation_recipient_ready value = 1 compare = equals }
				}
			}
			PREV = { set_variable = { var = SHL_period_cash value = 0 } }
			set_variable = { var = SHL_obligation_work value = 1 }
			if = {
				limit = { check_variable = { var = SHL_shift_policy value = 1 compare = equals } }
				set_variable = { var = SHL_obligation_work value = 2 }
			}
			set_variable = { var = SHL_obligation_remaining value = 2 }
			subtract_from_variable = { var = SHL_obligation_remaining value = SHL_obligation_progress }
			clamp_variable = { var = SHL_obligation_work min = 0 max = SHL_obligation_remaining }
			add_to_variable = { var = SHL_obligation_progress value = SHL_obligation_work }
			if = {
				limit = { check_variable = { var = SHL_obligation_kind value = 3 compare = equals } }
				SHL_deliver_arsenal_batch = yes
			}
		}
		clear_variable = SHL_obligation_recipient_ready
		clear_variable = SHL_obligation_work
		clear_variable = SHL_obligation_remaining
	}
}

# ----- Division of the fire -----
# Furnace revenue is divided between the houses, the quarters, the Conclave
# treasury and the front. The four shares always sum to 100 and change only
# through SHL_fire_transfer; the furnace cycle turns them into pressure, unrest
# and rifles, and SHL_fire_dynamic reads values derived in SHL_fire_refresh.

# COUNTRY SHL. One-time setup from the fresh-campaign startup.
SHL_fire_initialize = {
	if = {
		limit = { NOT = { has_country_flag = SHL_fire_initialized } }
		set_country_flag = SHL_fire_initialized
		set_variable = { var = SHL_share_crest value = 45 }
		set_variable = { var = SHL_share_quart value = 25 }
		set_variable = { var = SHL_share_state value = 20 }
		set_variable = { var = SHL_share_front value = 10 }
		SHL_fire_refresh = yes
	}
}

SHL_fire_refresh = {
	if = {
		limit = { has_country_flag = SHL_fire_initialized }
		set_variable = { var = SHL_fire_political_power value = SHL_share_crest }
		subtract_from_variable = { var = SHL_fire_political_power value = 30 }
		divide_variable = { var = SHL_fire_political_power value = 250 }
		set_variable = { var = SHL_fire_stability value = SHL_share_quart }
		subtract_from_variable = { var = SHL_fire_stability value = 25 }
		divide_variable = { var = SHL_fire_stability value = 250 }
		set_variable = { var = SHL_fire_tax_collection value = SHL_share_state }
		subtract_from_variable = { var = SHL_fire_tax_collection value = 20 }

# COUNTRY SHL. Temporary inputs: SHL_fire_from, SHL_fire_to (1 houses, 2 quarters,
# 3 treasury, 4 front) and SHL_fire_amount. The amount is limited by the source
# share and by the 70-point ceiling of the receiver, so the sum stays 100.
SHL_fire_transfer = {
	if = {
		limit = {
			has_country_flag = SHL_fire_initialized
			NOT = { check_variable = { var = SHL_fire_from value = SHL_fire_to compare = equals } }
		}
		if = {
			limit = { check_variable = { var = SHL_fire_from value = 1 compare = equals } }
			if = {
				limit = { check_variable = { var = SHL_share_crest value = SHL_fire_amount compare = less_than } }
				set_temp_variable = { var = SHL_fire_amount value = SHL_share_crest }
			}
		}
		else_if = {
			limit = { check_variable = { var = SHL_fire_from value = 2 compare = equals } }
			if = {
				limit = { check_variable = { var = SHL_share_quart value = SHL_fire_amount compare = less_than } }
				set_temp_variable = { var = SHL_fire_amount value = SHL_share_quart }
			}
		}
		else_if = {
			limit = { check_variable = { var = SHL_fire_from value = 3 compare = equals } }
			if = {
				limit = { check_variable = { var = SHL_share_state value = SHL_fire_amount compare = less_than } }
				set_temp_variable = { var = SHL_fire_amount value = SHL_share_state }
			}
		}
		else_if = {
			limit = { check_variable = { var = SHL_fire_from value = 4 compare = equals } }
			if = {
				limit = { check_variable = { var = SHL_share_front value = SHL_fire_amount compare = less_than } }
				set_temp_variable = { var = SHL_fire_amount value = SHL_share_front }
			}
		}
		if = {
			limit = { check_variable = { var = SHL_fire_to value = 1 compare = equals } }
			set_temp_variable = { var = SHL_fire_room value = 70 }
			subtract_from_temp_variable = { var = SHL_fire_room value = SHL_share_crest }
		}
		else_if = {
			limit = { check_variable = { var = SHL_fire_to value = 2 compare = equals } }
			set_temp_variable = { var = SHL_fire_room value = 70 }
			subtract_from_temp_variable = { var = SHL_fire_room value = SHL_share_quart }
		}
		else_if = {
			limit = { check_variable = { var = SHL_fire_to value = 3 compare = equals } }
			set_temp_variable = { var = SHL_fire_room value = 70 }
			subtract_from_temp_variable = { var = SHL_fire_room value = SHL_share_state }
		}
		else_if = {
			limit = { check_variable = { var = SHL_fire_to value = 4 compare = equals } }
			set_temp_variable = { var = SHL_fire_room value = 70 }
			subtract_from_temp_variable = { var = SHL_fire_room value = SHL_share_front }
		}
		if = {
			limit = { check_variable = { var = SHL_fire_room value = SHL_fire_amount compare = less_than } }
			set_temp_variable = { var = SHL_fire_amount value = SHL_fire_room }
		}
		if = {
			limit = { check_variable = { var = SHL_fire_amount value = 0 compare = greater_than } }
			if = {
				limit = { check_variable = { var = SHL_fire_from value = 1 compare = equals } }
				subtract_from_variable = { var = SHL_share_crest value = SHL_fire_amount }
			}
			else_if = {
				limit = { check_variable = { var = SHL_fire_from value = 2 compare = equals } }
				subtract_from_variable = { var = SHL_share_quart value = SHL_fire_amount }
			}
			else_if = {
				limit = { check_variable = { var = SHL_fire_from value = 3 compare = equals } }
				subtract_from_variable = { var = SHL_share_state value = SHL_fire_amount }
			}
			else_if = {
				limit = { check_variable = { var = SHL_fire_from value = 4 compare = equals } }
				subtract_from_variable = { var = SHL_share_front value = SHL_fire_amount }
			}
			if = {
				limit = { check_variable = { var = SHL_fire_to value = 1 compare = equals } }
				add_to_variable = { var = SHL_share_crest value = SHL_fire_amount }
			}
			else_if = {
				limit = { check_variable = { var = SHL_fire_to value = 2 compare = equals } }
				add_to_variable = { var = SHL_share_quart value = SHL_fire_amount }
			}
			else_if = {
				limit = { check_variable = { var = SHL_fire_to value = 3 compare = equals } }
				add_to_variable = { var = SHL_share_state value = SHL_fire_amount }
			}
			else_if = {
				limit = { check_variable = { var = SHL_fire_to value = 4 compare = equals } }
				add_to_variable = { var = SHL_share_front value = SHL_fire_amount }
			}
		}
		SHL_fire_refresh = yes
	}
}

SHL_fire_align_step = {
	set_temp_variable = { var = SHL_fire_gap_crest value = SHL_share_crest }
	subtract_from_temp_variable = { var = SHL_fire_gap_crest value = SHL_fire_target_crest }
	set_temp_variable = { var = SHL_fire_gap_quart value = SHL_share_quart }
	subtract_from_temp_variable = { var = SHL_fire_gap_quart value = SHL_fire_target_quart }
	set_temp_variable = { var = SHL_fire_gap_state value = SHL_share_state }
	subtract_from_temp_variable = { var = SHL_fire_gap_state value = SHL_fire_target_state }
	set_temp_variable = { var = SHL_fire_gap_front value = SHL_share_front }
	subtract_from_temp_variable = { var = SHL_fire_gap_front value = SHL_fire_target_front }
	set_temp_variable = { var = SHL_fire_from value = 1 }
	set_temp_variable = { var = SHL_fire_best value = SHL_fire_gap_crest }
	if = {
		limit = { check_variable = { var = SHL_fire_gap_quart value = SHL_fire_best compare = greater_than } }
		set_temp_variable = { var = SHL_fire_from value = 2 }
		set_temp_variable = { var = SHL_fire_best value = SHL_fire_gap_quart }
	}
	if = {
		limit = { check_variable = { var = SHL_fire_gap_state value = SHL_fire_best compare = greater_than } }
		set_temp_variable = { var = SHL_fire_from value = 3 }
		set_temp_variable = { var = SHL_fire_best value = SHL_fire_gap_state }
	}
	if = {
		limit = { check_variable = { var = SHL_fire_gap_front value = SHL_fire_best compare = greater_than } }
		set_temp_variable = { var = SHL_fire_from value = 4 }
		set_temp_variable = { var = SHL_fire_best value = SHL_fire_gap_front }
	}
	set_temp_variable = { var = SHL_fire_to value = 1 }
	set_temp_variable = { var = SHL_fire_worst value = SHL_fire_gap_crest }
	if = {
		limit = { check_variable = { var = SHL_fire_gap_quart value = SHL_fire_worst compare = less_than } }
		set_temp_variable = { var = SHL_fire_to value = 2 }
		set_temp_variable = { var = SHL_fire_worst value = SHL_fire_gap_quart }
	}
	if = {
		limit = { check_variable = { var = SHL_fire_gap_state value = SHL_fire_worst compare = less_than } }
		set_temp_variable = { var = SHL_fire_to value = 3 }
		set_temp_variable = { var = SHL_fire_worst value = SHL_fire_gap_state }
	}
	if = {
		limit = { check_variable = { var = SHL_fire_gap_front value = SHL_fire_worst compare = less_than } }
		set_temp_variable = { var = SHL_fire_to value = 4 }
		set_temp_variable = { var = SHL_fire_worst value = SHL_fire_gap_front }
	}
	set_temp_variable = { var = SHL_fire_amount value = 4 }
	if = {
		limit = { check_variable = { var = SHL_fire_best value = SHL_fire_amount compare = less_than } }
		set_temp_variable = { var = SHL_fire_amount value = SHL_fire_best }
	}
	set_temp_variable = { var = SHL_fire_need value = SHL_fire_worst }
	multiply_temp_variable = { var = SHL_fire_need value = -1 }
	if = {
		limit = { check_variable = { var = SHL_fire_need value = SHL_fire_amount compare = less_than } }
		set_temp_variable = { var = SHL_fire_amount value = SHL_fire_need }
	}
	if = {
		limit = { check_variable = { var = SHL_fire_amount value = 0 compare = greater_than } }
		SHL_fire_transfer = yes
	}
}

# COUNTRY SHL. A random incident at most once per draw; the cooldown and the
# open-crisis flag keep incidents from piling up.

		limit = {
			SHL_country_operable = yes
			NOT = { has_country_flag = SHL_crisis_cooldown }

		random_list = {
			92 = { }

				modifier = {
					factor = 0.5

				}
				set_country_flag = SHL_crisis_open

					20 = { country_event = { id = ADISCORD_shl_crisis.2 hours = 1 } }
					20 = { country_event = { id = ADISCORD_shl_crisis.3 hours = 1 } }
