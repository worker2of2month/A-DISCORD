# Stelander succession split, conservation, and single postwar finalization.

STP_snapshot_internal_split = {
	save_global_event_target_as = STP_crisis_main_side

	clr_country_flag = STP_prewar_owned_state_1
	clr_country_flag = STP_prewar_owned_state_2
	clr_country_flag = STP_prewar_owned_state_3
	clr_country_flag = STP_prewar_owned_state_28
	clr_country_flag = STP_prewar_owned_state_29
	clr_country_flag = STP_prewar_owned_state_43
	clr_country_flag = STP_prewar_owned_state_44
	clr_country_flag = STP_prewar_owned_state_45
	clr_country_flag = STP_prewar_owned_state_46
	clr_country_flag = STP_prewar_owned_state_53
	clr_country_flag = STP_prewar_owned_state_88

	set_variable = { var = STP_pre_guard_divisions value = num_divisions }
	delete_units = {
		division_template = "Capital Guard"
		disband = yes
	}
	set_variable = { var = STP_post_guard_divisions value = num_divisions }
	if = {
		limit = {
			check_variable = {
				var = STP_post_guard_divisions
				value = var:STP_pre_guard_divisions
				compare = less_than
			}
		}
		set_country_flag = STP_capital_guard_existed
	}
	else = {
		clr_country_flag = STP_capital_guard_existed
	}

	# Snapshot the free stockpile only after the exact guard disband refunds equipment.
	set_variable = {
		var = STP_prewar_stockpile_infantry
		value = num_equipment@infantry_equipment
	}
	set_variable = {
		var = STP_prewar_stockpile_support
		value = num_equipment@support_equipment
	}
	set_variable = {
		var = STP_prewar_resistance_escrow_infantry
		value = var:STP_resistance_escrow_infantry
	}
	set_variable = {
		var = STP_prewar_resistance_escrow_support
		value = var:STP_resistance_escrow_support
	}

	if = { limit = { owns_state = 1 } set_country_flag = STP_prewar_owned_state_1 }
	if = { limit = { owns_state = 2 } set_country_flag = STP_prewar_owned_state_2 }
	if = { limit = { owns_state = 3 } set_country_flag = STP_prewar_owned_state_3 }
	if = { limit = { owns_state = 28 } set_country_flag = STP_prewar_owned_state_28 }
	if = { limit = { owns_state = 29 } set_country_flag = STP_prewar_owned_state_29 }
	if = { limit = { owns_state = 43 } set_country_flag = STP_prewar_owned_state_43 }
	if = { limit = { owns_state = 44 } set_country_flag = STP_prewar_owned_state_44 }
	if = { limit = { owns_state = 45 } set_country_flag = STP_prewar_owned_state_45 }
	if = { limit = { owns_state = 46 } set_country_flag = STP_prewar_owned_state_46 }
	if = { limit = { owns_state = 53 } set_country_flag = STP_prewar_owned_state_53 }
	if = { limit = { owns_state = 88 } set_country_flag = STP_prewar_owned_state_88 }

	# Civil-war countries must be outside all external factions. The finalizer
	# consumes this flag before deciding whether NOD can restore the mandate.
	if = {
		limit = { is_subject_of = NOD }
		set_country_flag = STP_nodrul_mandate_pending
		NOD = {
			set_autonomy = {
				target = ROOT
				autonomy_state = autonomy_free
			}
		}
	}
	if = {
		limit = { is_in_faction = yes }
		leave_faction = yes
	}
}

STP_select_party_starter_states = {
	STP_clear_internal_starter_state_flags = yes
	clr_country_flag = STP_civil_war_starter_selected
	if = {
		limit = { owns_state = 1 }
		1 = { set_state_flag = STP_party_starter_state }
		set_country_flag = STP_civil_war_starter_selected
		set_variable = { var = STP_civil_war_starter_capital value = 1 }
	}
	if = {
		limit = { owns_state = 2 }
		2 = { set_state_flag = STP_party_starter_state }
		set_country_flag = STP_civil_war_starter_selected
		if = {
			limit = { NOT = { owns_state = 1 } }
			set_variable = { var = STP_civil_war_starter_capital value = 2 }
		}
	}
	if = {
		limit = { owns_state = 3 }
		3 = { set_state_flag = STP_party_starter_state }
		set_country_flag = STP_civil_war_starter_selected
		if = {
			limit = { NOT = { owns_state = 1 } NOT = { owns_state = 2 } }
			set_variable = { var = STP_civil_war_starter_capital value = 3 }
		}
	}
	if = {
		limit = { owns_state = 28 }
		28 = { set_state_flag = STP_party_starter_state }
		set_country_flag = STP_civil_war_starter_selected
		set_variable = { var = STP_civil_war_starter_capital value = 28 }
	}
	if = {
		limit = { NOT = { has_country_flag = STP_civil_war_starter_selected } }
		STP_select_resistance_starter_states = yes
		if = {
			limit = { has_country_flag = STP_civil_war_starter_selected }
			1 = { clr_state_flag = STP_resistance_starter_state }
			2 = { clr_state_flag = STP_resistance_starter_state }
			3 = { clr_state_flag = STP_resistance_starter_state }
			28 = { clr_state_flag = STP_resistance_starter_state }
			29 = { clr_state_flag = STP_resistance_starter_state }
			43 = { clr_state_flag = STP_resistance_starter_state }
			44 = { clr_state_flag = STP_resistance_starter_state }
			45 = { clr_state_flag = STP_resistance_starter_state }
			46 = { clr_state_flag = STP_resistance_starter_state }
			53 = { clr_state_flag = STP_resistance_starter_state }
			88 = { clr_state_flag = STP_resistance_starter_state }
			if = { limit = { owns_state = 1 } 1 = { set_state_flag = STP_party_starter_state } }
			else_if = { limit = { owns_state = 2 } 2 = { set_state_flag = STP_party_starter_state } }
			else_if = { limit = { owns_state = 3 } 3 = { set_state_flag = STP_party_starter_state } }
			else_if = { limit = { owns_state = 28 } 28 = { set_state_flag = STP_party_starter_state } }
			else_if = { limit = { owns_state = 29 } 29 = { set_state_flag = STP_party_starter_state } }
			else_if = { limit = { owns_state = 43 } 43 = { set_state_flag = STP_party_starter_state } }
			else_if = { limit = { owns_state = 44 } 44 = { set_state_flag = STP_party_starter_state } }
			else_if = { limit = { owns_state = 45 } 45 = { set_state_flag = STP_party_starter_state } }
			else_if = { limit = { owns_state = 46 } 46 = { set_state_flag = STP_party_starter_state } }
			else_if = { limit = { owns_state = 53 } 53 = { set_state_flag = STP_party_starter_state } }
			else_if = { limit = { owns_state = 88 } 88 = { set_state_flag = STP_party_starter_state } }
		}
	}
}

	clr_country_flag = STP_main_campaign_side

	clr_country_flag = STP_internal_outcome_pending
	clr_country_flag = STP_civil_war_starter_selected

	}
}

STP_start_resistance_revolt_chauvinism = {
	STP_snapshot_internal_split = yes
	clear_global_event_target = STP_crisis_party_side
	clear_global_event_target = STP_crisis_resistance_side
	clr_country_flag = STP_crisis_resistance_side
	set_country_flag = STP_internal_war_started
	set_country_flag = STP_crisis_party_side
	save_global_event_target_as = STP_crisis_party_side
	set_politics = { ruling_party = hedonism elections_allowed = no }
	STP_select_resistance_starter_states = yes

	if = {
		limit = { check_variable = { var = STP_node_officers value = 0 compare = equals } }
		start_civil_war = {
			ideology = chauvinism
			ruling_party = hedonism
			size = 0
			army_ratio = 0
			keep_all_characters = yes
			capital = var:STP_civil_war_starter_capital
			states = all
			states_filter = { has_state_flag = STP_resistance_starter_state }
			STP_prepare_resistance_revolter = yes
		}
	}
	else_if = {
		limit = { check_variable = { var = STP_node_officers value = 1 compare = equals } }
		start_civil_war = {
			ideology = chauvinism
			ruling_party = hedonism
			size = 0

			capital = var:STP_civil_war_starter_capital
			states = all
			states_filter = { has_state_flag = STP_resistance_starter_state }
			STP_prepare_resistance_revolter = yes
		}
	}
	else_if = {
		limit = { check_variable = { var = STP_node_officers value = 2 compare = equals } }
		start_civil_war = {
			ideology = chauvinism
			ruling_party = hedonism
			size = 0
			army_ratio = 0.35
			keep_all_characters = yes
			capital = var:STP_civil_war_starter_capital
			states = all
			states_filter = { has_state_flag = STP_resistance_starter_state }
			STP_prepare_resistance_revolter = yes
		}
	}
	else = {
		start_civil_war = {
			ideology = chauvinism
			ruling_party = hedonism
			size = 0
			army_ratio = 0.5
			keep_all_characters = yes
			capital = var:STP_civil_war_starter_capital
			states = all
			states_filter = { has_state_flag = STP_resistance_starter_state }
			STP_prepare_resistance_revolter = yes
		}
	}

	every_country = {
		limit = {
			original_tag = STP
			has_war_with = ROOT
			NOT = { has_country_flag = STP_main_campaign_side }
			has_country_flag = STP_crisis_resistance_side
		}
		save_global_event_target_as = STP_crisis_resistance_side
	}
	STP_apply_civil_war_state_map = yes
	STP_reconcile_internal_stockpiles = yes
	STP_transfer_resistance_escrow = yes
	STP_materialize_resistance_packages = yes
	STP_assign_internal_characters = yes
	if = {
		limit = {
			has_event_target = STP_crisis_party_side
			country_exists = event_target:STP_crisis_party_side
			has_event_target = STP_crisis_resistance_side
			country_exists = event_target:STP_crisis_resistance_side
		}
		event_target:STP_crisis_party_side = {
			country_event = { id = stp_crisis.51 days = 3 }
		}
		event_target:STP_crisis_resistance_side = {
			country_event = { id = stp_crisis.51 days = 3 }
		}
	}
}

STP_apply_civil_war_state_map = {
	if = {
		limit = {
			has_event_target = STP_crisis_party_side
			country_exists = event_target:STP_crisis_party_side
			has_event_target = STP_crisis_resistance_side
			country_exists = event_target:STP_crisis_resistance_side
		}
		set_variable = { var = STP_resistance_mapped_states value = 0 }

		# The three party-base states are never earned by a resistance node.
		if = {
			limit = { has_country_flag = STP_prewar_owned_state_1 }
			event_target:STP_crisis_party_side = { transfer_state = 1 }
		}
		if = {
			limit = { has_country_flag = STP_prewar_owned_state_2 }
			event_target:STP_crisis_party_side = { transfer_state = 2 }
		}
		if = {
			limit = { has_country_flag = STP_prewar_owned_state_3 }
			event_target:STP_crisis_party_side = { transfer_state = 3 }
		}

		if = {
			limit = {
				has_country_flag = STP_prewar_owned_state_28
				has_country_flag = STP_outcome_shabrat_main_war
			}
			event_target:STP_crisis_party_side = { transfer_state = 28 }
		}
		else_if = {
			limit = {
				has_country_flag = STP_prewar_owned_state_28
				check_variable = { var = STP_node_palace value = 2 compare = equals }
				has_country_flag = STP_capital_guard_loyal_to_resistance
			}
			event_target:STP_crisis_resistance_side = { transfer_state = 28 }
			add_to_variable = { var = STP_resistance_mapped_states value = 1 }
		}
		else_if = {
			limit = { has_country_flag = STP_prewar_owned_state_28 }
			event_target:STP_crisis_party_side = { transfer_state = 28 }
		}

		if = {
			limit = {
				has_country_flag = STP_prewar_owned_state_29
				check_variable = { var = STP_node_officers value = 1 compare = greater_than_or_equals }
			}
			event_target:STP_crisis_resistance_side = { transfer_state = 29 }
			add_to_variable = { var = STP_resistance_mapped_states value = 1 }
		}
		else_if = {
			limit = { has_country_flag = STP_prewar_owned_state_29 }
			event_target:STP_crisis_party_side = { transfer_state = 29 }
		}

		if = {
			limit = {
				has_country_flag = STP_prewar_owned_state_43
				check_variable = { var = STP_node_mountains value = 1 compare = greater_than_or_equals }
			}
			event_target:STP_crisis_resistance_side = { transfer_state = 43 }
			add_to_variable = { var = STP_resistance_mapped_states value = 1 }
		}
		else_if = {
			limit = { has_country_flag = STP_prewar_owned_state_43 }
			event_target:STP_crisis_party_side = { transfer_state = 43 }
		}

		if = {
			limit = {
				has_country_flag = STP_prewar_owned_state_44
				check_variable = { var = STP_node_mountains value = 2 compare = greater_than_or_equals }
			}
			event_target:STP_crisis_resistance_side = { transfer_state = 44 }
			add_to_variable = { var = STP_resistance_mapped_states value = 1 }
		}
		else_if = {
			limit = { has_country_flag = STP_prewar_owned_state_44 }
			event_target:STP_crisis_party_side = { transfer_state = 44 }
		}

		if = {
			limit = {
				has_country_flag = STP_prewar_owned_state_45
				check_variable = { var = STP_node_market value = 2 compare = equals }
			}
			event_target:STP_crisis_resistance_side = { transfer_state = 45 }
			add_to_variable = { var = STP_resistance_mapped_states value = 1 }
		}
		else_if = {
			limit = { has_country_flag = STP_prewar_owned_state_45 }
			event_target:STP_crisis_party_side = { transfer_state = 45 }
		}

		if = {
			limit = {
				has_country_flag = STP_prewar_owned_state_46
				check_variable = { var = STP_node_street value = 1 compare = greater_than_or_equals }
			}
			event_target:STP_crisis_resistance_side = { transfer_state = 46 }
			add_to_variable = { var = STP_resistance_mapped_states value = 1 }
		}
		else_if = {
			limit = { has_country_flag = STP_prewar_owned_state_46 }
			event_target:STP_crisis_party_side = { transfer_state = 46 }
		}

		if = {
			limit = {
				has_country_flag = STP_prewar_owned_state_53
				check_variable = { var = STP_node_mountains value = 3 compare = equals }
			}
			event_target:STP_crisis_resistance_side = { transfer_state = 53 }
			add_to_variable = { var = STP_resistance_mapped_states value = 1 }
		}
		else_if = {
			limit = { has_country_flag = STP_prewar_owned_state_53 }
			event_target:STP_crisis_party_side = { transfer_state = 53 }
		}

		if = {
			limit = {
				has_country_flag = STP_prewar_owned_state_88
				has_country_flag = STP_garrison_88_loyal_to_resistance
			}
			event_target:STP_crisis_resistance_side = { transfer_state = 88 }
			add_to_variable = { var = STP_resistance_mapped_states value = 1 }
		}
		else_if = {
			limit = { has_country_flag = STP_prewar_owned_state_88 }
			event_target:STP_crisis_party_side = { transfer_state = 88 }
		}

		# A zero-node resistance still receives one manifest state, never a random state.
		if = {
			limit = {
				check_variable = {
					var = STP_resistance_mapped_states
					value = 0
					compare = equals
				}
			}
			if = {
				limit = { has_country_flag = STP_prewar_owned_state_43 }
				event_target:STP_crisis_resistance_side = {
					transfer_state = 43
					set_country_flag = STP_resistance_isolated_fallback
				}
			}
		}

		event_target:STP_crisis_party_side = {
			if = { limit = { owns_state = 28 } set_capital = { state = 28 } }
			else = { set_capital = { state = 1 } }
		}
		event_target:STP_crisis_resistance_side = {
			if = { limit = { owns_state = 28 } set_capital = { state = 28 } }
			else_if = { limit = { owns_state = 29 } set_capital = { state = 29 } }
			else_if = { limit = { owns_state = 43 } set_capital = { state = 43 } }
			else_if = { limit = { owns_state = 44 } set_capital = { state = 44 } }
			else_if = { limit = { owns_state = 45 } set_capital = { state = 45 } }
			else_if = { limit = { owns_state = 46 } set_capital = { state = 46 } }
			else_if = { limit = { owns_state = 53 } set_capital = { state = 53 } }
			else_if = { limit = { owns_state = 88 } set_capital = { state = 88 } }
			else_if = { limit = { owns_state = 1 } set_capital = { state = 1 } }
			else_if = { limit = { owns_state = 2 } set_capital = { state = 2 } }
			else_if = { limit = { owns_state = 3 } set_capital = { state = 3 } }
		}
	}

	1 = { clr_state_flag = STP_party_starter_state clr_state_flag = STP_resistance_starter_state }
	2 = { clr_state_flag = STP_party_starter_state clr_state_flag = STP_resistance_starter_state }
	3 = { clr_state_flag = STP_party_starter_state clr_state_flag = STP_resistance_starter_state }
	28 = { clr_state_flag = STP_party_starter_state clr_state_flag = STP_resistance_starter_state }
	29 = { clr_state_flag = STP_party_starter_state clr_state_flag = STP_resistance_starter_state }
	43 = { clr_state_flag = STP_party_starter_state clr_state_flag = STP_resistance_starter_state }
	44 = { clr_state_flag = STP_party_starter_state clr_state_flag = STP_resistance_starter_state }
	45 = { clr_state_flag = STP_party_starter_state clr_state_flag = STP_resistance_starter_state }
	46 = { clr_state_flag = STP_party_starter_state clr_state_flag = STP_resistance_starter_state }
	53 = { clr_state_flag = STP_party_starter_state clr_state_flag = STP_resistance_starter_state }
	88 = { clr_state_flag = STP_party_starter_state clr_state_flag = STP_resistance_starter_state }

STP_reconcile_internal_stockpiles = {
	if = {
		limit = {
			has_event_target = STP_crisis_party_side
			country_exists = event_target:STP_crisis_party_side
			has_event_target = STP_crisis_resistance_side
			country_exists = event_target:STP_crisis_resistance_side
		}
		event_target:STP_crisis_party_side = {
			set_variable = {
				var = STP_split_stockpile_infantry
				value = num_equipment@infantry_equipment
			}
			set_variable = {
				var = STP_split_stockpile_support
				value = num_equipment@support_equipment
			}
			ROOT = {
				set_variable = {
					var = STP_split_total_infantry
					value = PREV.STP_split_stockpile_infantry
				}
				set_variable = {
					var = STP_split_total_support
					value = PREV.STP_split_stockpile_support
				}
			}
		}
		event_target:STP_crisis_resistance_side = {
			set_variable = {
				var = STP_split_stockpile_infantry
				value = num_equipment@infantry_equipment
			}
			set_variable = {
				var = STP_split_stockpile_support
				value = num_equipment@support_equipment
			}
			ROOT = {
				add_to_variable = {
					var = STP_split_total_infantry
					value = PREV.STP_split_stockpile_infantry
				}
				add_to_variable = {
					var = STP_split_total_support
					value = PREV.STP_split_stockpile_support
				}
			}
		}
		set_variable = {
			var = STP_split_delta_infantry
			value = var:STP_prewar_stockpile_infantry
		}
		subtract_from_variable = {
			var = STP_split_delta_infantry
			value = var:STP_split_total_infantry
		}
		set_variable = {
			var = STP_split_delta_support
			value = var:STP_prewar_stockpile_support
		}
		subtract_from_variable = {
			var = STP_split_delta_support
			value = var:STP_split_total_support
		}
		add_equipment_to_stockpile = {
			type = infantry_equipment
			amount = var:STP_split_delta_infantry
		}
		add_equipment_to_stockpile = {
			type = support_equipment
			amount = var:STP_split_delta_support
		}
	}
}

				type = support_equipment
				amount = ROOT.STP_prewar_resistance_escrow_support
				producer = ROOT
			}
		}
		# Clear every copied source only after the one real-stock transfer above.
		event_target:STP_crisis_party_side = {

			set_variable = { var = STP_resistance_escrow_support value = 0 }

			set_variable = { var = STP_resistance_escrow_infantry value = 0 }

		}
		set_variable = { var = STP_resistance_escrow_infantry value = 0 }
		set_variable = { var = STP_resistance_escrow_support value = 0 }
	}
}

STP_materialize_resistance_packages = {
	if = {
		limit = {
			has_event_target = STP_crisis_party_side
			country_exists = event_target:STP_crisis_party_side
			has_event_target = STP_crisis_resistance_side
			country_exists = event_target:STP_crisis_resistance_side
		}
		# Mountains spend the eligibility budget before the street network.
		if = {
			limit = {
				check_variable = { var = STP_node_mountains value = 1 compare = greater_than_or_equals }
				check_variable = { var = STP_militia_budget_infantry value = 600 compare = greater_than_or_equals }
			}
			subtract_from_variable = { var = STP_militia_budget_infantry value = 600 }
			event_target:STP_crisis_resistance_side = {
				STP_create_empty_mountain_militia = yes
			}
		}
		if = {
			limit = {
				check_variable = { var = STP_node_mountains value = 2 compare = greater_than_or_equals }
				check_variable = { var = STP_militia_budget_infantry value = 600 compare = greater_than_or_equals }
			}
			subtract_from_variable = { var = STP_militia_budget_infantry value = 600 }
			event_target:STP_crisis_resistance_side = {
				STP_create_empty_mountain_militia = yes
			}
		}
		if = {
			limit = {
				check_variable = { var = STP_node_mountains value = 3 compare = equals }
				check_variable = { var = STP_militia_budget_infantry value = 600 compare = greater_than_or_equals }
			}
			subtract_from_variable = { var = STP_militia_budget_infantry value = 600 }
			event_target:STP_crisis_resistance_side = {
				STP_create_empty_mountain_militia = yes
			}
		}
		if = {
			limit = {
				check_variable = { var = STP_node_street value = 1 compare = greater_than_or_equals }
				check_variable = { var = STP_militia_budget_infantry value = 400 compare = greater_than_or_equals }
			}
			subtract_from_variable = { var = STP_militia_budget_infantry value = 400 }
			event_target:STP_crisis_resistance_side = {
				STP_create_empty_urban_militia = yes
			}
		}
		if = {
			limit = {
				check_variable = { var = STP_node_street value = 2 compare = equals }
				check_variable = { var = STP_militia_budget_infantry value = 400 compare = greater_than_or_equals }
			}
			subtract_from_variable = { var = STP_militia_budget_infantry value = 400 }
			event_target:STP_crisis_resistance_side = {
				STP_create_empty_urban_militia = yes
			}
		}

		# The original guard is recreated once, empty, on its chosen loyalty side.
		if = {
			limit = {
				has_country_flag = STP_capital_guard_existed
				has_country_flag = STP_capital_guard_loyal_to_resistance
			}
			event_target:STP_crisis_resistance_side = {
				STP_create_empty_capital_guard = yes
			}
		}
		else_if = {
			limit = { has_country_flag = STP_capital_guard_existed }
			event_target:STP_crisis_party_side = {
				STP_create_empty_capital_guard = yes
			}
		}

		# Officer preparation is fixed by tier and never randomized.
		if = {
			limit = { check_variable = { var = STP_node_officers value = 1 compare = equals } }
			event_target:STP_crisis_resistance_side = {
				add_command_power = 10
				add_timed_idea = { idea = STP_officer_preparation_1 days = 30 }
			}
		}
		else_if = {
			limit = { check_variable = { var = STP_node_officers value = 2 compare = equals } }
			event_target:STP_crisis_resistance_side = {
				add_command_power = 20
				add_timed_idea = { idea = STP_officer_preparation_2 days = 60 }
			}
		}
		else_if = {
			limit = { check_variable = { var = STP_node_officers value = 3 compare = equals } }
			event_target:STP_crisis_resistance_side = {
				add_command_power = 30
				add_timed_idea = { idea = STP_officer_preparation_3 days = 90 }
			}
		}

		if = {
			limit = { check_variable = { var = STP_node_market value = 1 compare = greater_than_or_equals } }
			event_target:STP_crisis_party_side = {
				activate_mission = STP_party_market_sabotage
			}
		}
		if = {
			limit = { check_variable = { var = STP_node_market value = 2 compare = equals } }
			event_target:STP_crisis_party_side = {
				set_variable = {
					var = STP_market_transfer_infantry
					value = num_equipment@infantry_equipment
				}
				clamp_variable = {
					var = STP_market_transfer_infantry
					min = 0
					max = 200
				}
				set_variable = {
					var = STP_market_transfer_support
					value = num_equipment@support_equipment
				}
				clamp_variable = {
					var = STP_market_transfer_support
					min = 0
					max = 50
				}
				set_variable = {
					var = STP_market_transfer_infantry_negative
					value = var:STP_market_transfer_infantry
				}
				multiply_variable = {
					var = STP_market_transfer_infantry_negative
					value = -1
				}
				set_variable = {
					var = STP_market_transfer_support_negative
					value = var:STP_market_transfer_support
				}
				multiply_variable = {
					var = STP_market_transfer_support_negative
					value = -1
				}
				add_equipment_to_stockpile = {
					type = infantry_equipment
					amount = var:STP_market_transfer_infantry_negative
				}
				add_equipment_to_stockpile = {
					type = support_equipment
					amount = var:STP_market_transfer_support_negative
				}
				event_target:STP_crisis_resistance_side = {
					add_equipment_to_stockpile = {
						type = infantry_equipment
						amount = PREV.STP_market_transfer_infantry
						producer = ROOT
					}
					add_equipment_to_stockpile = {
						type = support_equipment
						amount = PREV.STP_market_transfer_support
						producer = ROOT
					}
				}
			}
		}
	}
}

# The civil war never enters the peace-conference system. The surviving
# canonical side first closes the exact internal war, absorbs the defeated
# revolter and only then runs the shared post-war finalizer.
STP_resolve_scripted_internal_victory = {
	if = {
		limit = {
			has_event_target = STP_internal_capitulated_side
			country_exists = event_target:STP_internal_capitulated_side
			OR = {
				has_country_flag = STP_crisis_party_side
				has_country_flag = STP_crisis_resistance_side
			}
		}
		if = {
			limit = { has_war_with = event_target:STP_internal_capitulated_side }
			white_peace = { tag = event_target:STP_internal_capitulated_side }
		}
		annex_country = {
			target = event_target:STP_internal_capitulated_side
			transfer_troops = yes
		}
		if = {
			limit = { has_country_flag = STP_crisis_party_side }
			set_country_flag = STP_winner_hedersett
		}
		else_if = {
			limit = { has_country_flag = STP_resistance_leader_sotnikov }
			set_country_flag = STP_winner_sotnikov
		}
		else = { set_country_flag = STP_winner_shabrat }
		STP_finalize_internal_outcome = yes
	}
}

STP_finalize_internal_outcome = {
	if = {
		limit = {
			NOT = { has_country_flag = STP_internal_outcome_finalizing }
			NOT = { has_country_flag = STP_internal_outcome_finalized }
		}
		set_country_flag = STP_internal_outcome_finalizing
		save_global_event_target_as = STP_postwar_country
		set_country_flag = STP_postwar_campaign_side
		set_temp_variable = { var = ADISCORD_STP_VAL_effect_value value = 3 } STP_set_crisis_phase = yes

		# Consume the saved mandate state while all role targets still exist.
		STP_resolve_postwar_nodrul_mandate = yes

		clr_country_flag = STP_main_campaign_side
		clr_country_flag = STP_crisis_party_side
		clr_country_flag = STP_crisis_resistance_side
		clr_country_flag = STP_health_calendar_started

		if = {
			limit = {
				has_event_target = STP_crisis_main_side
				country_exists = event_target:STP_crisis_main_side
			}
			event_target:STP_crisis_main_side = {
				clr_country_flag = STP_main_campaign_side
				clr_country_flag = STP_crisis_party_side
				clr_country_flag = STP_crisis_resistance_side
			}
		}
		if = {
			limit = {
				has_event_target = STP_crisis_party_side
				country_exists = event_target:STP_crisis_party_side
			}
			event_target:STP_crisis_party_side = {
				clr_country_flag = STP_main_campaign_side
				clr_country_flag = STP_crisis_party_side
				clr_country_flag = STP_crisis_resistance_side
			}
		}
		if = {
			limit = {
				has_event_target = STP_crisis_resistance_side
				country_exists = event_target:STP_crisis_resistance_side
			}
			event_target:STP_crisis_resistance_side = {
				clr_country_flag = STP_main_campaign_side
				clr_country_flag = STP_crisis_party_side
				clr_country_flag = STP_crisis_resistance_side
			}
		}

		STP_assign_postwar_leader = yes
		load_focus_tree = {
			tree = ADISCORD_STP_postwar_focus
			keep_completed = no
		}
		STP_clear_external_crisis_participants = yes

		if = {
			limit = { has_country_flag = STP_no_war_outcome }
			set_country_flag = {
				flag = VAL_STP_deferred_invoice_window
				days = 60
				value = 1
			}
		}
		VAL_STP_start_war_countdown_120 = yes

		set_country_flag = STP_internal_outcome_finalized
		clr_country_flag = STP_internal_outcome_finalizing
		clr_country_flag = STP_internal_outcome_pending

		clear_global_event_target = STP_crisis_main_side
		clear_global_event_target = STP_crisis_party_side
		clear_global_event_target = STP_crisis_resistance_side
	}
}

STP_try_finalize_internal_war = {
	if = {
		limit = {
			NOT = { has_country_flag = STP_internal_outcome_finalizing }
			NOT = { has_country_flag = STP_internal_outcome_finalized }
			has_event_target = STP_crisis_party_side
			has_event_target = STP_crisis_resistance_side
		}
		clr_country_flag = STP_party_role_still_viable
		clr_country_flag = STP_resistance_role_still_viable

		if = {
			limit = { country_exists = event_target:STP_crisis_party_side }
			event_target:STP_crisis_party_side = {
				if = {
					limit = {
						NOT = { has_capitulated = yes }
						STP_controls_any_internal_manifest_state = yes
					}
					ROOT = { set_country_flag = STP_party_role_still_viable }
				}
			}
		}
		if = {
			limit = { country_exists = event_target:STP_crisis_resistance_side }
			event_target:STP_crisis_resistance_side = {
				if = {
					limit = {
						NOT = { has_capitulated = yes }
						STP_controls_any_internal_manifest_state = yes
					}
					ROOT = { set_country_flag = STP_resistance_role_still_viable }
				}
			}
		}

		if = {
			limit = {
				has_country_flag = STP_party_role_still_viable
				NOT = { has_country_flag = STP_resistance_role_still_viable }
				country_exists = event_target:STP_crisis_party_side
			}
			event_target:STP_crisis_party_side = {
				set_country_flag = STP_winner_hedersett
				STP_finalize_internal_outcome = yes
			}
		}
		else_if = {
			limit = {
				has_country_flag = STP_resistance_role_still_viable
				NOT = { has_country_flag = STP_party_role_still_viable }
				country_exists = event_target:STP_crisis_resistance_side
			}
			event_target:STP_crisis_resistance_side = {
				if = {
					limit = { has_country_flag = STP_resistance_leader_sotnikov }
					set_country_flag = STP_winner_sotnikov
				}
				else = {
					set_country_flag = STP_winner_shabrat
				}
				STP_finalize_internal_outcome = yes
			}
		}
	}
}

NOD_refresh_army_equipment_ratio = {
	set_temp_variable = {
		var = NOD_army_equipment_actual
		value = num_equipment_in_armies@infantry_equipment
	}
	add_to_temp_variable = {
		var = NOD_army_equipment_actual
		value = num_equipment_in_armies@ADISCORD_squad_weapons_equipment
	}
	add_to_temp_variable = {
		var = NOD_army_equipment_actual
		value = num_equipment_in_armies@support_equipment
	}
	add_to_temp_variable = {
		var = NOD_army_equipment_actual
		value = num_equipment_in_armies@artillery_equipment
	}
	set_temp_variable = {
		var = NOD_army_equipment_target
		value = num_target_equipment_in_armies@infantry_equipment
	}
	add_to_temp_variable = {
		var = NOD_army_equipment_target
		value = num_target_equipment_in_armies@ADISCORD_squad_weapons_equipment
	}
	add_to_temp_variable = {
		var = NOD_army_equipment_target
		value = num_target_equipment_in_armies@support_equipment
	}
	add_to_temp_variable = {
		var = NOD_army_equipment_target
		value = num_target_equipment_in_armies@artillery_equipment
	}
	multiply_temp_variable = {
		var = NOD_army_equipment_target
		value = 0.85
	}
	set_variable = { var = NOD_army_equipment_ratio value = 0 }
	if = {
		limit = {
			check_variable = {
				var = NOD_army_equipment_target
				value = 0
				compare = less_than_or_equals
			}
		}
		set_variable = { var = NOD_army_equipment_ratio value = 100 }
	}
	else_if = {
		limit = {
			check_variable = {
				var = NOD_army_equipment_actual
				value = NOD_army_equipment_target
				compare = greater_than_or_equals
			}
		}
		set_variable = { var = NOD_army_equipment_ratio value = 85 }
	}
	else = {
		# Reuse the 85% target to expose a second, lower threshold for the
		# exceptional early-defense reaction without adding a polling tick.
		multiply_temp_variable = {
			var = NOD_army_equipment_target
			value = 0.823529
		}
		if = {
			limit = {
				check_variable = {
					var = NOD_army_equipment_actual
					value = NOD_army_equipment_target
					compare = greater_than_or_equals
				}
			}
			set_variable = { var = NOD_army_equipment_ratio value = 70 }
		}
	}
}

NOD_evaluate_limited_war_losses = {
	clr_country_flag = NOD_limited_war_pyrrhic
	set_temp_variable = {
		var = NOD_limited_current_deployed_manpower
		value = deployed_army_manpower_k
	}
	set_temp_variable = { var = NOD_limited_losses value = casualties_k }
	subtract_from_temp_variable = {
		var = NOD_limited_losses
		value = NOD_limited_start_casualties
	}
	clamp_temp_variable = { var = NOD_limited_losses min = 0 }
	set_temp_variable = {
		var = NOD_limited_pyrrhic_manpower_threshold
		value = NOD_limited_start_deployed_manpower
	}
	multiply_temp_variable = {
		var = NOD_limited_pyrrhic_manpower_threshold
		value = 0.08
	}
	if = {
		limit = {
			check_variable = {
				var = NOD_limited_losses
				value = NOD_limited_pyrrhic_manpower_threshold
				compare = greater_than
			}
		}
		set_country_flag = NOD_limited_war_pyrrhic
	}
	if = {
		limit = {
			has_event_target = NOD_limited_war_target_country
			country_exists = event_target:NOD_limited_war_target_country
		}
		set_temp_variable = {
			var = NOD_limited_target_losses
			value = event_target:NOD_limited_war_target_country.casualties_k
		}
		subtract_from_temp_variable = {
			var = NOD_limited_target_losses
			value = event_target:NOD_limited_war_target_country.NOD_limited_start_casualties
		}
		clamp_temp_variable = { var = NOD_limited_target_losses min = 0 }
		multiply_temp_variable = {
			var = NOD_limited_target_losses
			value = 1.5
		}
		if = {
			limit = {
				check_variable = {
					var = NOD_limited_losses
					value = NOD_limited_target_losses
					compare = greater_than
				}
			}
			set_country_flag = NOD_limited_war_pyrrhic
		}
	}
	if = {
		limit = {
			has_country_flag = NOD_limited_resolution_victory
			NOT = { has_country_flag = NOD_limited_war_pyrrhic }
		}
		set_temp_variable = { var = ADISCORD_STP_VAL_effect_value value = 10 } NOD_change_crisis_attention = yes
	}
	else_if = {
		limit = {
			has_country_flag = NOD_limited_resolution_victory
			has_country_flag = NOD_limited_war_pyrrhic
		}
		set_temp_variable = { var = ADISCORD_STP_VAL_effect_value value = -5 } NOD_change_crisis_attention = yes
	}
}
