	every_country = {
		limit = { is_ai = no }
		scoped_sound_effect = superevent_vorkerland_civilwar_sound_e

# Central victory records the claimant and unlocks border-by-border
# reunification decisions. Normal capitulation and peace mechanics retain all
# actual state transfers; these effects deliberately annex and puppet nobody.
ADISCORD_vorkerland_apply_worker_map = {
	WKR = { set_country_flag = ADISCORD_vorkerland_central_unifier }
	ADISCORD_vorkerland_begin_reunification = yes
}

ADISCORD_vorkerland_apply_vlad_map = {
	VAD = {
		set_country_flag = ADISCORD_vorkerland_central_unifier
		if = { limit = { has_global_flag = ADISCORD_vorkerland_worker_rescued_by_vlad } set_country_flag = ADISCORD_vorkerland_worker_preserved }
		if = { limit = { has_global_flag = ADISCORD_vorkerland_worker_missing } set_country_flag = ADISCORD_vorkerland_worker_unaccounted_for }
		if = { limit = { has_global_flag = ADISCORD_vorkerland_worker_killed } set_country_flag = ADISCORD_vorkerland_worker_martyr }
	}
	ADISCORD_vorkerland_begin_reunification = yes
}

ADISCORD_vorkerland_apply_dorian_map = {
	TVA = { set_country_flag = ADISCORD_vorkerland_central_unifier }
	ADISCORD_vorkerland_begin_reunification = yes
}
