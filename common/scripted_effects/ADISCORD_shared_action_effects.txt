# Shared action-accounting API. Country decisions should use these effects
# instead of changing the treasury without recording the transaction.
ADISCORD_economy_spend_50 = {
	ADISCORD_economy_initialize_country = yes
	add_to_variable = { var = ADISCORD_economy_treasury value = -50 }
	add_to_variable = { var = ADISCORD_economy_current_month_action_costs value = 50 }
	ADISCORD_economy_mark_dirty = yes
}

# Propaganda campaigns use a country-scoped pool. HOI4 has no native slot
# effects, so keep the accounting in a scripted variable.
ADISCORD_campaign_slot_grant = {
	add_to_variable = { var = ADISCORD_available_campaign_slots value = 1 }
}

ADISCORD_campaign_slot_consume = {
	if = {
		limit = { ADISCORD_has_campaign_slot = yes }
		add_to_variable = { var = ADISCORD_available_campaign_slots value = -1 }
	}
}

ADISCORD_campaign_slot_release = {
	add_to_variable = { var = ADISCORD_available_campaign_slots value = 1 }
}
