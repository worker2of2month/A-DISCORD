		if = {
			limit = {
				PREV = {
					tag = ROOT
					has_war_with = FROM
				}
				NOT = { has_war_with = FROM }
				NOT = {
					AND = {
						OR = { tag = STP tag = SLI }
						PREV = {
							OR = {
								tag = NOD
								is_subject_of = NOD
							}
						}

					}
				}

				targeted_alliance = ROOT
				enemy = FROM
				hostility_reason = asked_to_join
				single_target_only = yes
			}
		}
		else_if = {
			limit = {
				PREV = {
					tag = FROM
					has_war_with = ROOT
				}
				NOT = { has_war_with = ROOT }
				NOT = {
					AND = {
						OR = { tag = STP tag = SLI }
						PREV = {
							OR = {
								tag = NOD
								is_subject_of = NOD
							}
						}
						ROOT = { STP_cw_northern_enemy_blocks_subject_entry = yes }
					}
				}
			}
			add_to_war = {
				targeted_alliance = FROM
				enemy = ROOT
				hostility_reason = asked_to_join
				single_target_only = yes
			}
