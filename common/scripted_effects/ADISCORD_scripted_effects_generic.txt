# COUNTRY candidate, PREV is its subject/overlord already fighting.
# ROOT/FROM retain the opponents supplied by on_war_relation_added.
ADISCORD_join_subject_war = {
	if = {
		limit = {
			exists = yes
			NOT = { tag = ROOT }
			NOT = { tag = FROM }
			NOT = { has_war_with = PREV }
		}
		if = {
			limit = {
				PREV = {
					tag = ROOT
					has_war_with = FROM
				}
				NOT = { has_war_with = FROM }
				NOT = {
					AND = {
						OR = {
							tag = STP
							tag = SLI
						}
						PREV = {
							OR = {
								tag = NOD
								is_subject_of = NOD
							}
						}
						FROM = { STP_cw_northern_enemy_blocks_subject_entry = yes }
					}
				}
			}
			add_to_war = {
				targeted_alliance = ROOT
				enemy = FROM
				hostility_reason = asked_to_join
				single_target_only = yes
			}
		}
		else_if = {
			limit = {
				PREV = {
					tag = FROM
					has_war_with = ROOT
				}
				NOT = { has_war_with = ROOT }
				NOT = {
					AND = {
						OR = {
							tag = STP
							tag = SLI
						}
						PREV = {
							OR = {
								tag = NOD
								is_subject_of = NOD
							}
						}
						ROOT = { STP_cw_northern_enemy_blocks_subject_entry = yes }
					}
				}
			}
			add_to_war = {
				targeted_alliance = FROM
				enemy = ROOT
				hostility_reason = asked_to_join
				single_target_only = yes
			}
		}
	}
}

ADISCORD_expand_naval_yards = {
	random_owned_controlled_state = {
		limit = {
			is_coastal = yes
			naval_base > 0
			free_building_slots = {
				building = dockyard
				size > 1
				include_locked = yes
			}
		}
		add_extra_state_shared_building_slots = 2
		add_building_construction = {
			type = dockyard
			level = 2
			instant_build = yes
		}
	}
	hidden_effect = { ADISCORD_economy_mark_dirty = yes }
}
