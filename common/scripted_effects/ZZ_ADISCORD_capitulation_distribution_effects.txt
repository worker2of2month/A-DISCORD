# The deterministic capitulation fallback still needs a single country to
# receive unoccupied land and dissolve the defeated tag. Before that happens,
# preserve the front line: every independent co-belligerent keeps states that
# it actually controls. Subjects leave their gains to their overlord, matching
# normal peace-conference ownership and protecting scripted subject borders.
ADISCORD_distribute_occupied_states_to_cobelligerents = {
	every_owned_state = {
		limit = {
			CONTROLLER = {
				exists = yes
				is_subject = no
				NOT = { has_capitulated = yes }
				NOT = { tag = FROM }
				has_war_together_with = FROM
			}
		}
		CONTROLLER = { transfer_state = PREV }
	}
}

# Central Vorkerland outcomes require all central states to remain reachable by
# the claimant war graph. Restrict the same occupation split to central actors;
# an unrelated regional intervention cannot strand a required victory state.
ADISCORD_vorkerland_distribute_central_states_to_cobelligerents = {
	every_owned_state = {
		limit = {
			CONTROLLER = {
				exists = yes
				is_subject = no
				NOT = { has_capitulated = yes }
				NOT = { tag = FROM }
				has_war_together_with = FROM
				ADISCORD_vorkerland_is_central_claimant = yes
			}
		}
		CONTROLLER = { transfer_state = PREV }
	}
}

# Run in the legitimate living claimant's scope. The package routers execute
# before the human handoff and final annex, so mixed historic settlements keep
# their recorded owners (for example RIV/SWB -> WKR and REV/OSV -> TVA).
ADISCORD_vorkerland_dissolve_premature_wrk_as_claimant = {
	if = {
		limit = {
			ADISCORD_vorkerland_is_main_claimant = yes
			exists = yes
			is_subject = no
			NOT = { has_capitulated = yes }
			WRK = { exists = yes ADISCORD_vorkerland_is_premature_wrk = yes }
		}
		set_global_flag = ADISCORD_vorkerland_premature_wrk_recovery_v1_attempted
		if = { limit = { WKR = { exists = yes } } WKR = { ADISCORD_vorkerland_restore_premature_wrk_packages_to_wkr = yes } }
		if = { limit = { VAD = { exists = yes } } VAD = { ADISCORD_vorkerland_restore_premature_wrk_packages_to_vad = yes } }
		if = { limit = { TVA = { exists = yes } } TVA = { ADISCORD_vorkerland_restore_premature_wrk_packages_to_tva = yes } }
		set_global_flag = ADISCORD_vorkerland_premature_wrk_subject_cleanup_active
		WRK = { ADISCORD_vorkerland_release_losing_claimant_subjects = yes }
		if = {
			limit = { WRK = { exists = yes is_ai = no } }
			change_tag_from = WRK
		}
		if = {
			limit = { country_exists = WRK }
			annex_country = { target = WRK transfer_troops = yes }
		}
		set_global_flag = ADISCORD_vorkerland_premature_wrk_recovery_v1_completed
		clr_global_flag = ADISCORD_vorkerland_premature_wrk_recovery_v1_failed
		clr_global_flag = ADISCORD_vorkerland_premature_wrk_subject_cleanup_active
		log = "[ADISCORD][VORKERLAND][RECOVERY] Premature WRK release dissolved after package-aware owner/controller repair."
	}
}

# Run from the released subject's on_action scope when dormant WRK appeared as
# its overlord. Only one living claimant receives the package-aware dissolve,
# in a fixed priority order; the dissolve itself releases every other WRK
# subject behind a same-tick recursion guard before annexing the false shell.
ADISCORD_vorkerland_route_premature_wrk_release_to_living_claimant = {
	if = {
		limit = { WKR = { exists = yes is_subject = no NOT = { has_capitulated = yes } } }
		WKR = { ADISCORD_vorkerland_dissolve_premature_wrk_as_claimant = yes }
	}
	else_if = {
		limit = { VAD = { exists = yes is_subject = no NOT = { has_capitulated = yes } } }
		VAD = { ADISCORD_vorkerland_dissolve_premature_wrk_as_claimant = yes }
	}
	else_if = {
		limit = { TVA = { exists = yes is_subject = no NOT = { has_capitulated = yes } } }
		TVA = { ADISCORD_vorkerland_dissolve_premature_wrk_as_claimant = yes }
	}
	else = {
		set_global_flag = ADISCORD_vorkerland_premature_wrk_recovery_v1_failed
		log = "[ADISCORD][VORKERLAND][RECOVERY] Premature WRK subject relation had no living claimant receiver."
	}
}

# Run in the central winner's country scope. ROOT remains the capitulated
# country from on_capitulation. Temporary claimants are reserved for phase.6;
# minor tags are marked for the delayed post-peace-cache finalizer.
ADISCORD_vorkerland_settle_central_capitulation = {
	if = {
		limit = { ROOT = { ADISCORD_vorkerland_is_temporary_claimant = yes } }
		# A defeated central claimant must survive as a diagnostic shell until the
		# phase controller has one verified winner. Its subjects and armed forces
		# are released/transferred by postwar formation, never by this fallback.
		ROOT = { set_country_flag = ADISCORD_vorkerland_claimant_cleanup_reserved }
		ADISCORD_vorkerland_begin_reunification = yes
	}
	else = {
		set_country_flag = {
			flag = ADISCORD_vorkerland_central_recovery
			days = 35
			value = 1
		}
		add_manpower = 1500
		if = { limit = { tag = WKR } ROOT = { set_country_flag = ADISCORD_vorkerland_central_minor_winner_wkr } }
		if = { limit = { tag = VAD } ROOT = { set_country_flag = ADISCORD_vorkerland_central_minor_winner_vad } }
		if = { limit = { tag = TVA } ROOT = { set_country_flag = ADISCORD_vorkerland_central_minor_winner_tva } }
		if = {
			limit = {
				tag = WRK
				has_global_flag = ADISCORD_vorkerland_phase_postwar_integration
				has_global_flag = ADISCORD_vorkerland_reunification_verified
			}
			ROOT = { set_country_flag = ADISCORD_vorkerland_central_minor_winner_wrk }
		}
		ROOT = {
			set_country_flag = ADISCORD_vorkerland_central_minor_settlement_pending
			country_event = { id = ADISCORD_vorkerland_collapse.65 days = 1 }
		}
		# Capitulated zero-state countries may discard their own delayed events.
		# Mirror the controller onto the living winner so both new and old saves
		# receive a bounded, country-scoped settlement pass.
		ADISCORD_vorkerland_schedule_central_minor_settlement_recovery = yes
	}
}

ADISCORD_vorkerland_schedule_central_minor_settlement_recovery = {
	if = {
		limit = {
			ADISCORD_vorkerland_has_matching_pending_central_minor_settlement = yes
			NOT = { has_country_flag = ADISCORD_vorkerland_central_minor_settlement_controller_scheduled }
		}
		set_country_flag = {
			flag = ADISCORD_vorkerland_central_minor_settlement_controller_scheduled
			days = 5
			value = 1
		}
		country_event = { id = ADISCORD_vorkerland_collapse.69 days = 1 }
	}
}
