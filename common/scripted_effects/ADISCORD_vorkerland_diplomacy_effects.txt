ADISCORD_vorkerland_attempt_vad_solar_intervention = {
	if = {
		limit = {
			tag = VAD
			has_global_flag = ADISCORD_vorkerland_vad_solar_intervention_reserved
			has_global_flag = ADISCORD_vorkerland_phase_central_preparation
			NOT = { has_global_flag = ADISCORD_vorkerland_focus_central_showdown_requested }
			NOT = { has_global_flag = ADISCORD_vorkerland_showdown_queue_initialized }
			NOT = { has_global_flag = ADISCORD_vorkerland_central_showdown_started }
			NOT = { has_global_flag = ADISCORD_vorkerland_vad_solar_intervention_active }
			is_subject = no
			NOT = { has_capitulated = yes }
			NOT = { has_war = yes }
			ADISCORD_vorkerland_vad_has_solar_intervention_border = yes
		}
		set_country_flag = ADISCORD_vorkerland_vad_solar_intervention_attempted
		if = {
			limit = { has_global_flag = ADISCORD_vorkerland_solar_winner_sra country_exists = SRA }
			set_global_flag = ADISCORD_vorkerland_vad_solar_intervention_active
			set_global_flag = ADISCORD_vorkerland_vad_solar_intervention_target_sra
			SRA = { ADISCORD_vorkerland_leave_inherited_faction = yes }
			declare_war_on = { target = SRA type = take_state_focus generator = { 76 104 198 307 310 } }
			set_country_flag = ADISCORD_vorkerland_vad_solar_intervention_verify_pending
			country_event = { id = ADISCORD_vorkerland_diplomacy.6 days = 1 }
		}
		else_if = {
			limit = { has_global_flag = ADISCORD_vorkerland_solar_winner_csl country_exists = CSL }
			set_global_flag = ADISCORD_vorkerland_vad_solar_intervention_active
			set_global_flag = ADISCORD_vorkerland_vad_solar_intervention_target_csl
			CSL = { ADISCORD_vorkerland_leave_inherited_faction = yes }
			declare_war_on = { target = CSL type = take_state_focus generator = { 76 104 198 307 310 } }
			set_country_flag = ADISCORD_vorkerland_vad_solar_intervention_verify_pending
			country_event = { id = ADISCORD_vorkerland_diplomacy.6 days = 1 }
		}
	}
	ADISCORD_vorkerland_cancel_vad_solar_intervention_reservation = yes
}

ADISCORD_vorkerland_verify_vad_solar_intervention = {
	if = {
		limit = {
			tag = VAD
			has_global_flag = ADISCORD_vorkerland_vad_solar_intervention_active
			has_country_flag = ADISCORD_vorkerland_vad_solar_intervention_verify_pending
			NOT = { has_global_flag = ADISCORD_vorkerland_sol_restoration_pending }
			OR = {
				AND = { has_global_flag = ADISCORD_vorkerland_vad_solar_intervention_target_sra has_war_with = SRA }
				AND = { has_global_flag = ADISCORD_vorkerland_vad_solar_intervention_target_csl has_war_with = CSL }
			}
		}
		clr_country_flag = ADISCORD_vorkerland_vad_solar_intervention_verify_pending
		clr_country_flag = ADISCORD_vorkerland_vad_solar_intervention_verify_retry
	}
	else_if = {
		limit = {
			tag = VAD
			has_global_flag = ADISCORD_vorkerland_vad_solar_intervention_active
			has_country_flag = ADISCORD_vorkerland_vad_solar_intervention_verify_pending
			NOT = { has_global_flag = ADISCORD_vorkerland_sol_restoration_pending }
			NOT = { has_country_flag = ADISCORD_vorkerland_vad_solar_intervention_verify_retry }
		}
		set_country_flag = ADISCORD_vorkerland_vad_solar_intervention_verify_retry
		if = {
			limit = { has_global_flag = ADISCORD_vorkerland_vad_solar_intervention_target_sra country_exists = SRA }
			SRA = { ADISCORD_vorkerland_leave_inherited_faction = yes }
			declare_war_on = { target = SRA type = take_state_focus generator = { 76 104 198 307 310 } }
		}
		else_if = {
			limit = { has_global_flag = ADISCORD_vorkerland_vad_solar_intervention_target_csl country_exists = CSL }
			CSL = { ADISCORD_vorkerland_leave_inherited_faction = yes }
			declare_war_on = { target = CSL type = take_state_focus generator = { 76 104 198 307 310 } }
		}
		country_event = { id = ADISCORD_vorkerland_diplomacy.9 days = 1 }
	}
	else_if = {
		limit = {
			tag = VAD
			has_global_flag = ADISCORD_vorkerland_vad_solar_intervention_active
			has_country_flag = ADISCORD_vorkerland_vad_solar_intervention_verify_pending
			has_country_flag = ADISCORD_vorkerland_vad_solar_intervention_verify_retry
			NOT = { has_global_flag = ADISCORD_vorkerland_sol_restoration_pending }
		}
		clr_country_flag = ADISCORD_vorkerland_vad_solar_intervention_verify_pending
		ADISCORD_vorkerland_clear_failed_vad_solar_intervention = yes
		log = "[ADISCORD][VORKERLAND][DIPLOMACY] VAD Solar intervention failed after one retry."
	}
}

# A failed claimant cannot leave the central-preparation gate permanently
# occupied. This is safe from capitulation, peace, and old-save repair hooks.
ADISCORD_vorkerland_clear_failed_vad_solar_intervention = {
	if = {
		limit = {
			has_global_flag = ADISCORD_vorkerland_vad_solar_intervention_active
			OR = {
				has_global_flag = ADISCORD_vorkerland_sol_restoration_failed
				NOT = { country_exists = VAD }
				VAD = { has_capitulated = yes }
				AND = {
					VAD = {
						exists = yes
						NOT = { has_capitulated = yes }
						NOT = { has_country_flag = ADISCORD_vorkerland_vad_solar_intervention_verify_pending }
					}
					NOT = { has_global_flag = ADISCORD_vorkerland_sol_restoration_verified }
					NOT = { has_global_flag = ADISCORD_vorkerland_sol_restoration_pending }
					OR = {
						AND = {
							has_global_flag = ADISCORD_vorkerland_vad_solar_intervention_target_sra
							OR = {
								NOT = { country_exists = SRA }
								VAD = { NOT = { has_war_with = SRA } }
							}
						}
						AND = {
							has_global_flag = ADISCORD_vorkerland_vad_solar_intervention_target_csl
							OR = {
								NOT = { country_exists = CSL }
								VAD = { NOT = { has_war_with = CSL } }
							}
						}
						NOT = {
							OR = {
								has_global_flag = ADISCORD_vorkerland_vad_solar_intervention_target_sra
								has_global_flag = ADISCORD_vorkerland_vad_solar_intervention_target_csl
							}
						}
					}
				}
			}
		}
		clr_global_flag = ADISCORD_vorkerland_vad_solar_intervention_active
		clr_global_flag = ADISCORD_vorkerland_vad_solar_intervention_target_sra
		clr_global_flag = ADISCORD_vorkerland_vad_solar_intervention_target_csl
		clr_global_flag = ADISCORD_vorkerland_vad_solar_intervention_reserved
		clr_global_flag = ADISCORD_vorkerland_sol_restoration_pending
		clr_global_flag = ADISCORD_vorkerland_sol_restoration_materialized
		clr_global_flag = ADISCORD_vorkerland_sol_restoration_materialization_retry
		clr_global_flag = ADISCORD_vorkerland_sol_restoration_retry
		clr_global_flag = ADISCORD_vorkerland_sol_restoration_fresh_release
		VAD = {
			clr_country_flag = ADISCORD_vorkerland_vad_solar_intervention_verify_pending
			clr_country_flag = ADISCORD_vorkerland_vad_solar_intervention_verify_retry
		}
		set_global_flag = ADISCORD_vorkerland_vad_solar_intervention_failed
	}
}

			NOT = { has_global_flag = ADISCORD_vorkerland_sol_restoration_materialized }
			owns_state = 76 controls_state = 76
			owns_state = 104 controls_state = 104
			owns_state = 198 controls_state = 198
			owns_state = 307 controls_state = 307
			owns_state = 310 controls_state = 310
			NOT = { has_global_flag = ADISCORD_vorkerland_sol_restoration_verified }
		}
		76 = { add_core_of = SOL }
		104 = { add_core_of = SOL }
		198 = { add_core_of = SOL }
		307 = { add_core_of = SOL }
		310 = { add_core_of = SOL }
		if = {
			limit = { NOT = { country_exists = SOL } }
			set_global_flag = ADISCORD_vorkerland_sol_restoration_fresh_release
			release_autonomy = {
				target = SOL
				autonomy_state = autonomy_puppet
				freedom_level = 0.10
			}
		}
		else = {
			SOL = {
				if = { limit = { is_subject = yes } overlord = { set_autonomy = { target = SOL autonomy_state = autonomy_free } } }

			}
			puppet = SOL
			set_autonomy = { target = SOL autonomy_state = autonomy_puppet freedom_level = 0.10 }
			SOL = {
				transfer_state = 76
				transfer_state = 104
				transfer_state = 198
				transfer_state = 307
				transfer_state = 310
			}
		}
		set_global_flag = ADISCORD_vorkerland_sol_restoration_materialized
		clr_global_flag = ADISCORD_vorkerland_sol_restoration_materialization_retry
		country_event = { id = ADISCORD_vorkerland_diplomacy.5 days = 1 }
	}

			tag = VAD
			has_global_flag = ADISCORD_vorkerland_vad_solar_intervention_active
			has_global_flag = ADISCORD_vorkerland_sol_restoration_pending

			NOT = { has_global_flag = ADISCORD_vorkerland_sol_restoration_failed }
		}
		set_global_flag = ADISCORD_vorkerland_sol_restoration_materialization_retry
		country_event = { id = ADISCORD_vorkerland_diplomacy.7 days = 1 }
	}
	else_if = {
		limit = {
			tag = VAD
			has_global_flag = ADISCORD_vorkerland_vad_solar_intervention_active
			has_global_flag = ADISCORD_vorkerland_sol_restoration_pending
			NOT = { has_global_flag = ADISCORD_vorkerland_sol_restoration_materialized }
			has_global_flag = ADISCORD_vorkerland_sol_restoration_materialization_retry
		}
		set_global_flag = ADISCORD_vorkerland_sol_restoration_failed

ADISCORD_vorkerland_join_regional_allies_to_showdown = {

		limit = {
			has_global_flag = ADISCORD_vorkerland_central_showdown_started
			ADISCORD_vorkerland_central_showdown_edges_verified = yes
			OR = {
				has_global_flag = ADISCORD_vorkerland_vad_sol_alliance_accepted
				has_global_flag = ADISCORD_vorkerland_sol_restoration_verified
			}
			country_exists = SOL
			country_exists = VAD
		}
		SOL = {
			if = { limit = { VAD = { has_war_with = WKR } NOT = { has_war_with = WKR } } add_to_war = { targeted_alliance = VAD enemy = WKR hostility_reason = asked_to_join } }
			if = { limit = { VAD = { has_war_with = TVA } NOT = { has_war_with = TVA } } add_to_war = { targeted_alliance = VAD enemy = TVA hostility_reason = asked_to_join } }
		}
	}
	if = {
		limit = {
			has_global_flag = ADISCORD_vorkerland_central_showdown_started
			ADISCORD_vorkerland_central_showdown_edges_verified = yes
			has_global_flag = ADISCORD_vorkerland_wkr_vla_alliance_accepted
			country_exists = VLA
			country_exists = WKR
		}
		VLA = {
			if = { limit = { WKR = { has_war_with = VAD } NOT = { has_war_with = VAD } } add_to_war = { targeted_alliance = WKR enemy = VAD hostility_reason = asked_to_join } }
			if = { limit = { WKR = { has_war_with = TVA } NOT = { has_war_with = TVA } } add_to_war = { targeted_alliance = WKR enemy = TVA hostility_reason = asked_to_join } }
		}
	}
}

ADISCORD_vorkerland_disengage_regional_allies_before_formation = {
	if = {
		limit = { country_exists = SOL }
		SOL = {
			if = { limit = { country_exists = WKR has_war_with = WKR } white_peace = WKR }
			if = { limit = { country_exists = VAD has_war_with = VAD } white_peace = VAD }
			if = { limit = { country_exists = TVA has_war_with = TVA } white_peace = TVA }
		}
	}
	if = {
		limit = { country_exists = VLA }
		VLA = {
			if = { limit = { country_exists = WKR has_war_with = WKR } white_peace = WKR }
			if = { limit = { country_exists = VAD has_war_with = VAD } white_peace = VAD }
			if = { limit = { country_exists = TVA has_war_with = TVA } white_peace = TVA }
		}
	}
}
