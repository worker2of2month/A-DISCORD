ADISCORD_STP_lock_regular_army_templates = {
	set_division_template_lock = {
		division_template = "Police division"
		is_locked = yes
	}
	set_division_force_allow_recruiting = {
		division_template = "Police division"
		force_allow_recruiting = no
	}
	set_division_template_lock = {
		division_template = "Regular army"
		is_locked = yes
	}
	set_division_force_allow_recruiting = {
		division_template = "Regular army"
		force_allow_recruiting = no
	}

	# The guard is permanently unique; reassert its OOB restrictions for old saves.
	set_division_template_lock = {
		division_template = "Capital Guard"
		is_locked = yes
	}
	set_division_force_allow_recruiting = {
		division_template = "Capital Guard"
		force_allow_recruiting = no
	}
}

ADISCORD_STP_unlock_regular_army_templates = {
	# Compatibility with saves made while STP used the country-wide lock.
	country_lock_all_division_template = no

	set_division_template_lock = {
		division_template = "Police division"
		is_locked = no
	}
	set_division_force_allow_recruiting = {
		division_template = "Police division"
		force_allow_recruiting = yes
	}
	set_division_template_lock = {
		division_template = "Regular army"
		is_locked = no
	}
	set_division_force_allow_recruiting = {
		division_template = "Regular army"
		force_allow_recruiting = yes
	}
}

ADISCORD_STP_migrate_army_template_lock = {
	# The old country-wide lock also disabled the Thunder at Our Gates HQ template.
	country_lock_all_division_template = no

	if = {
		limit = { has_idea = STP_hedonism_with_no_bondaries }
		ADISCORD_STP_lock_regular_army_templates = yes
	}
	else = {
		ADISCORD_STP_unlock_regular_army_templates = yes
	}
}
