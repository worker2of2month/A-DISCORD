# A-DISCORD strategic economy MVP.
# The economy is a budget layer. It feeds the existing ADISCORD_economic_development progress and ideas.
ADISCORD_economy_initialize_country = {
	ADISCORD_initialize_society_development_variables = yes
	if = {
		limit = { NOT = { has_variable = ADISCORD_economy_initialized } }
		ADISCORD_economy_set_default_values = yes
		ADISCORD_economy_initialize_from_development = yes
		ADISCORD_economy_set_simulation_tier = yes
		ADISCORD_economy_update_model_and_cycle = yes
		ADISCORD_economy_recalculate_policy_modifiers = yes
		ADISCORD_economy_full_refresh = yes
		ADISCORD_economy_light_update = yes
		ADISCORD_economy_calculate_development_multiplier = yes
		ADISCORD_economy_refresh_spending_ideas = yes
		ADISCORD_economy_update_gui = yes
	}
	else = {
		ADISCORD_economy_migrate_schema = yes
		ADISCORD_economy_set_simulation_tier = yes
	}
}

ADISCORD_economy_migrate_schema = {
	# Versioned, one-shot save migration. Ordinary ticks must not repeat the
	# full has_variable/clamp initialization path.
	if = {
		limit = {
			OR = {
				NOT = { has_variable = ADISCORD_economy_schema_version }
				check_variable = { var = ADISCORD_economy_schema_version value = 3 compare = less_than }
			}
		}
		ADISCORD_economy_initialize_variables = yes
		if = {
			limit = { check_variable = { var = ADISCORD_economy_base_monthly_development_gain value = 0 compare = less_than_or_equals } }
			set_variable = { var = ADISCORD_economy_base_monthly_development_gain value = 1 }
		}
		if = { limit = { NOT = { has_variable = ADISCORD_economy_tax_change_cooldown } } set_variable = { var = ADISCORD_economy_tax_change_cooldown value = 0 } }
		if = { limit = { NOT = { has_variable = ADISCORD_economy_army_budget_change_cooldown } } set_variable = { var = ADISCORD_economy_army_budget_change_cooldown value = 0 } }
		if = { limit = { NOT = { has_variable = ADISCORD_economy_construction_budget_change_cooldown } } set_variable = { var = ADISCORD_economy_construction_budget_change_cooldown value = 0 } }
		if = { limit = { NOT = { has_variable = ADISCORD_economy_gui_page } } set_variable = { var = ADISCORD_economy_gui_page value = 0 } }
		set_variable = { var = ADISCORD_economy_schema_version value = 3 }
		set_variable = { var = ADISCORD_economy_needs_full_refresh value = 1 }
	}
}

ADISCORD_economy_set_simulation_tier = {
	if = { limit = { ADISCORD_economy_is_player_tier_country = yes } set_variable = { var = ADISCORD_economy_simulation_tier value = 0 } }
	else_if = { limit = { ADISCORD_economy_is_primary_tier_country = yes } set_variable = { var = ADISCORD_economy_simulation_tier value = 1 } }
	else_if = { limit = { ADISCORD_economy_is_secondary_tier_country = yes } set_variable = { var = ADISCORD_economy_simulation_tier value = 2 } }
	else = { set_variable = { var = ADISCORD_economy_simulation_tier value = 3 } }
}

ADISCORD_economy_calculate_personal_income = {
	# Stable taxable-base proxy. Civilian industry represents registered urban
	# employment while development and state capacity determine collection.
	set_variable = { var = ADISCORD_economy_personal_income value = 1.2 }
	set_temp_variable = { var = ADISCORD_economy_personal_factory_temp value = num_of_civilian_factories }
	multiply_temp_variable = { var = ADISCORD_economy_personal_factory_temp value = 0.06 }
	add_to_variable = { var = ADISCORD_economy_personal_income value = ADISCORD_economy_personal_factory_temp }
	if = { limit = { ADISCORD_society_development_at_least_2 = yes } add_to_variable = { var = ADISCORD_economy_personal_income value = 0.3 } }
	if = { limit = { ADISCORD_society_development_at_least_3 = yes } add_to_variable = { var = ADISCORD_economy_personal_income value = 0.4 } }

ADISCORD_economy_calculate_factory_income = {
	# Military factories are normally an expense, not state income. Income appears only through arms-export/private-contractor profiles.
	set_variable = { var = ADISCORD_economy_factory_income value = 0 }
	if = {
		limit = {
			OR = {
				ADISCORD_economy_model_is_decentralized_market = yes
				ADISCORD_economy_model_is_mixed = yes
				ADISCORD_economy_model_is_oligarchic_clan = yes
				has_idea = VAL_worldwide_famous_weponry
				ADISCORD_economy_has_taxation_industrial_tariffs = yes
				ADISCORD_economy_has_taxation_extraction_quotas = yes
			}
		}
		set_variable = { var = ADISCORD_economy_factory_income value = num_of_military_factories }
		multiply_variable = { var = ADISCORD_economy_factory_income value = 0.015 }
	}
	if = { limit = { ADISCORD_economy_has_taxation_industrial_tariffs = yes } add_to_variable = { var = ADISCORD_economy_factory_income value = 0.4 } }
	if = { limit = { ADISCORD_economy_has_taxation_extraction_quotas = yes } add_to_variable = { var = ADISCORD_economy_factory_income value = 0.3 } }
	if = { limit = { has_idea = VAL_worldwide_famous_weponry } add_to_variable = { var = ADISCORD_economy_factory_income value = 0.9 } }
	if = {
		limit = {
			OR = {
				ADISCORD_economy_model_is_decentralized_market = yes
				ADISCORD_economy_model_is_mixed = yes
				ADISCORD_economy_model_is_oligarchic_clan = yes
				has_idea = VAL_worldwide_famous_weponry
				ADISCORD_economy_has_taxation_industrial_tariffs = yes
				ADISCORD_economy_has_taxation_extraction_quotas = yes
			}
		}
		set_temp_variable = { var = ADISCORD_economy_factory_income_temp value = ADISCORD_industrial_cluster_count }
		multiply_temp_variable = { var = ADISCORD_economy_factory_income_temp value = 0.08 }
		add_to_variable = { var = ADISCORD_economy_factory_income value = ADISCORD_economy_factory_income_temp }
	}
	# Mobilization economy converts arms output into war capacity, not normal treasury profit.
	if = { limit = { ADISCORD_economy_model_is_mobilization = yes } set_variable = { var = ADISCORD_economy_factory_income value = 0 } }
	clamp_variable = { var = ADISCORD_economy_factory_income min = 0 max = 50 }
}

		limit = { check_variable = { var = ADISCORD_economy_tax_burden_mode value = 1 compare = less_than_or_equals } }
		multiply_variable = { var = ADISCORD_economy_personal_income value = 0.65 }
		multiply_variable = { var = ADISCORD_economy_business_income value = 0.85 }
		multiply_variable = { var = ADISCORD_economy_factory_income value = 0.90 }
		add_to_variable = { var = ADISCORD_economy_consumer_goods_income value = 0.3 }
	}
	else_if = {
		limit = { check_variable = { var = ADISCORD_economy_tax_burden_mode value = 2 compare = less_than_or_equals } }
		multiply_variable = { var = ADISCORD_economy_personal_income value = 0.85 }
		multiply_variable = { var = ADISCORD_economy_business_income value = 0.95 }
		multiply_variable = { var = ADISCORD_economy_factory_income value = 0.95 }
		add_to_variable = { var = ADISCORD_economy_consumer_goods_income value = 0.1 }
	}
	else_if = {
		limit = {
			check_variable = { var = ADISCORD_economy_tax_burden_mode value = 4 compare = greater_than_or_equals }
			check_variable = { var = ADISCORD_economy_tax_burden_mode value = 5 compare = less_than }
		}
		multiply_variable = { var = ADISCORD_economy_personal_income value = 1.15 }
		multiply_variable = { var = ADISCORD_economy_business_income value = 1.08 }
		multiply_variable = { var = ADISCORD_economy_factory_income value = 1.05 }
		add_to_variable = { var = ADISCORD_economy_consumer_goods_income value = -0.1 }
	}
	else_if = {
		limit = { check_variable = { var = ADISCORD_economy_tax_burden_mode value = 5 compare = greater_than_or_equals } }
		multiply_variable = { var = ADISCORD_economy_personal_income value = 1.35 }
		multiply_variable = { var = ADISCORD_economy_business_income value = 1.15 }
		multiply_variable = { var = ADISCORD_economy_factory_income value = 1.10 }
		add_to_variable = { var = ADISCORD_economy_consumer_goods_income value = -0.3 }
	}
	clamp_variable = { var = ADISCORD_economy_personal_income min = 0 max = 100 }
	clamp_variable = { var = ADISCORD_economy_business_income min = 0 max = 150 }
	clamp_variable = { var = ADISCORD_economy_consumer_goods_income min = 0 max = 100 }
	clamp_variable = { var = ADISCORD_economy_factory_income min = 0 max = 150 }
}

ADISCORD_economy_calculate_army_expenses = {
	# Actual fielded manpower is charged through stable threshold triggers. The
	# tiers are cumulative, so very large armies become materially expensive.
	set_variable = { var = ADISCORD_economy_army_expenses value = 0.75 }
	if = { limit = { has_army_manpower = { size > 9999 } } add_to_variable = { var = ADISCORD_economy_army_expenses value = 0.35 } }
	if = { limit = { has_army_manpower = { size > 24999 } } add_to_variable = { var = ADISCORD_economy_army_expenses value = 0.50 } }
	if = { limit = { has_army_manpower = { size > 49999 } } add_to_variable = { var = ADISCORD_economy_army_expenses value = 0.75 } }
	if = { limit = { has_army_manpower = { size > 99999 } } add_to_variable = { var = ADISCORD_economy_army_expenses value = 1.00 } }
	if = { limit = { has_army_manpower = { size > 249999 } } add_to_variable = { var = ADISCORD_economy_army_expenses value = 1.50 } }
	if = { limit = { ADISCORD_army_development_at_least_2 = yes } add_to_variable = { var = ADISCORD_economy_army_expenses value = 0.5 } }
	if = { limit = { ADISCORD_army_development_at_least_3 = yes } add_to_variable = { var = ADISCORD_economy_army_expenses value = 0.5 } }
	if = { limit = { ADISCORD_army_development_at_least_4 = yes } add_to_variable = { var = ADISCORD_economy_army_expenses value = 0.75 } }
	if = { limit = { ADISCORD_army_development_at_least_5 = yes } add_to_variable = { var = ADISCORD_economy_army_expenses value = 0.75 } }
	if = { limit = { has_idea = ADISCORD_military_organization_contract_brigades } add_to_variable = { var = ADISCORD_economy_army_expenses value = 0.5 } }
	if = { limit = { has_idea = ADISCORD_military_organization_general_staff } add_to_variable = { var = ADISCORD_economy_army_expenses value = 0.75 } }
	if = { limit = { has_idea = ADISCORD_military_organization_total_defense_grid } add_to_variable = { var = ADISCORD_economy_army_expenses value = 1.0 } }
	if = { limit = { has_idea = ADISCORD_military_organization_militia_autonomy } multiply_variable = { var = ADISCORD_economy_army_expenses value = 0.85 } }
	if = { limit = { has_war = yes } multiply_variable = { var = ADISCORD_economy_army_expenses value = 1.25 } }
	if = { limit = { ADISCORD_army_development_at_least_4 = yes } multiply_variable = { var = ADISCORD_economy_army_expenses value = 1.10 } }
	if = { limit = { ADISCORD_army_development_at_least_5 = yes } multiply_variable = { var = ADISCORD_economy_army_expenses value = 1.05 } }
	if = { limit = { check_variable = { var = ADISCORD_economy_army_spending_mode value = 1 compare = less_than_or_equals } } multiply_variable = { var = ADISCORD_economy_army_expenses value = 0.50 } }
	else_if = { limit = { check_variable = { var = ADISCORD_economy_army_spending_mode value = 2 compare = less_than_or_equals } } multiply_variable = { var = ADISCORD_economy_army_expenses value = 0.75 } }
	else_if = { limit = { check_variable = { var = ADISCORD_economy_army_spending_mode value = 4 compare = greater_than_or_equals } check_variable = { var = ADISCORD_economy_army_spending_mode value = 5 compare = less_than } } multiply_variable = { var = ADISCORD_economy_army_expenses value = 1.50 } }
	else_if = { limit = { check_variable = { var = ADISCORD_economy_army_spending_mode value = 5 compare = greater_than_or_equals } } multiply_variable = { var = ADISCORD_economy_army_expenses value = 2.50 } }
	clamp_variable = { var = ADISCORD_economy_army_expenses min = 0 max = 5000 }
}

	multiply_variable = { var = ADISCORD_economy_credit_temp value = 5 }

	multiply_variable = { var = ADISCORD_economy_credit_temp value = 5 }
	add_to_variable = { var = ADISCORD_economy_creditworthiness value = ADISCORD_economy_credit_temp }
	if = { limit = { ADISCORD_economy_model_is_decentralized_market = yes } add_to_variable = { var = ADISCORD_economy_creditworthiness value = 8 } }

	set_variable = { var = ADISCORD_economy_credit_temp value = ADISCORD_economy_inflation }
	subtract_from_variable = { var = ADISCORD_economy_creditworthiness value = ADISCORD_economy_credit_temp }
	set_variable = { var = ADISCORD_economy_credit_temp value = ADISCORD_economy_fiscal_stress }
	divide_variable = { var = ADISCORD_economy_credit_temp value = 2 }
	subtract_from_variable = { var = ADISCORD_economy_creditworthiness value = ADISCORD_economy_credit_temp }
	set_variable = { var = ADISCORD_economy_credit_temp value = ADISCORD_economy_debt_ratio }
	divide_variable = { var = ADISCORD_economy_credit_temp value = 3 }

	subtract_from_variable = { var = ADISCORD_economy_creditworthiness value = ADISCORD_economy_restructuring_penalty }

	clamp_variable = { var = ADISCORD_economy_creditworthiness min = 0 max = 100 }

ADISCORD_economy_apply_monthly_balance = {
	# V19 TDA-core: one country, one month, one applied balance.
	# If the balance pushes the treasury below zero, auto-loan only covers the uncovered part.
	set_variable = { var = ADISCORD_economy_last_auto_borrowing value = 0 }
	set_variable = { var = ADISCORD_economy_last_uncovered_deficit value = 0 }
	set_variable = { var = ADISCORD_economy_last_monthly_balance_applied value = ADISCORD_economy_monthly_balance }
	set_variable = { var = ADISCORD_economy_treasury_before_tick value = ADISCORD_economy_treasury }
	set_variable = { var = ADISCORD_economy_last_month_treasury_before value = ADISCORD_economy_accounting_period_treasury_start }
	set_variable = { var = ADISCORD_economy_last_month_income value = ADISCORD_economy_monthly_income }
	set_variable = { var = ADISCORD_economy_last_month_expenses value = ADISCORD_economy_monthly_expenses }
	set_variable = { var = ADISCORD_economy_last_month_balance value = ADISCORD_economy_monthly_balance }
	set_variable = { var = ADISCORD_economy_last_month_debt_added value = ADISCORD_economy_current_month_debt_added }
	set_variable = { var = ADISCORD_economy_last_month_debt_paid value = ADISCORD_economy_current_month_debt_paid }
	set_variable = { var = ADISCORD_economy_last_month_action_costs value = ADISCORD_economy_current_month_action_costs }
	set_variable = { var = ADISCORD_economy_last_month_action_income value = ADISCORD_economy_current_month_action_income }
	set_variable = { var = ADISCORD_economy_last_month_building_costs value = ADISCORD_economy_current_month_building_costs }

	if = {
		limit = { check_variable = { var = ADISCORD_economy_monthly_balance value = 0 compare = greater_than_or_equals } }
		set_variable = { var = ADISCORD_economy_balance_temp value = ADISCORD_economy_monthly_balance }
		multiply_variable = { var = ADISCORD_economy_balance_temp value = ADISCORD_economy_final_reserve_growth_factor_bp }
		divide_variable = { var = ADISCORD_economy_balance_temp value = 100 }
		add_to_variable = { var = ADISCORD_economy_treasury value = ADISCORD_economy_balance_temp }
		set_variable = { var = ADISCORD_economy_last_monthly_balance_applied value = ADISCORD_economy_balance_temp }
		set_variable = { var = ADISCORD_economy_deficit_streak value = 0 }
		add_to_variable = { var = ADISCORD_economy_surplus_streak value = 1 }
		add_to_variable = { var = ADISCORD_economy_deficit_pressure value = -2 }
	}
	else = {
		add_to_variable = { var = ADISCORD_economy_treasury value = ADISCORD_economy_monthly_balance }
		add_to_variable = { var = ADISCORD_economy_deficit_pressure value = 1 }
		add_to_variable = { var = ADISCORD_economy_deficit_streak value = 1 }
		set_variable = { var = ADISCORD_economy_surplus_streak value = 0 }

		# Automatic loan only for uncovered deficit. No huge packages, no inflation spiral.
		if = {
			limit = { check_variable = { var = ADISCORD_economy_treasury value = 0 compare = less_than } }
			set_variable = { var = ADISCORD_economy_uncovered_deficit_temp value = 0 }
			subtract_from_variable = { var = ADISCORD_economy_uncovered_deficit_temp value = ADISCORD_economy_treasury }
			set_variable = { var = ADISCORD_economy_last_uncovered_deficit value = ADISCORD_economy_uncovered_deficit_temp }
			if = {
				limit = {
					check_variable = { var = ADISCORD_economy_auto_loan_enabled value = 1 compare = greater_than_or_equals }
				}
				set_variable = { var = ADISCORD_economy_debt_capacity_room_temp value = ADISCORD_economy_debt_capacity }
				subtract_from_variable = { var = ADISCORD_economy_debt_capacity_room_temp value = ADISCORD_economy_debt }
				clamp_variable = { var = ADISCORD_economy_debt_capacity_room_temp min = 0 max = 5000 }
				set_variable = { var = ADISCORD_economy_auto_borrow_temp value = ADISCORD_economy_uncovered_deficit_temp }
				set_temp_variable = { var = ADISCORD_economy_auto_borrow_over_cap_temp value = ADISCORD_economy_auto_borrow_temp }
				subtract_from_variable = { var = ADISCORD_economy_auto_borrow_over_cap_temp value = ADISCORD_economy_debt_capacity_room_temp }
				if = {
					limit = {
						check_variable = { var = ADISCORD_economy_auto_borrow_over_cap_temp value = 0 compare = greater_than }
					}
					subtract_from_variable = { var = ADISCORD_economy_auto_borrow_temp value = ADISCORD_economy_auto_borrow_over_cap_temp }
				}
				clamp_variable = { var = ADISCORD_economy_auto_borrow_temp min = 0 max = 5000 }
				if = {
					limit = { check_variable = { var = ADISCORD_economy_auto_borrow_temp value = 0.1 compare = greater_than } }
					add_to_variable = { var = ADISCORD_economy_debt value = ADISCORD_economy_auto_borrow_temp }
					set_variable = { var = ADISCORD_economy_last_auto_borrowing value = ADISCORD_economy_auto_borrow_temp }
					add_to_variable = { var = ADISCORD_economy_last_month_debt_added value = ADISCORD_economy_auto_borrow_temp }
				}
			}
			set_variable = { var = ADISCORD_economy_treasury value = 0 }
			add_to_variable = { var = ADISCORD_economy_fiscal_stress value = 0.25 }
			if = {
				limit = {
					check_variable = { var = ADISCORD_economy_auto_loan_enabled value = 1 compare = less_than }
				}
				add_to_variable = { var = ADISCORD_economy_deficit_pressure value = 1.5 }
				add_to_variable = { var = ADISCORD_economy_fiscal_stress value = 0.75 }
			}
			set_temp_variable = { var = ADISCORD_economy_uncovered_after_auto_borrow_temp value = ADISCORD_economy_last_uncovered_deficit }
			subtract_from_variable = { var = ADISCORD_economy_uncovered_after_auto_borrow_temp value = ADISCORD_economy_last_auto_borrowing }
			if = {
				limit = {
					check_variable = { var = ADISCORD_economy_auto_loan_enabled value = 1 compare = greater_than_or_equals }
					check_variable = { var = ADISCORD_economy_uncovered_after_auto_borrow_temp value = 0.1 compare = greater_than }
				}
				add_to_variable = { var = ADISCORD_economy_deficit_pressure value = 1.5 }
				add_to_variable = { var = ADISCORD_economy_fiscal_stress value = 0.75 }
			}
		}
	}

	set_variable = { var = ADISCORD_economy_treasury_after_tick value = ADISCORD_economy_treasury }
	set_variable = { var = ADISCORD_economy_actual_treasury_delta value = ADISCORD_economy_treasury_after_tick }
	subtract_from_variable = { var = ADISCORD_economy_actual_treasury_delta value = ADISCORD_economy_treasury_before_tick }
	set_variable = { var = ADISCORD_economy_last_month_treasury_after value = ADISCORD_economy_treasury_after_tick }
	set_variable = { var = ADISCORD_economy_accounting_expected_treasury_after_temp value = ADISCORD_economy_last_month_treasury_before }
	add_to_variable = { var = ADISCORD_economy_accounting_expected_treasury_after_temp value = ADISCORD_economy_last_monthly_balance_applied }
	add_to_variable = { var = ADISCORD_economy_accounting_expected_treasury_after_temp value = ADISCORD_economy_last_month_action_income }
	subtract_from_variable = { var = ADISCORD_economy_accounting_expected_treasury_after_temp value = ADISCORD_economy_last_month_action_costs }
	subtract_from_variable = { var = ADISCORD_economy_accounting_expected_treasury_after_temp value = ADISCORD_economy_last_month_building_costs }
	add_to_variable = { var = ADISCORD_economy_accounting_expected_treasury_after_temp value = ADISCORD_economy_last_month_debt_added }
	subtract_from_variable = { var = ADISCORD_economy_accounting_expected_treasury_after_temp value = ADISCORD_economy_last_month_debt_paid }
	set_variable = { var = ADISCORD_economy_last_month_unexplained_delta value = ADISCORD_economy_last_month_treasury_after }
	subtract_from_variable = { var = ADISCORD_economy_last_month_unexplained_delta value = ADISCORD_economy_accounting_expected_treasury_after_temp }
	set_variable = { var = ADISCORD_economy_accounting_period_treasury_start value = ADISCORD_economy_last_month_treasury_after }
	set_variable = { var = ADISCORD_economy_current_month_debt_added value = 0 }
	set_variable = { var = ADISCORD_economy_current_month_debt_paid value = 0 }
	set_variable = { var = ADISCORD_economy_current_month_action_costs value = 0 }
	set_variable = { var = ADISCORD_economy_current_month_action_income value = 0 }
	set_variable = { var = ADISCORD_economy_current_month_building_costs value = 0 }

	clamp_variable = { var = ADISCORD_economy_deficit_streak min = 0 max = 24 }
	clamp_variable = { var = ADISCORD_economy_surplus_streak min = 0 max = 24 }
	clamp_variable = { var = ADISCORD_economy_deficit_pressure min = 0 max = 100 }
	ADISCORD_economy_clamp_treasury = yes
}

	# Secondary AI uses one country-scoped aggregate transaction instead of
	# twelve hidden monthly loops. Reuse the bounded deficit/loan semantics,
	# then restore the monthly forecast used by AI strategies.
	set_temp_variable = { var = ADISCORD_economy_saved_monthly_income value = ADISCORD_economy_monthly_income }

	clamp_variable = { var = ADISCORD_economy_surplus_streak min = 0 max = 24 }
}

ADISCORD_economy_update_fiscal_stress = {
	# TDA-lite pass: fiscal stress is a slow warning indicator, not a self-feeding death spiral.
	set_variable = { var = ADISCORD_economy_fiscal_stress_delta value = 0 }
	if = { limit = { check_variable = { var = ADISCORD_economy_last_uncovered_deficit value = 0.1 compare = greater_than } } add_to_variable = { var = ADISCORD_economy_fiscal_stress_delta value = 0.5 } }
	if = { limit = { check_variable = { var = ADISCORD_economy_debt_ratio value = 70 compare = greater_than_or_equals } } add_to_variable = { var = ADISCORD_economy_fiscal_stress_delta value = 0.5 } }
	if = { limit = { check_variable = { var = ADISCORD_economy_debt_ratio value = 100 compare = greater_than_or_equals } } add_to_variable = { var = ADISCORD_economy_fiscal_stress_delta value = 1.0 } }
	if = { limit = { check_variable = { var = ADISCORD_economy_deficit_streak value = 3 compare = greater_than_or_equals } } add_to_variable = { var = ADISCORD_economy_fiscal_stress_delta value = 0.5 } }
	if = { limit = { check_variable = { var = ADISCORD_economy_deficit_streak value = 6 compare = greater_than_or_equals } } add_to_variable = { var = ADISCORD_economy_fiscal_stress_delta value = 1.0 } }
	if = { limit = { check_variable = { var = ADISCORD_economy_inflation value = 25 compare = greater_than_or_equals } } add_to_variable = { var = ADISCORD_economy_fiscal_stress_delta value = 0.5 } }
	if = { limit = { check_variable = { var = ADISCORD_economy_debt_service value = 20 compare = greater_than_or_equals } } add_to_variable = { var = ADISCORD_economy_fiscal_stress_delta value = 0.5 } }
	if = { limit = { check_variable = { var = ADISCORD_economy_monthly_balance value = 0 compare = greater_than_or_equals } } add_to_variable = { var = ADISCORD_economy_fiscal_stress_delta value = -0.75 } }
	if = { limit = { check_variable = { var = ADISCORD_economy_surplus_streak value = 3 compare = greater_than_or_equals } } add_to_variable = { var = ADISCORD_economy_fiscal_stress_delta value = -0.75 } }
	if = {
		limit = { check_variable = { var = ADISCORD_economy_fiscal_stress_delta value = 0 compare = greater_than } }
		multiply_variable = { var = ADISCORD_economy_fiscal_stress_delta value = ADISCORD_economy_final_fiscal_stress_gain_factor_bp }
		divide_variable = { var = ADISCORD_economy_fiscal_stress_delta value = 100 }
		multiply_variable = { var = ADISCORD_economy_fiscal_stress_delta value = 100 }
		divide_variable = { var = ADISCORD_economy_fiscal_stress_delta value = ADISCORD_economy_final_fiscal_stress_resistance_factor_bp }
	}
	clamp_variable = { var = ADISCORD_economy_fiscal_stress_delta min = -1.5 max = 2.0 }
	add_to_variable = { var = ADISCORD_economy_fiscal_stress value = ADISCORD_economy_fiscal_stress_delta }
	if = { limit = { check_variable = { var = ADISCORD_economy_restructuring_penalty value = 0 compare = greater_than } check_variable = { var = ADISCORD_economy_monthly_balance value = 0 compare = greater_than_or_equals } } add_to_variable = { var = ADISCORD_economy_restructuring_penalty value = -1 } }
	clamp_variable = { var = ADISCORD_economy_fiscal_stress min = 0 max = 100 }
}


ADISCORD_economy_update_inflation = {
	# TDA-lite pass: inflation moves slowly and is primarily caused by explicit emission, shocks and crises.
	# Planned-bureaucratic systems suppress visible market inflation; deficits become shortages/overload instead.
	if = {
		limit = { ADISCORD_economy_model_is_planned_bureaucratic = yes }
		set_variable = { var = ADISCORD_economy_emission_pressure value = 0 }
		if = { limit = { check_variable = { var = ADISCORD_economy_inflation value = 0 compare = greater_than } } add_to_variable = { var = ADISCORD_economy_inflation value = -2 } }
		if = { limit = { check_variable = { var = ADISCORD_economy_money_printing_level value = 1 compare = greater_than_or_equals } } add_to_variable = { var = ADISCORD_economy_stretched_score value = ADISCORD_economy_money_printing_level } }
		if = { limit = { check_variable = { var = ADISCORD_economy_deficit_pressure value = 20 compare = greater_than_or_equals } } add_to_variable = { var = ADISCORD_economy_stretched_score value = 1 } }
	}
	else = {
		set_variable = { var = ADISCORD_economy_emission_pressure value = ADISCORD_economy_money_printing_level }

	if = {
		limit = { ADISCORD_economy_can_expand_money_emission = yes }

		# injection is bounded by reserve capacity and carries persistent prices.
		set_variable = { var = ADISCORD_economy_emission_cash_temp value = 35 }
		set_variable = { var = ADISCORD_economy_emission_room_temp value = ADISCORD_economy_treasury_cap }

		if = {
			limit = { check_variable = { var = ADISCORD_economy_emission_overflow_temp value = 0 compare = greater_than } }
			subtract_from_variable = { var = ADISCORD_economy_emission_cash_temp value = ADISCORD_economy_emission_overflow_temp }

ADISCORD_economy_apply_tax_burden_side_effects = {
	# TDA-lite pass: taxation levels affect the tax base first. They should not directly create runaway inflation.
	# Extreme taxation creates mild price-shock/fiscal-pressure only; inflation is handled by ADISCORD_economy_update_inflation.
	if = {
		limit = { check_variable = { var = ADISCORD_economy_tax_burden_mode value = 1 compare = less_than_or_equals } }
		add_to_variable = { var = ADISCORD_economy_price_shock value = -0.25 }
	}
	else_if = {
		limit = {
			check_variable = { var = ADISCORD_economy_tax_burden_mode value = 4 compare = greater_than_or_equals }
			check_variable = { var = ADISCORD_economy_tax_burden_mode value = 5 compare = less_than }
		}
		add_to_variable = { var = ADISCORD_economy_price_shock value = 0.25 }
		if = { limit = { has_war = yes } add_to_variable = { var = ADISCORD_economy_war_fatigue_score value = 0.5 } }
	}
	else_if = {
		limit = { check_variable = { var = ADISCORD_economy_tax_burden_mode value = 5 compare = greater_than_or_equals } }
		add_to_variable = { var = ADISCORD_economy_price_shock value = 0.75 }
		add_to_variable = { var = ADISCORD_economy_fiscal_stress value = 0.5 }
		if = { limit = { has_war = yes } add_to_variable = { var = ADISCORD_economy_war_fatigue_score value = 1.0 } }
	}
	clamp_variable = { var = ADISCORD_economy_price_shock min = 0 max = 100 }
	clamp_variable = { var = ADISCORD_economy_fiscal_stress min = 0 max = 100 }
	clamp_variable = { var = ADISCORD_economy_war_fatigue_score min = 0 max = 100 }
}

ADISCORD_economy_update_ai_state = {
	# Hysteresis: crisis and recovery have stricter exit thresholds than entry
	# thresholds, preventing budget modes from changing direction every tick.
	if = {
		limit = { ADISCORD_economy_ai_is_crisis = yes }
		if = {
			limit = {
				check_variable = { var = ADISCORD_economy_monthly_balance value = 0 compare = greater_than_or_equals }
				check_variable = { var = ADISCORD_economy_surplus_streak value = 2 compare = greater_than_or_equals }
				check_variable = { var = ADISCORD_economy_debt_ratio value = 70 compare = less_than }
				check_variable = { var = ADISCORD_economy_fiscal_stress value = 45 compare = less_than }
			}
			set_variable = { var = ADISCORD_economy_ai_state value = 3 }
		}
	}
	else_if = {
		limit = { ADISCORD_economy_ai_is_recovery = yes }
		if = {
			limit = {
				OR = {
					check_variable = { var = ADISCORD_economy_debt_ratio value = 110 compare = greater_than_or_equals }
					check_variable = { var = ADISCORD_economy_fiscal_stress value = 75 compare = greater_than_or_equals }
				}
			}
			set_variable = { var = ADISCORD_economy_ai_state value = 2 }
		}
		else_if = {
			limit = {
				OR = {
					check_variable = { var = ADISCORD_economy_monthly_balance value = 0 compare = less_than }
					check_variable = { var = ADISCORD_economy_deficit_streak value = 2 compare = greater_than_or_equals }
				}
			}
			set_variable = { var = ADISCORD_economy_ai_state value = 1 }
		}
		else_if = {
			limit = {
				check_variable = { var = ADISCORD_economy_surplus_streak value = 6 compare = greater_than_or_equals }
				check_variable = { var = ADISCORD_economy_debt_ratio value = 35 compare = less_than }
				check_variable = { var = ADISCORD_economy_fiscal_stress value = 20 compare = less_than }
				check_variable = { var = ADISCORD_economy_treasury value = 75 compare = greater_than_or_equals }
			}
			set_variable = { var = ADISCORD_economy_ai_state value = 0 }
		}
	}
	else_if = {
		limit = { ADISCORD_economy_ai_is_stressed = yes }
		if = {
			limit = {
				OR = {
					check_variable = { var = ADISCORD_economy_debt_ratio value = 100 compare = greater_than_or_equals }
					check_variable = { var = ADISCORD_economy_fiscal_stress value = 70 compare = greater_than_or_equals }
					check_variable = { var = ADISCORD_economy_deficit_streak value = 6 compare = greater_than_or_equals }
					AND = {
						check_variable = { var = ADISCORD_economy_treasury value = 10 compare = less_than }
						check_variable = { var = ADISCORD_economy_monthly_balance value = 0 compare = less_than }
					}
				}
			}
			set_variable = { var = ADISCORD_economy_ai_state value = 2 }
		}
		else_if = {
			limit = {
				check_variable = { var = ADISCORD_economy_surplus_streak value = 3 compare = greater_than_or_equals }
				check_variable = { var = ADISCORD_economy_debt_ratio value = 50 compare = less_than }
				check_variable = { var = ADISCORD_economy_fiscal_stress value = 30 compare = less_than }
			}
			set_variable = { var = ADISCORD_economy_ai_state value = 0 }
		}
	}
	else = {
		if = {
			limit = {
				OR = {
					check_variable = { var = ADISCORD_economy_debt_ratio value = 110 compare = greater_than_or_equals }
					check_variable = { var = ADISCORD_economy_fiscal_stress value = 75 compare = greater_than_or_equals }
					check_variable = { var = ADISCORD_economy_deficit_streak value = 6 compare = greater_than_or_equals }
				}
			}
			set_variable = { var = ADISCORD_economy_ai_state value = 2 }
		}
		else_if = {
			limit = {
				OR = {
					check_variable = { var = ADISCORD_economy_deficit_streak value = 2 compare = greater_than_or_equals }
					check_variable = { var = ADISCORD_economy_debt_ratio value = 65 compare = greater_than_or_equals }
					check_variable = { var = ADISCORD_economy_fiscal_stress value = 40 compare = greater_than_or_equals }
					AND = {
						check_variable = { var = ADISCORD_economy_treasury value = 25 compare = less_than }
						check_variable = { var = ADISCORD_economy_monthly_balance value = 0 compare = less_than }
					}
				}
			}
			set_variable = { var = ADISCORD_economy_ai_state value = 1 }
		}
	}
}

ADISCORD_economy_ai_monthly_policy = {
	# Exactly one discretionary action per tick. Passive balance and automatic
	# deficit coverage remain separate accounting operations.
	if = {
		limit = { ADISCORD_economy_ai_is_crisis = yes }
		if = { limit = { ADISCORD_economy_can_restructure_debt = yes } ADISCORD_economy_restructure_debt = yes }
		else_if = { limit = { ADISCORD_economy_can_take_external_loan = yes ADISCORD_economy_has_low_treasury = yes } ADISCORD_economy_take_external_loan = yes }
		else_if = { limit = { ADISCORD_economy_can_issue_internal_bonds = yes ADISCORD_economy_has_low_treasury = yes } ADISCORD_economy_issue_internal_bonds = yes }
		else_if = { limit = { has_war = yes ADISCORD_economy_can_use_war_taxes = yes } ADISCORD_economy_war_taxes_action = yes }
		else_if = { limit = { ADISCORD_economy_can_stabilize_macro = yes } ADISCORD_economy_stabilization_package = yes }
		else_if = { limit = { ADISCORD_economy_can_decrease_construction_spending = yes } ADISCORD_economy_decrease_construction_spending = yes }
		else_if = { limit = { has_war = no ADISCORD_economy_can_decrease_army_spending = yes } ADISCORD_economy_decrease_army_spending = yes }
		else_if = { limit = { ADISCORD_economy_can_increase_tax_burden = yes } ADISCORD_economy_increase_tax_burden = yes }
	}
	else_if = {
		limit = { ADISCORD_economy_ai_is_stressed = yes }
		if = { limit = { has_war = yes ADISCORD_economy_can_use_war_taxes = yes } ADISCORD_economy_war_taxes_action = yes }
		else_if = { limit = { check_variable = { var = ADISCORD_economy_monthly_balance value = 0 compare = less_than } ADISCORD_economy_can_increase_tax_burden = yes } ADISCORD_economy_increase_tax_burden = yes }
		else_if = { limit = { check_variable = { var = ADISCORD_economy_monthly_balance value = 0 compare = less_than } ADISCORD_economy_can_decrease_construction_spending = yes } ADISCORD_economy_decrease_construction_spending = yes }
		else_if = { limit = { has_war = no check_variable = { var = ADISCORD_economy_fiscal_stress value = 55 compare = greater_than_or_equals } ADISCORD_economy_can_decrease_army_spending = yes } ADISCORD_economy_decrease_army_spending = yes }
		else_if = { limit = { ADISCORD_economy_can_stabilize_macro = yes } ADISCORD_economy_stabilization_package = yes }
	}
	else_if = {
		limit = { ADISCORD_economy_ai_is_recovery = yes }
		if = { limit = { ADISCORD_economy_can_early_repay_debt = yes } ADISCORD_economy_early_repay_debt = yes }
		else_if = { limit = { check_variable = { var = ADISCORD_economy_tax_burden_mode value = 3 compare = greater_than } ADISCORD_economy_can_decrease_tax_burden = yes } ADISCORD_economy_decrease_tax_burden = yes }
		else_if = { limit = { check_variable = { var = ADISCORD_economy_construction_spending_mode value = 3 compare = less_than } ADISCORD_economy_can_increase_construction_spending = yes } ADISCORD_economy_increase_construction_spending = yes }
		else_if = { limit = { check_variable = { var = ADISCORD_economy_army_spending_mode value = 3 compare = less_than } ADISCORD_economy_can_increase_army_spending = yes } ADISCORD_economy_increase_army_spending = yes }
	}
	else = {
		if = { limit = { ADISCORD_economy_can_early_repay_debt = yes check_variable = { var = ADISCORD_economy_surplus_streak value = 3 compare = greater_than_or_equals } } ADISCORD_economy_early_repay_debt = yes }
		else_if = { limit = { check_variable = { var = ADISCORD_economy_construction_spending_mode value = 3 compare = less_than } ADISCORD_economy_can_increase_construction_spending = yes } ADISCORD_economy_increase_construction_spending = yes }
		else_if = { limit = { check_variable = { var = ADISCORD_economy_army_spending_mode value = 3 compare = less_than } ADISCORD_economy_can_increase_army_spending = yes } ADISCORD_economy_increase_army_spending = yes }
		else_if = { limit = { ADISCORD_economy_can_expand_public_capital = yes check_variable = { var = ADISCORD_economy_surplus_streak value = 4 compare = greater_than_or_equals } } ADISCORD_economy_invest_reserves = yes }
		else_if = { limit = { check_variable = { var = ADISCORD_economy_tax_burden_mode value = 3 compare = greater_than } ADISCORD_economy_can_decrease_tax_burden = yes } ADISCORD_economy_decrease_tax_burden = yes }
	}
}

ADISCORD_economy_update_model_and_cycle = {
	# V7: explicit economic-system laws are authoritative when present.
	# If an old save has no ADISCORD_economic_system_* law, fall back to the old inferred model logic.
	set_variable = { var = ADISCORD_economy_model value = 2 }

	if = { limit = { has_idea = ADISCORD_economic_system_agrarian } set_variable = { var = ADISCORD_economy_model value = 0 } }
	if = { limit = { has_idea = ADISCORD_economic_system_industrializing } set_variable = { var = ADISCORD_economy_model value = 2 } }
	if = { limit = { has_idea = ADISCORD_economic_system_free_market } set_variable = { var = ADISCORD_economy_model value = 1 } }
	if = { limit = { has_idea = ADISCORD_economic_system_mixed } set_variable = { var = ADISCORD_economy_model value = 2 } }
	if = { limit = { has_idea = ADISCORD_economic_system_state_coordinated } set_variable = { var = ADISCORD_economy_model value = 3 } }
	if = { limit = { has_idea = ADISCORD_economic_system_planned_bureaucratic } set_variable = { var = ADISCORD_economy_model value = 4 } }
	if = { limit = { has_idea = ADISCORD_economic_system_mobilization } set_variable = { var = ADISCORD_economy_model value = 5 } }
	if = { limit = { has_idea = ADISCORD_economic_system_oligarchic_clan } set_variable = { var = ADISCORD_economy_model value = 6 } }
	if = { limit = { has_idea = ADISCORD_economic_system_technocratic } set_variable = { var = ADISCORD_economy_model value = 7 } }

	if = {
		limit = {
			NOT = {
				OR = {
					has_idea = ADISCORD_economic_system_agrarian
					has_idea = ADISCORD_economic_system_industrializing
					has_idea = ADISCORD_economic_system_free_market
					has_idea = ADISCORD_economic_system_mixed
					has_idea = ADISCORD_economic_system_state_coordinated
					has_idea = ADISCORD_economic_system_planned_bureaucratic
					has_idea = ADISCORD_economic_system_mobilization
					has_idea = ADISCORD_economic_system_oligarchic_clan
					has_idea = ADISCORD_economic_system_technocratic
				}
			}
		}

		set_variable = { var = ADISCORD_economy_model value = 2 }
		if = { limit = { ADISCORD_economy_has_industrial_artisan_markets = yes ADISCORD_state_development_at_most_2 = yes } set_variable = { var = ADISCORD_economy_model value = 0 } }
		if = { limit = { ADISCORD_economy_has_taxation_light_dues = yes ADISCORD_economy_has_labor_loose_contracts = yes } set_variable = { var = ADISCORD_economy_model value = 1 } }
		if = { limit = { ADISCORD_economy_has_industrial_state_planning = yes ADISCORD_state_development_at_least_3 = yes } set_variable = { var = ADISCORD_economy_model value = 3 } }
		if = { limit = { ADISCORD_economy_has_industrial_state_planning = yes ADISCORD_state_development_at_least_4 = yes } set_variable = { var = ADISCORD_economy_model value = 4 } }
		if = { limit = { ADISCORD_economy_has_taxation_extraction_quotas = yes ADISCORD_economy_has_industrial_military_prioritization = yes } set_variable = { var = ADISCORD_economy_model value = 5 } }
		if = { limit = { ADISCORD_state_development_at_most_1 = yes ADISCORD_economy_has_taxation_industrial_tariffs = yes } set_variable = { var = ADISCORD_economy_model value = 6 } }
		if = { limit = { has_government = technocracy ADISCORD_economy_has_labor_technocratic_work_norms = yes } set_variable = { var = ADISCORD_economy_model value = 7 } }
	}

	set_variable = { var = ADISCORD_economy_cycle_phase value = 1 }
	if = { limit = { check_variable = { var = ADISCORD_economy_monthly_balance value = 0 compare = less_than } } set_variable = { var = ADISCORD_economy_cycle_phase value = 0 } }
	if = { limit = { check_variable = { var = ADISCORD_economy_monthly_balance value = 5 compare = greater_than_or_equals } check_variable = { var = ADISCORD_economy_stretched_level value = 2 compare = less_than } } set_variable = { var = ADISCORD_economy_cycle_phase value = 2 } }

	if = { limit = { check_variable = { var = ADISCORD_economy_stretched_level value = 3 compare = greater_than_or_equals } } set_variable = { var = ADISCORD_economy_cycle_phase value = 4 } }
	if = { limit = { has_war = yes OR = { ADISCORD_economy_has_taxation_extraction_quotas = yes ADISCORD_economy_has_industrial_military_prioritization = yes has_idea = ADISCORD_economic_system_mobilization } } set_variable = { var = ADISCORD_economy_cycle_phase value = 5 } }
	# State-planned economies do not use market business cycles in the dashboard.
	if = { limit = { ADISCORD_economy_model_is_planned_bureaucratic = yes } set_variable = { var = ADISCORD_economy_cycle_phase value = 6 } }
	if = { limit = { ADISCORD_economy_model_is_mobilization = yes } set_variable = { var = ADISCORD_economy_cycle_phase value = 7 } }
}
