# A-DISCORD strategic economy MVP.
# The economy is a budget layer. It feeds the existing ADISCORD_economic_development progress and ideas.
ADISCORD_economy_initialize_country = {
	if = {
		limit = {
			has_global_flag = ADISCORD_fresh_campaign_contract_v1
			NOT = { has_variable = ADISCORD_economy_initialized }
		}
		ADISCORD_initialize_society_development_variables = yes
		ADISCORD_economy_set_default_values = yes
		ADISCORD_economy_initialize_from_development = yes
		ADISCORD_economy_apply_country_starting_profile = yes
		ADISCORD_economy_set_simulation_tier = yes
		ADISCORD_economy_update_model_and_cycle = yes
		ADISCORD_economy_full_refresh = yes
		ADISCORD_economy_light_update = yes
		ADISCORD_economy_refresh_policy_previews = yes
		ADISCORD_economy_update_stretched = yes
		ADISCORD_economy_calculate_macro_indicators = yes
		ADISCORD_economy_update_model_and_cycle = yes
		ADISCORD_economy_calculate_development_multiplier = yes
		ADISCORD_economy_refresh_spending_ideas = yes
		ADISCORD_economy_update_gui = yes
	}
	else_if = {
		limit = { ADISCORD_economy_has_current_schema = yes }
		ADISCORD_economy_set_simulation_tier = yes
	}
}

# Country flavour is applied only during fresh-campaign initialization. Saves
# without the provenance marker remain untouched; current-schema saves only
# retain ordinary tier maintenance and never enter a migration path.
ADISCORD_economy_apply_country_starting_profile = {
	if = {
		limit = { tag = VAL }
		ADISCORD_economy_apply_val_starting_profile = yes
	}
	else_if = {
		limit = { tag = STP }
		ADISCORD_economy_apply_stp_starting_profile = yes
	}
}

ADISCORD_economy_apply_val_starting_profile = {
	set_variable = { var = ADISCORD_economy_treasury value = 180 }

ADISCORD_economy_migrate_schema = {
	# Quarantined legacy reference: validators require zero runtime callers.
	# Fresh initialization writes schema 15 directly and never enters this body.
	if = {
		limit = {
			OR = {
				NOT = { has_variable = ADISCORD_economy_schema_version }
				check_variable = { var = ADISCORD_economy_schema_version value = 11 compare = less_than }
			}
		}
		ADISCORD_economy_initialize_variables = yes
		if = {
			limit = { check_variable = { var = ADISCORD_economy_base_monthly_development_gain value = 0 compare = less_than_or_equals } }
			set_variable = { var = ADISCORD_economy_base_monthly_development_gain value = 1 }
		}
		if = { limit = { NOT = { has_variable = ADISCORD_economy_tax_change_cooldown } } set_variable = { var = ADISCORD_economy_tax_change_cooldown value = 0 } }
		if = { limit = { NOT = { has_variable = ADISCORD_economy_army_budget_change_cooldown } } set_variable = { var = ADISCORD_economy_army_budget_change_cooldown value = 0 } }
		if = { limit = { NOT = { has_variable = ADISCORD_economy_social_budget_change_cooldown } } set_variable = { var = ADISCORD_economy_social_budget_change_cooldown value = 0 } }
		if = {
			limit = { NOT = { has_variable = ADISCORD_economy_was_at_war } }
			set_variable = { var = ADISCORD_economy_was_at_war value = 0 }
			if = { limit = { has_war = yes } set_variable = { var = ADISCORD_economy_was_at_war value = 1 } }
		}
		if = { limit = { NOT = { has_variable = ADISCORD_economy_postwar_demobilization_months } } set_variable = { var = ADISCORD_economy_postwar_demobilization_months value = 0 } }
		# Schema 11 removes dead dashboard state, makes deficit borrowing mandatory,
		# and adds one cached reserve target. No recurring scan is introduced.
		ADISCORD_economy_recalculate_policy_modifiers = yes
		set_variable = { var = ADISCORD_economy_last_idea_signature value = -1 }
		set_variable = { var = ADISCORD_economy_schema_version value = 11 }
		set_variable = { var = ADISCORD_economy_needs_full_refresh value = 1 }
	}
	if = {
		limit = { check_variable = { var = ADISCORD_economy_schema_version value = 12 compare = less_than } }
		# Schema 12 retires construction as a policy without touching the ledger.
		# The previous policy level becomes the initial research course 1:1.
		set_variable = { var = ADISCORD_economy_research_spending_mode value = ADISCORD_economy_construction_spending_mode }
		clamp_variable = { var = ADISCORD_economy_research_spending_mode min = 1 max = 5 }
		set_variable = { var = ADISCORD_economy_research_budget_change_cooldown value = 0 }
		remove_ideas = {
			ADISCORD_economy_construction_spending_1
			ADISCORD_economy_construction_spending_2
			ADISCORD_economy_construction_spending_3
			ADISCORD_economy_construction_spending_4
			ADISCORD_economy_construction_spending_5
		}
		clear_variable = ADISCORD_economy_construction_spending_mode
		clear_variable = ADISCORD_economy_construction_budget_change_cooldown
		set_variable = { var = ADISCORD_economy_last_idea_signature value = -1 }
		set_variable = { var = ADISCORD_economy_schema_version value = 12 }
		set_variable = { var = ADISCORD_economy_needs_full_refresh value = 1 }
	}
	if = {
		limit = { check_variable = { var = ADISCORD_economy_schema_version value = 13 compare = less_than } }
		clear_variable = ADISCORD_economy_debt_capacity
		clear_variable = ADISCORD_economy_static_debt_capacity_bonus_bp
		clear_variable = ADISCORD_economy_policy_debt_capacity_bonus_bp
		clear_variable = ADISCORD_economy_final_debt_capacity_factor_bp
		clear_variable = ADISCORD_economy_debt_capacity_room_temp
		clear_variable = ADISCORD_economy_loan_debt_room_temp
		clear_variable = ADISCORD_economy_external_debt_room_temp
		clear_variable = ADISCORD_economy_capacity_factor_temp
		clear_variable = ADISCORD_economy_capacity_temp
		clear_variable = ADISCORD_economy_policy_preview_saved_capacity_factor_temp_temp
		clear_variable = ADISCORD_economy_policy_preview_saved_capacity_temp_temp
		clear_variable = ADISCORD_economy_policy_preview_saved_debt_capacity_temp
		clear_variable = ADISCORD_economy_debt_ratio
		clear_variable = ADISCORD_economy_pressure_temp
		clear_variable = ADISCORD_economy_last_uncovered_deficit
		clear_variable = ADISCORD_economy_last_period_unfunded_deficit
		set_variable = { var = ADISCORD_economy_schema_version value = 13 }
		set_variable = { var = ADISCORD_economy_needs_full_refresh value = 1 }
	}
	if = {
		limit = { check_variable = { var = ADISCORD_economy_schema_version value = 14 compare = less_than } }
		# Legacy saves have no trustworthy consecutive-settlement evidence. Preserve
		# only the safe healthy/strain/crisis band and start both bounded streaks at 0.
		set_variable = { var = ADISCORD_economy_debt_state value = 0 }
		if = {
			limit = { check_variable = { var = ADISCORD_economy_debt_crisis_level value = 2 compare = greater_than_or_equals } }
			set_variable = { var = ADISCORD_economy_debt_state value = 2 }
		}
		else_if = {
			limit = { check_variable = { var = ADISCORD_economy_debt_crisis_level value = 1 compare = greater_than_or_equals } }
			set_variable = { var = ADISCORD_economy_debt_state value = 1 }
		}
		set_variable = { var = ADISCORD_economy_debt_emergency_streak value = 0 }
		set_variable = { var = ADISCORD_economy_debt_default_streak value = 0 }
		set_variable = { var = ADISCORD_economy_pending_debt_notification_kind value = 0 }
		set_variable = { var = ADISCORD_economy_pending_debt_notification_amount value = 0 }
		set_variable = { var = ADISCORD_economy_pending_debt_notification_previous_state value = ADISCORD_economy_debt_state }
		set_variable = { var = ADISCORD_economy_pending_debt_notification_new_state value = ADISCORD_economy_debt_state }
		set_variable = { var = ADISCORD_economy_last_notified_debt_state value = ADISCORD_economy_debt_state }
		clear_variable = ADISCORD_economy_first_loan_notified
		if = {
			limit = { check_variable = { var = ADISCORD_economy_debt value = 0 compare = greater_than } }
			set_variable = { var = ADISCORD_economy_first_loan_notified value = 1 }
		}
		clear_variable = ADISCORD_economy_show_auto_loan_popup
		clear_variable = ADISCORD_economy_auto_loan_popup_amount
		set_variable = { var = ADISCORD_economy_debt_crisis_level value = ADISCORD_economy_debt_state }
		remove_ideas = {
			ADISCORD_economy_debt_strain
			ADISCORD_economy_debt_crisis
			ADISCORD_economy_debt_emergency
			ADISCORD_economy_debt_default
		}
		if = { limit = { check_variable = { var = ADISCORD_economy_debt_state value = 2 compare = equals } } add_ideas = ADISCORD_economy_debt_crisis }
		else_if = { limit = { check_variable = { var = ADISCORD_economy_debt_state value = 1 compare = equals } } add_ideas = ADISCORD_economy_debt_strain }
		set_variable = { var = ADISCORD_economy_schema_version value = 14 }
		set_variable = { var = ADISCORD_economy_needs_full_refresh value = 1 }
	}
	if = {
		limit = { check_variable = { var = ADISCORD_economy_schema_version value = 15 compare = less_than } }
		# Task 7 adds weekly source caches. Existing schema-14 saves must become
		# ineligible before one bounded rebuild publishes the readiness marker.
		set_variable = { var = ADISCORD_economy_weekly_source_cache_ready value = 0 }
		set_variable = { var = ADISCORD_economy_weekly_ready value = 0 }
		set_variable = { var = ADISCORD_economy_needs_full_refresh value = 1 }
		ADISCORD_economy_full_refresh_if_needed = yes
		set_variable = { var = ADISCORD_economy_schema_version value = 15 }
	}
	# A schema branch or a bounded dirty hook requests one heavy rebuild. Weekly
	# accounting never calls migration, so consuming it here cannot leak source
	# queries into the weekly graph.
	ADISCORD_economy_full_refresh_if_needed = yes
}

				check_variable = {
					var = ADISCORD_economy_simulation_tier
					value = ADISCORD_economy_simulation_tier_target_temp

				}
			}
		}

		ADISCORD_economy_refresh_ai_assistance = yes
	}
}

# Keep newly introduced transition effects above their first caller. Full game
# startup registers the whole database, while a scripted-effect hot reload can
# otherwise parse a call before it has seen the new effect name.
ADISCORD_economy_update_postwar_demobilization = {
	# Event-driven on_war/on_peace handles the normal path. The scheduled economy
	# pulse remains a save-safe fallback and still operates in country scope.
	if = {
		limit = { has_war = yes }
		set_variable = { var = ADISCORD_economy_was_at_war value = 1 }
		if = {
			limit = { check_variable = { var = ADISCORD_economy_postwar_demobilization_months value = 0 compare = greater_than } }
			set_variable = { var = ADISCORD_economy_postwar_demobilization_months value = 0 }
			remove_ideas = ADISCORD_economy_postwar_demobilization
		}
	}
	else_if = {
		limit = {
			has_war = no
			check_variable = { var = ADISCORD_economy_was_at_war value = 1 compare = greater_than_or_equals }
		}
		set_variable = { var = ADISCORD_economy_was_at_war value = 0 }
		set_variable = { var = ADISCORD_economy_postwar_demobilization_months value = 6 }
		if = {
			limit = { check_variable = { var = ADISCORD_economy_army_spending_mode value = 3 compare = greater_than } }
			set_variable = { var = ADISCORD_economy_army_spending_mode value = 3 }
			set_variable = { var = ADISCORD_economy_army_budget_change_cooldown value = 3 }
		}
		if = {
			limit = { OR = { ADISCORD_economy_has_idea_total_economic_mobilisation = yes ADISCORD_economy_has_idea_war_economy = yes } }
			add_ideas = partial_economic_mobilisation
		}
		if = {
			limit = {
				OR = {
					ADISCORD_economy_has_idea_extensive_conscription = yes
					ADISCORD_economy_has_idea_service_by_requirement = yes
					ADISCORD_economy_has_idea_all_adults_serve = yes
					ADISCORD_economy_has_idea_scraping_the_barrel = yes
				}
			}
			add_ideas = limited_conscription
		}
		add_timed_idea = { idea = ADISCORD_economy_postwar_demobilization days = 180 }
		if = { limit = { is_ai = no } country_event = { id = ADISCORD_economy.2 } }
	}
}

ADISCORD_economy_tick_postwar_demobilization = {
	if = {
		limit = { check_variable = { var = ADISCORD_economy_postwar_demobilization_months value = 0 compare = greater_than } }
		set_temp_variable = { var = ADISCORD_economy_demobilization_time_temp value = -1 }
		multiply_temp_variable = { var = ADISCORD_economy_demobilization_time_temp value = ADISCORD_economy_tick_scale }
		add_to_variable = { var = ADISCORD_economy_postwar_demobilization_months value = ADISCORD_economy_demobilization_time_temp }
		clamp_variable = { var = ADISCORD_economy_postwar_demobilization_months min = 0 max = 6 }
	}
}

ADISCORD_economy_calculate_army_expenses = {
	if = {
		limit = { check_variable = { var = ADISCORD_economy_policy_preview_uses_cached_base_temp value = 1 compare = greater_than_or_equals } }
		set_variable = { var = ADISCORD_economy_army_expenses value = ADISCORD_economy_army_expense_policy_base }
	}
	else = {
		# Actual fielded manpower is charged through stable threshold triggers. The
		# tiers are cumulative, so very large armies become materially expensive.
		set_variable = { var = ADISCORD_economy_army_expenses value = 0.75 }
		if = { limit = { has_army_manpower = { size > 9999 } } add_to_variable = { var = ADISCORD_economy_army_expenses value = 0.35 } }
		if = { limit = { has_army_manpower = { size > 24999 } } add_to_variable = { var = ADISCORD_economy_army_expenses value = 0.50 } }
		if = { limit = { has_army_manpower = { size > 49999 } } add_to_variable = { var = ADISCORD_economy_army_expenses value = 0.75 } }
		if = { limit = { has_army_manpower = { size > 99999 } } add_to_variable = { var = ADISCORD_economy_army_expenses value = 1.00 } }
		if = { limit = { has_army_manpower = { size > 249999 } } add_to_variable = { var = ADISCORD_economy_army_expenses value = 1.50 } }
		if = { limit = { ADISCORD_army_development_at_least_2 = yes } add_to_variable = { var = ADISCORD_economy_army_expenses value = 0.5 } }
		if = { limit = { ADISCORD_army_development_at_least_3 = yes } add_to_variable = { var = ADISCORD_economy_army_expenses value = 0.5 } }
		if = { limit = { ADISCORD_army_development_at_least_4 = yes } add_to_variable = { var = ADISCORD_economy_army_expenses value = 0.75 } }
		if = { limit = { ADISCORD_army_development_at_least_5 = yes } add_to_variable = { var = ADISCORD_economy_army_expenses value = 0.75 } }
		add_to_variable = { var = ADISCORD_economy_army_expenses value = ADISCORD_economy_cached_army_organization_flat_expense }
		multiply_variable = { var = ADISCORD_economy_army_expenses value = ADISCORD_economy_cached_army_organization_factor }
		if = { limit = { has_war = yes } multiply_variable = { var = ADISCORD_economy_army_expenses value = 1.25 } }

		if = { limit = { ADISCORD_army_development_at_least_5 = yes } multiply_variable = { var = ADISCORD_economy_army_expenses value = 1.05 } }
		set_variable = { var = ADISCORD_economy_army_expense_policy_base value = ADISCORD_economy_army_expenses }

	if = { limit = { check_variable = { var = ADISCORD_economy_army_spending_mode value = 1 compare = equals } } multiply_variable = { var = ADISCORD_economy_army_expenses value = 0.50 } }
	else_if = { limit = { check_variable = { var = ADISCORD_economy_army_spending_mode value = 2 compare = equals } } multiply_variable = { var = ADISCORD_economy_army_expenses value = 0.75 } }
	else_if = { limit = { check_variable = { var = ADISCORD_economy_army_spending_mode value = 3 compare = equals } } multiply_variable = { var = ADISCORD_economy_army_expenses value = 1.00 } }
	else_if = { limit = { check_variable = { var = ADISCORD_economy_army_spending_mode value = 4 compare = equals } } multiply_variable = { var = ADISCORD_economy_army_expenses value = 1.50 } }

ADISCORD_economy_update_debt_state_after_settlement = {
	# A settlement advances each bounded streak once at most. Its paired else
	# resets immediately when the exact qualifying condition is false.
	if = {
		limit = { check_variable = { var = ADISCORD_economy_interest_share_income value = 40 compare = greater_than_or_equals } }
		add_to_variable = { var = ADISCORD_economy_debt_emergency_streak value = 1 }
		if = {
			limit = { check_variable = { var = ADISCORD_economy_debt_emergency_streak value = 4 compare = greater_than_or_equals } }
			set_variable = { var = ADISCORD_economy_debt_state value = 3 }
		}
	}
	else = { set_variable = { var = ADISCORD_economy_debt_emergency_streak value = 0 } }
	if = {
		limit = {
			check_variable = { var = ADISCORD_economy_interest_share_income value = 40 compare = greater_than_or_equals }
			check_variable = { var = ADISCORD_economy_weekly_balance value = 0 compare = less_than }
		}
		add_to_variable = { var = ADISCORD_economy_debt_default_streak value = 1 }
		if = {
			limit = { check_variable = { var = ADISCORD_economy_debt_default_streak value = 13 compare = greater_than_or_equals } }
			set_variable = { var = ADISCORD_economy_debt_state value = 4 }
		}
	}
	else = { set_variable = { var = ADISCORD_economy_debt_default_streak value = 0 } }
	# The lower chain cannot overwrite an earned emergency/default in the same
	# settlement. It does make genuine improvements visible immediately.
	if = {
		limit = {
			check_variable = { var = ADISCORD_economy_debt_default_streak value = 13 compare = less_than }
			check_variable = { var = ADISCORD_economy_debt_emergency_streak value = 4 compare = less_than }
		}
		if = {
			limit = { check_variable = { var = ADISCORD_economy_interest_share_income value = 25 compare = greater_than_or_equals } }
			set_variable = { var = ADISCORD_economy_debt_state value = 2 }
		}
		else_if = {
			limit = { check_variable = { var = ADISCORD_economy_interest_share_income value = 10 compare = greater_than_or_equals } }
			set_variable = { var = ADISCORD_economy_debt_state value = 1 }
		}
		else = { set_variable = { var = ADISCORD_economy_debt_state value = 0 } }
	}
	clamp_variable = { var = ADISCORD_economy_debt_emergency_streak min = 0 max = 4 }
	clamp_variable = { var = ADISCORD_economy_debt_default_streak min = 0 max = 13 }
	set_variable = { var = ADISCORD_economy_debt_crisis_level value = ADISCORD_economy_debt_state }
	ADISCORD_economy_sync_debt_state_idea = yes
}

ADISCORD_economy_queue_debt_notification = {
	# Cache first, then let an upward state replace kind 1. The borrowed amount
	# remains intact so a combined notice still explains the automatic loan.
	set_variable = { var = ADISCORD_economy_pending_debt_notification_kind value = 0 }
	set_variable = { var = ADISCORD_economy_pending_debt_notification_new_state value = ADISCORD_economy_debt_state }
	if = {
		limit = {
			check_variable = { var = ADISCORD_economy_pending_debt_notification_amount value = 0 compare = greater_than }
			NOT = { has_variable = ADISCORD_economy_first_loan_notified }
		}
		set_variable = { var = ADISCORD_economy_pending_debt_notification_kind value = 1 }
		set_variable = { var = ADISCORD_economy_first_loan_notified value = 1 }
	}
	if = {
		limit = {
			check_variable = { var = ADISCORD_economy_debt_state value = ADISCORD_economy_pending_debt_notification_previous_state compare = greater_than }
			check_variable = { var = ADISCORD_economy_debt_state value = ADISCORD_economy_last_notified_debt_state compare = greater_than }
		}
		if = { limit = { check_variable = { var = ADISCORD_economy_debt_state value = 4 compare = equals } } set_variable = { var = ADISCORD_economy_pending_debt_notification_kind value = 5 } }
		else_if = { limit = { check_variable = { var = ADISCORD_economy_debt_state value = 3 compare = equals } } set_variable = { var = ADISCORD_economy_pending_debt_notification_kind value = 4 } }
		else_if = { limit = { check_variable = { var = ADISCORD_economy_debt_state value = 2 compare = equals } } set_variable = { var = ADISCORD_economy_pending_debt_notification_kind value = 3 } }
		else_if = { limit = { check_variable = { var = ADISCORD_economy_debt_state value = 1 compare = equals } } set_variable = { var = ADISCORD_economy_pending_debt_notification_kind value = 2 } }
		set_variable = { var = ADISCORD_economy_last_notified_debt_state value = ADISCORD_economy_debt_state }
	}
	if = {
		limit = {
			check_variable = { var = ADISCORD_economy_debt_state value = ADISCORD_economy_pending_debt_notification_previous_state compare = less_than }
			check_variable = { var = ADISCORD_economy_last_notified_debt_state value = ADISCORD_economy_debt_state compare = greater_than }
		}
		set_variable = { var = ADISCORD_economy_last_notified_debt_state value = ADISCORD_economy_debt_state }
	}
	if = {
		limit = {
			is_ai = no
			check_variable = { var = ADISCORD_economy_pending_debt_notification_kind value = 0 compare = greater_than }
		}
		country_event = { id = ADISCORD_economy.3 }
	}
}

# Repayment actions may immediately lower an already-persistent debt state and
# its debuff. Task 6 owns every upward transition, settlement streak, and
# notification; this reconciler deliberately cannot create states 3 or 4.
ADISCORD_economy_reconcile_debt_state_after_action = {
	if = {
		limit = {
			check_variable = { var = ADISCORD_economy_interest_share_income value = 10 compare = less_than }
			check_variable = { var = ADISCORD_economy_debt_state value = 0 compare = greater_than }
		}
		set_variable = { var = ADISCORD_economy_debt_state value = 0 }
		if = {
			limit = { check_variable = { var = ADISCORD_economy_last_notified_debt_state value = 0 compare = greater_than } }
			set_variable = { var = ADISCORD_economy_last_notified_debt_state value = 0 }
		}
		remove_ideas = { ADISCORD_economy_debt_strain ADISCORD_economy_debt_crisis ADISCORD_economy_debt_emergency ADISCORD_economy_debt_default }
	}
	else_if = {
		limit = {
			check_variable = { var = ADISCORD_economy_interest_share_income value = 25 compare = less_than }
			check_variable = { var = ADISCORD_economy_debt_state value = 1 compare = greater_than }
		}
		set_variable = { var = ADISCORD_economy_debt_state value = 1 }
		if = {
			limit = { check_variable = { var = ADISCORD_economy_last_notified_debt_state value = 1 compare = greater_than } }
			set_variable = { var = ADISCORD_economy_last_notified_debt_state value = 1 }
		}
		remove_ideas = { ADISCORD_economy_debt_strain ADISCORD_economy_debt_crisis ADISCORD_economy_debt_emergency ADISCORD_economy_debt_default }
		add_ideas = ADISCORD_economy_debt_strain
	}
	else_if = {
		limit = {
			check_variable = { var = ADISCORD_economy_interest_share_income value = 40 compare = less_than }
			check_variable = { var = ADISCORD_economy_debt_state value = 2 compare = greater_than }
		}
		set_variable = { var = ADISCORD_economy_debt_state value = 2 }
		if = {
			limit = { check_variable = { var = ADISCORD_economy_last_notified_debt_state value = 2 compare = greater_than } }

ADISCORD_economy_calculate_weekly_budget = {
	# 52 weekly settlements must equal the former twelve monthly settlements:
	# 12 / 52 = 3 / 13.
	set_variable = { var = ADISCORD_economy_weekly_income value = ADISCORD_economy_monthly_income }
	multiply_variable = { var = ADISCORD_economy_weekly_income value = 3 }
	divide_variable = { var = ADISCORD_economy_weekly_income value = 13 }
	# Flat country modifiers are applied after the 3/13 conversion: +20 is exactly +20 per week.
	add_to_variable = { var = ADISCORD_economy_weekly_income value = ADISCORD_economy_final_weekly_income_bonus }
	set_variable = { var = ADISCORD_economy_weekly_expenses value = ADISCORD_economy_monthly_expenses }
	multiply_variable = { var = ADISCORD_economy_weekly_expenses value = 3 }
	divide_variable = { var = ADISCORD_economy_weekly_expenses value = 13 }
	set_variable = { var = ADISCORD_economy_weekly_balance value = ADISCORD_economy_weekly_income }
	subtract_from_variable = { var = ADISCORD_economy_weekly_balance value = ADISCORD_economy_weekly_expenses }
	# Display-only weekly equivalent of the last macro inflation change. The source
	# remains authoritative; 3/13 is the same exact month-to-week conversion used
	# by the cash forecast, and this value has no gameplay consumer.
	set_variable = { var = ADISCORD_economy_weekly_inflation_change value = ADISCORD_economy_inflation_delta_temp }
	multiply_variable = { var = ADISCORD_economy_weekly_inflation_change value = 3 }
	divide_variable = { var = ADISCORD_economy_weekly_inflation_change value = 13 }
	# Four weeks of ordinary spending is a readable player target, not a new
	# subsystem. It reuses an already-calculated scalar and adds no scan.
	set_variable = { var = ADISCORD_economy_safe_reserve value = ADISCORD_economy_weekly_expenses }
	multiply_variable = { var = ADISCORD_economy_safe_reserve value = 4 }
	clamp_variable = { var = ADISCORD_economy_safe_reserve min = 50 max = 250 }
	# Display-only runway: 260 means at least five years, including a non-negative
	# current balance. The denominator floor prevents a near-zero deficit from
	# producing an unstable UI value; neither cache is consumed by gameplay.
	set_variable = { var = ADISCORD_economy_deficit_runway value = 260 }
	if = {
		limit = { check_variable = { var = ADISCORD_economy_weekly_balance value = 0 compare = less_than } }
		set_temp_variable = { var = ADISCORD_economy_deficit_runway_denominator_temp value = 0 }
		subtract_from_temp_variable = { var = ADISCORD_economy_deficit_runway_denominator_temp value = ADISCORD_economy_weekly_balance }
		clamp_temp_variable = { var = ADISCORD_economy_deficit_runway_denominator_temp min = 0.01 max = 5000 }
		set_variable = { var = ADISCORD_economy_deficit_runway value = ADISCORD_economy_treasury }
		divide_variable = { var = ADISCORD_economy_deficit_runway value = ADISCORD_economy_deficit_runway_denominator_temp }
	}
	clamp_variable = { var = ADISCORD_economy_deficit_runway min = 0 max = 260 }
	# Mirrors the live inflation ideas' consumer-goods burden bands. This is a
	# display multiplier, not an additional expense or modifier source.
	set_variable = { var = ADISCORD_economy_inflation_expense_multiplier value = 1 }
	if = { limit = { check_variable = { var = ADISCORD_economy_inflation value = 10 compare = greater_than_or_equals } } set_variable = { var = ADISCORD_economy_inflation_expense_multiplier value = 1.01 } }
	if = { limit = { check_variable = { var = ADISCORD_economy_inflation value = 25 compare = greater_than_or_equals } } set_variable = { var = ADISCORD_economy_inflation_expense_multiplier value = 1.04 } }
	if = { limit = { check_variable = { var = ADISCORD_economy_inflation value = 50 compare = greater_than_or_equals } } set_variable = { var = ADISCORD_economy_inflation_expense_multiplier value = 1.08 } }
	if = { limit = { check_variable = { var = ADISCORD_economy_inflation value = 75 compare = greater_than_or_equals } } set_variable = { var = ADISCORD_economy_inflation_expense_multiplier value = 1.15 } }
}

ADISCORD_economy_update_monthly_budget_trend = {
	# Macro pressure remains monthly even though cash now settles weekly.
	if = {
		limit = { check_variable = { var = ADISCORD_economy_monthly_balance value = 0 compare = greater_than_or_equals } }
		set_variable = { var = ADISCORD_economy_deficit_streak value = 0 }
		add_to_variable = { var = ADISCORD_economy_surplus_streak value = 1 }
		add_to_variable = { var = ADISCORD_economy_deficit_pressure value = -2 }
	}
	else = {
		# Only new pressure is modified. A surplus always clears two points, so a
		# penalty cannot accidentally weaken recovery as well as worsening deficits.
		set_temp_variable = { var = ADISCORD_economy_deficit_pressure_gain_temp value = 1 }

ADISCORD_economy_update_fiscal_stress = {
	# TDA-lite pass: fiscal stress is a slow warning indicator, not a self-feeding death spiral.
	set_variable = { var = ADISCORD_economy_fiscal_stress_delta value = 0 }
	if = { limit = { check_variable = { var = ADISCORD_economy_interest_share_income value = 10 compare = greater_than_or_equals } } add_to_variable = { var = ADISCORD_economy_fiscal_stress_delta value = 0.5 } }
	if = { limit = { check_variable = { var = ADISCORD_economy_interest_share_income value = 25 compare = greater_than_or_equals } } add_to_variable = { var = ADISCORD_economy_fiscal_stress_delta value = 1.0 } }
	if = { limit = { check_variable = { var = ADISCORD_economy_deficit_streak value = 3 compare = greater_than_or_equals } } add_to_variable = { var = ADISCORD_economy_fiscal_stress_delta value = 0.5 } }
	if = { limit = { check_variable = { var = ADISCORD_economy_deficit_streak value = 6 compare = greater_than_or_equals } } add_to_variable = { var = ADISCORD_economy_fiscal_stress_delta value = 1.0 } }
	if = { limit = { check_variable = { var = ADISCORD_economy_inflation value = 25 compare = greater_than_or_equals } } add_to_variable = { var = ADISCORD_economy_fiscal_stress_delta value = 0.5 } }
	if = { limit = { check_variable = { var = ADISCORD_economy_debt_service value = 20 compare = greater_than_or_equals } } add_to_variable = { var = ADISCORD_economy_fiscal_stress_delta value = 0.5 } }
	if = { limit = { check_variable = { var = ADISCORD_economy_monthly_balance value = 0 compare = greater_than_or_equals } } add_to_variable = { var = ADISCORD_economy_fiscal_stress_delta value = -0.75 } }
	if = { limit = { check_variable = { var = ADISCORD_economy_surplus_streak value = 3 compare = greater_than_or_equals } } add_to_variable = { var = ADISCORD_economy_fiscal_stress_delta value = -0.75 } }
	if = {
		limit = { check_variable = { var = ADISCORD_economy_fiscal_stress_delta value = 0 compare = greater_than } }
		multiply_variable = { var = ADISCORD_economy_fiscal_stress_delta value = ADISCORD_economy_final_fiscal_stress_gain_factor_bp }
		divide_variable = { var = ADISCORD_economy_fiscal_stress_delta value = 100 }
		multiply_variable = { var = ADISCORD_economy_fiscal_stress_delta value = 100 }
		divide_variable = { var = ADISCORD_economy_fiscal_stress_delta value = ADISCORD_economy_final_fiscal_stress_resistance_factor_bp }
	}
	multiply_variable = { var = ADISCORD_economy_fiscal_stress_delta value = ADISCORD_economy_tick_scale }
	if = {
		limit = { check_variable = { var = ADISCORD_economy_tick_scale value = 6 compare = greater_than_or_equals } }
		clamp_variable = { var = ADISCORD_economy_fiscal_stress_delta min = -9 max = 12 }
	}
	else = { clamp_variable = { var = ADISCORD_economy_fiscal_stress_delta min = -1.5 max = 2.0 } }
	add_to_variable = { var = ADISCORD_economy_fiscal_stress value = ADISCORD_economy_fiscal_stress_delta }
	if = { limit = { check_variable = { var = ADISCORD_economy_restructuring_penalty value = 0 compare = greater_than } check_variable = { var = ADISCORD_economy_monthly_balance value = 0 compare = greater_than_or_equals } } add_to_variable = { var = ADISCORD_economy_restructuring_penalty value = -1 } }
	clamp_variable = { var = ADISCORD_economy_fiscal_stress min = 0 max = 100 }
}


ADISCORD_economy_update_inflation = {
	# TDA-lite pass: inflation moves slowly and is primarily caused by explicit emission, shocks and crises.
	# Planned-bureaucratic systems suppress visible market inflation; deficits become shortages/overload instead.
	if = {
		limit = { ADISCORD_economy_model_is_planned_bureaucratic = yes }
		set_variable = { var = ADISCORD_economy_emission_pressure value = 0 }
		set_variable = { var = ADISCORD_economy_inflation_delta_temp value = 0 }
		if = {
			limit = { check_variable = { var = ADISCORD_economy_inflation value = 0 compare = greater_than } }
			set_variable = { var = ADISCORD_economy_inflation_delta_temp value = -2 }
			multiply_variable = { var = ADISCORD_economy_inflation_delta_temp value = ADISCORD_economy_tick_scale }
			set_temp_variable = { var = ADISCORD_economy_planned_inflation_floor_temp value = ADISCORD_economy_inflation }
			add_to_temp_variable = { var = ADISCORD_economy_planned_inflation_floor_temp value = ADISCORD_economy_inflation_delta_temp }
			if = {
				limit = { check_variable = { var = ADISCORD_economy_planned_inflation_floor_temp value = 0 compare = less_than } }
				set_variable = { var = ADISCORD_economy_inflation_delta_temp value = 0 }
				subtract_from_variable = { var = ADISCORD_economy_inflation_delta_temp value = ADISCORD_economy_inflation }
			}
			add_to_variable = { var = ADISCORD_economy_inflation value = ADISCORD_economy_inflation_delta_temp }
		}
		# Planned systems convert monetary and fiscal pressure into persistent
		# shortages. This is a base component of overload, not a write into the
		# derived stretched score that will be rebuilt later in the tick.
		set_temp_variable = { var = ADISCORD_economy_planned_shortage_delta_temp value = ADISCORD_economy_money_printing_level }
		if = { limit = { check_variable = { var = ADISCORD_economy_deficit_pressure value = 20 compare = greater_than_or_equals } } add_to_temp_variable = { var = ADISCORD_economy_planned_shortage_delta_temp value = 1 } }
		if = { limit = { check_variable = { var = ADISCORD_economy_monthly_balance value = 0 compare = greater_than_or_equals } } add_to_temp_variable = { var = ADISCORD_economy_planned_shortage_delta_temp value = -1 } }
		if = { limit = { check_variable = { var = ADISCORD_economy_money_printing_level value = 0 compare = less_than_or_equals } } add_to_temp_variable = { var = ADISCORD_economy_planned_shortage_delta_temp value = -1 } }
		multiply_temp_variable = { var = ADISCORD_economy_planned_shortage_delta_temp value = ADISCORD_economy_tick_scale }
		add_to_variable = { var = ADISCORD_economy_planned_shortage_pressure value = ADISCORD_economy_planned_shortage_delta_temp }
	}
	else = {
		set_temp_variable = { var = ADISCORD_economy_market_shortage_decay_temp value = -2 }
		multiply_temp_variable = { var = ADISCORD_economy_market_shortage_decay_temp value = ADISCORD_economy_tick_scale }
		add_to_variable = { var = ADISCORD_economy_planned_shortage_pressure value = ADISCORD_economy_market_shortage_decay_temp }
		set_variable = { var = ADISCORD_economy_emission_pressure value = ADISCORD_economy_money_printing_level }
		multiply_variable = { var = ADISCORD_economy_emission_pressure value = 2 }
		multiply_variable = { var = ADISCORD_economy_emission_pressure value = 100 }
		divide_variable = { var = ADISCORD_economy_emission_pressure value = ADISCORD_economy_final_money_printing_efficiency_factor_bp }
		set_variable = { var = ADISCORD_economy_inflation_pressure_temp value = ADISCORD_economy_emission_pressure }
		set_variable = { var = ADISCORD_economy_inflation_component_temp value = ADISCORD_economy_deficit_pressure }
		divide_variable = { var = ADISCORD_economy_inflation_component_temp value = 40 }
		add_to_variable = { var = ADISCORD_economy_inflation_pressure_temp value = ADISCORD_economy_inflation_component_temp }
		set_variable = { var = ADISCORD_economy_inflation_component_temp value = ADISCORD_economy_price_shock }
		divide_variable = { var = ADISCORD_economy_inflation_component_temp value = 5 }
		add_to_variable = { var = ADISCORD_economy_inflation_pressure_temp value = ADISCORD_economy_inflation_component_temp }
		set_variable = { var = ADISCORD_economy_inflation_component_temp value = ADISCORD_economy_debt_crisis_level }
		multiply_variable = { var = ADISCORD_economy_inflation_component_temp value = 0.5 }
		add_to_variable = { var = ADISCORD_economy_inflation_pressure_temp value = ADISCORD_economy_inflation_component_temp }
		if = { limit = { has_war = yes } add_to_variable = { var = ADISCORD_economy_inflation_pressure_temp value = 0.25 } }
		if = { limit = { ADISCORD_economy_model_is_mobilization = yes } add_to_variable = { var = ADISCORD_economy_inflation_pressure_temp value = 0.5 } }
		multiply_variable = { var = ADISCORD_economy_inflation_pressure_temp value = ADISCORD_economy_final_inflation_pressure_factor_bp }
		divide_variable = { var = ADISCORD_economy_inflation_pressure_temp value = 100 }
		multiply_variable = { var = ADISCORD_economy_inflation_pressure_temp value = 100 }
		divide_variable = { var = ADISCORD_economy_inflation_pressure_temp value = ADISCORD_economy_final_price_stability_factor_bp }
		set_variable = { var = ADISCORD_economy_inflation_delta_temp value = ADISCORD_economy_inflation_pressure_temp }
		divide_variable = { var = ADISCORD_economy_inflation_delta_temp value = 6 }
		# Like TDA, ordinary inactivity is not an anti-inflation policy. Market
		# inflation is sticky and falls only through explicit stabilisation actions.
		clamp_variable = { var = ADISCORD_economy_inflation_delta_temp min = 0 max = 1.5 }
		multiply_variable = { var = ADISCORD_economy_inflation_delta_temp value = ADISCORD_economy_tick_scale }
		add_to_variable = { var = ADISCORD_economy_inflation value = ADISCORD_economy_inflation_delta_temp }
	}
	if = {
		limit = { check_variable = { var = ADISCORD_economy_price_shock value = 0 compare = greater_than } }
		set_temp_variable = { var = ADISCORD_economy_price_shock_decay_temp value = -0.5 }
		multiply_temp_variable = { var = ADISCORD_economy_price_shock_decay_temp value = ADISCORD_economy_tick_scale }
		add_to_variable = { var = ADISCORD_economy_price_shock value = ADISCORD_economy_price_shock_decay_temp }
	}
	if = {
		limit = { check_variable = { var = ADISCORD_economy_deficit_pressure value = 0 compare = greater_than } check_variable = { var = ADISCORD_economy_monthly_balance value = 0 compare = greater_than_or_equals } }
		set_temp_variable = { var = ADISCORD_economy_deficit_pressure_decay_temp value = -2 }
		multiply_temp_variable = { var = ADISCORD_economy_deficit_pressure_decay_temp value = ADISCORD_economy_tick_scale }
		add_to_variable = { var = ADISCORD_economy_deficit_pressure value = ADISCORD_economy_deficit_pressure_decay_temp }
	}
	clamp_variable = { var = ADISCORD_economy_inflation min = 0 max = 100 }
	clamp_variable = { var = ADISCORD_economy_emission_pressure min = 0 max = 100 }
	clamp_variable = { var = ADISCORD_economy_price_shock min = 0 max = 100 }
	clamp_variable = { var = ADISCORD_economy_was_at_war min = 0 max = 1 }
	clamp_variable = { var = ADISCORD_economy_postwar_demobilization_months min = 0 max = 6 }
	clamp_variable = { var = ADISCORD_economy_deficit_pressure min = 0 max = 100 }
	clamp_variable = { var = ADISCORD_economy_planned_shortage_pressure min = 0 max = 50 }
}

	if = {
		limit = { check_variable = { var = ADISCORD_economic_development_progress value = 100 compare = greater_than_or_equals } }
		if = {
			limit = { ADISCORD_economy_can_upgrade_economic_development = yes }
			set_variable = { var = ADISCORD_economic_development_progress value = 0 }
			ADISCORD_increase_economic_development = yes
		}
		else = {
			set_variable = { var = ADISCORD_economic_development_progress value = 99 }
		}
	}
	ADISCORD_economy_refresh_existing_economic_development_idea = yes
}

ADISCORD_economy_update_war_fatigue = {

	if = { limit = { has_war = yes } add_to_temp_variable = { var = ADISCORD_economy_war_fatigue_time_temp value = 2 } }
	else = { add_to_temp_variable = { var = ADISCORD_economy_war_fatigue_time_temp value = -1 } }
	if = {
		limit = {
			has_war = no
			check_variable = { var = ADISCORD_economy_postwar_demobilization_months value = 0 compare = greater_than }
		}
		add_to_temp_variable = { var = ADISCORD_economy_war_fatigue_time_temp value = -4 }
	}
	if = { limit = { check_variable = { var = ADISCORD_economy_army_spending_mode value = 5 compare = greater_than_or_equals } has_war = yes } add_to_temp_variable = { var = ADISCORD_economy_war_fatigue_time_temp value = 2 } }
	# Mobilized institutions are exhausting only while they are actually serving
	# a war. In peace the score must decay predictably instead of rising invisibly.
	if = { limit = { has_war = yes ADISCORD_economy_has_military_total_defense_grid = yes } add_to_temp_variable = { var = ADISCORD_economy_war_fatigue_time_temp value = 1 } }
	if = { limit = { has_war = yes ADISCORD_economy_has_labor_mobilized_labor = yes } add_to_temp_variable = { var = ADISCORD_economy_war_fatigue_time_temp value = 1 } }
	if = { limit = { has_war = yes ADISCORD_economy_has_taxation_extraction_quotas = yes } add_to_temp_variable = { var = ADISCORD_economy_war_fatigue_time_temp value = 1 } }

	add_to_variable = { var = ADISCORD_economy_war_fatigue_score value = ADISCORD_economy_war_fatigue_time_temp }

	else_if = { limit = { check_variable = { var = ADISCORD_economy_monthly_casualties_delta_k value = 25 compare = greater_than_or_equals } } add_to_variable = { var = ADISCORD_economy_war_fatigue_score value = 2 } }
	else_if = { limit = { check_variable = { var = ADISCORD_economy_monthly_casualties_delta_k value = 5 compare = greater_than_or_equals } } add_to_variable = { var = ADISCORD_economy_war_fatigue_score value = 1 } }

		add_to_variable = { var = ADISCORD_economy_war_fatigue_score value = ADISCORD_economy_war_fatigue_delta_temp }
	}
	clamp_variable = { var = ADISCORD_economy_war_fatigue_score min = 0 max = 100 }

ADISCORD_economy_war_taxes_action = {

	if = {
		limit = { ADISCORD_economy_can_use_war_taxes = yes }
		set_temp_variable = { var = ADISCORD_economy_war_tax_cash_temp value = 40 }
		if = { limit = { ADISCORD_economy_model_is_state_coordinated = yes } add_to_temp_variable = { var = ADISCORD_economy_war_tax_cash_temp value = 15 } }
		if = { limit = { ADISCORD_economy_model_is_planned_bureaucratic = yes } add_to_temp_variable = { var = ADISCORD_economy_war_tax_cash_temp value = 20 } }
		if = { limit = { ADISCORD_economy_model_is_mobilization = yes } add_to_temp_variable = { var = ADISCORD_economy_war_tax_cash_temp value = 25 } }
		set_temp_variable = { var = ADISCORD_economy_war_tax_room_temp value = ADISCORD_economy_treasury_cap }
		subtract_from_variable = { var = ADISCORD_economy_war_tax_room_temp value = ADISCORD_economy_treasury }
		set_temp_variable = { var = ADISCORD_economy_war_tax_overflow_temp value = ADISCORD_economy_war_tax_cash_temp }
		subtract_from_variable = { var = ADISCORD_economy_war_tax_overflow_temp value = ADISCORD_economy_war_tax_room_temp }
		if = { limit = { check_variable = { var = ADISCORD_economy_war_tax_overflow_temp value = 0 compare = greater_than } } subtract_from_variable = { var = ADISCORD_economy_war_tax_cash_temp value = ADISCORD_economy_war_tax_overflow_temp } }
		add_to_variable = { var = ADISCORD_economy_treasury value = ADISCORD_economy_war_tax_cash_temp }
		add_to_variable = { var = ADISCORD_economy_current_month_action_income value = ADISCORD_economy_war_tax_cash_temp }
		set_temp_variable = { var = ADISCORD_economy_war_tax_overload_temp value = ADISCORD_economy_war_tax_cash_temp }
		divide_temp_variable = { var = ADISCORD_economy_war_tax_overload_temp value = 4 }
		add_to_variable = { var = ADISCORD_economy_action_overload_residue value = ADISCORD_economy_war_tax_overload_temp }
		add_to_variable = { var = ADISCORD_economy_war_fatigue_score value = 5 }
		if = { limit = { ADISCORD_economy_model_is_mobilization = yes } add_to_variable = { var = ADISCORD_economy_war_fatigue_score value = 3 } }
		if = { limit = { ADISCORD_economy_model_is_decentralized_market = yes } add_to_variable = { var = ADISCORD_economy_action_overload_residue value = 5 } }
		set_variable = { var = ADISCORD_economy_recent_war_financing value = 1 }
		add_timed_idea = { idea = ADISCORD_economy_war_taxes days = 30 }
		ADISCORD_economy_mark_dirty = yes
	}
}

ADISCORD_economy_update_ai_state = {
	# Hysteresis: crisis and recovery have stricter exit thresholds than entry
	# thresholds, preventing budget modes from changing direction every tick.
	if = {
		limit = { ADISCORD_economy_ai_is_crisis = yes }
		if = {
			limit = {
				check_variable = { var = ADISCORD_economy_monthly_balance value = 0 compare = greater_than_or_equals }
				check_variable = { var = ADISCORD_economy_surplus_streak value = 2 compare = greater_than_or_equals }
				check_variable = { var = ADISCORD_economy_debt_income_ratio value = 70 compare = less_than }
				check_variable = { var = ADISCORD_economy_fiscal_stress value = 45 compare = less_than }
			}
			set_variable = { var = ADISCORD_economy_ai_state value = 3 }
		}
	}
	else_if = {
		limit = { ADISCORD_economy_ai_is_recovery = yes }
		if = {
			limit = {
				OR = {
					check_variable = { var = ADISCORD_economy_debt_income_ratio value = 110 compare = greater_than_or_equals }
					check_variable = { var = ADISCORD_economy_fiscal_stress value = 75 compare = greater_than_or_equals }
				}
			}
			set_variable = { var = ADISCORD_economy_ai_state value = 2 }
		}
		else_if = {
			limit = {
				OR = {
					check_variable = { var = ADISCORD_economy_monthly_balance value = 0 compare = less_than }
					check_variable = { var = ADISCORD_economy_deficit_streak value = 2 compare = greater_than_or_equals }
				}
			}
			set_variable = { var = ADISCORD_economy_ai_state value = 1 }
		}
		else_if = {
			limit = {
				check_variable = { var = ADISCORD_economy_surplus_streak value = 6 compare = greater_than_or_equals }
				check_variable = { var = ADISCORD_economy_debt_income_ratio value = 35 compare = less_than }
				check_variable = { var = ADISCORD_economy_fiscal_stress value = 20 compare = less_than }
				check_variable = { var = ADISCORD_economy_treasury value = 75 compare = greater_than_or_equals }
			}
			set_variable = { var = ADISCORD_economy_ai_state value = 0 }
		}
	}
	else_if = {
		limit = { ADISCORD_economy_ai_is_stressed = yes }
		if = {
			limit = {
				OR = {
					check_variable = { var = ADISCORD_economy_debt_income_ratio value = 100 compare = greater_than_or_equals }
					check_variable = { var = ADISCORD_economy_fiscal_stress value = 70 compare = greater_than_or_equals }
					check_variable = { var = ADISCORD_economy_deficit_streak value = 6 compare = greater_than_or_equals }
					AND = {
						check_variable = { var = ADISCORD_economy_treasury value = 10 compare = less_than }
						check_variable = { var = ADISCORD_economy_monthly_balance value = 0 compare = less_than }
					}
				}
			}
			set_variable = { var = ADISCORD_economy_ai_state value = 2 }
		}
		else_if = {
			limit = {
				check_variable = { var = ADISCORD_economy_surplus_streak value = 3 compare = greater_than_or_equals }
				check_variable = { var = ADISCORD_economy_debt_income_ratio value = 50 compare = less_than }
				check_variable = { var = ADISCORD_economy_fiscal_stress value = 30 compare = less_than }
			}
			set_variable = { var = ADISCORD_economy_ai_state value = 0 }
		}
	}
	else = {
		if = {
			limit = {
				OR = {
					check_variable = { var = ADISCORD_economy_debt_income_ratio value = 110 compare = greater_than_or_equals }
					check_variable = { var = ADISCORD_economy_fiscal_stress value = 75 compare = greater_than_or_equals }
					check_variable = { var = ADISCORD_economy_deficit_streak value = 6 compare = greater_than_or_equals }
				}
			}
			set_variable = { var = ADISCORD_economy_ai_state value = 2 }
		}
		else_if = {
			limit = {
				OR = {
					check_variable = { var = ADISCORD_economy_deficit_streak value = 2 compare = greater_than_or_equals }
					check_variable = { var = ADISCORD_economy_debt_income_ratio value = 65 compare = greater_than_or_equals }
					check_variable = { var = ADISCORD_economy_fiscal_stress value = 40 compare = greater_than_or_equals }
					AND = {
						check_variable = { var = ADISCORD_economy_treasury value = 25 compare = less_than }
						check_variable = { var = ADISCORD_economy_monthly_balance value = 0 compare = less_than }
					}
				}
			}
			set_variable = { var = ADISCORD_economy_ai_state value = 1 }
		}
	}
}

			limit = { ADISCORD_economy_ai_is_crisis = yes }
			if = { limit = { ADISCORD_economy_can_restructure_debt = yes } ADISCORD_economy_restructure_debt = yes }
			else_if = { limit = { check_variable = { var = ADISCORD_economy_inflation value = 15 compare = greater_than_or_equals } ADISCORD_economy_can_stabilize_macro = yes } ADISCORD_economy_stabilization_package = yes }
			else_if = { limit = { has_war = yes ADISCORD_economy_can_use_war_taxes = yes } ADISCORD_economy_war_taxes_action = yes }
			else_if = { limit = { check_variable = { var = ADISCORD_economy_monthly_balance value = 0 compare = less_than } ADISCORD_economy_can_increase_tax_burden = yes } ADISCORD_economy_increase_tax_burden = yes }
			else_if = { limit = { check_variable = { var = ADISCORD_economy_monthly_balance value = 0 compare = less_than } check_variable = { var = ADISCORD_economy_social_spending_mode value = 2 compare = greater_than } ADISCORD_economy_can_decrease_social_spending = yes } ADISCORD_economy_decrease_social_spending = yes }

			else_if = { limit = { check_variable = { var = ADISCORD_economy_monthly_balance value = 0 compare = less_than } ADISCORD_economy_can_decrease_research_spending = yes } ADISCORD_economy_decrease_research_spending = yes }
			else_if = { limit = { ADISCORD_economy_can_issue_internal_bonds = yes ADISCORD_economy_has_low_treasury = yes } ADISCORD_economy_issue_internal_bonds = yes }
			else_if = { limit = { ADISCORD_economy_can_take_external_loan = yes ADISCORD_economy_has_low_treasury = yes } ADISCORD_economy_take_external_loan = yes }
		}
		else_if = {
			limit = { ADISCORD_economy_ai_is_stressed = yes }
			if = { limit = { check_variable = { var = ADISCORD_economy_inflation value = 15 compare = greater_than_or_equals } ADISCORD_economy_can_stabilize_macro = yes } ADISCORD_economy_stabilization_package = yes }

			else_if = { limit = { has_war = no check_variable = { var = ADISCORD_economy_fiscal_stress value = 55 compare = greater_than_or_equals } ADISCORD_economy_can_decrease_army_spending = yes } ADISCORD_economy_decrease_army_spending = yes }
			else_if = { limit = { check_variable = { var = ADISCORD_economy_monthly_balance value = 0 compare = less_than } ADISCORD_economy_can_decrease_research_spending = yes } ADISCORD_economy_decrease_research_spending = yes }

			else_if = { limit = { ADISCORD_economy_can_early_repay_debt = yes } ADISCORD_economy_early_repay_debt = yes }
			else_if = { limit = { check_variable = { var = ADISCORD_economy_tax_burden_mode value = 3 compare = greater_than } ADISCORD_economy_can_decrease_tax_burden = yes } ADISCORD_economy_decrease_tax_burden = yes }

ADISCORD_economy_update_model_and_cycle = {
	# V7: explicit economic-system laws are authoritative when present.
	# Fresh initialization already assigns model 2; a current-schema country
	# without a law keeps its existing model instead of entering a save backfill.
	# The category is a law list, so exactly one branch can be active. A single
	# if/else_if chain avoids querying the same nine idea IDs a second time.
	if = { limit = { ADISCORD_economy_has_idea_economic_system_agrarian = yes } set_variable = { var = ADISCORD_economy_model value = 0 } }
	else_if = { limit = { ADISCORD_economy_has_idea_economic_system_industrializing = yes } set_variable = { var = ADISCORD_economy_model value = 2 } }
	else_if = { limit = { ADISCORD_economy_has_idea_economic_system_free_market = yes } set_variable = { var = ADISCORD_economy_model value = 1 } }
	else_if = { limit = { ADISCORD_economy_has_idea_economic_system_mixed = yes } set_variable = { var = ADISCORD_economy_model value = 2 } }
	else_if = { limit = { ADISCORD_economy_has_idea_economic_system_state_coordinated = yes } set_variable = { var = ADISCORD_economy_model value = 3 } }
	else_if = { limit = { ADISCORD_economy_has_idea_economic_system_planned_bureaucratic = yes } set_variable = { var = ADISCORD_economy_model value = 4 } }
	else_if = { limit = { ADISCORD_economy_has_idea_economic_system_mobilization = yes } set_variable = { var = ADISCORD_economy_model value = 5 } }
	else_if = { limit = { ADISCORD_economy_has_idea_economic_system_oligarchic_clan = yes } set_variable = { var = ADISCORD_economy_model value = 6 } }
	else_if = { limit = { ADISCORD_economy_has_idea_economic_system_technocratic = yes } set_variable = { var = ADISCORD_economy_model value = 7 } }

	set_variable = { var = ADISCORD_economy_cycle_phase value = 1 }
	if = { limit = { check_variable = { var = ADISCORD_economy_monthly_balance value = 0 compare = less_than } } set_variable = { var = ADISCORD_economy_cycle_phase value = 0 } }
	if = { limit = { check_variable = { var = ADISCORD_economy_monthly_balance value = 5 compare = greater_than_or_equals } check_variable = { var = ADISCORD_economy_stretched_level value = 2 compare = less_than } } set_variable = { var = ADISCORD_economy_cycle_phase value = 2 } }
	# Four readable conditions replace model-specific and wartime pseudo-cycles.
	# Economic model and war status remain separate facts in the dashboard.
	if = {
		limit = {
			OR = {
				check_variable = { var = ADISCORD_economy_inflation value = 30 compare = greater_than_or_equals }
				check_variable = { var = ADISCORD_economy_stretched_level value = 3 compare = greater_than_or_equals }
				check_variable = { var = ADISCORD_economy_debt_crisis_level value = 3 compare = greater_than_or_equals }
			}
		}
		set_variable = { var = ADISCORD_economy_cycle_phase value = 3 }
	}
}
