# Scripted settlement for the Bezhaysk campaign.
# The campaign is against the feudal bloc as it exists when BJK falls.
# Only members that are still tied to BJK and fighting the winner are annexed.

ADISCORD_bezhaysk_settle_sts_victory = {
	if = {
		limit = { BLD = { exists = yes has_war_with = STS OR = { is_subject_of = BJK is_in_faction_with = BJK } } }
		if = { limit = { has_war_with = BLD } white_peace = BLD }
		annex_country = { target = BLD transfer_troops = no }
	}
	if = {
		limit = { BHG = { exists = yes has_war_with = STS OR = { is_subject_of = BJK is_in_faction_with = BJK } } }
		if = { limit = { has_war_with = BHG } white_peace = BHG }
		annex_country = { target = BHG transfer_troops = no }
	}
	if = {
		limit = { BGT = { exists = yes has_war_with = STS OR = { is_subject_of = BJK is_in_faction_with = BJK } } }
		if = { limit = { has_war_with = BGT } white_peace = BGT }
		annex_country = { target = BGT transfer_troops = no }
	}
	if = {
		limit = { BBV = { exists = yes has_war_with = STS OR = { is_subject_of = BJK is_in_faction_with = BJK } } }
		if = { limit = { has_war_with = BBV } white_peace = BBV }
		annex_country = { target = BBV transfer_troops = no }
	}
	if = {
		limit = { BCM = { exists = yes has_war_with = STS OR = { is_subject_of = BJK is_in_faction_with = BJK } } }
		if = { limit = { has_war_with = BCM } white_peace = BCM }
		annex_country = { target = BCM transfer_troops = no }
	}
	if = {
		limit = { BJK = { exists = yes } }
		if = { limit = { has_war_with = BJK } white_peace = BJK }
		annex_country = { target = BJK transfer_troops = no }
	}
	clr_country_flag = ADISCORD_bezhaysk_campaign_active
	ADISCORD_economy_mark_dirty = yes
}

# Joint Kefreyt-Nodrul settlement.
# Each feudal holding remains an indivisible package. The side controlling its
# capital receives the whole holding. If neither partner controls the capital,
# the side credited with BJK's capitulation receives the unresolved package.
ADISCORD_bezhaysk_settle_val_nod_joint_victory = {
	VAL = { set_country_flag = ADISCORD_bezhaysk_joint_settlement }
	NOD = { set_country_flag = ADISCORD_bezhaysk_joint_settlement }

	if = {
		limit = { BLD = { exists = yes OR = { has_war_with = VAL has_war_with = NOD } OR = { is_subject_of = BJK is_in_faction_with = BJK } } }
		if = { limit = { NOD = { has_war_with = BLD } } NOD = { white_peace = BLD } }
		if = { limit = { VAL = { has_war_with = BLD } } VAL = { white_peace = BLD } }
		if = {
			limit = { 31 = { controller = { OR = { tag = NOD is_subject_of = NOD } } } }
			NOD = { annex_country = { target = BLD transfer_troops = no } }
		}
		else_if = {
			limit = { 31 = { controller = { OR = { tag = VAL is_subject_of = VAL } } } }
			VAL = { annex_country = { target = BLD transfer_troops = no } }
		}
		else_if = {
			limit = { BJK = { has_country_flag = ADISCORD_bezhaysk_joint_default_nod } }
			NOD = { annex_country = { target = BLD transfer_troops = no } }
		}
		else = { VAL = { annex_country = { target = BLD transfer_troops = no } } }
	}

	if = {
		limit = { BHG = { exists = yes OR = { has_war_with = VAL has_war_with = NOD } OR = { is_subject_of = BJK is_in_faction_with = BJK } } }

		if = { limit = { VAL = { has_war_with = BHG } } VAL = { white_peace = BHG } }

			limit = { 5 = { controller = { OR = { tag = NOD is_subject_of = NOD } } } }

			NOD = { annex_country = { target = BGT transfer_troops = no } }
		}
		else_if = {

			VAL = { annex_country = { target = BGT transfer_troops = no } }
		}
