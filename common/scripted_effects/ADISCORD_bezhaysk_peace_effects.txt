# Scripted settlement for the Bezhaysk campaign.
# The campaign is against the feudal bloc as it exists when BJK falls.
# Each settlement snapshots its participants before changing their relations.

# COUNTRY holding Gornin; the recipient is saved by the unification effect.
ADISCORD_bezhaysk_transfer_gornin = {
	if = {
		limit = { has_character = STP_ilya_gornin }
		# A live country-leader office prevents the engine from moving its holder.
		STP_ilya_gornin = {
			if = {
				limit = { has_ideology = steland_liberationism }
				remove_country_leader_role = { ideology = steland_liberationism }
			}
		}
		set_nationality = {
			character = STP_ilya_gornin
			target_country = event_target:ADISCORD_bezhaysk_unified_administration
		}
	}
}

# COUNTRY VAL, after the settlement has assigned overlords and before clearing
# ADISCORD_bezhaysk_settlement_members. Only VAL's awarded holdings are merged.
ADISCORD_bezhaysk_unify_val_administration = {
	# VAL is the empty-recipient sentinel. Prefer the capital domain when awarded;
	# otherwise the first VAL holding becomes the centre of the joint settlement.
	save_event_target_as = ADISCORD_bezhaysk_unified_administration
	if = {
		limit = {
			is_in_array = { array = ADISCORD_bezhaysk_settlement_members value = BJK }
			BJK = {
				exists = yes
				is_subject_of = VAL
				has_country_flag = ADISCORD_bezhaysk_val_administration
			}
		}
		BJK = { save_event_target_as = ADISCORD_bezhaysk_unified_administration }
	}
	for_each_scope_loop = {
		array = ADISCORD_bezhaysk_settlement_members
		if = {
			limit = {
				exists = yes
				is_subject_of = VAL
				has_country_flag = ADISCORD_bezhaysk_val_administration
			}
			if = {
				limit = { event_target:ADISCORD_bezhaysk_unified_administration = { tag = VAL } }
				save_event_target_as = ADISCORD_bezhaysk_unified_administration
			}
			else_if = {
				limit = { NOT = { tag = event_target:ADISCORD_bezhaysk_unified_administration } }
				# Preserve the holdings' existing core status without coring foreign conquests.
				every_owned_state = {
					if = {
						limit = { is_core_of = PREV }
						add_core_of = event_target:ADISCORD_bezhaysk_unified_administration
					}
				}
				clr_country_flag = ADISCORD_bezhaysk_val_administration
				event_target:ADISCORD_bezhaysk_unified_administration = {
					annex_country = {
						target = PREV
						transfer_troops = yes
					}
				}
				# Native annexation can leave impassable owned states behind.
				every_owned_state = {
					if = {
						limit = {
							OR = {
								is_controlled_by = PREV
								controller = {
									OR = {
										tag = VAL
										is_subject_of = VAL
									}
								}
							}
						}
						set_state_controller_to = event_target:ADISCORD_bezhaysk_unified_administration
					}
					event_target:ADISCORD_bezhaysk_unified_administration = { transfer_state = PREV }
				}
			}
		}
	}
	event_target:ADISCORD_bezhaysk_unified_administration = {
		if = {
			limit = { is_subject_of = VAL }
			set_cosmetic_tag = BJK_VAL_administration
			STP = { ADISCORD_bezhaysk_transfer_gornin = yes }
			STS = { ADISCORD_bezhaysk_transfer_gornin = yes }
			if = {
				limit = {
					has_character = STP_ilya_gornin
					STP_ilya_gornin = { NOT = { has_ideology = contractual_etatism } }
				}
				add_country_leader_role = {
					character = STP_ilya_gornin
					promote_leader = yes
					country_leader = {
						ideology = contractual_etatism
						expire = "2200.1.1.1"
					}
				}
			}
			set_politics = {
				ruling_party = etatism
				elections_allowed = no
			}
			if = {
				limit = { has_character = STP_ilya_gornin }
				promote_character = {
					character = STP_ilya_gornin
					ideology = contractual_etatism
				}
			}
			ADISCORD_economy_mark_dirty = yes
		}
	}
}

# Capital control is evaluated before white peace to award each whole holding.
# Unresolved holdings follow the credited victor. VAL's awards then merge;
# NOD's awards retain their separate protected administrations.
ADISCORD_bezhaysk_settle_val_nod_joint_victory = {
	# Snapshot every recipient before faction changes or white peace alter the coalition.
	clear_temp_array = ADISCORD_bezhaysk_settlement_members
	clear_temp_array = ADISCORD_bezhaysk_settlement_nod_awards
	clear_temp_array = ADISCORD_bezhaysk_settlement_val_awards
	if = {
		limit = { BLD = { exists = yes OR = { has_war_with = VAL has_war_with = NOD is_subject_of = BJK is_in_faction_with = BJK tag = BJK } } }
		add_to_temp_array = { array = ADISCORD_bezhaysk_settlement_members value = BLD }
		if = {
			limit = { 31 = { controller = { OR = { tag = NOD is_subject_of = NOD } } } }
			add_to_temp_array = { array = ADISCORD_bezhaysk_settlement_nod_awards value = BLD }
		}
		else_if = {
			limit = { 31 = { controller = { OR = { tag = VAL is_subject_of = VAL } } } }
			add_to_temp_array = { array = ADISCORD_bezhaysk_settlement_val_awards value = BLD }
		}
	}
	if = {
		limit = { BHG = { exists = yes OR = { has_war_with = VAL has_war_with = NOD is_subject_of = BJK is_in_faction_with = BJK tag = BJK } } }
		add_to_temp_array = { array = ADISCORD_bezhaysk_settlement_members value = BHG }
		if = {
			limit = { 5 = { controller = { OR = { tag = NOD is_subject_of = NOD } } } }
			add_to_temp_array = { array = ADISCORD_bezhaysk_settlement_nod_awards value = BHG }
		}
