# CIN, OSF and APH operations retained independently from the STP campaign.

VAL_clear_foreign_operation = {
	clr_country_flag = VAL_foreign_operation_active
	clr_country_flag = VAL_operation_CIN_study_market
	clr_country_flag = VAL_operation_CIN_arms_brokerage
	clr_country_flag = VAL_operation_CIN_infrastructure_concession
	clr_country_flag = VAL_operation_CIN_hire_local_captain
	clr_country_flag = VAL_operation_CIN_prepare_separate_terms

VAL_resolve_cin_operation = {
	VAL_select_cin_contract_posture = yes
	if = { limit = { check_variable = { var = VAL_CIN_last_family value = ADISCORD_VAL_family_code compare = equals } } add_to_variable = { var = VAL_CIN_repeat_streak value = 1 } }
	else = { set_variable = { var = VAL_CIN_last_family value = ADISCORD_VAL_family_code } set_variable = { var = VAL_CIN_repeat_streak value = 1 } }
	if = {
		limit = { check_variable = { var = VAL_CIN_repeat_streak value = 3 compare = greater_than_or_equals } }
		if = { limit = { check_variable = { var = ADISCORD_VAL_family_code value = 1 } } set_country_flag = { flag = VAL_CIN_block_market days = 90 value = 1 } }
		else_if = { limit = { check_variable = { var = ADISCORD_VAL_family_code value = 2 } } set_country_flag = { flag = VAL_CIN_block_arms days = 90 value = 1 } }
		else_if = { limit = { check_variable = { var = ADISCORD_VAL_family_code value = 3 } } set_country_flag = { flag = VAL_CIN_block_infrastructure days = 90 value = 1 } }
		else_if = { limit = { check_variable = { var = ADISCORD_VAL_family_code value = 4 } } set_country_flag = { flag = VAL_CIN_block_captain days = 90 value = 1 } }
		else = { set_country_flag = { flag = VAL_CIN_block_terms days = 90 value = 1 } }
	}
	if = { limit = { check_variable = { var = ADISCORD_VAL_family_code value = 1 } NOT = { has_country_flag = VAL_CIN_success_market } } set_country_flag = VAL_CIN_success_market add_to_variable = { var = VAL_CIN_influence value = 1 } }
	else_if = { limit = { check_variable = { var = ADISCORD_VAL_family_code value = 2 } NOT = { has_country_flag = VAL_CIN_success_arms } } set_country_flag = VAL_CIN_success_arms add_to_variable = { var = VAL_CIN_influence value = 1 } }
	else_if = { limit = { check_variable = { var = ADISCORD_VAL_family_code value = 3 } NOT = { has_country_flag = VAL_CIN_success_infrastructure } } set_country_flag = VAL_CIN_success_infrastructure add_to_variable = { var = VAL_CIN_influence value = 1 } }
	else_if = { limit = { check_variable = { var = ADISCORD_VAL_family_code value = 4 } NOT = { has_country_flag = VAL_CIN_success_captain } } set_country_flag = VAL_CIN_success_captain add_to_variable = { var = VAL_CIN_influence value = 1 } }
	else_if = { limit = { check_variable = { var = ADISCORD_VAL_family_code value = 5 } NOT = { has_country_flag = VAL_CIN_success_terms } } set_country_flag = VAL_CIN_success_terms add_to_variable = { var = VAL_CIN_influence value = 1 } }
	clamp_variable = { var = VAL_CIN_influence min = 0 max = 3 }
	set_country_flag = { flag = VAL_CIN_target_cooldown days = 70 value = 1 }
	VAL_recalculate_stp_campaign_readiness = yes
	VAL_clear_foreign_operation = yes
}

VAL_select_cin_contract_posture = {
	if = { limit = { check_variable = { var = VAL_CIN_contract_posture value = 0 compare = equals } } random_list = { 30 = { set_variable = { var = VAL_CIN_contract_posture value = 1 } } 25 = { modifier = { factor = 2 CIN = { has_war = yes } } set_variable = { var = VAL_CIN_contract_posture value = 2 } } 25 = { modifier = { factor = 2 CIN = { has_stability < 0.4 } } set_variable = { var = VAL_CIN_contract_posture value = 3 } } 20 = { modifier = { factor = 2 CIN = { has_equipment = { infantry_equipment < 500 } } } set_variable = { var = VAL_CIN_contract_posture value = 4 } } } }
}

	if = { limit = { has_country_flag = VAL_operation_CIN_study_market } set_temp_variable = { var = ADISCORD_VAL_family_code value = 1 } VAL_resolve_cin_operation = yes }
