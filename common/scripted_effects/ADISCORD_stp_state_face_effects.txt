# ADISCORD / STP State Face
# Standalone leader-health API retained for the user-authored STP content.
# There is no automatic calendar, death event, civil war or focus-tree outcome here.

STP_initialize_leader_health = {
	if = {
		limit = { NOT = { has_variable = STP_leader_health_stage } }
		set_variable = { var = STP_leader_health_stage value = 1 }
	}
	clamp_variable = { var = STP_leader_health_stage min = 1 max = 5 }
	STP_sync_leader_health = yes
}

STP_sync_leader_health = {
	set_variable = { var = STP_state_face_stage value = STP_leader_health_stage }

	if = {
		limit = { check_variable = { var = STP_leader_health_stage value = 1 compare = equals } }
		set_variable = { var = STP_leader_health value = 100 }
		clr_country_flag = STP_ivanov_dead
	}
	else_if = {
		limit = { check_variable = { var = STP_leader_health_stage value = 2 compare = equals } }
		set_variable = { var = STP_leader_health value = 75 }
		clr_country_flag = STP_ivanov_dead
	}
	else_if = {
		limit = { check_variable = { var = STP_leader_health_stage value = 3 compare = equals } }
		set_variable = { var = STP_leader_health value = 45 }
		clr_country_flag = STP_ivanov_dead
	}
	else_if = {
		limit = { check_variable = { var = STP_leader_health_stage value = 4 compare = equals } }
		set_variable = { var = STP_leader_health value = 15 }
		clr_country_flag = STP_ivanov_dead
	}
	else = {
		set_variable = { var = STP_leader_health value = 0 }
		set_country_flag = STP_ivanov_dead
	}

	STP_refresh_leader_health_modifier = yes
}

STP_refresh_leader_health_modifier = {
	set_variable = { var = STP_fading_pp_factor value = 0 }
	set_variable = { var = STP_fading_stability_factor value = 0 }
	set_variable = { var = STP_fading_command_power_factor value = 0 }
	set_variable = { var = STP_fading_planning_factor value = 0 }
	set_variable = { var = STP_fading_org_factor value = 0 }

	if = {
		limit = { check_variable = { var = STP_leader_health_stage value = 2 compare = equals } }
		set_variable = { var = STP_fading_pp_factor value = -0.05 }
		set_variable = { var = STP_fading_stability_factor value = -0.02 }
		set_variable = { var = STP_fading_command_power_factor value = -0.05 }
	}
	else_if = {
		limit = { check_variable = { var = STP_leader_health_stage value = 3 compare = equals } }
		set_variable = { var = STP_fading_pp_factor value = -0.10 }
		set_variable = { var = STP_fading_stability_factor value = -0.05 }
		set_variable = { var = STP_fading_command_power_factor value = -0.10 }
		set_variable = { var = STP_fading_planning_factor value = -0.05 }
	}
	else_if = {
		limit = { check_variable = { var = STP_leader_health_stage value = 4 compare = equals } }
		set_variable = { var = STP_fading_pp_factor value = -0.20 }
		set_variable = { var = STP_fading_stability_factor value = -0.10 }
		set_variable = { var = STP_fading_command_power_factor value = -0.20 }
		set_variable = { var = STP_fading_planning_factor value = -0.10 }
		set_variable = { var = STP_fading_org_factor value = -0.05 }
	}

	if = {
		limit = {
			NOT = { has_country_flag = STP_ivanov_dead }
			NOT = { has_dynamic_modifier = { modifier = STP_fading_father } }
		}
		add_dynamic_modifier = { modifier = STP_fading_father }
	}
	if = {
		limit = { has_country_flag = STP_ivanov_dead }
		remove_dynamic_modifier = { modifier = STP_fading_father }
	}
}

# Call with ADISCORD_STP_VAL_effect_value = 1..5.
STP_set_health_stage = {
	set_temp_variable = { var = STP_requested_health_stage value = ADISCORD_STP_VAL_effect_value }
	if = {
		limit = {
			check_variable = { var = STP_requested_health_stage value = 1 compare = greater_than_or_equals }
			check_variable = { var = STP_requested_health_stage value = 5 compare = less_than_or_equals }
		}
		set_variable = { var = STP_leader_health_stage value = STP_requested_health_stage }
		STP_sync_leader_health = yes
	}
}
