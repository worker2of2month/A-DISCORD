# Southern final war. Side numbers: 1 = Shahrabad, 2 = the northern champion.
# Settlement terms: 1 = annexation, 2 = subjects. Subject types follow the
# winner's path: 0 ordinary puppet, 1 house concession for the defeated leader,
# 2 compact charter, 3 namestnik office, 4 oil charter.

# ----- Crisis clock -----

# GLOBAL. Open once at the actual peace/return settlement, or at the first
# eligible furnace cycle. Repeated calls preserve the original deadline.
ADISCORD_south_begin_preparation = {
	if = {
		limit = {
			ADISCORD_south_crisis_open = yes
			NOT = { has_global_flag = ADISCORD_south_preparation_started }
		}
		set_global_flag = ADISCORD_south_preparation_started
		ADISCORD_south_wake_regional_countries = yes
	}
}

# COUNTRY. Simultaneous missions display the dated global warning.
ADISCORD_south_show_preparation = {
	if = {
		limit = { exists = yes }
		activate_mission = ADISCORD_south_preparation
	}
}

# GLOBAL. A new return campaign invalidates the earlier peace's preparation.
# Ordinary load, focus completion and periodic checks never call this reset.
ADISCORD_south_suspend_preparation = {
	clr_global_flag = ADISCORD_south_preparation_started
	clr_global_flag = ADISCORD_south_crisis_warned
	NAM = { ADISCORD_south_clear_preparation = yes }
	SHL = { ADISCORD_south_clear_preparation = yes }
	EFL = { ADISCORD_south_clear_preparation = yes }
	AZH = { ADISCORD_south_clear_preparation = yes }
}

ADISCORD_south_clear_preparation = {
	remove_mission = ADISCORD_south_preparation
	clr_country_flag = ADISCORD_south_offensive_order
}

# COUNTRY. The order belongs to this warning, not to historical focus completion.
ADISCORD_south_prepare_offensive = {
	if = {
		limit = {
			ADISCORD_south_final_focus_available = yes
			ADISCORD_south_declarer_valid = yes
		}
		set_country_flag = ADISCORD_south_offensive_order
		ADISCORD_south_final_war_start = yes
	}
}

# COUNTRY SHL. Called once per 28-day furnace cycle, the only periodic driver.
ADISCORD_south_crisis_cycle = {
	ADISCORD_south_begin_preparation = yes
	if = {
		limit = {
			ADISCORD_south_crisis_open = yes
			ADISCORD_south_preparation_elapsed = yes
			NOT = { ADISCORD_south_north_candidate_exists = yes }
			SHL = { ADISCORD_south_champion_viable = yes }
		}
		set_global_flag = ADISCORD_south_final_resolved
		ADISCORD_south_suspend_preparation = yes
		set_global_flag = ADISCORD_south_hegemon_SHL
		SHL = { SHL_announce_crown_of_the_south = yes }
		set_global_flag = ADISCORD_south_unopposed
		set_variable = { var = global.ADISCORD_south_winner_side value = 1 }
		SHL = { mark_focus_tree_layout_dirty = yes }
		news_event = { id = ADISCORD_south.31 hours = 1 }
	}
	else_if = {
		limit = { ADISCORD_south_crisis_open = yes }
		if = {
			limit = {
				NOT = { has_global_flag = ADISCORD_south_crisis_warned }
				ADISCORD_south_preparation_can_finish = yes
			}
			set_global_flag = ADISCORD_south_crisis_warned
			NAM = { ADISCORD_south_show_preparation = yes }
			SHL = { ADISCORD_south_show_preparation = yes }
			EFL = { ADISCORD_south_show_preparation = yes }
			AZH = { ADISCORD_south_show_preparation = yes }
			news_event = { id = ADISCORD_south.3 hours = 1 }
		}
		if = {
			limit = {
				ADISCORD_south_final_can_start = yes
			}
			ADISCORD_south_dispatch_prepared_war = yes
		}
	}
}

# COUNTRY SHL. Orders take effect after the warning; an unordered automatic
# attack waits one additional furnace cycle.
ADISCORD_south_dispatch_prepared_war = {
	if = {
		limit = { ADISCORD_south_final_can_start = yes }
		if = {
			limit = {
				NAM = {
					ADISCORD_south_declarer_valid = yes
					ADISCORD_south_final_can_start = yes
					has_country_flag = ADISCORD_south_offensive_order
				}
			}
			NAM = { ADISCORD_south_final_war_start = yes }
		}
		else_if = {
			limit = { SHL = { has_country_flag = ADISCORD_south_offensive_order } }
			SHL = { ADISCORD_south_final_war_start = yes }
		}
		else_if = {
			limit = {
				has_global_flag = {
					flag = ADISCORD_south_crisis_warned
					days > 83
				}
			}
			SHL = { ADISCORD_south_final_war_start = yes }
		}
	}
}

# ----- Campaign start -----

# GLOBAL. Svetlogorye leads while sovereign; otherwise the Restitution Alliance.
ADISCORD_south_select_north_leader = {
	NAM = { clr_country_flag = ADISCORD_south_north_leader }
	EFL = { clr_country_flag = ADISCORD_south_north_leader }
	AZH = { clr_country_flag = ADISCORD_south_north_leader }
	if = {
		limit = { NAM = { ADISCORD_south_champion_viable = yes } }
		NAM = { set_country_flag = ADISCORD_south_north_leader }
	}
	else_if = {
		limit = { EFL = { ADISCORD_south_champion_viable = yes } }
		EFL = { set_country_flag = ADISCORD_south_north_leader }
	}
	else_if = {
		limit = { AZH = { ADISCORD_south_champion_viable = yes } }
		AZH = { set_country_flag = ADISCORD_south_north_leader }
	}
}

# COUNTRY declarer: SHL, or the country selected as the northern leader. The
# crisis clock calls this in SHL scope.
ADISCORD_south_final_war_start = {
	if = {
		limit = {
			ADISCORD_south_final_can_start = yes
			ADISCORD_south_declarer_valid = yes
		}
		set_global_flag = ADISCORD_south_final_started
		ADISCORD_south_suspend_preparation = yes
		ADISCORD_south_select_north_leader = yes
		if = {
			limit = { tag = SHL }
			set_variable = { var = global.ADISCORD_south_declarer_side value = 1 }
		}
		else = {
			set_variable = { var = global.ADISCORD_south_declarer_side value = 2 }
		}
		ADISCORD_south_wake_regional_countries = yes
		NAM = { mark_focus_tree_layout_dirty = yes }

		SHL = {
			ADISCORD_south_prepare_champion = yes
			set_variable = { var = ADISCORD_south_side value = 1 }
			SHL_use_campaign_reserve = yes
			# Reveals the wartime Front of the South branch.
			mark_focus_tree_layout_dirty = yes
		}
		NAM = {
			if = {
				limit = { has_country_flag = ADISCORD_south_north_leader }
				ADISCORD_south_prepare_champion = yes
				set_variable = { var = ADISCORD_south_side value = 2 }
			}
		}
		EFL = {
			if = {
				limit = { has_country_flag = ADISCORD_south_north_leader }
				ADISCORD_south_prepare_champion = yes
				set_variable = { var = ADISCORD_south_side value = 2 }
			}
		}
		AZH = {
			if = {
				limit = { has_country_flag = ADISCORD_south_north_leader }
				ADISCORD_south_prepare_champion = yes
				set_variable = { var = ADISCORD_south_side value = 2 }
			}
		}

		if = {
			limit = { tag = SHL }
			ADISCORD_south_shl_declare_on_north = yes
		}
		else = {
			declare_war_on = { target = SHL type = annex_everything }
		}

		ADISCORD_south_assign_bound_sides = yes
		ADISCORD_south_record_key_states = yes
		ADISCORD_south_offer_side_choices = yes
		ADISCORD_south_raise_oil_uprising = yes

		SHL = {
			ADISCORD_south_add_path_spirit = yes
			# The deadline compares the age of the campaign flag, so a relaunch
			# cannot be settled by the previous campaign's callback.
			country_event = { id = ADISCORD_south.20 days = 900 }
			country_event = { id = ADISCORD_south.1 hours = 1 }
			news_event = { id = ADISCORD_south.2 hours = 2 }
		}
		NAM = {
			if = {
				limit = { has_country_flag = ADISCORD_south_north_leader }
				ADISCORD_south_add_path_spirit = yes
				country_event = { id = ADISCORD_south.1 hours = 1 }
			}
		}
		EFL = {
			if = {
				limit = { has_country_flag = ADISCORD_south_north_leader }
				ADISCORD_south_add_path_spirit = yes
				country_event = { id = ADISCORD_south.1 hours = 1 }
			}
		}
		AZH = {
			if = {
				limit = { has_country_flag = ADISCORD_south_north_leader }
				ADISCORD_south_add_path_spirit = yes
				country_event = { id = ADISCORD_south.1 hours = 1 }
			}
		}
	}
}

# COUNTRY champion. Leads its own faction and holds mandatory-major status only
# for this campaign, so the native peace waits for the champion's defeat.
ADISCORD_south_prepare_champion = {
	ADISCORD_release_non_participating_minor_optimization = yes
	if = {
		limit = {
			is_in_faction = yes
			is_faction_leader = no
		}
		leave_faction = yes
	}
	if = {
		limit = { is_in_faction = no }
		if = {
			limit = { tag = SHL }
			create_faction_from_template = {
				template = faction_template_ADISCORD_standard
				name = ADISCORD_south_shl_faction
			}
		}
		else = {
			create_faction_from_template = {
				template = faction_template_ADISCORD_standard
				name = ADISCORD_south_north_faction
			}
		}
	}
	if = {
		limit = {
			OR = {
				is_major = no
				has_country_flag = RUS_crisis_added_major
				RUS_crisis_other_major_required = yes
			}
		}
		set_country_flag = ADISCORD_south_added_major
		set_major = yes
	}
}

# COUNTRY. Joins Shahrabad's faction when free and enters the current war.

	ADISCORD_release_non_participating_minor_optimization = yes

	if = {
		limit = {
			is_in_faction = no
			is_subject = no
		}
		SHL = { add_to_faction = PREV }
	}
	if = {
		limit = {
			NAM = { has_country_flag = ADISCORD_south_north_leader }
			NOT = { has_war_with = NAM }
		}

			targeted_alliance = SHL

# ----- Path spirits -----

# COUNTRY champion. The path chooses the war, not only its name.
ADISCORD_south_add_path_spirit = {
	if = {
		limit = { tag = SHL }
		if = {
			limit = { check_variable = { var = SHL_political_course value = 1 compare = equals } }
			add_ideas = ADISCORD_south_shl_house_contracts
		}
		else_if = {
			limit = { check_variable = { var = SHL_political_course value = 2 compare = equals } }
			add_ideas = ADISCORD_south_shl_city_levies
		}
		else_if = {
			limit = { check_variable = { var = SHL_political_course value = 3 compare = equals } }
			add_ideas = ADISCORD_south_shl_last_pour
		}
		else_if = {
			limit = { check_variable = { var = SHL_political_course value = 4 compare = equals } }
			add_ideas = ADISCORD_south_shl_promised_fiefs
		}
		else_if = {
			limit = { check_variable = { var = SHL_political_course value = 5 compare = equals } }
			add_ideas = ADISCORD_south_shl_oil_uprising
		}
	}
	else_if = {
		limit = { tag = NAM }
		if = {
			limit = { has_completed_focus = NAM_service_state }
			add_ideas = ADISCORD_south_nam_emergency_writs
		}
		else_if = {
			limit = { has_completed_focus = NAM_federal_charter }
			add_ideas = ADISCORD_south_nam_charter_levies
		}
	}
	else = {
		add_ideas = ADISCORD_south_restitution_line
	}
	ADISCORD_economy_mark_dirty = yes
}

# COUNTRY. Every terminal path removes the campaign spirits.
ADISCORD_south_remove_path_spirit = {
	remove_ideas = ADISCORD_south_shl_house_contracts
	remove_ideas = ADISCORD_south_shl_city_levies
	remove_ideas = ADISCORD_south_shl_last_pour
	remove_ideas = ADISCORD_south_shl_promised_fiefs
	remove_ideas = ADISCORD_south_shl_oil_uprising
	remove_ideas = ADISCORD_south_nam_emergency_writs
	remove_ideas = ADISCORD_south_nam_charter_levies
	remove_ideas = ADISCORD_south_restitution_line
	ADISCORD_economy_mark_dirty = yes
}

# ----- Settlement -----

# GLOBAL. Shahrabad wins the south.
ADISCORD_south_final_resolve_shl_victory = {
	if = {
		limit = { ADISCORD_south_final_active = yes }
		# The resolved flag precedes white peace so on_peace cannot re-enter.
		set_global_flag = ADISCORD_south_final_resolved
		ADISCORD_south_suspend_preparation = yes
		set_global_flag = ADISCORD_south_hegemon_SHL
		SHL = { SHL_announce_crown_of_the_south = yes }
		set_variable = { var = global.ADISCORD_south_winner_side value = 1 }
		set_variable = { var = global.ADISCORD_south_terms value = 1 }
		SHL = { mark_focus_tree_layout_dirty = yes }
		set_variable = { var = global.ADISCORD_south_subject_type value = 0 }
		if = {
			limit = { SHL = { check_variable = { var = SHL_political_course value = 1 compare = equals } } }
			set_variable = { var = global.ADISCORD_south_terms value = 2 }
			set_variable = { var = global.ADISCORD_south_subject_type value = 1 }
		}
		else_if = {
			limit = { SHL = { check_variable = { var = SHL_political_course value = 2 compare = equals } } }
			set_variable = { var = global.ADISCORD_south_terms value = 2 }
			set_variable = { var = global.ADISCORD_south_subject_type value = 2 }
		}
		ADISCORD_south_end_final_wars = yes
		ADISCORD_south_mark_losers = yes
		ADISCORD_south_close_campaign_state = yes
		SHL = {
			country_event = { id = ADISCORD_south.30 days = 1 }
			news_event = { id = ADISCORD_south.31 hours = 2 }
		}
	}
}

# GLOBAL. The northern side wins. Only Svetlogorye can claim the south; the

# COUNTRY target (event ROOT); FROM is the champion. Both sides are rechecked
# because the answer may come long after the demand.
ADISCORD_south_accept_offer = {
	if = {
		limit = {
			ADISCORD_south_nam_story_clear = yes
			ADISCORD_south_offer_answer_valid = yes
		}
		if = {
			limit = { check_variable = { var = ADISCORD_south_offer_kind value = 3 compare = equals } }
			if = {
				limit = {
					OR = {
						is_in_faction = no
						ADISCORD_south_alliance_switch_possible = yes
					}
					FROM = {
						OR = {
							is_in_faction = no
							is_faction_leader = yes
						}
					}
				}
				# leave_faction applies at once, so the follower joins the
				# champion in the same tick, as in the resource-war alliance setup.
				if = {
					limit = { is_in_faction = yes }
					leave_faction = yes
				}
				FROM = {
					ADISCORD_south_ensure_campaign_faction = yes
					add_to_faction = PREV
				}
			}
		}
		else_if = {
			limit = { check_variable = { var = ADISCORD_south_offer_kind value = 2 compare = equals } }
			clear_variable = ADISCORD_south_offer_kind
			FROM = { annex_country = { target = PREV transfer_troops = yes } }
		}
		else = {
			# Subjugation waits one day after the faction exit for the native faction cache.
			if = {
				limit = {
					is_in_faction = yes
					is_faction_leader = yes
				}
				dismantle_faction = yes
			}
			else_if = {
				limit = { is_in_faction = yes }
				leave_faction = yes
			}
			if = {
				limit = { FROM = { tag = SHL } }
				set_country_flag = ADISCORD_south_submits_to_SHL
			}
			else = {
				set_country_flag = ADISCORD_south_submits_to_NAM
			}
			country_event = { id = ADISCORD_south.41 days = 1 }
		}
	}
	clear_variable = ADISCORD_south_offer_kind
}

# GLOBAL. Called at the final war start. A signed Svetlogorsk district compact
# leaves the underground without support; half of the delivered rifles return.
ADISCORD_south_raise_oil_uprising = {
	if = {
		limit = {
			SHL = { has_country_flag = ADISCORD_south_underground_armed }
			NAM = {
				has_country_flag = ADISCORD_south_north_leader
				owns_state = 688
				controls_state = 688
			}
			NOT = { country_exists = SLF }
		}
		SHL = { clr_country_flag = ADISCORD_south_underground_armed }
		# The Weber contact's early rifles follow the main delivery on both paths.
		if = {
			limit = { NAM = { has_country_flag = NAM_district_compact } }
			SHL = {
				add_equipment_to_stockpile = { type = infantry_equipment amount = 13500 producer = SHL }
				if = {
					limit = { has_country_flag = SHL_weber_contact }
					add_equipment_to_stockpile = { type = infantry_equipment amount = 1500 producer = SHL }
				}
				country_event = { id = ADISCORD_south.44 hours = 2 }
			}
		}
		else = {
			SLF = { transfer_state = 688 }
			688 = {
				add_core_of = SLF
				set_state_controller_to = SLF
			}
			SLF = {
				set_capital = { state = 688 }
				add_ideas = ADISCORD_nam_insurgent_columns
				add_war_support = 0.20
				load_oob = "SLF_nam_resource_war"
				add_equipment_to_stockpile = { type = infantry_equipment amount = 27000 producer = SHL }
				if = {
					limit = { SHL = { has_country_flag = SHL_weber_contact } }
					add_equipment_to_stockpile = { type = infantry_equipment amount = 1500 producer = SHL }
				}
				add_equipment_to_stockpile = { type = support_equipment amount = 120 producer = SLF }
				set_variable = { var = ADISCORD_south_side value = 1 }
				add_to_war = {
					targeted_alliance = SHL
					enemy = NAM
					hostility_reason = asked_to_join
				}
				news_event = { id = ADISCORD_south.45 hours = 2 }
			}
		}
		SHL = { clr_country_flag = SHL_weber_contact }
	}
	else = {
		SHL = { clr_country_flag = SHL_weber_contact }
	}
}
