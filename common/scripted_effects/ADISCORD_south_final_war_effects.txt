# Southern final war. Side numbers: 1 = Shahrabad, 2 = the northern champion.
# Settlement terms: 1 = annexation, 2 = subjects, 3 = furnace administrations.
# Subject types follow the
# winner's path: 0 ordinary puppet, 1 house concession for the defeated leader,
# 2 compact charter, 3 namestnik office, 4 oil charter, 5 economic contract.

# ----- Crisis deadline -----

# GLOBAL. First land contact after the resource settlement starts the clock.
# A later loss of the border leaves its date intact.
ADISCORD_south_begin_preparation = {
	if = {
		limit = {
			NOT = { has_global_flag = ADISCORD_south_preparation_started }
			ADISCORD_south_crisis_open = yes
		}
		if = {
			limit = { ADISCORD_south_north_candidate_exists = yes }
			if = {
				limit = { has_global_flag = ADISCORD_south_unopposed_since }
				clr_global_flag = ADISCORD_south_unopposed_since
			}
			if = {
				limit = { ADISCORD_south_sides_share_border = yes }
				set_global_flag = ADISCORD_south_preparation_started
				ADISCORD_south_wake_regional_countries = yes
				hidden_effect = {
					ADISCORD_south_watch_deadline = yes
					SHL = { ADISCORD_south_announce_deadline = yes }
					NAM = { ADISCORD_south_announce_deadline = yes }
					EFL = { ADISCORD_south_announce_deadline = yes }
					AZH = { ADISCORD_south_announce_deadline = yes }
				}
			}
		}
		else_if = {
			limit = { NOT = { has_global_flag = ADISCORD_south_unopposed_since } }
			set_global_flag = ADISCORD_south_unopposed_since
		}
	}
}

# COUNTRY. Shahrabad and the current northern candidate learn the deadline date.
ADISCORD_south_announce_deadline = {
	if = {
		limit = {
			ADISCORD_south_champion_viable = yes
			ADISCORD_south_declarer_valid = yes
		}
		country_event = { id = ADISCORD_south.6 hours = 6 }
	}
}

# Shahrabad's victories bring the deadline date closer in 15-day units, at
# most six per campaign, counted from the start of the game.
ADISCORD_south_pull_clock_campaign = {
	ADISCORD_south_pull_clock_unit = yes
	ADISCORD_south_pull_clock_unit = yes
	ADISCORD_south_pull_clock_unit = yes
}

ADISCORD_south_pull_clock_war = {
	ADISCORD_south_pull_clock_unit = yes
	ADISCORD_south_pull_clock_unit = yes
}

# GLOBAL. Each unit is checked against the date it would move, so a deadline
# closer than 136 days stays where it is. The counter is never cleared: a
# repelled campaign keeps the date it already reached.
ADISCORD_south_pull_clock_unit = {
	if = {
		limit = {
			NOT = { has_global_flag = ADISCORD_south_final_started }
			NOT = { has_global_flag = ADISCORD_south_final_resolved }
			NOT = { check_variable = { var = global.ADISCORD_south_crisis_pull value = 6 compare = greater_than_or_equals } }
			ADISCORD_south_deadline_can_pull = yes
		}
		add_to_variable = { var = global.ADISCORD_south_crisis_pull value = 1 }
	}
}

# GLOBAL, from hidden contexts. Starts the deadline watch unless one is
# already pending. Shahrabad's furnace cycle runs every 28 days and skips the
# crisis while Shahrabad is capitulated, so it cannot hit a calendar date.
# The watch is one hidden SHL event that reschedules itself: every 28 days
# while the deadline is further away, daily in its last 31 days. A step reads
# the date, three flags and one global variable, iterates no countries or
# states, and the chain ends when the deadline fires or the preparation
# closes. Each step refreshes the watch flag; a flag older than the 28-day
# step marks a chain lost with its country, which a later call may replace.
# Only the chain clears the flag, so a pending callback is never duplicated.
ADISCORD_south_watch_deadline = {
	if = {
		limit = {
			ADISCORD_south_crisis_open = yes
			has_global_flag = ADISCORD_south_preparation_started
			NOT = { has_global_flag = ADISCORD_south_crisis_deadline_passed }
			OR = {
				NOT = { has_global_flag = ADISCORD_south_deadline_watch }
				has_global_flag = {
					flag = ADISCORD_south_deadline_watch
					days > 29
				}
			}
			SHL = { exists = yes }
		}
		clr_global_flag = ADISCORD_south_deadline_watch
		set_global_flag = ADISCORD_south_deadline_watch
		SHL = { country_event = { id = ADISCORD_south.5 days = 1 } }
	}
}

# COUNTRY SHL, from the watch event. One step of the bounded chain.
ADISCORD_south_deadline_watch_step = {
	clr_global_flag = ADISCORD_south_deadline_watch
	if = {
		limit = { ADISCORD_south_deadline_due = yes }
		ADISCORD_south_check_deadline = yes
	}
	else_if = {
		limit = {
			ADISCORD_south_crisis_open = yes
			has_global_flag = ADISCORD_south_preparation_started
			NOT = { has_global_flag = ADISCORD_south_crisis_deadline_passed }
		}
		set_global_flag = ADISCORD_south_deadline_watch
		if = {
			limit = { ADISCORD_south_deadline_watch_daily = yes }
			country_event = { id = ADISCORD_south.5 days = 1 }
		}
		else = {
			country_event = { id = ADISCORD_south.5 days = 28 }
		}
	}
}

# GLOBAL, from hidden events only. Marks the passed deadline once; the war is
# dispatched an hour later by a hidden SHL event. The furnace cycle calls this
# too, so a deadline met on its tick starts the war in the same cycle.
ADISCORD_south_check_deadline = {
	if = {
		limit = { ADISCORD_south_deadline_due = yes }
		set_global_flag = ADISCORD_south_crisis_deadline_passed
		SHL = { country_event = { id = ADISCORD_south.4 hours = 1 } }
	}
}

# GLOBAL. A new return campaign invalidates the earlier peace's preparation.

# COUNTRY SHL. Called once per 28-day furnace cycle, the periodic driver of
# the crisis; the deadline watch only marks the calendar date.
ADISCORD_south_crisis_cycle = {
	ADISCORD_south_begin_preparation = yes
	ADISCORD_south_check_deadline = yes
	ADISCORD_south_watch_deadline = yes
	if = {
		limit = {
			ADISCORD_south_crisis_open = yes
			NOT = { ADISCORD_south_north_candidate_exists = yes }
			OR = {
				ADISCORD_south_preparation_elapsed = yes
				has_global_flag = {
					flag = ADISCORD_south_unopposed_since
					days > 179
				}
			}
			SHL = { ADISCORD_south_champion_viable = yes }
		}
		set_global_flag = ADISCORD_south_final_resolved
		ADISCORD_south_suspend_preparation = yes
		set_global_flag = ADISCORD_south_hegemon_SHL
		SHL = { SHL_announce_crown_of_the_south = yes }
		set_global_flag = ADISCORD_south_unopposed
		set_variable = { var = global.ADISCORD_south_winner_side value = 1 }
		SHL = { mark_focus_tree_layout_dirty = yes }
		news_event = { id = ADISCORD_south.31 hours = 1 }
	}
	else_if = {
		limit = { ADISCORD_south_crisis_open = yes }
		if = {
			limit = {
				NOT = { has_global_flag = ADISCORD_south_crisis_warned }
				NOT = { has_global_flag = ADISCORD_south_crisis_deadline_passed }
				ADISCORD_south_preparation_can_finish = yes
			}
			set_global_flag = ADISCORD_south_crisis_warned
			NAM = { ADISCORD_south_show_preparation = yes }
			SHL = { ADISCORD_south_show_preparation = yes }
			EFL = { ADISCORD_south_show_preparation = yes }
			AZH = { ADISCORD_south_show_preparation = yes }
			news_event = { id = ADISCORD_south.3 hours = 1 }
		}
		if = {
			limit = {
				ADISCORD_south_final_can_start = yes
			}
			ADISCORD_south_dispatch_prepared_war = yes
		}
	}
	# A war declared by this same cycle starts paying from the next one.
	if = {
		limit = {
			ADISCORD_south_final_active = yes
			has_global_flag = {
				flag = ADISCORD_south_final_started
				days > 1
			}
		}
		ADISCORD_south_strain_cycle = yes
	}
}

# COUNTRY SHL. Orders take effect after the warning; an unordered automatic
# attack waits one additional furnace cycle, or starts at once after the
# crisis deadline.
ADISCORD_south_dispatch_prepared_war = {
	if = {
		limit = { ADISCORD_south_final_can_start = yes }
		if = {
			limit = {
				NAM = {
					ADISCORD_south_declarer_valid = yes
					ADISCORD_south_final_can_start = yes
					has_country_flag = ADISCORD_south_offensive_order
				}
			}
			NAM = { ADISCORD_south_final_war_start = yes }
		}
		else_if = {
			limit = { SHL = { has_country_flag = ADISCORD_south_offensive_order } }
			SHL = { ADISCORD_south_final_war_start = yes }
		}
		else_if = {
			limit = {
				OR = {
					has_global_flag = {
						flag = ADISCORD_south_crisis_warned
						days > 83
					}
					has_global_flag = ADISCORD_south_crisis_deadline_passed
				}
			}
			SHL = { ADISCORD_south_final_war_start = yes }
		}
	}
}

	}

# COUNTRY declarer: SHL, or the country selected as the northern leader. The
# deadline dispatch calls this in SHL scope.
ADISCORD_south_final_war_start = {
	if = {
		limit = {
			ADISCORD_south_final_can_start = yes
			ADISCORD_south_declarer_valid = yes
		}
		ADISCORD_south_begin_hostilities = yes
	}
}

# COUNTRY declarer. Shared setup keeps debug wars on the same settlement path.
ADISCORD_south_begin_hostilities = {
	if = {
		limit = {
			ADISCORD_south_crisis_open = yes
			ADISCORD_south_participants_ready = yes
			ADISCORD_south_declarer_valid = yes
		}
		set_global_flag = ADISCORD_south_final_started
		# Read before suspension clears the deadline marker.
		if = {
			limit = { has_global_flag = ADISCORD_south_crisis_deadline_passed }
			ADISCORD_south_mark_unprepared = yes
		}
		ADISCORD_south_suspend_preparation = yes
		ADISCORD_south_select_north_leader = yes
		if = {
			limit = { tag = SHL }
			set_variable = { var = global.ADISCORD_south_declarer_side value = 1 }
		}
		else = {
			set_variable = { var = global.ADISCORD_south_declarer_side value = 2 }
		}
		ADISCORD_south_wake_regional_countries = yes
		NAM = { mark_focus_tree_layout_dirty = yes }

		SHL = {
			ADISCORD_south_prepare_champion = yes
			set_variable = { var = ADISCORD_south_side value = 1 }
			SHL_use_campaign_reserve = yes
			ADISCORD_south_load_war_tree = yes
		}
		NAM = {
			if = {
				limit = { has_country_flag = ADISCORD_south_north_leader }
				ADISCORD_south_prepare_champion = yes
				set_variable = { var = ADISCORD_south_side value = 2 }
				ADISCORD_south_load_war_tree = yes
			}
		}
		EFL = {
			if = {
				limit = { has_country_flag = ADISCORD_south_north_leader }
				ADISCORD_south_prepare_champion = yes
				set_variable = { var = ADISCORD_south_side value = 2 }
			}
		}
		AZH = {
			if = {
				limit = { has_country_flag = ADISCORD_south_north_leader }
				ADISCORD_south_prepare_champion = yes
				set_variable = { var = ADISCORD_south_side value = 2 }
			}
		}

		if = {
			limit = { tag = SHL }
			ADISCORD_south_shl_declare_on_north = yes
		}
		else = {
			declare_war_on = { target = SHL type = annex_everything }
		}

		ADISCORD_south_assign_bound_sides = yes
		ADISCORD_south_record_key_states = yes
		ADISCORD_south_offer_side_choices = yes
		ADISCORD_south_raise_oil_uprising = yes

		set_temp_variable = { ADISCORD_superevent_request = 76 }
		ADISCORD_superevent_enqueue = yes

		SHL = {
			ADISCORD_south_add_path_spirit = yes
			# The war has no deadline: only a champion's capitulation settles it.
			country_event = { id = ADISCORD_south.1 hours = 1 }
			news_event = { id = ADISCORD_south.2 hours = 2 }
		}
		NAM = {
			if = {
				limit = { has_country_flag = ADISCORD_south_north_leader }
				ADISCORD_south_add_path_spirit = yes
				country_event = { id = ADISCORD_south.1 hours = 1 }
			}
		}
		EFL = {
			if = {
				limit = { has_country_flag = ADISCORD_south_north_leader }
				ADISCORD_south_add_path_spirit = yes
				country_event = { id = ADISCORD_south.1 hours = 1 }
			}
		}
		AZH = {
			if = {
				limit = { has_country_flag = ADISCORD_south_north_leader }
				ADISCORD_south_add_path_spirit = yes
				country_event = { id = ADISCORD_south.1 hours = 1 }
			}
		}
	}
}

# COUNTRY champion. Leads its own faction and holds mandatory-major status only
# for this campaign, so the native peace waits for the champion's defeat.
ADISCORD_south_prepare_champion = {
	ADISCORD_release_non_participating_minor_optimization = yes
	if = {
		limit = {
			is_in_faction = yes
			is_faction_leader = no
		}
		leave_faction = yes
	}
	ADISCORD_south_ensure_campaign_faction = yes
	if = {
		limit = {
			OR = {
				is_major = no
				has_country_flag = RUS_crisis_added_major
				RUS_crisis_other_major_required = yes
			}
		}
		set_country_flag = ADISCORD_south_added_major
		set_major = yes
	}
	ADISCORD_south_open_champion_front = yes
	if = {
		limit = { difficulty > 1 }
		hidden_effect = { country_event = { id = ADISCORD_south.7 days = 45 } }
	}
	if = {
		limit = {
			is_ai = yes
			difficulty > 2
		}
		ADISCORD_south_issue_campaign_reserve = yes
		if = {
			limit = { difficulty > 3 }
			ADISCORD_south_issue_campaign_reserve = yes
		}
	}
}

# The event creates a fixed composition at delivery, independent of templates
# edited during preparation or inherited from the resource war.
ADISCORD_south_raise_second_wave = {
	if = {
		limit = { difficulty > 2 }
		add_manpower = 100000
		set_temp_variable = { var = ADISCORD_south_second_wave_regiments value = 7 }
	}
	else = {
		add_manpower = 30000
		set_temp_variable = { var = ADISCORD_south_second_wave_regiments value = 4 }
	}
	random_owned_controlled_state = {
		prioritize = { 699 700 701 69 }
		save_event_target_as = ADISCORD_south_second_wave_state
	}
	division_template = {
		name = "Southern Reserve Regiment"
		template_counter = 3
		division_names_group = ADISCORD_INF_01
		regiments = {
			infantry = { x = 0 y = 0 }
			infantry = { x = 0 y = 1 }
			infantry = { x = 1 y = 0 }
			infantry = { x = 1 y = 1 }
		}
	}
	for_loop_effect = {
		end = ADISCORD_south_second_wave_regiments
		value = ADISCORD_south_second_wave_index
		if = {
			limit = { ADISCORD_south_can_equip_reserve_regiment = yes }
			add_manpower = -4000
			add_equipment_to_stockpile = { type = infantry_equipment amount = -3600 }
			add_equipment_to_stockpile = { type = ADISCORD_squad_weapons_equipment amount = -32 }
			event_target:ADISCORD_south_second_wave_state = {
				create_unit = {
					division = "division_template = \"Southern Reserve Regiment\" start_experience_factor = 0.20 start_equipment_factor = 1.0 start_manpower_factor = 1.0"
					owner = PREV
					allow_spawning_on_enemy_provs = no
				}
			}
		}
	}
	division_template = {
		name = "Southern Reserve Militia"
		template_counter = 80
		division_names_group = ADISCORD_INF_01
		regiments = {
			ADISCORD_militia = { x = 0 y = 0 }
			ADISCORD_militia = { x = 0 y = 1 }
			ADISCORD_militia = { x = 1 y = 0 }
		}
	}
	event_target:ADISCORD_south_second_wave_state = {
		for_loop_effect = {
			end = 10
			value = ADISCORD_south_second_wave_militia_index
			create_unit = {
				division = "division_template = \"Southern Reserve Militia\" start_experience_factor = 0.10 start_equipment_factor = 1.0 start_manpower_factor = 1.0"
				owner = PREV
				allow_spawning_on_enemy_provs = no
			}
		}
	}
}

# GLOBAL. Fallback for a campaign war ended by another script before either
# champion capitulated; the war itself has no deadline. Only captured enemy key
# states count. Neutral ground and late side choices cannot manufacture an
# advance; equal gains mean peace.
ADISCORD_south_final_resolve_by_control = {
	if = {
		limit = {
			NOT = { SHL = { ADISCORD_south_champion_viable = yes } }
			NOT = { ADISCORD_south_north_leader_exists = yes }
		}
		ADISCORD_south_final_resolve_without_hegemon = yes
	}
	else_if = {
		limit = { NOT = { SHL = { ADISCORD_south_champion_viable = yes } } }
		ADISCORD_south_final_resolve_north_victory = yes
	}
	else_if = {
		limit = { NOT = { ADISCORD_south_north_leader_exists = yes } }
		ADISCORD_south_final_resolve_shl_victory = yes
	}
	else = {
		set_variable = { var = global.ADISCORD_south_shl_gains value = 0 }
		set_variable = { var = global.ADISCORD_south_north_gains value = 0 }
		for_each_scope_loop = {
			array = global.ADISCORD_south_key_states
			ADISCORD_south_count_key_state = yes
		}
		if = {
			limit = { check_variable = { var = global.ADISCORD_south_shl_gains value = global.ADISCORD_south_north_gains compare = greater_than } }
			ADISCORD_south_final_resolve_shl_victory = yes
		}
		else_if = {
			limit = { check_variable = { var = global.ADISCORD_south_north_gains value = global.ADISCORD_south_shl_gains compare = greater_than } }
			ADISCORD_south_final_resolve_north_victory = yes
		}
		else = {
			ADISCORD_south_final_repel = yes
		}
	}
}

			limit = { has_focus_tree = NAM_focus }

		}
		else_if = {
			limit = { has_focus_tree = NAM_postwar_focus }

# ----- Refused consolidation offers -----
# A refusal war belongs to the champion that sent the offer. When the refusing
# side falls, a member that refused annexation is annexed by the champion and
# every other member submits on the champion's current terms: peace and the
# faction exit come on the capitulation tick, the subjugation one day later
# through ADISCORD_south_finalize_submission (ADISCORD_south.47). A defeated
# champion makes white peace with the whole refusing side.

# GLOBAL, from on_capitulation; ROOT capitulated, FROM is the victor. A refusing
# member that falls before its partners stays occupied until the last one.
ADISCORD_south_refusal_capitulation = {
	if = {
		limit = {
			ROOT = { ADISCORD_south_refusal_side_SHL = yes }
			FROM = {
				OR = {
					tag = SHL
					is_in_faction_with = SHL
					is_subject_of = SHL
				}
			}
		}
		set_global_flag = skip_default_capitulation
		if = {
			limit = { ROOT = { ADISCORD_south_refusal_side_fallen_SHL = yes } }
			SHL = { ADISCORD_south_impose_refused_terms = yes }
		}
	}
	else_if = {
		limit = {
			ROOT = { ADISCORD_south_refusal_side_NAM = yes }
			FROM = {
				OR = {
					tag = NAM
					is_in_faction_with = NAM
					is_subject_of = NAM
				}
			}
		}
		set_global_flag = skip_default_capitulation
		if = {
			limit = { ROOT = { ADISCORD_south_refusal_side_fallen_NAM = yes } }
			NAM = { ADISCORD_south_impose_refused_terms = yes }
		}
	}
	else_if = {
		limit = {
			ROOT = { tag = SHL }
			FROM = { ADISCORD_south_refusal_side_SHL = yes }
		}
		set_global_flag = skip_default_capitulation
		SHL = { ADISCORD_south_end_refusal_war = yes }
	}
	else_if = {
		limit = {
			ROOT = { tag = NAM }
			FROM = { ADISCORD_south_refusal_side_NAM = yes }
		}
		set_global_flag = skip_default_capitulation
		NAM = { ADISCORD_south_end_refusal_war = yes }
	}
}

# COUNTRY champion; ROOT is the last capitulated member of the refusing side.
# Members are recorded before the first peace: white peace runs on_peace, which
# clears refusal records, and may end the war of later members.
ADISCORD_south_impose_refused_terms = {
	save_event_target_as = ADISCORD_south_refusal_champion
	clear_temp_array = ADISCORD_south_refusal_members
	clear_temp_array = ADISCORD_south_refusal_annexed
	every_enemy_country = {
		limit = {
			OR = {
				tag = ROOT
				is_in_faction_with = ROOT
				is_subject_of = ROOT
				ROOT = { is_subject_of = PREV }
			}
		}
		add_to_temp_array = { array = ADISCORD_south_refusal_members value = THIS }
		if = {
			limit = {
				OR = {
					AND = {
						PREV = { tag = SHL }
						has_country_flag = { flag = ADISCORD_south_refused_SHL value > 1 }
					}
					AND = {
						PREV = { tag = NAM }
						has_country_flag = { flag = ADISCORD_south_refused_NAM value > 1 }
					}
				}
			}
			add_to_temp_array = { array = ADISCORD_south_refusal_annexed value = THIS }
		}
	}
	for_each_scope_loop = {
		array = ADISCORD_south_refusal_annexed
		ADISCORD_south_clear_refusal_records = yes
		event_target:ADISCORD_south_refusal_champion = {
			if = {
				limit = { has_war_with = PREV }
				white_peace = PREV
			}
			annex_country = {
				target = PREV
				transfer_troops = no
			}
		}
	}
	for_each_scope_loop = {
		array = ADISCORD_south_refusal_members
		if = {
			limit = { exists = yes }
			ADISCORD_south_submit_refusal_member = yes
		}
	}
	clear_temp_array = ADISCORD_south_refusal_members
	clear_temp_array = ADISCORD_south_refusal_annexed
}

# COUNTRY defeated member. The receipt flag is the ordinary pending submission;

				end_civil_wars = no

		}

	ADISCORD_south_clear_refusal_records = yes

	}

		set_country_flag = ADISCORD_south_submits_to_NAM

}

	multiply_variable = { var = ADISCORD_south_strain_stability value = -0.002 }
	set_variable = { var = ADISCORD_south_strain_war_support value = ADISCORD_south_war_strain }
	multiply_variable = { var = ADISCORD_south_strain_war_support value = -0.001 }
	set_variable = { var = ADISCORD_south_strain_factory value = ADISCORD_south_war_strain }
	multiply_variable = { var = ADISCORD_south_strain_factory value = -0.001 }
	if = {
		limit = { NOT = { has_dynamic_modifier = { modifier = ADISCORD_south_war_strain_dynamic } } }
		add_dynamic_modifier = { modifier = ADISCORD_south_war_strain_dynamic }
	}
	force_update_dynamic_modifier = yes
	ADISCORD_economy_mark_dirty = yes
}

# COUNTRY champion. Sets temp ADISCORD_south_lost_key_states and
# ADISCORD_south_held_key_states: own key states of the war-start snapshot held
# by the enemy side and by the own side. Neutral control counts for neither.
ADISCORD_south_count_own_key_states = {
	set_temp_variable = { var = ADISCORD_south_lost_key_states value = 0 }
	set_temp_variable = { var = ADISCORD_south_held_key_states value = 0 }
	set_temp_variable = { var = ADISCORD_south_count_side value = ADISCORD_south_side }
	for_each_scope_loop = {
		array = global.ADISCORD_south_key_states
		ADISCORD_south_count_own_key_state = yes
	}
}

# STATE. Temporary variables keep the tally of the counting champion.
ADISCORD_south_count_own_key_state = {
	if = {
		limit = { check_variable = { var = ADISCORD_south_initial_side value = ADISCORD_south_count_side compare = equals } }
		if = {
			limit = {
				OR = {
					AND = {
						check_variable = { var = ADISCORD_south_count_side value = 1 compare = equals }
						ADISCORD_south_state_held_by_north_side = yes
					}
					AND = {
						check_variable = { var = ADISCORD_south_count_side value = 2 compare = equals }
						ADISCORD_south_state_held_by_shl_side = yes
					}
				}
			}
			add_to_temp_variable = { var = ADISCORD_south_lost_key_states value = 1 }
		}
		else_if = {
			limit = {
				OR = {
					AND = {
						check_variable = { var = ADISCORD_south_count_side value = 1 compare = equals }
						ADISCORD_south_state_held_by_shl_side = yes
					}
					AND = {
						check_variable = { var = ADISCORD_south_count_side value = 2 compare = equals }
						ADISCORD_south_state_held_by_north_side = yes
					}
				}
			}
			add_to_temp_variable = { var = ADISCORD_south_held_key_states value = 1 }
		}
	}
}

# ----- Expeditions and defectors -----
# A champion pays when it sends a request; the price waits in escrow under
# ADISCORD_south_expedition_pending or ADISCORD_south_defector_pending until the
# other country answers. Refusal, a failed recheck, a lost answer (fallback
# event ADISCORD_south.66) and the end of the campaign refund it exactly once.

# COUNTRY champion. One fixed, locked template for every expedition.
ADISCORD_south_ensure_expedition_template = {
	if = {
		limit = { NOT = { has_template = "Desert Expedition" } }
		division_template = {
			name = "Desert Expedition"
			template_counter = 3
			division_names_group = ADISCORD_INF_01
			is_locked = yes
			force_allow_recruiting = no
			regiments = {
				infantry = {
					x = 0
					y = 0
				}
				infantry = {
					x = 0
					y = 1
				}
				infantry = {
					x = 1
					y = 0
				}
				infantry = {
					x = 1
					y = 1
				}
				ADISCORD_line_artillery = {
					x = 2
					y = 0
				}
			}
			support = {
				engineer = {
					x = 0
					y = 0
				}
			}
		}
	}
}

# COUNTRY champion. Price of a request, debited when it is sent.
ADISCORD_south_pay_expedition = {
	add_political_power = -50
	subtract_from_variable = { var = ADISCORD_economy_treasury value = 120 }
	add_to_variable = { var = ADISCORD_economy_current_month_action_costs value = 120 }
	ADISCORD_economy_mark_dirty = yes
}

ADISCORD_south_refund_expedition = {
	if = {
		limit = { has_country_flag = ADISCORD_south_expedition_pending }
		clr_country_flag = ADISCORD_south_expedition_pending
		add_political_power = 50
		add_to_variable = { var = ADISCORD_economy_treasury value = 120 }
		add_to_variable = { var = ADISCORD_economy_current_month_action_income value = 120 }
		ADISCORD_economy_mark_dirty = yes
	}
}

ADISCORD_south_refund_defector = {
	if = {
		limit = { has_country_flag = ADISCORD_south_defector_pending }
		clr_country_flag = ADISCORD_south_defector_pending
		clear_variable = ADISCORD_south_defector_target
		add_political_power = 80
	}
}

# COUNTRY ally (event ROOT), FROM champion. Payment and delivery settle together.
ADISCORD_south_settle_expedition = {
	if = {
		limit = { ADISCORD_south_expedition_possible = yes }
		FROM = {
			clr_country_flag = ADISCORD_south_expedition_pending
			ADISCORD_south_ensure_expedition_template = yes
		}
		add_manpower = -6000
		add_equipment_to_stockpile = {
			type = infantry_equipment
			amount = -1000
		}
		FROM = {
			capital_scope = {
				ADISCORD_south_create_expedition_division = yes
				ADISCORD_south_create_expedition_division = yes
			}
		}
		if = {
			limit = { FROM = { ADISCORD_south_large_expedition = yes } }
			add_manpower = -3000
			add_equipment_to_stockpile = {
				type = infantry_equipment
				amount = -500
			}
			FROM = {
				capital_scope = { ADISCORD_south_create_expedition_division = yes }
			}
		}
		FROM = { country_event = { id = ADISCORD_south.61 hours = 1 } }
	}
	else = {
		FROM = {
			ADISCORD_south_refund_expedition = yes
			country_event = { id = ADISCORD_south.62 hours = 1 }
		}
	}
}

# STATE capital of the champion, owner PREV. The division arrives with the men
# and rifles the ally paid (about 3000 and 500); the champion's stockpile
# reinforces the rest.
ADISCORD_south_create_expedition_division = {
	create_unit = {
		division = "division_template = \"Desert Expedition\" start_experience_factor = 0.2 start_equipment_factor = 0.11575 start_manpower_factor = 0.625"
		owner = PREV
		allow_spawning_on_enemy_provs = no
	}
}

# COUNTRY defector (event ROOT); FROM is the envoy. Its original side is checked before payment settles.
ADISCORD_south_settle_defection = {
	if = {
		limit = { ADISCORD_south_defection_possible = yes }
		FROM = {
			clr_country_flag = ADISCORD_south_defector_pending
			clear_variable = ADISCORD_south_defector_target
		}
		set_country_flag = ADISCORD_south_defected
		ADISCORD_south_leave_coalition = yes
		if = {
			limit = { FROM = { ADISCORD_south_on_shl_side = yes } }
			every_enemy_country = {
				limit = { ADISCORD_south_on_shl_side = yes }
				white_peace = PREV
			}
			clear_variable = ADISCORD_south_side
			ADISCORD_south_join_shl_side = yes
		}
		else = {
			every_enemy_country = {
				limit = { ADISCORD_south_on_north_side = yes }
				white_peace = PREV
			}
			clear_variable = ADISCORD_south_side
			ADISCORD_south_join_north_side = yes
		}
		FROM = { country_event = { id = ADISCORD_south.64 hours = 1 } }
	}
	else = {
		FROM = {
			if = {
				limit = {
					has_country_flag = ADISCORD_south_defector_pending
					check_variable = { var = ADISCORD_south_defector_target value = ROOT compare = equals }
				}
				ADISCORD_south_refund_defector = yes
				country_event = { id = ADISCORD_south.65 hours = 1 }
			}
		}
	}
}
