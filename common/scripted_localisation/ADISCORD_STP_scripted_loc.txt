	name = STPGetSuspicionValue

	text = {
		trigger = { check_variable = { var = STP_party_suspicion value = 70 compare = greater_than_or_equals } }
		localization_key = STP_SUSPICION_VALUE_HIGH

		localization_key = STP_SUSPICION_VALUE_MEDIUM
	}
	text = {
		localization_key = STP_SUSPICION_VALUE_LOW
	}
}
