	name = ADISCORDSouthNorthSide
	# During the war the flagged leader; before it, the first viable candidate.
	text = {
		trigger = {
			OR = {
				NAM = { has_country_flag = ADISCORD_south_north_leader }
				AND = {
					NOT = { ADISCORD_south_final_active = yes }
					NAM = { ADISCORD_south_champion_viable = yes }
				}
			}
		}
		localization_key = ADISCORD_south_north_side_nam

				}

		}
		localization_key = ADISCORD_south_north_side_efl

		trigger = {
