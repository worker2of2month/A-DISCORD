on_actions = {
	# ROOT is the attacking side and FROM is the defending side. Publishing from
	# the relation edge keeps news tied to wars that really exist in the engine,
	# rather than to focus promises or intervention preparation.
	on_war_relation_added = {
		effect = {
			# Vorkerland: the federal collapse becomes an actual war between the
			# three central claimants. The earlier tower-collapse news is separate.
			if = {
				limit = {
					has_global_flag = ADISCORD_vorkerland_collapse_started
					NOT = { has_global_flag = ADISCORD_news_vorkerland_fighting_published }
					ROOT = { OR = { tag = WKR tag = VAD tag = TVA } }
					FROM = { OR = { tag = WKR tag = VAD tag = TVA } }
				}
				set_global_flag = ADISCORD_news_vorkerland_fighting_published
				ROOT = { news_event = { id = ADISCORD_world_news.1 hours = 1 } }
			}

			# Stelander already has a richer public outbreak report (.70) with
			# election-dependent text. Schedule it from the actual STP/STS war
			# relation after the split stack unwinds; opening the news window
			# inside declare_war_on corrupts the observer client. The fallback call in
			# STP_cw_begin_hostilities checks the same publication flag.
			if = {
				limit = {
					has_global_flag = STP_cw_started
					NOT = { has_global_flag = ADISCORD_news_stelander_civil_war_published }
					OR = {
						AND = { ROOT = { tag = STP } FROM = { tag = STS } }
						AND = { ROOT = { tag = STS } FROM = { tag = STP } }
					}
				}
				set_global_flag = ADISCORD_news_stelander_civil_war_published
				ROOT = { news_event = { id = ADISCORD_STP_cw.70 hours = 1 } }
			}

			# The VAL-SRP war uses the detailed Occidia invasion report. Its fallback
			# in the ultimatum callback shares this guard; both wait past declare_war_on.
			if = {
				limit = {
					has_global_flag = STP_cw_started
					NOT = { has_global_flag = ADISCORD_news_kefreyt_intervention_published }
					OR = {
						AND = { ROOT = { tag = VAL } FROM = { tag = SRP } }
						AND = { FROM = { tag = VAL } ROOT = { tag = SRP } }
					}
				}
				set_global_flag = ADISCORD_news_kefreyt_intervention_published
				ROOT = { news_event = { id = ADISCORD_STP_cw.73 hours = 1 } }
			}

			# Itora's intervention is only news once troops actually enter a war
			# against one of the northern Vorkerland claimants.
			if = {
				limit = {
					has_global_flag = ADISCORD_vorkerland_collapse_started
					NOT = { has_global_flag = ADISCORD_news_itora_intervention_published }
					OR = {
						AND = {
							ROOT = { tag = IVN }
							FROM = { OR = { tag = ZAO tag = WPA tag = WPS tag = PWR tag = PSD } }
						}
						AND = {
							FROM = { tag = IVN }
							ROOT = { OR = { tag = ZAO tag = WPA tag = WPS tag = PWR tag = PSD } }
						}
					}
				}
				set_global_flag = ADISCORD_news_itora_intervention_published
				ROOT = { news_event = { id = ADISCORD_world_news.5 hours = 1 } }
			}
		}
	}
}
