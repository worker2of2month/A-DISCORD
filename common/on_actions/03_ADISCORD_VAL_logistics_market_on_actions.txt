on_actions = {
	on_startup = {
		effect = {
			if = {
				limit = { VAL = { exists = yes } }
				VAL = {
					VAL_initialize_logistics_market = yes
					VAL_trade_routes_map_refresh_cache = yes
				}
			}
		}
	}

	on_war_relation_added = {
		effect = {
			if = {
				limit = {
					VAL = { exists = yes }
					OR = {
						ROOT = { OR = { tag = VAL tag = STP tag = STS tag = NOD tag = WKR tag = VAD tag = TVA } }
						FROM = { OR = { tag = VAL tag = STP tag = STS tag = NOD tag = WKR tag = VAD tag = TVA } }
					}
				}
				VAL = { VAL_open_refugee_waves = yes }
			}
		}
	}

	on_weekly_VAL = {
		effect = {
			VAL_logistics_market_weekly = yes
		}
	}

	on_state_control_changed = {
		effect = {
			if = {
				limit = {
					# Only these authored route nodes can change corridor availability.
					FROM.FROM = { OR = { state = 29 state = 33 state = 43 state = 44 state = 46 state = 59 state = 60 state = 61 state = 88 } }
					VAL = { exists = yes VAL_trade_corridors_unlocked = yes }
				}
				VAL = { VAL_refresh_trade_network = yes }
			}
		}
	}
}
