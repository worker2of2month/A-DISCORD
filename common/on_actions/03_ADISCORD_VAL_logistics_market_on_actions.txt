on_actions = {
	on_startup = {
		effect = {
			if = {
				limit = { VAL = { exists = yes } }
				VAL = { VAL_initialize_logistics_market = yes }
			}
		}
	}

	on_war_relation_added = {
		effect = {
			if = {
				limit = {
					VAL = { exists = yes }
					OR = {
						ROOT = { OR = { tag = VAL tag = STP tag = STS tag = NOD tag = WKR tag = VAD tag = TVA } }
						FROM = { OR = { tag = VAL tag = STP tag = STS tag = NOD tag = WKR tag = VAD tag = TVA } }
					}
				}
				VAL = { VAL_open_refugee_waves = yes }
			}
		}
	}

	on_weekly_VAL = {
		effect = {
			VAL_logistics_market_weekly = yes
		}
	}

	on_state_control_changed = {
		effect = {
			if = {
				limit = { VAL = { exists = yes VAL_trade_corridors_unlocked = yes } }
				VAL = { VAL_refresh_trade_network = yes }
			}
		}
	}
}
