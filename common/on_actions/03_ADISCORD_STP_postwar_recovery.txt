on_actions = {
	# The normal STP/STS settlement checks closure in the same tick as white_peace.
	# Clausewitz can keep the war relation alive until the next tick, so retry only
	# for a confirmed postwar victor. This also repairs saves made in that gap.
	on_daily = {
		effect = {
			if = {
				limit = {
					OR = { tag = STP tag = STS }
					has_global_flag = STP_cw_started
					NOT = { has_global_flag = STP_cw_union_wars_finished }
					has_country_flag = STP_cw_won_union_battle
					has_country_flag = STP_cw_postwar
				}
				STP_cw_check_union_wars_finished = yes
			}
		}
	}
}
