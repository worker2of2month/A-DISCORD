on_actions = {
	# Loaded after country handlers. skip_default_capitulation reserves their
	# scripted settlements; other capitulations use white peace and annexation.
	on_capitulation = {
		effect = {
			if = {
				limit = {
					NOT = { has_global_flag = skip_default_capitulation }
					FROM = { exists = yes has_capitulated = no }
					# Neutral faction members cannot veto a war they never joined.
					NOT = {
						ROOT = {
							is_in_faction = yes
							any_other_country = {
								is_in_faction_with = PREV
								has_capitulated = no
								has_war_with = FROM
							}
						}
					}
				}
				# Snapshot before white_peace can remove every faction war relation
				# or annex_country can remove ROOT and its faction identity.
				# Temporary arrays belong only to this callback, not to the save.
				clear_temp_array = ADISCORD_peace_defeated_countries
				clear_temp_array = ADISCORD_peace_awarded_states
				FROM = {
					every_enemy_country = {
						limit = {
							OR = { tag = ROOT has_capitulated = yes }
							OR = {
								tag = ROOT
								is_in_faction_with = ROOT
								is_puppet_of = ROOT
							}
						}
						add_to_temp_array = { array = ADISCORD_peace_defeated_countries value = THIS }
						every_owned_state = {
							add_to_temp_array = { array = ADISCORD_peace_awarded_states value = THIS }
						}
					}
				}
				for_each_scope_loop = {
					array = ADISCORD_peace_defeated_countries
					FROM = { if = { limit = { has_war_with = PREV } white_peace = PREV } }
					if = {
						limit = { has_country_flag = ADISCORD_vorkerland_coalition_mandatory_major }
						set_major = no
						clr_country_flag = ADISCORD_vorkerland_coalition_mandatory_major
						clr_country_flag = ADISCORD_vorkerland_regional_auxiliary
					}
					FROM = { annex_country = { target = PREV transfer_troops = no } }
				}
				# Only states owned by these defeated participants were captured.
				# Include impassable remnants without sweeping cores of neighbours.
				for_each_scope_loop = {
					array = ADISCORD_peace_awarded_states
					FROM = { transfer_state = PREV }
				}
				clear_temp_array = ADISCORD_peace_defeated_countries
				clear_temp_array = ADISCORD_peace_awarded_states
			}

			# Vorkerland's central router deliberately reserves a defeated claimant
			# from the generic fallback while it updates the reunification phase. Once
			# that bookkeeping has run, consume the defeated claimant in the same
			# capitulation tick instead of leaving a permanent diagnostic shell.
			if = {
				limit = {
					ROOT = {
						has_country_flag = ADISCORD_vorkerland_claimant_cleanup_reserved
						ADISCORD_vorkerland_has_surviving_coalition_partner = no
						ADISCORD_vorkerland_is_main_claimant = yes
						has_capitulated = yes
					}
				}

				# Ostfort and Tehlar are authored Worx protectorates. If TVA loses,
				# fold them into the defeated directorate before its victor annexes it;
				# otherwise an overlord annex can free them as one-state postwar shells.
				if = {
					limit = {
						ROOT = { tag = TVA }
						WTD = {
							exists = yes
							has_capitulated = yes
							has_country_flag = ADISCORD_vorkerland_worx_aligned_technocrats
						}
					}
					TVA = {
						if = { limit = { has_war_with = WTD } white_peace = WTD }
						WTD = { if = { limit = { has_country_flag = ADISCORD_vorkerland_coalition_mandatory_major } set_major = no clr_country_flag = ADISCORD_vorkerland_coalition_mandatory_major clr_country_flag = ADISCORD_vorkerland_regional_auxiliary } }
						annex_country = { target = WTD transfer_troops = no }
					}
				}
				if = {
					limit = {
						ROOT = { tag = TVA }
						TGD = {
							exists = yes
							has_capitulated = yes
							has_country_flag = ADISCORD_vorkerland_joined_worx_directorate
						}
					}
					TVA = {
						if = { limit = { has_war_with = TGD } white_peace = TGD }
						TGD = { if = { limit = { has_country_flag = ADISCORD_vorkerland_coalition_mandatory_major } set_major = no clr_country_flag = ADISCORD_vorkerland_coalition_mandatory_major clr_country_flag = ADISCORD_vorkerland_regional_auxiliary } }
						annex_country = { target = TGD transfer_troops = no }
					}
				}

				# The feature router may credit the capitulation to the claimant itself,
				# one of its subjects, or the three fixed auxiliary tags below. Mirror
				# that normalization here so the land always reaches the living claimant.
				FROM = {
					if = {
						limit = {
							ADISCORD_vorkerland_is_main_claimant = yes
							exists = yes
							is_subject = no
							NOT = { has_capitulated = yes }
						}
						if = { limit = { has_war_with = ROOT } white_peace = ROOT }
						ROOT = { if = { limit = { has_country_flag = ADISCORD_vorkerland_coalition_mandatory_major } set_major = no clr_country_flag = ADISCORD_vorkerland_coalition_mandatory_major clr_country_flag = ADISCORD_vorkerland_regional_auxiliary } }
						annex_country = { target = ROOT transfer_troops = yes }
					}
					else_if = {
						limit = { ADISCORD_vorkerland_has_live_claimant_leader = yes }
						faction_leader = {
							if = { limit = { has_war_with = ROOT } white_peace = ROOT }
							ROOT = { if = { limit = { has_country_flag = ADISCORD_vorkerland_coalition_mandatory_major } set_major = no clr_country_flag = ADISCORD_vorkerland_coalition_mandatory_major clr_country_flag = ADISCORD_vorkerland_regional_auxiliary } }
						annex_country = { target = ROOT transfer_troops = yes }
						}
					}
					else_if = {
						limit = {
							is_subject = yes
							NOT = { has_capitulated = yes }
							OVERLORD = {
								exists = yes
								ADISCORD_vorkerland_is_main_claimant = yes
								is_subject = no
								NOT = { has_capitulated = yes }
							}
						}
						OVERLORD = {
							if = { limit = { has_war_with = ROOT } white_peace = ROOT }
							ROOT = { if = { limit = { has_country_flag = ADISCORD_vorkerland_coalition_mandatory_major } set_major = no clr_country_flag = ADISCORD_vorkerland_coalition_mandatory_major clr_country_flag = ADISCORD_vorkerland_regional_auxiliary } }
						annex_country = { target = ROOT transfer_troops = yes }
						}
					}
					else_if = {
						limit = { tag = WTD TVA = { exists = yes is_subject = no NOT = { has_capitulated = yes } } }
						TVA = {
							if = { limit = { has_war_with = ROOT } white_peace = ROOT }
							ROOT = { if = { limit = { has_country_flag = ADISCORD_vorkerland_coalition_mandatory_major } set_major = no clr_country_flag = ADISCORD_vorkerland_coalition_mandatory_major clr_country_flag = ADISCORD_vorkerland_regional_auxiliary } }
						annex_country = { target = ROOT transfer_troops = yes }
						}
					}
					else_if = {
						limit = { tag = VLA WKR = { exists = yes is_subject = no NOT = { has_capitulated = yes } } }
						WKR = {
							if = { limit = { has_war_with = ROOT } white_peace = ROOT }
							ROOT = { if = { limit = { has_country_flag = ADISCORD_vorkerland_coalition_mandatory_major } set_major = no clr_country_flag = ADISCORD_vorkerland_coalition_mandatory_major clr_country_flag = ADISCORD_vorkerland_regional_auxiliary } }
						annex_country = { target = ROOT transfer_troops = yes }
						}
					}
					else_if = {
						limit = { tag = SOL VAD = { exists = yes is_subject = no NOT = { has_capitulated = yes } } }
						VAD = {
							if = { limit = { has_war_with = ROOT } white_peace = ROOT }
							ROOT = { if = { limit = { has_country_flag = ADISCORD_vorkerland_coalition_mandatory_major } set_major = no clr_country_flag = ADISCORD_vorkerland_coalition_mandatory_major clr_country_flag = ADISCORD_vorkerland_regional_auxiliary } }
						annex_country = { target = ROOT transfer_troops = yes }
						}
					}
					else_if = {
						limit = {
							tag = WRK
							exists = yes
							is_subject = no
							NOT = { has_capitulated = yes }
							has_global_flag = ADISCORD_vorkerland_phase_postwar_integration
							has_global_flag = ADISCORD_vorkerland_reunification_verified
						}
						if = { limit = { has_war_with = ROOT } white_peace = ROOT }
						ROOT = { if = { limit = { has_country_flag = ADISCORD_vorkerland_coalition_mandatory_major } set_major = no clr_country_flag = ADISCORD_vorkerland_coalition_mandatory_major clr_country_flag = ADISCORD_vorkerland_regional_auxiliary } }
						annex_country = { target = ROOT transfer_troops = yes }
					}
				}
			}

			if = {
				limit = { has_global_flag = ADISCORD_vorkerland_central_showdown_started NOT = { has_global_flag = ADISCORD_vorkerland_collapse_finished } }
				ADISCORD_vorkerland_begin_reunification = yes
			}

			# The reservation is per capitulation and must never leak into the next one.
			clr_global_flag = skip_default_capitulation
		}
	}
}
