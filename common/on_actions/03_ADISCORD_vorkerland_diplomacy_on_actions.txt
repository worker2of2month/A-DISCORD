on_actions = {
	on_capitulation = {
		effect = {
			if = {
				limit = {
					ROOT = { OR = { tag = SOL tag = SRA tag = CSL tag = VLA tag = EBA tag = TGD } }
					NOT = { has_global_flag = ADISCORD_vorkerland_regional_diplomacy_check_scheduled }
				}
				set_global_flag = {
					flag = ADISCORD_vorkerland_regional_diplomacy_check_scheduled
					days = 3
				}
				if = {
					limit = { country_exists = WKR }
					WKR = { country_event = { id = ADISCORD_vorkerland_diplomacy.1 days = 1 } }
				}
				else_if = {
					limit = { country_exists = VAD }
					VAD = { country_event = { id = ADISCORD_vorkerland_diplomacy.1 days = 1 } }
				}
				else_if = {
					limit = { country_exists = TVA }
					TVA = { country_event = { id = ADISCORD_vorkerland_diplomacy.1 days = 1 } }
				}
			}

			# VAD's intervention has an exact settlement instead of a peace
			# conference: only the five Solar states are rebuilt as loyalist SOL.
			if = {
				limit = {
					has_global_flag = ADISCORD_vorkerland_vad_solar_intervention_active
					ROOT = {
						OR = {
							AND = { tag = SRA has_global_flag = ADISCORD_vorkerland_vad_solar_intervention_target_sra }
							AND = { tag = CSL has_global_flag = ADISCORD_vorkerland_vad_solar_intervention_target_csl }
						}
					}
					FROM = { tag = VAD NOT = { has_capitulated = yes } }
				}
				set_global_flag = skip_default_capitulation
				set_global_flag = ADISCORD_vorkerland_sol_restoration_pending
				clr_global_flag = ADISCORD_vorkerland_sol_restoration_materialized
				clr_global_flag = ADISCORD_vorkerland_sol_restoration_materialization_retry
				clr_global_flag = ADISCORD_vorkerland_sol_restoration_retry
				clr_global_flag = ADISCORD_vorkerland_sol_restoration_failed
				clr_global_flag = ADISCORD_vorkerland_sol_restoration_fresh_release
				FROM = {
					clr_country_flag = ADISCORD_vorkerland_vad_solar_intervention_verify_pending
					clr_country_flag = ADISCORD_vorkerland_vad_solar_intervention_verify_retry
					annex_country = { target = ROOT transfer_troops = no }
					transfer_state = 76
					transfer_state = 104
					transfer_state = 198
					transfer_state = 307
					transfer_state = 310
					country_event = { id = ADISCORD_vorkerland_diplomacy.4 days = 1 }
				}
			}

			if = {
				limit = {
					has_global_flag = ADISCORD_vorkerland_wkr_solyarino_intervention_active
					country_exists = WKR
					ROOT = { OR = { tag = SOL tag = SRA tag = CSL } }
				}
				WKR = { country_event = { id = ADISCORD_vorkerland_diplomacy.12 days = 1 } }
			}
			ADISCORD_vorkerland_clear_failed_vad_solar_intervention = yes
		}
	}

	on_peace = {
		effect = {
			ADISCORD_vorkerland_clear_failed_vad_solar_intervention = yes
			ADISCORD_vorkerland_clear_wkr_solyarino_intervention = yes
		}
	}
}
