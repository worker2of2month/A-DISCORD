####Country Development

on_actions = {
	on_startup = {
		effect = {
			create_entity = { ##poluted zone
			    entity = ADISCORD_global_entity
			    id = 1
			    x = 42
			    y = 21
			    z = -100.000
			    province = 16531
			    state = 42
			    rotation = 1.2
			    scale = 0.03
			    min_zoom = 5555.0
			    #visible = scripted_trigger_name
			}
			create_entity = {
			    entity = ADISCORD_global_entity
			    id = 2
			    x = 42
			    y = 21
			    z = -100.000
			    province = 12863
			    state = 42
			    rotation = 1.2
			    scale = 0.02
			    min_zoom = 5555.0
			    #visible = scripted_trigger_name
			}
			#create_entity = { ##clouds
			#    entity = ADISCORD_global_entity_2
			#    id = 2
			#    x = 42
			#    y = 21
			#    z = -100.000
			#    province = 6713
			#    state = 42
			#    rotation = 1.2
			#    scale = 1.02
			#    min_zoom = 5555.0
			#    #visible = scripted_trigger_name
			#}
			# Scenario history is the sole provenance producer. A pre-contract save
			# cannot enter this branch, while the completion sentinel prevents a
			# fresh campaign reload from overwriting research or development.
			if = {
				limit = {
					has_global_flag = ADISCORD_fresh_campaign_contract_v1
					NOT = { has_global_flag = ADISCORD_starting_technology_profiles_applied }
				}
				every_country = {
					ADISCORD_grant_starting_technology_profile = yes
				}
				every_country = {
					if = {
						limit = { OR = { ADISCORD_is_playable_development_country = yes is_ai = no } }
						ADISCORD_initialize_default_country_development = yes
					}
				}
				STP = {
					# Fresh history starts with per-template locks; Army HQ remains usable.
					ADISCORD_STP_lock_regular_army_templates = yes

					# Starting society development levels for StepanLand.
					ADISCORD_set_society_development_2 = yes
					ADISCORD_set_social_system_development_2 = yes
					ADISCORD_set_army_development_2 = yes
					ADISCORD_set_cultural_development_1 = yes
					ADISCORD_set_state_development_3 = yes
					ADISCORD_set_economic_development_4 = yes
					ADISCORD_initialize_society_development_variables = yes
					ADISCORD_update_total_country_development = yes
				}
				VAL = {
					# Starting society development levels for ValeraLand.
					# Contract state: strong mercenary army, strong arms economy,
					# weak social base, harsh culture, decent but not excellent state apparatus.
					ADISCORD_set_society_development_2 = yes
					ADISCORD_set_social_system_development_1 = yes
					ADISCORD_set_army_development_4 = yes
					ADISCORD_set_cultural_development_2 = yes
					ADISCORD_set_state_development_2 = yes
					ADISCORD_set_economic_development_4 = yes
					ADISCORD_initialize_society_development_variables = yes
					ADISCORD_update_total_country_development = yes
				}
				WRK = {
					# Ruined great-power baseline: substantial institutions and army,
					# but only middling post-collapse economic capacity. This prevents
					# the large inherited factory base from entering an automatic debt spiral.
					ADISCORD_set_society_development_2 = yes
					ADISCORD_set_social_system_development_2 = yes
					ADISCORD_set_army_development_3 = yes
					ADISCORD_set_cultural_development_2 = yes
					ADISCORD_set_state_development_3 = yes
					ADISCORD_set_economic_development_3 = yes
					ADISCORD_initialize_society_development_variables = yes
					ADISCORD_update_total_country_development = yes
				}
				every_country = {
					limit = {
						OR = {
							ADISCORD_economy_is_player_tier_country = yes
							ADISCORD_economy_is_primary_tier_country = yes
							ADISCORD_economy_is_secondary_tier_country = yes
						}
					}
					ADISCORD_economy_initialize_country = yes
					# V19 TDA-core: disabled hidden startup economic pseudo-building distribution.
					# ADISCORD_distribute_starting_buildings_by_country = yes
				}
				set_global_flag = ADISCORD_starting_technology_profiles_applied
			}
		}
	}
	# Event-driven war-state edges keep demobilization responsive without adding
	# a daily poll. Only an already current-schema country can enter these paths.
	on_war = {
		effect = {
			if = {
				limit = { ADISCORD_economy_has_current_schema = yes }
				ADISCORD_economy_update_postwar_demobilization = yes
				ADISCORD_economy_update_gui = yes
			}
		}
	}
	on_peace = {
		effect = {
			if = {
				limit = {
					ADISCORD_economy_has_current_schema = yes
					has_war = no
				}
				ADISCORD_economy_update_postwar_demobilization = yes
				# The transition can replace economy and conscription laws. Refresh
				# their scalar caches once, then reuse the ordinary budget-control
				# refresh instead of waiting up to a month for the menu to catch up.
				ADISCORD_economy_recalculate_policy_modifiers = yes
				ADISCORD_economy_refresh_after_budget_control_change = yes
			}
		}
	}
	on_weekly = {
		effect = {
			# Country-scoped cash settlement. Forecast inputs are cached country
			# variables; no country or state iteration is allowed in this pulse.
			if = {
				limit = { ADISCORD_economy_should_weekly_update = yes }
				ADISCORD_economy_weekly_update = yes
			}
		}
	}
	on_state_control_changed = {
		# ROOT is the new controller and FROM is the old controller. A control
		# change can enable or disable regional cluster output, so invalidate both
		# country caches cheaply and let the next scheduled/open-window refresh do
		# the single owned-state recount.
		effect = {
			if = {
				limit = { ADISCORD_economy_has_current_schema = yes }
				ADISCORD_economy_mark_dirty = yes
			}
			FROM = {
				if = {
					limit = { ADISCORD_economy_has_current_schema = yes }
					ADISCORD_economy_mark_dirty = yes
				}
			}
		}
	}
	on_monthly = {
		effect = {
			# HOI4 monthly pulse is already country-scoped. Do not every_country here.
			if = {
				limit = { ADISCORD_economy_should_monthly_update = yes }
				ADISCORD_economy_monthly_update = yes
			}
			if = {
				limit = {
					has_global_flag = ADISCORD_fresh_campaign_contract_v1
					OR = { ADISCORD_is_playable_development_country = yes is_ai = no }
				}
				ADISCORD_tick_all_society_development_monthly = yes
			}
		}
	}
	on_yearly = {
		effect = {
			# Country-scoped yearly update. The trigger decides whether this country is secondary-tier.
			if = {
				limit = { ADISCORD_economy_should_yearly_update = yes }
				ADISCORD_economy_yearly_update = yes
			}
			if = {
				limit = {
					has_global_flag = ADISCORD_fresh_campaign_contract_v1
					ADISCORD_is_secondary_development_country = yes
					is_ai = yes
				}
				ADISCORD_tick_all_society_development_yearly = yes
			}
		}
	}
}
