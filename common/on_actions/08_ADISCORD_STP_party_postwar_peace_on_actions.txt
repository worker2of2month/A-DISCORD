# Party postwar Kefreyt campaign. This runs before the generic scripted-peace router.
on_actions = {
	on_capitulation_immediate = {
		effect = {
			if = {
				limit = {
					ROOT = {
						tag = VAL
						has_war_with = STP
						NOT = { any_enemy_country = { NOT = { OR = { tag = STP is_subject_of = STP } } } }
					}
					STP = {
						exists = yes
						has_capitulated = no
						has_country_flag = STP_cw_postwar
						has_country_flag = STP_pw_party_kefreyt_campaign_active
					}
					FROM = { OR = { tag = STP is_subject_of = STP } }
				}
				set_global_flag = skip_default_capitulation
				STP = { STP_pw_party_settle_kefreyt_victory = yes }
				clr_global_flag = skip_default_capitulation
			}
		}
	}
}
