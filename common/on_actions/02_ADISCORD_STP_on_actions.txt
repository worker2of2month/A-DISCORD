on_actions = {
	on_weekly_VAL = {
		effect = {
			if = { limit = { STP_cw_kefreyt_aid_open = yes NOT = { has_country_flag = STP_ps_val_offer_sent } STP = { has_country_flag = { flag = STP_sided_with_the_party_flag days > 7 } } }
				set_country_flag = STP_ps_val_offer_sent
				country_event = { id = ADISCORD_STP_preparation.6 }
			}
		}
	}
	on_daily_NOD = {
		effect = {
			if = { limit = { has_country_flag = STP_ps_expedition_ready }
				if = { limit = { OR = { NOD_cw_intervention_possible = yes AND = { has_country_flag = NOD_cw_entered has_war_with = STS } } } STP_ps_apply_expedition = yes }
				else = { clr_country_flag = STP_ps_expedition_ready }
			}
			if = { limit = { has_country_flag = STP_ps_exile_received STP_ps_nod_can_host_exiles = no } STP_ps_close_exile = yes }
			if = { limit = { has_idea = STP_ps_engineers_idea STP_ps_northern_front_active = no } remove_ideas = STP_ps_engineers_idea }
			if = { limit = { has_idea = STP_ps_expedition_idea NOT = { has_war_with = STS } } remove_ideas = STP_ps_expedition_idea }
		}
	}
	on_daily_STP = {
		effect = {
			if = { limit = { STP_ps_war_active = yes }
				STP_ps_begin_defence = yes
				STP_ps_poll_nodes = yes
			}
			else_if = { limit = { OR = { has_variable = STP_ps_reorg_deposit has_dynamic_modifier = { modifier = STP_ps_defence_dynamic } } } STP_ps_close_defence = yes }
		}
	}
	on_daily = {
		effect = {
			if = {
				limit = {
					OR = { tag = STP tag = STS }
					has_global_flag = STP_cw_started
					NOT = { has_global_flag = STP_cw_union_wars_finished }
					has_country_flag = STP_cw_won_union_battle
					has_country_flag = STP_cw_postwar
				}
				STP_cw_check_union_wars_finished = yes
			}
		}
	}
	# Country removal has no shared callback here; reconcile only an open offer.
	on_weekly_STS = {
		effect = {
			if = { limit = { STP_pw_can_reconstruct = yes NOT = { has_variable = STP_pw_recovery_progress } } STP_pw_begin_recovery = yes }
			if = { limit = { has_country_flag = STP_cw_postwar } STP_pw_reconcile_postwar_interactivity = yes }
			if = {
				limit = { exists = yes has_country_flag = STP_pc_lib_package_pending }
				STP_pc_offer_liberation_package = yes
			}
		}
	}
	# The split creates STS before the actual STP-STS war relation. Reconcile the
	# newborn country at that late edge, when its territory, templates and army
	# already exist but the original STP technology state is still authoritative.
	on_war_relation_added = {
		effect = {
			if = { limit = { OR = { AND = { ROOT = { tag = STP } FROM = { tag = STS } } AND = { ROOT = { tag = STS } FROM = { tag = STP } } } } STP = { STP_ps_begin_defence = yes } }
			if = {
				limit = {
					has_global_flag = STP_cw_started
					NOT = { has_global_flag = STP_cw_union_wars_finished }
					OR = {
						AND = { ROOT = { tag = STP } FROM = { tag = STS } }
						AND = { ROOT = { tag = STS } FROM = { tag = STP } }
					}
					STS = {
						exists = yes
						has_country_flag = STP_cw_participant
						NOT = { has_country_flag = STP_cw_sts_runtime_bootstrap_applied }
					}
				}
				STS = {
					inherit_technology = STP
					mark_technology_tree_layout_dirty = yes
					# Production lines are country state and do not follow inherit_technology.
					# Keep the newborn two-factory queue bounded and immediately useful.
					add_equipment_production = {
						equipment = { type = infantry_equipment_0 creator = "STS" }
						requested_factories = 1
						progress = 0.30
						efficiency = 50
					}
					add_equipment_production = {
						equipment = { type = ADISCORD_squad_weapons_equipment_0 creator = "STS" }
						requested_factories = 1
						progress = 0.30
						efficiency = 50
					}
					set_country_flag = STP_cw_sts_runtime_bootstrap_applied
				}
			}
			if = {
				limit = {
					has_global_flag = STP_cw_started
					NOT = { has_global_flag = STP_cw_union_wars_finished }
					OR = {
						AND = { ROOT = { tag = STP } FROM = { tag = STS } }
						AND = { ROOT = { tag = STS } FROM = { tag = STP } }
					}
					STS = {
						exists = yes
						has_country_flag = STP_cw_participant
						NOT = { has_country_flag = STP_cw_old_fada_guard_positioned }
						53 = { is_owned_by = STS }
						any_country_division = {
							division_has_battalion_in_template = ADISCORD_territorial
							unit_strength > 0.99
						}
					}
				}
				# Old Fada's prewar police unit was dissolved into the shared reserve.
				# Move one already-paid newborn brigade back there one-for-one instead
				# of creating additional manpower or equipment for the resistance.
				# destroy_unit/create_unit during declare_war_on corrupts the heap
				# on the observer client; wait until the war relation exists.
				STS = { country_event = { id = ADISCORD_STP_cw.31 hours = 1 } }
			}
		}
	}
	on_weekly = {
		effect = {
			if = {
				limit = { tag = NOD check_variable = { var = STP_cw_northern_campaign_status value = 1 compare = equals } }
				STP_cw_poll_northern_campaign = yes
			}
			if = {
				limit = { tag = NOD OR = { NOD_cw_intervention_possible = yes has_country_flag = NOD_cw_intervention_approved } }
				STP_cw_poll_nod_intervention = yes
			}
			# Foreign conquest can temporarily invalidate the split's anchor guards.
			# Preserve the declared result and retry only this country, without resets.
			if = {
				limit = { tag = STP has_country_flag = STP_cw_elections_finished NOT = { has_country_flag = STP_cw_participant } NOT = { has_global_flag = STP_cw_started } }
				STP_cw_start = yes
			}
			if = {
				limit = {
					tag = STP
					has_country_flag = STP_cw_participant
					OR = {
						has_global_flag = STP_cw_started
						has_country_flag = STP_sided_with_the_party_flag
						has_country_flag = STP_sided_with_Maksim_flag
					}
					NOT = { has_global_flag = STP_cw_union_wars_finished }
				}
				STP_cw_begin_hostilities = yes
			}
			if = {
				limit = {
					tag = STS
					has_country_flag = STP_cw_participant
					OR = {
						has_global_flag = STP_cw_started
						has_country_flag = STP_sided_with_Maksim_flag
						has_country_flag = STP_sided_with_the_party_flag
						STP = { has_country_flag = STP_sided_with_the_party_flag }
						STP = { has_country_flag = STP_sided_with_Maksim_flag }
					}
					NOT = { has_war_with = STP }
					NOT = { has_global_flag = STP_cw_union_wars_finished }
				}
				STP_cw_begin_hostilities = yes
			}
			if = {
				limit = {
					tag = STS
					has_global_flag = STP_cw_started
					NOT = { has_country_flag = STP_cw_nod_entered }
					OR = {
						has_active_mission = STP_cw_nod_warning
						NOT = { has_global_flag = STP_cw_union_wars_finished }
					}
				}
				STP_cw_poll_nod_intervention = yes
			}
			# A busy foreign contract delays the offer; it does not silently choose neutrality.
			if = {
				limit = {
					tag = VAL
					VAL_cw_external_course_available = yes
					NOT = { has_country_flag = VAL_cw_offer_pending }
					NOT = { has_country_flag = VAL_cw_offer_shown }
				}
				set_country_flag = VAL_cw_offer_pending
				country_event = { id = ADISCORD_STP_cw.22 days = 1 }
			}
		}
	}
	on_weekly_STP = {
		effect = {
			if = { limit = { STP_pf_active = yes }
				STP_pf_initialize = yes
				if = { limit = { OR = {
					AND = { STP_pf_nod_contacts = yes check_variable = { var = STP_pf_nod_connected value = 0 compare = equals } }
					AND = { STP_pf_nod_contacts = no check_variable = { var = STP_pf_nod_connected value = 1 compare = equals } }
				} } STP_refresh_apparatus_loyalty = yes }
			}
			else = { STP_pf_clear = yes }
		}
	}

}
