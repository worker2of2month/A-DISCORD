on_actions = {
	on_startup = {
		effect = {
			# General history raises the contract only while creating a fresh campaign.
			# This module records its own one-shot initialization before scanning.
			if = {
				limit = {
					has_global_flag = ADISCORD_fresh_campaign_contract_v1
					NOT = { has_global_flag = ADISCORD_minor_optimization_fresh_campaign_v1 }
				}
				set_global_flag = ADISCORD_minor_optimization_fresh_campaign_v1
				every_country = {
					ADISCORD_apply_non_participating_minor_optimization = yes
					ADISCORD_economy_refresh_ai_assistance = yes
				}
			}
		}
	}

	on_war = {
		effect = {
			# Country-scoped emergency release: an organically involved minor must
			# become fully functional before it has to replace wartime losses.
			ADISCORD_release_non_participating_minor_optimization = yes
			ADISCORD_economy_refresh_ai_assistance = yes
		}
	}

	on_peace = {
		effect = {
			# War-state changes remove civil-war and retreat help immediately.
			ADISCORD_economy_refresh_ai_assistance = yes
		}
	}

	on_state_control_changed = {
		effect = {
			# State-control edges are the bounded territorial input for surrender
			# progress. Refresh only the new and former controllers; never scan.
			if = {
				limit = { ADISCORD_economy_ai_assistance_needs_edge_evaluation = yes }
				ADISCORD_economy_refresh_ai_assistance = yes
			}
			FROM = {
				if = {
					limit = { ADISCORD_economy_ai_assistance_needs_edge_evaluation = yes }
					ADISCORD_economy_refresh_ai_assistance = yes
				}
			}
		}
	}
}
