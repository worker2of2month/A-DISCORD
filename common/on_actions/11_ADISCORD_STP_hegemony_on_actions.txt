on_actions = {
	on_war_relation_added = {
		effect = {
			if = {
				limit = {
					STS = { exists = yes has_country_flag = STP_heg_kefreyt_final_started }
					ROOT = { tag = STS }
					FROM = { OR = { tag = VAL is_in_faction_with = VAL } }
				}
				FROM = { set_country_flag = STP_heg_kefreyt_final_member }
			}
			else_if = {
				limit = {
					STS = { exists = yes has_country_flag = STP_heg_kefreyt_final_started }
					FROM = { tag = STS }
					ROOT = { OR = { tag = VAL is_in_faction_with = VAL } }
				}
				ROOT = { set_country_flag = STP_heg_kefreyt_final_member }
			}
		}
	}

	on_capitulation = {
		effect = {
			if = {
				limit = {
					STS = { exists = yes has_country_flag = STP_heg_northern_final_started }
					ROOT = {
						OR = { tag = NOD tag = YPR tag = TFF }
						has_war_with = STS
					}
				}
				ROOT = { set_country_flag = STP_heg_defeated_by_sts }
			}
			if = {
				limit = {
					STS = { exists = yes has_country_flag = STP_heg_kefreyt_final_started }
					ROOT = { tag = VAL has_war_with = STS }
				}
				ROOT = { set_country_flag = STP_heg_defeated_by_sts }
			}
		}
	}
}
