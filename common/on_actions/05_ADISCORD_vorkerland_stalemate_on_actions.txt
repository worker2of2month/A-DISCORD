on_actions = {
	# ROOT is the attacking side and FROM is the defending side. This event-only
	# hook replaces recurring checks and cannot repair an already-running save.
	on_war_relation_added = {
		effect = {
			if = {
				limit = {
					ROOT = { ADISCORD_vorkerland_is_main_claimant = yes }
					FROM = { ADISCORD_vorkerland_is_central_minor = yes }
				}
				ROOT = { ADISCORD_vorkerland_schedule_central_stalemate_deadline = yes }
			}
			if = {
				limit = {
					ROOT = { tag = SOL }
					FROM = { OR = { tag = SRA tag = CSL } }
				}
				ROOT = { ADISCORD_vorkerland_schedule_solarino_stalemate_deadline = yes }
			}
			if = {
				limit = {
					ROOT = { ADISCORD_vorkerland_is_main_claimant = yes }
					FROM = { ADISCORD_vorkerland_is_main_claimant = yes }
					has_global_flag = ADISCORD_vorkerland_phase_central_showdown
					NOT = { has_global_flag = ADISCORD_vorkerland_central_initiative_selected }
					WKR = {
						exists = yes
						is_ai = yes
					}
					VAD = {
						exists = yes
						is_ai = yes
					}
					TVA = {
						exists = yes
						is_ai = yes
					}
				}

				# The central triangle is otherwise highly deterministic. A single bounded
				# operational window varies AI campaigns without scripting a winner.
				# The observed dominant claimant keeps a smaller share of the roll so the
				# two challengers can convert otherwise competitive starts into victories.
				set_global_flag = ADISCORD_vorkerland_central_initiative_selected
				random_list = {
					40 = {
						WKR = {
							set_country_flag = ADISCORD_vorkerland_central_initiative_wkr
							add_timed_idea = {
								idea = ADISCORD_vorkerland_northern_operational_initiative
								days = 210
							}
						}
					}
					20 = {
						VAD = {
							set_country_flag = ADISCORD_vorkerland_central_initiative_vad
							add_timed_idea = {
								idea = ADISCORD_vorkerland_northern_operational_initiative
								days = 210
							}
						}
					}
					40 = {
						TVA = {
							set_country_flag = ADISCORD_vorkerland_central_initiative_tva
							add_timed_idea = {
								idea = ADISCORD_vorkerland_northern_operational_initiative
								days = 210
							}
						}
					}
				}
			}
		}
	}
}
