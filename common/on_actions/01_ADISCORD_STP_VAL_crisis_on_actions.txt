on_actions = {
	on_startup = {
		effect = {
			STP = {
				ADISCORD_STP_VAL_initialize_schema = yes
				STP_refresh_crisis_modifier = yes

				if = {
					limit = {
						has_country_flag = STP_main_campaign_side
						NOT = { has_country_flag = STP_health_calendar_started }
						NOT = { has_country_flag = STP_ivanov_dead }
					}
					set_country_flag = STP_health_calendar_started
					complete_national_focus = STP_Nectar_of_the_Gods
					activate_mission = STP_health_stage_1_to_2
				}

				if = {
					limit = {
						has_power_balance = { id = STP_inner_party_opinions_bop }
					}
					remove_power_balance = { id = STP_inner_party_opinions_bop }
				}
			}
			VAL = {
				ADISCORD_STP_VAL_initialize_schema = yes
				remove_ideas = VAL_mercenary_state
				VAL_refresh_contract_modifier = yes
				if = { limit = { VAL_stp_crisis_at_least_rupture = yes } VAL_select_negotiation_posture = yes }
			}
			NOD = {
				ADISCORD_STP_VAL_initialize_schema = yes
				NOD_select_crisis_posture = yes
			}
		}
	}

	on_war_relation_added = {
		effect = {
			# ROOT is the literal attacker and FROM the literal defender here.
			# Only an illicit pre-rupture VAL attack can invoke this reaction.
			if = {
				limit = {
					ROOT = { tag = VAL }
					FROM = {
						tag = STP
						has_country_flag = STP_main_campaign_side
						check_variable = { var = STP_crisis_phase value = 2 compare = less_than }
					}
					country_exists = NOD
				}
				ROOT = {
					set_country_flag = VAL_STP_early_aggression_active
					save_global_event_target_as = STP_VAL_early_attacker
				}
				FROM = { save_global_event_target_as = STP_VAL_early_defender }
				NOD = {
					NOD_refresh_army_equipment_ratio = yes
					NOD_handle_early_val_aggression = yes
				}
			}
			if = {
				limit = {
					VAL = { has_country_flag = VAL_northern_campaign_active }
					OR = {
						ROOT = { has_country_flag = VAL_northern_campaign_participant }
						FROM = { has_country_flag = VAL_northern_campaign_participant }
					}
					NOT = {
						AND = {
							ROOT = { has_country_flag = VAL_northern_campaign_participant }
							FROM = { has_country_flag = VAL_northern_campaign_participant }
						}
					}
				}
				VAL = { VAL_mark_northern_campaign_contaminated = yes }
			}
			NOD = {
				clr_country_flag = NOD_crisis_posture_lock
				NOD_check_limited_war_integrity = yes
				NOD_select_crisis_posture = yes
			}
		}
	}

	on_peace = {
		effect = {
			NOD = {
				clr_country_flag = NOD_crisis_posture_lock
				NOD_check_limited_war_integrity = yes
				NOD_select_crisis_posture = yes
			}
			VAL = {
				if = {
					limit = {
						has_event_target = STP_postwar_country
						country_exists = event_target:STP_postwar_country
						NOT = { has_country_flag = VAL_STP_regime_change_processed }
						event_target:STP_postwar_country = { has_country_flag = STP_internal_outcome_finalized }
					}
					VAL_handle_stp_regime_change = yes
					set_country_flag = VAL_STP_regime_change_processed
					if = {
						limit = {
							has_country_flag = VAL_STP_concession_arms_debt
							NOT = { has_country_flag = VAL_STP_debt_schedule_started }
						}
						set_country_flag = VAL_STP_debt_schedule_started
						activate_mission = VAL_STP_arms_debt_day_30
						activate_mission = VAL_STP_arms_debt_day_90
						activate_mission = VAL_STP_arms_debt_day_150
					}
					if = {
						limit = { has_country_flag = VAL_STP_concession_advisers }
						remove_decision = VAL_STP_adviser_factory_obligation
						activate_mission = VAL_STP_adviser_factory_obligation
					}
				}
				VAL_check_northern_campaign_integrity = yes
			}
			if = {
				limit = {
					has_event_target = STP_crisis_party_side
					country_exists = event_target:STP_crisis_party_side
				}
				event_target:STP_crisis_party_side = {
					if = {
						limit = {
							has_country_flag = STP_crisis_party_side
							NOT = { has_country_flag = STP_internal_outcome_finalizing }
							NOT = { has_country_flag = STP_internal_outcome_finalized }
						}
						STP_try_finalize_internal_war = yes
					}
				}
			}
			if = {
				limit = {
					has_event_target = STP_crisis_resistance_side
					country_exists = event_target:STP_crisis_resistance_side
				}
				event_target:STP_crisis_resistance_side = {
					if = {
						limit = {
							has_country_flag = STP_crisis_resistance_side
							NOT = { has_country_flag = STP_internal_outcome_finalizing }
							NOT = { has_country_flag = STP_internal_outcome_finalized }
						}
						STP_try_finalize_internal_war = yes
					}
				}
			}
		}
	}

	on_capitulation = {
		effect = {
			if = {
				limit = {
					VAL = { has_country_flag = VAL_northern_campaign_active }
					ROOT = { has_country_flag = VAL_northern_campaign_participant }
				}
				set_global_flag = skip_default_capitulation
				set_global_flag = STP_VAL_skip_capitulation_claimed
				ROOT = { save_global_event_target_as = VAL_northern_capitulated_country }
				if = {
					limit = { FROM = { OR = { tag = VAL has_war_together_with = VAL } } }
					VAL = { set_country_flag = VAL_northern_capitulation_by_val_side }
				}
				VAL = { VAL_handle_northern_campaign_capitulation = yes }
			}
			# Final VAL-STP war: suppress the conference only for the exact
			# saved participants, then resolve borders and dependency by script.
			if = {
				limit = {
					VAL = {
						has_country_flag = VAL_STP_final_war_active
						has_event_target = STP_postwar_country
					}
					ROOT = { tag = event_target:STP_postwar_country }
					FROM = { OR = { tag = VAL has_war_together_with = VAL } }
				}
				set_global_flag = skip_default_capitulation
				set_global_flag = STP_VAL_skip_capitulation_claimed
				VAL = { STP_VAL_resolve_final_val_victory = yes }
			}
			else_if = {
				limit = {
					ROOT = { tag = VAL has_country_flag = VAL_STP_final_war_active }
					VAL = { has_event_target = STP_postwar_country }
					FROM = {
						OR = {
							tag = event_target:STP_postwar_country
							has_war_together_with = event_target:STP_postwar_country
						}
					}
				}
				set_global_flag = skip_default_capitulation
				set_global_flag = STP_VAL_skip_capitulation_claimed
				VAL = { STP_VAL_resolve_final_stp_victory = yes }
			}
			NOD = {
				clr_country_flag = NOD_crisis_posture_lock
				NOD_check_limited_war_integrity = yes
				NOD_select_crisis_posture = yes
			}
			VAL = {
				if = {
					limit = {
						has_country_flag = VAL_foreign_operation_active
						has_event_target = VAL_STP_operation_target
						event_target:VAL_STP_operation_target = { has_capitulated = yes }
					}
					VAL_finish_stp_operation = yes
				}
				VAL_check_northern_campaign_integrity = yes
			}
			if = {
				limit = {
					has_event_target = STP_crisis_party_side
					country_exists = event_target:STP_crisis_party_side
				}
				event_target:STP_crisis_party_side = {
					if = {
						limit = {
							has_country_flag = STP_crisis_party_side
							NOT = { has_country_flag = STP_internal_outcome_finalizing }
							NOT = { has_country_flag = STP_internal_outcome_finalized }
						}
						STP_try_finalize_internal_war = yes
					}
				}
			}
			if = {
				limit = {
					has_event_target = STP_crisis_resistance_side
					country_exists = event_target:STP_crisis_resistance_side
				}
				event_target:STP_crisis_resistance_side = {
					if = {
						limit = {
							has_country_flag = STP_crisis_resistance_side
							NOT = { has_country_flag = STP_internal_outcome_finalizing }
							NOT = { has_country_flag = STP_internal_outcome_finalized }
						}
						STP_try_finalize_internal_war = yes
					}
				}
			}
			if = {
				limit = { has_global_flag = STP_VAL_skip_capitulation_claimed }
				clr_global_flag = STP_VAL_skip_capitulation_claimed
				clr_global_flag = skip_default_capitulation
			}
		}
	}

	on_leave_faction = {
		effect = {
			NOD = {
				clr_country_flag = NOD_crisis_posture_lock
				NOD_check_limited_war_integrity = yes
				NOD_select_crisis_posture = yes
			}
		}
	}

	on_annex = {
		effect = {
			NOD = {
				clr_country_flag = NOD_crisis_posture_lock
				NOD_check_limited_war_integrity = yes
				NOD_select_crisis_posture = yes
			}
			VAL = {
				if = {
					limit = {
						has_country_flag = VAL_foreign_operation_active
						OR = {
							NOT = { has_event_target = VAL_STP_operation_target }
							NOT = { country_exists = event_target:VAL_STP_operation_target }
						}
					}
					VAL_clear_foreign_operation = yes
				}
			}
		}
	}

	on_state_control_changed = {
		effect = {
			NOD = {
				NOD_check_limited_war_control = yes
				NOD_check_limited_war_integrity = yes
			}
			VAL = { VAL_check_resource_corridor_control = yes }
			VAL = { VAL_check_northern_campaign_control = yes }
		}
	}
}
