on_actions = {
	on_startup = {
		effect = {
			if = {
				limit = {
					NOT = { has_global_flag = ADISCORD_vorkerland_collapse_scheduled }
				}
				set_global_flag = ADISCORD_vorkerland_collapse_scheduled
				if = {
					limit = { RUS = { exists = yes } }
					RUS = { ADISCORD_vorkerland_apply_dirty_modifiers = yes }
				}
				if = {
					limit = { WRK = { exists = yes } }
					WRK = {
						country_event = {
							id = ADISCORD_vorkerland_collapse.1
							days = 120
							random_days = 60
						}
					}
				}
			}
			# Save-compatible recovery for an interrupted exclusion-zone reveal.
			# The timed global guard makes this run once per load, not once per
			# country, and the spawn events themselves skip tags already present.
			if = {
				limit = {
					has_global_flag = ADISCORD_vorkerland_dirty_opened
					NOT = { has_global_flag = ADISCORD_vorkerland_dirty_cascade_complete }
					NOT = { has_global_flag = ADISCORD_vorkerland_dirty_cascade_recovery_scheduled }
					RUS = { exists = yes }
				}
				set_global_flag = {
					flag = ADISCORD_vorkerland_dirty_cascade_recovery_scheduled
					days = 15
				}
				RUS = {
					country_event = { id = ADISCORD_vorkerland_collapse.11 days = 1 }
					country_event = { id = ADISCORD_vorkerland_collapse.17 days = 3 }
					country_event = { id = ADISCORD_vorkerland_collapse.12 days = 5 }
					country_event = { id = ADISCORD_vorkerland_collapse.18 days = 7 }
					country_event = { id = ADISCORD_vorkerland_collapse.13 days = 9 }
					country_event = { id = ADISCORD_vorkerland_collapse.19 days = 11 }
				}
			}
		}
	}
	on_capitulation = {
		effect = {
			# Reserve only an intervention capitulation delivered by one of its
			# canonical sides. The late global fallback must not turn the two
			# fixed-state objectives into a broad annexation.
			if = {
				limit = {
					has_global_flag = ADISCORD_vorkerland_ivanland_intervention_active
					ROOT = { OR = { tag = IBL tag = PWR } }
					FROM = { OR = { tag = IVN has_war_together_with = IVN } }
				}
				set_global_flag = skip_default_capitulation
				if = {
					limit = { IVN = { controls_state = 90 controls_state = 91 } }
					IVN = { ADISCORD_vorkerland_ivanland_intervention_success = yes }
				}
			}
			else_if = {
				limit = {
					has_global_flag = ADISCORD_vorkerland_ivanland_intervention_active
					ROOT = { tag = IVN }
					FROM = {
						OR = {
							tag = IBL
							tag = PWR
							has_war_together_with = IBL
							has_war_together_with = PWR
						}
					}
				}
				set_global_flag = skip_default_capitulation
				IVN = { ADISCORD_vorkerland_ivanland_intervention_failure = yes }
			}
			# A third party may still settle its own regional war normally. Remove
			# IVN from that war first so the global fallback cannot award it land.
			if = {
				limit = {
					has_global_flag = ADISCORD_vorkerland_ivanland_intervention_active
					ROOT = {
						OR = { tag = IBL tag = PWR }
						has_war_with = IVN
					}
					FROM = { NOT = { OR = { tag = IVN has_war_together_with = IVN } } }
				}
				IVN = { white_peace = ROOT }
			}
			ADISCORD_vorkerland_check_central_outcome = yes
		}
	}
	on_monthly = {
		effect = {
			if = {
				limit = { OR = { tag = ROM tag = TRU tag = ZAO } }
				ADISCORD_vorkerland_sync_independence_cosmetic = yes
			}
			if = {
				limit = {
					OR = { tag = WRK tag = VAD }
					OR = {
						has_global_flag = ADISCORD_vorkerland_collapse_started
						has_variable = ADISCORD_vorkerland_civil_war_exhaustion
					}
				}
				ADISCORD_vorkerland_update_civil_war_exhaustion = yes
			}
			if = {
				limit = { tag = RUS }
				ADISCORD_vorkerland_check_central_outcome = yes
			}
			if = {
				limit = {
					tag = PIV
					NOT = { has_idea = ADISCORD_vorkerland_piv_macri_volunteer_mission }
					EBA = {
						exists = yes
						is_subject = no
						has_war = yes
						NOT = { has_capitulated = yes }
					}
				}
				add_ideas = ADISCORD_vorkerland_piv_macri_volunteer_mission
			}
			if = {
				limit = {
					tag = PIV
					has_idea = ADISCORD_vorkerland_piv_macri_volunteer_mission
					OR = {
						NOT = { country_exists = EBA }
						EBA = { has_war = no }
						EBA = { has_capitulated = yes }
					}
				}
				remove_ideas = ADISCORD_vorkerland_piv_macri_volunteer_mission
			}
		}
	}
	on_puppet = {
		effect = {
			if = {
				limit = { OR = { tag = ROM tag = TRU tag = ZAO } }
				ADISCORD_vorkerland_sync_independence_cosmetic = yes
			}
		}
	}
	on_release_as_puppet = {
		effect = {
			if = {
				limit = { OR = { tag = ROM tag = TRU tag = ZAO } }
				ADISCORD_vorkerland_sync_independence_cosmetic = yes
			}
		}
	}
	on_release_as_free = {
		effect = {
			if = {
				limit = { OR = { tag = ROM tag = TRU tag = ZAO } }
				ADISCORD_vorkerland_sync_independence_cosmetic = yes
			}
		}
	}
	on_state_control_changed = {
		effect = {
			RUS = { ADISCORD_vorkerland_check_khan_border_war = yes }
			ADISCORD_vorkerland_check_central_outcome = yes
		}
	}
}
