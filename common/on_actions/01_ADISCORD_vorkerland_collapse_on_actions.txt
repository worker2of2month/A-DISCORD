on_actions = {
	on_startup = {
		effect = {
			# Country history seeds this marker only while creating a fresh campaign.
			# Once consumed, later loads cannot schedule the collapse again.
			WRK = {
				if = {
					limit = {
						has_country_flag = ADISCORD_vorkerland_fresh_start_collapse_pending_v1
						NOT = { has_global_flag = ADISCORD_vorkerland_collapse_scheduled }
						NOT = { has_global_flag = ADISCORD_vorkerland_collapse_started }
						NOT = { has_global_flag = ADISCORD_vorkerland_collapse_finished }
					}
					clr_country_flag = ADISCORD_vorkerland_fresh_start_collapse_pending_v1
					set_global_flag = ADISCORD_vorkerland_collapse_scheduled
					if = {
						limit = { RUS = { exists = yes } }
						RUS = { ADISCORD_vorkerland_apply_dirty_modifiers = yes }
					}
					ADISCORD_vorkerland_set_phase_prewar = yes
					country_event = {
						id = ADISCORD_vorkerland_phase.1
						days = 175
						random_days = 21
					}
				}
			}
		}
	}
	on_monthly_WKR = {
		effect = {
			if = {
				limit = {
					ADISCORD_vorkerland_is_live_claimant = yes
					has_global_flag = ADISCORD_vorkerland_collapse_started
					NOT = { has_global_flag = ADISCORD_vorkerland_collapse_finished }
				}
				ADISCORD_vorkerland_refresh_war_economy_dynamic_state = yes
			}
		}
	}
	on_monthly_VAD = {
		effect = {
			if = {
				limit = {
					ADISCORD_vorkerland_is_live_claimant = yes
					has_global_flag = ADISCORD_vorkerland_collapse_started
					NOT = { has_global_flag = ADISCORD_vorkerland_collapse_finished }
				}
				ADISCORD_vorkerland_refresh_war_economy_dynamic_state = yes
			}
		}
	}
	on_monthly_TVA = {
		effect = {
			if = {
				limit = {
					ADISCORD_vorkerland_is_live_claimant = yes
					has_global_flag = ADISCORD_vorkerland_collapse_started
					NOT = { has_global_flag = ADISCORD_vorkerland_collapse_finished }
				}
				ADISCORD_vorkerland_refresh_war_economy_dynamic_state = yes
			}
		}
	}
	on_war = {
		effect = {
			if = {
				limit = {
					ADISCORD_vorkerland_is_main_claimant = yes
					NOT = { has_global_flag = ADISCORD_vorkerland_central_war_finished }
				}
				ADISCORD_vorkerland_update_civil_war_exhaustion = yes
				ADISCORD_vorkerland_reconcile_campaign_state = yes
			}
			if = {
				limit = { tag = EBA }
				PIV = {
					if = {
						limit = {
							exists = yes
							NOT = { has_idea = ADISCORD_vorkerland_piv_macri_volunteer_mission }
						}
						add_ideas = ADISCORD_vorkerland_piv_macri_volunteer_mission
					}
				}
			}
		}
	}
	on_peace = {
		effect = {
			if = {
				limit = {
					has_global_flag = ADISCORD_vorkerland_rom_northern_intervention_active
					NOT = { has_global_flag = ADISCORD_vorkerland_rom_northern_intervention_resolved }
				}
				ADISCORD_vorkerland_schedule_rom_northern_intervention_check = yes
			}
			if = {
				limit = {
					ADISCORD_vorkerland_is_main_claimant = yes
					NOT = { has_global_flag = ADISCORD_vorkerland_central_war_finished }
				}
				ADISCORD_vorkerland_update_civil_war_exhaustion = yes
				ADISCORD_vorkerland_reconcile_campaign_state = yes
			}
			if = {
				limit = { tag = EBA has_war = no }
				PIV = {
					if = {
						limit = {
							exists = yes
							has_idea = ADISCORD_vorkerland_piv_macri_volunteer_mission
						}
						remove_ideas = ADISCORD_vorkerland_piv_macri_volunteer_mission
					}
				}
			}
		}
	}
	on_capitulation = {
		effect = {
			# A peace conference can materialise dormant WRK as a free or
			# puppeted shell. Reserve it from the generic annex fallback and
			# fold its land back into a living claimant the same tick.
			if = {
				limit = { ROOT = { ADISCORD_vorkerland_is_premature_wrk = yes } }
				set_global_flag = skip_default_capitulation
				set_global_flag = ADISCORD_vorkerland_premature_wrk_release_intercepted_v1
				if = {
					limit = {
						FROM = {
							ADISCORD_vorkerland_is_main_claimant = yes
							exists = yes
							is_subject = no
							NOT = { has_capitulated = yes }
						}
					}
					FROM = { ADISCORD_vorkerland_dissolve_premature_wrk_as_claimant = yes }
				}
				else = {
					ADISCORD_vorkerland_route_premature_wrk_release_to_living_claimant = yes
				}
			}
			if = {
				limit = {
					has_global_flag = ADISCORD_vorkerland_rom_northern_intervention_active
					NOT = { has_global_flag = ADISCORD_vorkerland_rom_northern_intervention_resolved }
					ROOT = { tag = ROM }
				}
				ADISCORD_vorkerland_rom_northern_intervention_failure = yes
			}
			else_if = {
				limit = {
					has_global_flag = ADISCORD_vorkerland_rom_northern_intervention_active
					NOT = { has_global_flag = ADISCORD_vorkerland_rom_northern_intervention_resolved }
				}
				ADISCORD_vorkerland_schedule_rom_northern_intervention_check = yes
			}
			if = {
				limit = {
					ROOT = { ADISCORD_vorkerland_is_main_claimant = yes }
					NOT = { has_global_flag = ADISCORD_vorkerland_central_war_finished }
				}
				ROOT = { ADISCORD_vorkerland_update_civil_war_exhaustion = yes }
			}
			if = {
				limit = { ROOT = { tag = EBA } }
				PIV = {
					if = {
						limit = {
							exists = yes
							has_idea = ADISCORD_vorkerland_piv_macri_volunteer_mission
						}
						remove_ideas = ADISCORD_vorkerland_piv_macri_volunteer_mission
					}
				}
			}
			# Volnograd rejected WRK only because Worker was confirmed dead. A WRK
			# victory ends the enforcement war immediately and installs the same
			# special district autonomy used before the collapse.
			if = {
				limit = {
					ROOT = { tag = VLA has_country_flag = ADISCORD_vorkerland_wrk_protectorate_enforcement }
					FROM = { tag = WRK }
				}
				set_global_flag = skip_default_capitulation
				FROM = { ADISCORD_vorkerland_wrk_complete_protectorate_enforcement = yes }
			}
			# No northern-front member may be consumed by the normal peace fallback.
			# IVN must physically control the complete 90/91/93/94 objective package;
			# the postwar client order is materialized only after this succeeds.
			if = {
				limit = {
					has_global_flag = ADISCORD_vorkerland_ivanland_intervention_active
					ROOT = { OR = { tag = ZAO tag = WPA tag = WPS tag = PWR tag = PSD } }
					FROM = { OR = { tag = IVN has_war_together_with = IVN } }
				}
				set_global_flag = skip_default_capitulation
				if = {
					limit = {
						IVN = {
							controls_state = 90
							controls_state = 91
							controls_state = 93
							controls_state = 94
						}
					}
					IVN = { ADISCORD_vorkerland_ivanland_intervention_success = yes }
				}
			}
			else_if = {
				limit = {
					has_global_flag = ADISCORD_vorkerland_ivanland_intervention_active
					ROOT = { tag = IVN }
					FROM = {
						OR = {
							tag = ZAO
							tag = WPA
							tag = WPS
							tag = PWR
							tag = PSD
							has_war_together_with = PWR
						}
					}
				}
				set_global_flag = skip_default_capitulation
				IVN = { ADISCORD_vorkerland_ivanland_intervention_failure = yes }
			}
			# A third party may still settle its own regional war normally. Remove
			# IVN from that war first so the global fallback cannot award it land.
			if = {
				limit = {
					has_global_flag = ADISCORD_vorkerland_ivanland_intervention_active
					ROOT = {
						OR = { tag = ZAO tag = WPA tag = WPS tag = PWR tag = PSD }
						has_war_with = IVN
					}
					FROM = { NOT = { OR = { tag = IVN has_war_together_with = IVN } } }
				}
				IVN = { white_peace = ROOT }
			}
			# Reserve every central defeat from the generic same-tick fallback, even
			# when the engine credits a sovereign/subject co-belligerent. Normalise the
			# settlement to that ally's live main claimant before preserving the shell.
			if = {
				limit = {
					has_global_flag = ADISCORD_vorkerland_collapse_wars_started
					OR = {
						NOT = { has_global_flag = ADISCORD_vorkerland_central_war_finished }
						AND = {
							FROM = { tag = WRK }
							has_global_flag = ADISCORD_vorkerland_phase_postwar_integration
							has_global_flag = ADISCORD_vorkerland_reunification_verified
						}
					}
					ROOT = { ADISCORD_vorkerland_is_main_claimant = yes }
					FROM = {
						OR = {
							AND = {
								ADISCORD_vorkerland_is_main_claimant = yes
								NOT = { has_capitulated = yes }
							}
							AND = {
								is_subject = yes
								NOT = { has_capitulated = yes }
								OVERLORD = {
									exists = yes
									ADISCORD_vorkerland_is_main_claimant = yes
									NOT = { has_capitulated = yes }
								}
							}
							AND = {
								tag = WTD
								has_war_together_with = TVA
								TVA = { exists = yes NOT = { has_capitulated = yes } }
							}
							AND = {
								tag = VLA
								has_war_together_with = WKR
								WKR = { exists = yes NOT = { has_capitulated = yes } }
							}
							AND = {
								tag = SOL
								has_war_together_with = VAD
								VAD = { exists = yes NOT = { has_capitulated = yes } }
							}
							AND = {
								tag = WRK
								has_global_flag = ADISCORD_vorkerland_phase_postwar_integration
								has_global_flag = ADISCORD_vorkerland_reunification_verified
							}
						}
					}
				}
				set_global_flag = skip_default_capitulation
				# Keep occupation gains inside the central claimant graph before the
				# defeated shell is reserved from the generic peace fallback.
				ROOT = { ADISCORD_vorkerland_distribute_central_states_to_cobelligerents = yes }
				FROM = { ADISCORD_vorkerland_story_offer_first_claimant_command_choice = yes }
				if = {
					limit = { FROM = { tag = WTD } }
					TVA = { ADISCORD_vorkerland_settle_central_capitulation = yes }
				}
				else_if = {
					limit = { FROM = { tag = VLA } }
					WKR = { ADISCORD_vorkerland_settle_central_capitulation = yes }
				}
				else_if = {
					limit = { FROM = { tag = SOL } }
					VAD = { ADISCORD_vorkerland_settle_central_capitulation = yes }
				}
				else = {
					FROM = {
						if = {
							limit = {
								tag = WRK
								has_global_flag = ADISCORD_vorkerland_phase_postwar_integration
								has_global_flag = ADISCORD_vorkerland_reunification_verified
							}
							ADISCORD_vorkerland_settle_central_capitulation = yes
						}
						else_if = {
							limit = { is_subject = yes }
							OVERLORD = { ADISCORD_vorkerland_settle_central_capitulation = yes }
						}
						else = { ADISCORD_vorkerland_settle_central_capitulation = yes }
					}
				}
			}
			ADISCORD_vorkerland_check_central_outcome = yes
		}
	}
	on_puppet = {
		effect = {
			if = {
				limit = { ADISCORD_vorkerland_release_requires_interception = yes }
				ADISCORD_vorkerland_intercept_premature_wrk_release = yes
			}
			else = {
				ADISCORD_vorkerland_sync_republics_from_ruins = yes
				if = {
					limit = { tag = VLA is_subject_of = WKR }
					WKR = { ADISCORD_vorkerland_wrk_activate_vla_auxiliaries = yes }
				}
				else_if = {
					limit = { tag = VLA is_subject_of = WRK }
					WRK = { ADISCORD_vorkerland_wrk_activate_vla_auxiliaries = yes }
				}
				if = {
					limit = { OR = { tag = ROM tag = TRU tag = ZAO tag = SOL } }
					ADISCORD_vorkerland_sync_independence_cosmetic = yes
				}
			}
		}
	}
	on_release_as_puppet = {

	on_state_control_changed = {
		effect = {
			# Iconic objectives. Each fires once per campaign; the global array is
			# the guard, so the literal id in the limit must match the id passed to
			# the effect. Tier 1 are the four claimant capitals and the Unity Tower,
			# tier 2 the industrial and port centres, tier 3 the contested hinges.
			if = {
				limit = { FROM.FROM = { state = 32 } ADISCORD_vorkerland_objective_transfer_is_valid = yes NOT = { is_in_array = { array = global.ADISCORD_vorkerland_objectives_taken value = 1 } } }
				set_temp_variable = { ADISCORD_vorkerland_objective_id = 1 }
				set_temp_variable = { ADISCORD_vorkerland_objective_value = 0.15 }
				ADISCORD_vorkerland_resolve_iconic_objective = yes
			}
			if = {
				limit = { FROM.FROM = { state = 75 } ADISCORD_vorkerland_objective_transfer_is_valid = yes NOT = { is_in_array = { array = global.ADISCORD_vorkerland_objectives_taken value = 2 } } }
				set_temp_variable = { ADISCORD_vorkerland_objective_id = 2 }
				set_temp_variable = { ADISCORD_vorkerland_objective_value = 0.15 }
				ADISCORD_vorkerland_resolve_iconic_objective = yes
			}
			if = {
				limit = { FROM.FROM = { state = 36 } ADISCORD_vorkerland_objective_transfer_is_valid = yes NOT = { is_in_array = { array = global.ADISCORD_vorkerland_objectives_taken value = 3 } } }
				set_temp_variable = { ADISCORD_vorkerland_objective_id = 3 }
				set_temp_variable = { ADISCORD_vorkerland_objective_value = 0.15 }
				ADISCORD_vorkerland_resolve_iconic_objective = yes
			}
			if = {
				limit = { FROM.FROM = { state = 40 } ADISCORD_vorkerland_objective_transfer_is_valid = yes NOT = { is_in_array = { array = global.ADISCORD_vorkerland_objectives_taken value = 4 } } }
				set_temp_variable = { ADISCORD_vorkerland_objective_id = 4 }
				set_temp_variable = { ADISCORD_vorkerland_objective_value = 0.15 }
				ADISCORD_vorkerland_resolve_iconic_objective = yes
			}
			if = {
				limit = { FROM.FROM = { state = 102 } ADISCORD_vorkerland_objective_transfer_is_valid = yes NOT = { is_in_array = { array = global.ADISCORD_vorkerland_objectives_taken value = 5 } } }
				set_temp_variable = { ADISCORD_vorkerland_objective_id = 5 }
				set_temp_variable = { ADISCORD_vorkerland_objective_value = 0.10 }
				ADISCORD_vorkerland_resolve_iconic_objective = yes
			}
			if = {
				limit = { FROM.FROM = { state = 33 } ADISCORD_vorkerland_objective_transfer_is_valid = yes NOT = { is_in_array = { array = global.ADISCORD_vorkerland_objectives_taken value = 6 } } }
				set_temp_variable = { ADISCORD_vorkerland_objective_id = 6 }
				set_temp_variable = { ADISCORD_vorkerland_objective_value = 0.10 }
				ADISCORD_vorkerland_resolve_iconic_objective = yes
			}
			if = {
				limit = { FROM.FROM = { state = 35 } ADISCORD_vorkerland_objective_transfer_is_valid = yes NOT = { is_in_array = { array = global.ADISCORD_vorkerland_objectives_taken value = 7 } } }
				set_temp_variable = { ADISCORD_vorkerland_objective_id = 7 }
				set_temp_variable = { ADISCORD_vorkerland_objective_value = 0.10 }
				ADISCORD_vorkerland_resolve_iconic_objective = yes
			}
			if = {
				limit = { FROM.FROM = { state = 39 } ADISCORD_vorkerland_objective_transfer_is_valid = yes NOT = { is_in_array = { array = global.ADISCORD_vorkerland_objectives_taken value = 8 } } }
				set_temp_variable = { ADISCORD_vorkerland_objective_id = 8 }
				set_temp_variable = { ADISCORD_vorkerland_objective_value = 0.10 }
				ADISCORD_vorkerland_resolve_iconic_objective = yes
			}
			if = {
				limit = { FROM.FROM = { state = 106 } ADISCORD_vorkerland_objective_transfer_is_valid = yes NOT = { is_in_array = { array = global.ADISCORD_vorkerland_objectives_taken value = 9 } } }
				set_temp_variable = { ADISCORD_vorkerland_objective_id = 9 }
				set_temp_variable = { ADISCORD_vorkerland_objective_value = 0.10 }
				ADISCORD_vorkerland_resolve_iconic_objective = yes
			}
			if = {
				limit = { FROM.FROM = { state = 81 } ADISCORD_vorkerland_objective_transfer_is_valid = yes NOT = { is_in_array = { array = global.ADISCORD_vorkerland_objectives_taken value = 10 } } }
				set_temp_variable = { ADISCORD_vorkerland_objective_id = 10 }
				set_temp_variable = { ADISCORD_vorkerland_objective_value = 0.06 }
				ADISCORD_vorkerland_resolve_iconic_objective = yes
			}
			if = {
				limit = { FROM.FROM = { state = 121 } ADISCORD_vorkerland_objective_transfer_is_valid = yes NOT = { is_in_array = { array = global.ADISCORD_vorkerland_objectives_taken value = 11 } } }
				set_temp_variable = { ADISCORD_vorkerland_objective_id = 11 }
				set_temp_variable = { ADISCORD_vorkerland_objective_value = 0.06 }
				ADISCORD_vorkerland_resolve_iconic_objective = yes
			}
			if = {
				limit = { FROM.FROM = { state = 104 } ADISCORD_vorkerland_objective_transfer_is_valid = yes NOT = { is_in_array = { array = global.ADISCORD_vorkerland_objectives_taken value = 12 } } }
				set_temp_variable = { ADISCORD_vorkerland_objective_id = 12 }
				set_temp_variable = { ADISCORD_vorkerland_objective_value = 0.06 }
				ADISCORD_vorkerland_resolve_iconic_objective = yes
			}
			# Monthly is too slow for a comeback mechanic: a claimant can be pushed
			# past the threshold and finished off inside one month.
			ROOT = { ADISCORD_vorkerland_refresh_organized_defence = yes }
			FROM = { ADISCORD_vorkerland_refresh_organized_defence = yes }
			if = {
				limit = {
					OR = {
						FROM.FROM = { state = 32 }
						FROM.FROM = { state = 33 }
						FROM.FROM = { state = 35 }
						FROM.FROM = { state = 36 }
						FROM.FROM = { state = 37 }
						FROM.FROM = { state = 38 }
						FROM.FROM = { state = 39 }
						FROM.FROM = { state = 40 }
						FROM.FROM = { state = 75 }
						FROM.FROM = { state = 81 }
						FROM.FROM = { state = 102 }
						FROM.FROM = { state = 104 }
						FROM.FROM = { state = 106 }
						FROM.FROM = { state = 121 }
						FROM.FROM = { state = 122 }
						FROM.FROM = { state = 123 }
						FROM.FROM = { state = 124 }
					}
				}
				ROOT = {
					if = {
						limit = { ADISCORD_vorkerland_is_main_claimant = yes }
						ADISCORD_vorkerland_recount_central_control = yes
					}
				}
				FROM = {
					if = {
						limit = { ADISCORD_vorkerland_is_main_claimant = yes }
						ADISCORD_vorkerland_recount_central_control = yes
					}
				}
				ADISCORD_vorkerland_refresh_legitimacy_leader = yes
				if = { limit = { country_exists = WKR } WKR = { ADISCORD_vorkerland_refresh_claimant_coalition = yes } }
				if = { limit = { country_exists = VAD } VAD = { ADISCORD_vorkerland_refresh_claimant_coalition = yes } }
				if = { limit = { country_exists = TVA } TVA = { ADISCORD_vorkerland_refresh_claimant_coalition = yes } }
			}
			ADISCORD_vorkerland_story_check_first_claimant_capital_fall = yes
			RUS = { ADISCORD_vorkerland_check_khan_border_war = yes }
			if = {
				limit = {
					has_global_flag = ADISCORD_vorkerland_ivanland_intervention_active
					IVN = {
						controls_state = 90
						controls_state = 91
						controls_state = 93
						controls_state = 94
					}
				}
				IVN = { ADISCORD_vorkerland_ivanland_intervention_success = yes }
			}
			if = {
				limit = {
					has_global_flag = ADISCORD_vorkerland_rom_northern_intervention_active
					NOT = { has_global_flag = ADISCORD_vorkerland_rom_northern_intervention_resolved }
					ROM = {
						exists = yes
						controls_state = 72
						controls_state = 196
						controls_state = 322
					}
				}
				ADISCORD_vorkerland_rom_northern_intervention_success = yes
			}
			else_if = {
				limit = {
					has_global_flag = ADISCORD_vorkerland_rom_northern_intervention_active
					NOT = { has_global_flag = ADISCORD_vorkerland_rom_northern_intervention_resolved }
				}
				ADISCORD_vorkerland_schedule_rom_northern_intervention_check = yes
			}
			ADISCORD_vorkerland_check_central_outcome = yes
		}
	}
