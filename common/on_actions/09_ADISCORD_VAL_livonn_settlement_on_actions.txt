on_actions = {
	# 02_ADISCORD_STP_on_actions.txt resolves the SRP military result first.
	# This later hook completes the territorial settlement by taking Livonn.
	on_capitulation = {
		effect = {
			if = {
				limit = {
					has_global_flag = STP_cw_started
					ROOT = { tag = SRP }
					VAL = {
						has_country_flag = VAL_cw_postwar
						NOT = { has_country_flag = VAL_cw_livonn_settlement_resolved }
					}
				}
				VAL = { VAL_cw_stage_livonn_settlement = yes }
			}
		}
	}

	# Save/load and on_action-order fallback. It is inert after the one-shot
	# staged/resolved flags have been written.
	on_weekly_VAL = {
		effect = {
			if = {
				limit = {
					has_country_flag = VAL_cw_postwar
					NOT = { has_country_flag = VAL_cw_livonn_settlement_staged }
					NOT = { has_country_flag = VAL_cw_livonn_settlement_resolved }
				}
				VAL_cw_stage_livonn_settlement = yes
			}
		}
	}
}
