on_actions = {
	on_startup = {
		effect = {
			OCA = { VAL_initialize_balchansk_administration = yes }
			if = { limit = { VAL = { exists = yes } } VAL = { VAL_initialize_command_balance = yes VAL_initialize_arsenal_recovery = yes VAL_grant_contract_warlord = yes VAL_call_subjects_to_wars = yes VAL_check_nam_concession = yes } }
			if = {
				limit = {
					has_global_flag = ADISCORD_fresh_campaign_contract_v1
					NOT = { has_global_flag = ADISCORD_val_rework_fresh_campaign_v1 }
					VAL = { exists = yes }
				}
				set_global_flag = ADISCORD_val_rework_fresh_campaign_v1
				VAL = { VAL_initialize_rework = yes }
			}
		}
	}

	on_war_relation_added = {
		effect = {
			if = { limit = { OR = { ROOT = { tag = VAL } FROM = { tag = VAL } } } VAL = { VAL_check_nam_concession = yes } }
			if = {
				limit = { OR = { ROOT = { tag = VAL } FROM = { tag = VAL } } }
				VAL = { VAL_call_subjects_to_wars = yes }
			}
		}
	}

	on_puppet = {
		effect = {
			if = { limit = { FROM = { tag = VAL } } VAL = { VAL_call_subjects_to_wars = yes } OCA = { VAL_reconcile_balchansk_income = yes } }
		}
	}

	on_monthly = {
		effect = {
			if = {
				limit = { tag = VAL has_country_flag = VAL_quarterly_contract_active }
				add_to_variable = { var = VAL_quarterly_contract_months_elapsed value = 1 }
				if = {
					limit = {
						is_ai = yes
						check_variable = { var = VAL_quarterly_contract_months_elapsed value = 2 compare = greater_than_or_equals }
						NOT = { has_country_flag = VAL_quarterly_contract_paid }
						NOT = { ADISCORD_has_4000_infantry_equipment = yes }
						NOT = { has_country_flag = VAL_ai_emergency_quartermaster_used }
					}
					add_equipment_to_stockpile = { type = infantry_equipment amount = 800 producer = VAL }
					set_country_flag = VAL_ai_emergency_quartermaster_used
				}
			}
		}
	}

	on_state_control_changed = {
		effect = {
			# ROOT is the new controller, FROM the previous controller, FROM.FROM the state.
			if = {
				limit = {
					FROM = { tag = VAL has_power_balance = { id = VAL_command_balance } has_war_with = ROOT }
					FROM.FROM = { is_owned_by = VAL is_core_of = VAL NOT = { has_state_flag = VAL_bop_core_lost } }
				}
				FROM = { add_power_balance_value = { id = VAL_command_balance value = -0.10 } }
				FROM.FROM = { set_state_flag = VAL_bop_core_lost }
			}
			else_if = {
				limit = { ROOT = { tag = VAL } FROM.FROM = { has_state_flag = VAL_bop_core_lost } }
				ROOT = {
					if = { limit = { has_power_balance = { id = VAL_command_balance } } add_power_balance_value = { id = VAL_command_balance value = 0.05 } }
				}
				FROM.FROM = { clr_state_flag = VAL_bop_core_lost }
			}

            if = {
                limit = { FROM.FROM = { OR = { state = 44 state = 88 } } OCA = { exists = yes } }
                OCA = { VAL_reconcile_balchansk_income = yes }
            }
			VAL = { VAL_check_nam_concession = yes }
			if = { limit = { VAL = { exists = yes has_country_flag = VAL_vorkerland_resource_access_active } 33 = { NOT = { is_controlled_by = WRK } } } VAL = { VAL_end_vorkerland_resource_access = yes } }
			if = {
				limit = {
					VAL = {
						exists = yes
						has_country_flag = VAL_westerholm_resource_rights_active
					}
					38 = { NOT = { controller = { tag = WRK } } }
				}
				VAL = { VAL_handle_westerholm_loss = yes }
			}
		}
	}

    on_weekly_VAL = {
        effect = {
            VAL_reconcile_supply_crisis = yes
            if = { limit = { has_idea = VAL_northern_volunteer_mandate } VAL_reconcile_northern_volunteers = yes } if = { limit = { check_variable = { var = VAL_frontier_stage value = 0 compare = greater_than } } VAL_frontier_weekly = yes } }
    }

}
