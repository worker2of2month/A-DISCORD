on_actions = {
	on_startup = {
		effect = {
			if = {
				limit = {
					has_global_flag = ADISCORD_fresh_campaign_contract_v1
					NOT = { has_global_flag = ADISCORD_val_rework_fresh_campaign_v1 }
					VAL = { exists = yes }
				}
				set_global_flag = ADISCORD_val_rework_fresh_campaign_v1
				VAL = { VAL_initialize_rework = yes }
			}
		}
	}

	on_monthly = {
		effect = {
			if = {
				limit = { tag = VAL has_country_flag = VAL_quarterly_contract_active }
				add_to_variable = { var = VAL_quarterly_contract_months_elapsed value = 1 }
				if = {
					limit = {
						is_ai = yes
						check_variable = { var = VAL_quarterly_contract_months_elapsed value = 2 compare = greater_than_or_equals }
						NOT = { has_country_flag = VAL_quarterly_contract_paid }
						NOT = { ADISCORD_has_4000_infantry_equipment = yes }
						NOT = { has_country_flag = VAL_ai_emergency_quartermaster_used }
					}
					add_equipment_to_stockpile = { type = infantry_equipment amount = 800 producer = VAL }
					set_country_flag = VAL_ai_emergency_quartermaster_used
				}
			}
		}
	}

	on_state_control_changed = {
		effect = {
			if = {
				limit = {
					VAL = {
						exists = yes
						has_country_flag = VAL_westerholm_resource_rights_active
					}
					202 = { NOT = { controller = { tag = WRK } } }
				}
				VAL = { VAL_handle_westerholm_loss = yes }
			}
		}
	}

    on_weekly_VAL = {
        effect = {
            VAL_reconcile_supply_crisis = yes
            if = { limit = { has_idea = VAL_northern_volunteer_mandate } VAL_reconcile_northern_volunteers = yes } if = { limit = { check_variable = { var = VAL_frontier_stage value = 0 compare = greater_than } } VAL_frontier_weekly = yes } }
    }
    on_capitulation_immediate = {
        effect = {
            if = { limit = { tag = VAL } clr_country_flag = VAL_occidian_settlement_pending }
            clr_country_flag = VAL_frontier_capitulation_pending
            if = {
                limit = {
                    VAL = { check_variable = { var = VAL_frontier_stage value = 3 compare = equals } }
                    OR = { AND = { tag = CIN OR = { has_country_flag = VAL_frontier_member VAL = { check_variable = { var = VAL_frontier_target value = 1 compare = equals } } } has_war_with = VAL capital_scope = { is_controlled_by = VAL } }
                        AND = { tag = OSF OR = { has_country_flag = VAL_frontier_member VAL = { check_variable = { var = VAL_frontier_target value = 2 compare = equals } } } has_war_with = VAL capital_scope = { is_controlled_by = VAL } }
                        AND = { tag = APH OR = { has_country_flag = VAL_frontier_member VAL = { check_variable = { var = VAL_frontier_target value = 3 compare = equals } } } has_war_with = VAL capital_scope = { is_controlled_by = VAL } }
                        AND = { tag = ERT VAL = { check_variable = { var = VAL_frontier_target value = 4 compare = equals } has_war_with = ERT } capital_scope = { is_controlled_by = VAL } }
                        AND = { OR = { tag = NOD tag = STP } has_country_flag = VAL_frontier_guarantor has_war_with = VAL capital_scope = { is_controlled_by = VAL } }
                        AND = { tag = VAL capital_scope = { OR = { controller = { has_country_flag = VAL_frontier_guarantor has_war_with = VAL } controller = { has_war_with = VAL OR = { has_country_flag = VAL_frontier_member AND = { tag = CIN VAL = { check_variable = { var = VAL_frontier_target value = 1 compare = equals } } } AND = { tag = OSF VAL = { check_variable = { var = VAL_frontier_target value = 2 compare = equals } } } AND = { tag = APH VAL = { check_variable = { var = VAL_frontier_target value = 3 compare = equals } } } AND = { tag = ERT VAL = { check_variable = { var = VAL_frontier_target value = 4 compare = equals } } } } } } } } }
                }
                if = { limit = { OR = { tag = CIN tag = OSF tag = APH tag = ERT } } set_country_flag = VAL_frontier_defeated }
                set_country_flag = VAL_frontier_capitulation_pending
                set_global_flag = skip_default_capitulation
                if = { limit = { tag = VAL } VAL_frontier_settle_defeat = yes }
                else = { VAL = { VAL_frontier_weekly = yes } }
                clr_global_flag = skip_default_capitulation
            }
            else_if = {
                limit = { tag = VAL check_variable = { var = VAL_frontier_stage value = 0 compare = greater_than } }
                # An unrelated winner keeps its own capitulation route.
                VAL_frontier_close = yes
            }
            else_if = {
                limit = { OR = { tag = CIN tag = OSF tag = APH } has_country_flag = VAL_frontier_member }
                # A third-party victory is outside VAL's territorial settlement.
                clr_country_flag = VAL_frontier_member
                clr_country_flag = VAL_frontier_defeated
                if = { limit = { has_country_flag = VAL_frontier_added_major } set_major = no clr_country_flag = VAL_frontier_added_major }
            }
            else_if = {
                limit = { OR = { tag = NOD tag = STP } has_country_flag = VAL_frontier_guarantor }
                clr_country_flag = VAL_frontier_guarantor
                if = { limit = { has_country_flag = VAL_frontier_added_major } set_major = no clr_country_flag = VAL_frontier_added_major }
            }
        }
    }
    on_capitulation = {
        effect = {
            if = {
                limit = { has_country_flag = VAL_frontier_capitulation_pending }
                clr_country_flag = VAL_frontier_capitulation_pending
                set_global_flag = skip_default_capitulation
            }
        }
    }

}
