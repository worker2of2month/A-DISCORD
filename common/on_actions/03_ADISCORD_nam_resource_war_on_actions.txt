					}

							state = 720
							state = 721
							state = 722
							state = 701
						}
					}
				}
				NAM = { NAM_rw_queue_front_check = yes }
			}
			# Any prepared border district may rebuild its line after recapture.
			if = {
				limit = {
					FROM = { tag = NAM }
					FROM.FROM = { has_state_flag = NAM_line_prepared }
				}
				FROM.FROM = { clr_state_flag = NAM_line_prepared }
			}
			# Oil throughput counts controlled fields, the pipeline end and the
			# sea terminal; the island steel concession needs Svetlogorsk control.

				limit = {
					FROM.FROM = {
						OR = {
							state = 67
							state = 688
							state = 689
							state = 690
							state = 719
							state = 700
						}
					}
				}
				NAM = { NAM_oil_refresh = yes }
			}
			if = {
				limit = {
					FROM = { tag = NAM }
					FROM.FROM = { state = 231 }
				}
				VAL = { NAM_steel_concession_validate = yes }
				GLP = { NAM_steel_concession_validate = yes }
			}
		}
