on_actions = {
	on_state_control_changed = {
		effect = {
			if = {
				limit = {
					OR = {
						ROOT = { tag = NAM }
						FROM = { tag = NAM }
					}
				}
				NAM = { NAM_invalidate_lost_compact = yes }
			}
			if = {
				limit = {
					ADISCORD_nam_resource_war_active = yes
					OR = {
						ROOT = { tag = NAM }
						FROM = { tag = NAM }
					}
				}
				NAM = { ADISCORD_nam_refresh_emergency_defence = yes }
			}
		}
	}
	# Pillars of power drift once a month; the hook serves one tag only.
	on_monthly_NAM = {
		effect = {
			NAM_groups_monthly = yes
			NAM_crisis_draw = yes
		}
	}
}
