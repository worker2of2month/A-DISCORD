on_actions = {
	on_war_relation_added = {
		effect = {
			if = {
				limit = {
					tag = NAM
					has_dynamic_modifier = { modifier = NAM_seal_discipline_dynamic }
				}
				FROM = { NAM_seal_record_war_entry = yes }
			}
			else_if = {
				limit = {
					FROM = {
						tag = NAM
						has_dynamic_modifier = { modifier = NAM_seal_discipline_dynamic }
					}
				}
				NAM_seal_record_war_entry = yes
			}
		}
	}

	on_daily_NAM = {
		effect = {
			# The post-war depot board replaces the spirit: its drain follows the
			# private share and never takes more than the stock. The countdown of
			# the owners' round is refreshed here for the board report only.
			if = {
				limit = { has_country_flag = NAM_depot_board_seeded }
				if = {
					limit = { has_country_flag = NAM_depot_board_open }
					set_variable = { var = NAM_depot_days_left value = NAM_depot_next_round_day }
					subtract_from_variable = { var = NAM_depot_days_left value = global.num_days }
				}
				if = {
					limit = { check_variable = { NAM_depot_daily_drain > 0 } }
					set_temp_variable = { var = NAM_depot_drain_today value = fuel_k }
					multiply_temp_variable = { var = NAM_depot_drain_today value = 1000 }
					clamp_temp_variable = {
						var = NAM_depot_drain_today
						min = 0
						max = NAM_depot_daily_drain
					}
					multiply_temp_variable = { var = NAM_depot_drain_today value = -1 }
					add_fuel = NAM_depot_drain_today
				}
			}
			else_if = {
				limit = { has_idea = NAM_private_tank_farms }
				if = {
					limit = { has_country_flag = { flag = NAM_depot_case value > 1 } }
					if = {
						limit = { has_fuel < 1000 }
						set_fuel = 0
					}
					else = { add_fuel = -1000 }
				}
				else_if = {
					limit = { has_fuel < 2000 }
					set_fuel = 0
				}
				else = { add_fuel = -2000 }
			}
		}
	}

	on_puppet = {
		effect = {
			NAM_sync_neighbor_governor = yes
			NAM_board_subject_change = yes
		}
	}

	on_release_as_puppet = {
		effect = {
			if = {
				limit = { tag = NAM }
				NAM_interregnum_close = yes
			}
			NAM_sync_neighbor_governor = yes
			NAM_board_subject_change = yes
		}
	}

	on_subject_free = {
		effect = {
			NAM_sync_neighbor_governor = yes
			NAM_board_subject_change = yes
		}
	}

	on_release_as_free = {
		effect = {
			NAM_sync_neighbor_governor = yes
			NAM_board_subject_change = yes
		}
	}

	on_liberate = {
		effect = {
			NAM_board_subject_change = yes
		}
	}

	on_subject_autonomy_level_change = {
		effect = {
			if = {
				limit = {
					tag = NAM
					is_subject = yes
				}
				NAM_interregnum_close = yes
			}
			NAM_board_subject_change = yes
		}
	}

	on_ruling_party_change = {
		effect = { NAM_sync_neighbor_governor = yes }
	}

	on_state_control_changed = {
		effect = {
			if = {
				limit = {
					NAM_neighbour_is_command = yes
					FROM.FROM = { is_owned_by = ROOT }
				}
				NAM_neighbour_bind_government = yes
				FROM.FROM = {
					if = {
						limit = { is_core_of = event_target:NAM_neighbour_resistance }
						NAM_neighbour_apply_district_charter = yes
					}
				}
			}
			if = {
				limit = {
					OR = {
						ROOT = { tag = NAM }
						FROM = { tag = NAM }
					}
				}
				NAM = { NAM_invalidate_lost_compact = yes }
			}
			if = {
				limit = {
					ADISCORD_nam_resource_war_active = yes
					OR = {
						ROOT = { tag = NAM }
						FROM = { tag = NAM }
					}
				}
				NAM = { ADISCORD_nam_refresh_emergency_defence = yes }
			}
			# The decisive ground of either side queues one delayed front check.
			if = {
				limit = {
					ADISCORD_nam_resource_war_active = yes
					OR = {
						ROOT = { tag = NAM }
						FROM = { tag = NAM }
					}
					FROM.FROM = {
						OR = {
							state = 67
							state = 688
							state = 689
							state = 690
							state = 719
							state = 720
							state = 721
							state = 722
							state = 701
						}
					}
				}
				NAM = { NAM_rw_queue_front_check = yes }
			}
			# Any prepared border district may rebuild its line after recapture.
			if = {
				limit = {
					FROM = { tag = NAM }
					FROM.FROM = { has_state_flag = NAM_line_prepared }
				}
				FROM.FROM = { clr_state_flag = NAM_line_prepared }
			}
			# Extraction depends on the fields and pipeline end. Contract berths use the
			# sea terminal; the island steel concession needs Svetlogorsk control.
			if = {
				limit = {
					FROM.FROM = {
						OR = {
							state = 67
							state = 688
							state = 689
							state = 690
							state = 719
							state = 700
						}
					}
				}
				NAM = { NAM_oil_refresh = yes }
			}
			if = {
				limit = {
					FROM = { tag = NAM }
					FROM.FROM = { state = 231 }
				}
				VAL = { NAM_steel_concession_validate = yes }
				GLP = { NAM_steel_concession_validate = yes }
			}
		}
	}
	# Pillars, incidents, the colonial concession, the restitution clock,
	# deliveries, seal discipline, oil contracts, the island levy, the
	# underground, the rivals, the junta's piracy and the humanists' free zones
	# advance once a month; the hook serves one tag and reads fixed tags and the
	# seven island states only.
	on_monthly_NAM = {
		effect = {
			NAM_groups_monthly = yes
			NAM_crisis_draw = yes
			NAM_colony_monthly = yes
			NAM_rw_monthly = yes
			NAM_underground_monthly = yes
			NAM_ladder_monthly = yes
			NAM_corp_monthly = yes
			NAM_consolidation_monthly = yes
		}
	}

	on_monthly_EFL = {
		effect = { NAM_rw_fill_tribute_tanks = yes }
	}

	on_monthly_AZH = {
		effect = { NAM_rw_fill_tribute_tanks = yes }
	}
}
