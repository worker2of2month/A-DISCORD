					}
				}
				NAM = { NAM_invalidate_lost_compact = yes }
			}

				limit = {

					}
				}
				NAM = { ADISCORD_nam_refresh_emergency_defence = yes }
			}
			# The decisive ground of either side queues one delayed front check.
			if = {
				limit = {
					ADISCORD_nam_resource_war_active = yes
					OR = {
						ROOT = { tag = NAM }
						FROM = { tag = NAM }
					}

						OR = {

				}
				NAM = { NAM_rw_queue_front_check = yes }
			}
			# A lost field shakes seal discipline; a lost line may be prepared
			# again after recapture.
			if = {
				limit = {
					FROM = { tag = NAM }
					FROM.FROM = {
						OR = {
							state = 67
							state = 688
							state = 689
							state = 690
							state = 719
							state = 700
						}
					}

					limit = {

						FROM.FROM = { NOT = { state = 700 } }

				}
			}
			# Oil throughput counts controlled fields, the pipeline end and the
			# sea terminal; the island steel concession needs Svetlogorsk control.

				limit = {
					OR = {
						ROOT = { tag = NAM }
						FROM = { tag = NAM }
					}

						OR = {
							state = 67

					}

				limit = {
					FROM = { tag = NAM }
					FROM.FROM = { state = 231 }
				}

				GLP = { NAM_steel_concession_validate = yes }
			}
		}
