on_actions = {
	on_daily = {
		effect = {
			if = {
				limit = {
					ADISCORD_nam_resource_war_ready = yes
					NOT = { has_global_flag = ADISCORD_nam_resource_war_scheduled }
				}
				set_global_flag = ADISCORD_nam_resource_war_scheduled
				NAM = {
					country_event = {
						id = ADISCORD_nam_resource_war.1
						days = 120
						random_days = 90
					}
				}
			}
		}
	}

	on_capitulation = {
		effect = {
			if = {
				limit = {
					ADISCORD_nam_resource_war_active = yes
					ROOT = { tag = NAM }
					FROM = { OR = { tag = EFL tag = AZH tag = SLF } }
				}
				set_global_flag = skip_default_capitulation
				NAM = { ADISCORD_nam_resource_war_resolve_coalition_victory = yes }
			}
			else_if = {
				limit = {
					ADISCORD_nam_resource_war_active = yes
					ROOT = { tag = EFL }
					FROM = { tag = NAM }
				}
				set_global_flag = skip_default_capitulation
				NAM = { ADISCORD_nam_resource_war_mark_eflor_defeated = yes }
			}
			else_if = {
				limit = {
					ADISCORD_nam_resource_war_active = yes
					ROOT = { tag = AZH }
					FROM = { tag = NAM }
				}
				set_global_flag = skip_default_capitulation
				NAM = { ADISCORD_nam_resource_war_mark_azhar_defeated = yes }
			}
			else_if = {
				limit = {
					ADISCORD_nam_resource_war_active = yes
					ROOT = { tag = SLF }
					FROM = { tag = NAM }
				}
				set_global_flag = skip_default_capitulation
				NAM = { ADISCORD_nam_resource_war_mark_rebels_defeated = yes }
			}
		}
	}
}
