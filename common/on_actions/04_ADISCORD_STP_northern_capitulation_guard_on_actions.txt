on_actions = {
	# 02_ADISCORD_STP_on_actions.txt marks a northern capitulation as managed in
	# the native immediate callback. Preserve that fact across duplicate/late
	# callbacks without leaving the global skip flag armed for another country.
	on_capitulation_immediate = {
		effect = {
			if = {
				limit = {
					ROOT = { has_country_flag = STP_cw_northern_capitulation_pending }
				}
				ROOT = {
					set_country_flag = {
						flag = STP_cw_northern_capitulation_reserved
						value = 1
						days = 2
					}
				}
			}
		}
	}

	# This file sorts after the STP settlement router and before the generic ZZ
	# annex fallback. Re-arm the shared reservation for every callback generated
	# by the same northern capitulation; ZZ clears the global flag afterwards.
	on_capitulation = {
		effect = {
			if = {
				limit = {
					ROOT = { has_country_flag = STP_cw_northern_capitulation_reserved }
				}
				set_global_flag = skip_default_capitulation
			}
		}
	}
}
