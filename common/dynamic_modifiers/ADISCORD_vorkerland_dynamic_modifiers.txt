# State-scoped autonomy ends when the owner loses regional-autonomy status.
# Conquering claimants and foreign countries must not inherit its bonuses.
ADISCORD_vorkerland_regional_autonomy = {
	enable = { always = yes }
	icon = GFX_modifiers_generic_military_plans
	remove_trigger = {
		OR = {
			has_global_flag = ADISCORD_vorkerland_collapse_finished
			NOT = { owner = { ADISCORD_vorkerland_holds_regional_autonomy = yes } }
		}
	}
	local_building_slots_factor = -0.25
	state_production_speed_buildings_factor = -0.15
	local_manpower = 0.30
	local_supplies = 0.20
	local_org_regain = 0.10
	local_supply_impact_factor = -0.15
}

# --- legitimacy_dynamic_modifiers ---
# Legitimacy is the swing mechanic of the civil war: a claimant moving from one
# end of the ledger to the other shifts forty percentage points across four
# axes, which is decisive over a campaign but never decisive in a single battle.
# The variable is written only by ADISCORD_vorkerland_add_legitimacy.
ADISCORD_vorkerland_legitimacy_dynamic = {
	icon = GFX_idea_generic_democratic_drift_bonus
	custom_modifier_tooltip = ADISCORD_vorkerland_legitimacy_score_tt

	stability_factor = ADISCORD_vorkerland_legitimacy_mod
	war_support_factor = ADISCORD_vorkerland_legitimacy_mod
	army_attack_factor = ADISCORD_vorkerland_legitimacy_mod
	army_defence_factor = ADISCORD_vorkerland_legitimacy_mod
}
