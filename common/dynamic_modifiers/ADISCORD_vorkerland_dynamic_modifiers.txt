# State-scoped autonomy ends when the owner loses regional-autonomy status.
# Conquering claimants and foreign countries must not inherit its bonuses.
ADISCORD_vorkerland_regional_autonomy = {
	enable = { always = yes }
	icon = GFX_modifiers_generic_military_plans
	remove_trigger = {
		OR = {
			has_global_flag = ADISCORD_vorkerland_collapse_finished
			NOT = { owner = { ADISCORD_vorkerland_holds_regional_autonomy = yes } }
		}
	}
	local_building_slots_factor = -0.25
	state_production_speed_buildings_factor = -0.15
	local_manpower = 0.30
	local_supplies = 0.20
	local_org_regain = 0.10
	local_supply_impact_factor = -0.15
}
