# A-Discord replaces the complete NCNS faction database. Keep one stable goal
# in every duration slot: an entirely empty selectable-goal pool makes the
# HOI4 1.19 faction AI enqueue a set-goal command with an invalid database
# pointer. The faction view also scans active goals for the native unlock
# effects when it builds the locked research and military tab tooltips. Keep
# those effects on the stable, already-serialized goals so existing saves have
# valid lookup targets without changing when the permanent goals complete.

	description = ADISCORD_faction_goal_operational_continuity_desc

	visible = { always = yes }
	available = { always = yes }
