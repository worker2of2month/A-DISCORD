# NCNS makes faction templates mandatory for safe scripted faction creation.
# A-Discord deliberately starts with the legacy behavior: a named alliance,
# no imported vanilla goals, and no vanilla ideology-dependent rules.
faction_template_ADISCORD_standard = {
	name = ADISCORD_faction_generic_name
	manifest = ADISCORD_faction_manifest_continuity
	icon = GFX_faction_logo_generic
	can_leader_join_other_factions = no

	visible = { always = no }
	available = { always = yes }

	# Fill every NCNS goal slot at creation. Leaving the mod-owned goal database
	# empty lets the faction AI enqueue an invalid set-goal command in HOI4 1.19.
	goals = {
		ADISCORD_faction_goal_operational_continuity
		ADISCORD_faction_goal_strategic_coordination
		ADISCORD_faction_goal_long_term_security
	}
}
