# Event-selected crisis courses. Every block is self-removing so a changed
# posture never leaves accumulated AI weights behind.

STP_AI_CAUTIOUS_SHABRAT = {
	allowed = { original_tag = STP }
	enable = {
		has_country_flag = STP_ai_course_cautious_shabrat
		has_country_flag = STP_main_campaign_side
		NOT = { has_country_flag = STP_ivanov_dead }
		NOT = { has_war = yes }
	}
	abort_when_not_enabled = yes
	ai_strategy = { type = equipment_production_factor id = infantry value = 15 }
	ai_strategy = { type = force_concentration_factor value = -10 }
}
