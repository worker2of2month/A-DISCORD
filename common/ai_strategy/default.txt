# Laws cost 150 PP. The generic reserve is 100, so an AI at war never
# accumulates enough to leave volunteer-only unless it is told to.
ADISCORD_ai_raise_conscription = {
	enable = {
		is_ai = yes
		has_capitulated = no
		OR = {
			AND = {
				has_idea = volunteer_only
				has_war_support > 0.1
				OR = {
					has_war = yes
					tag = WRK
					tag = WKR
					tag = VAD
					tag = TVA
					tag = STS
					tag = SRP
				}
			}
			AND = {
				has_war = yes
				has_idea = limited_conscription
				has_war_support > 0.2
			}
		}
	}
	abort_when_not_enabled = yes
	ai_strategy = { type = pp_spend_priority id = idea value = 150 }
	ai_strategy = { type = pp_spend_amount id = idea value = 150 }
}

# Auxiliary minima start at two, three and four military factories so their
# simultaneous requests leave at least one factory available for rifles.
# Wartime rifle shortages release healthy auxiliary reserves; separate restart
# thresholds avoid reallocating factories at each small stock fluctuation.
ADISCORD_produce_squad_weapons_low_stock = {
	enable = {
		num_of_military_factories > 1
		has_equipment = { ADISCORD_squad_weapons_equipment < 600 }
		OR = {
			has_war = no
			NOT = { stockpile_ratio = { archetype = infantry_equipment ratio < 0.10 } }
			stockpile_ratio = { archetype = ADISCORD_squad_weapons_equipment ratio < 0.05 }
		}
	}
	abort = {
		OR = {
			num_of_military_factories < 2
			has_equipment = { ADISCORD_squad_weapons_equipment > 1200 }
			AND = {
				has_war = yes
				stockpile_ratio = { archetype = infantry_equipment ratio < 0.03 }
				stockpile_ratio = { archetype = ADISCORD_squad_weapons_equipment ratio > 0.10 }
			}
		}
	}
	ai_strategy = { type = equipment_production_min_factories_archetype id = ADISCORD_squad_weapons_equipment value = 1 }
	ai_strategy = { type = equipment_variant_production_factor id = ADISCORD_squad_weapons_equipment value = 25 }
}

# Land-front demand must leave room for coastal operations. The target predicate
# is evaluated in the enemy country, with FROM referring to the planner.
ADISCORD_coastal_invasion_planning = {
	allowed = { OR = { original_tag = VAL original_tag = NOD original_tag = STP } }
	enable = {
		is_ai = yes
		has_war = yes
		has_capitulated = no
		has_tech = ADISCORD_tech_restored_dockyards
		num_divisions > 7
		has_navy_size = { size > 19 type = convoy }
	}
	abort_when_not_enabled = yes
	ai_strategy = { type = naval_invasion_focus value = 10 }
	ai_strategy = { type = naval_invasion_dominance_weight value = 30 }
	ai_strategy = {
		type = invasion_unit_request
		country_trigger = { has_war_with = FROM has_capitulated = no }
		value = 50
	}
}

ADISCORD_coastal_dockyard_capacity = {

# The convoy buffer reserves one dockyard below 100 transports. A country with
# only one dockyard must finish that reserve before maintaining its patrol fleet.
ADISCORD_coastal_patrol_replacement = {
	allowed = { OR = { original_tag = VAL original_tag = NOD original_tag = STP } }
	enable = {
		is_ai = yes
		has_capitulated = no
		num_of_naval_factories > 0
		has_navy_size = { size < 6 unit = ADISCORD_coastal_patrol_vessel }
		OR = {
			num_of_naval_factories > 1
			NOT = { has_equipment = { convoy_1 < 100 } }
		}
	}
	abort_when_not_enabled = yes
	ai_strategy = { type = equipment_production_min_factories_archetype id = ADISCORD_coastal_patrol_ship value = 1 }
}

# Allocation is independent of the number of war participants. Country profiles
# add their relative priorities; execution and supply checks remain native.
ADISCORD_active_enemy_front_concentration = {
	enable = {
		is_ai = yes
		has_war = yes
		has_capitulated = no
	}
	abort_when_not_enabled = yes

	ai_strategy = {
		type = front_unit_request
		country_trigger = {
			has_war_with = FROM
		}
		value = 50
	}
}

# FROM is the allocating country in a front's country_trigger. Keep a border
# when either side has a concrete war preparation, even during another war.
ADISCORD_wartime_neutral_borders = {
	enable = {
		is_ai = yes
		has_capitulated = no
		OR = {
			has_war = yes
			ADISCORD_ai_has_adjacent_war_preparation = yes
		}
	}
	abort_when_not_enabled = yes
	ai_strategy = {
		type = front_unit_request
		country_trigger = {
			NOT = { has_war_with = FROM }
			ADISCORD_ai_front_has_prewar_threat = no
		}
		value = -100
	}
}

ADISCORD_prewar_front_concentration = {
	enable = { is_ai = yes has_capitulated = no }
	abort_when_not_enabled = yes
	ai_strategy = {
		type = front_unit_request
		country_trigger = {
			exists = yes
			has_capitulated = no
			NOT = { has_war_with = FROM }
			NOT = { is_in_faction_with = FROM }
			ADISCORD_ai_front_has_prewar_threat = yes
		}
		value = 50
	}
}

# Reduce demand while the field army is being replenished. Role proportions and
# template XP remain available, including the first army of a new country.
ADISCORD_ai_rebuild_field_army = {
	enable = {
		is_ai = yes
		has_capitulated = no
		ADISCORD_ai_army_needs_replacements = yes
	}
	abort = {
		OR = {
			is_ai = no
			has_capitulated = yes
			ADISCORD_ai_army_replacement_pressure_cleared = yes
		}
	}
	ai_strategy = { type = ai_wanted_divisions_factor value = -60 }
}
