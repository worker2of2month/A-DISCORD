ADISCORD_produce_squad_weapons_low_stock = {
	enable = {
		num_of_military_factories > 1
		has_equipment = { ADISCORD_squad_weapons_equipment < 600 }
	}
	abort = {
		OR = {
			num_of_military_factories < 2
			has_equipment = { ADISCORD_squad_weapons_equipment > 1200 }
		}
	}
	ai_strategy = { type = equipment_production_min_factories_archetype id = ADISCORD_squad_weapons_equipment value = 1 }
	ai_strategy = { type = equipment_variant_production_factor id = ADISCORD_squad_weapons_equipment value = 25 }
}
