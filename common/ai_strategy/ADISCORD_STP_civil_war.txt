	abort_when_not_enabled = yes

	# leaves BBV and BHG as live neighbor demand during the northern war.

		country_trigger = {
			OR = {
				original_tag = BJK

				original_tag = BHG
				original_tag = BLD
				original_tag = BGT
				original_tag = BCM
			}
			NOT = { has_war_with = FROM }
		}
		value = -100
	}
}
