# War for the West: champions commit their armies against the whole opposing
# side. The corridor separates the champions, so most fronts face the other
# side's regional members rather than the rival champion itself.

ADISCORD_west_wrk_front_ivn = {
	allowed = { tag = WRK }
	enable = {
		ADISCORD_west_final_active = yes
		has_war_with = IVN
		NOT = {
			fighting_army_strength_ratio = {
				tag = IVN
				ratio < 0.67
			}
		}
	}
	abort_when_not_enabled = yes
	ai_strategy = {
		type = front_unit_request
		country_trigger = {
			ADISCORD_west_on_ivn_side = yes
			has_war_with = FROM
		}
		value = 100
	}
	ai_strategy = {
		type = front_control
		country_trigger = {
			ADISCORD_west_on_ivn_side = yes
			has_war_with = FROM
		}
		ratio = 0.01
		priority = 1500
		ordertype = front
		execution_type = balanced
		execute_order = yes
		manual_attack = yes
	}
}

# Production weights share the existing factory budget; no factory minimums
# accumulate on top of the generic infantry and support-equipment guarantees.
ADISCORD_wrk_postwar_production = {
	allowed = { tag = WRK }
	enable = {
		is_ai = yes
		ADISCORD_vorkerland_pw_sovereign = yes
	}
	abort_when_not_enabled = yes
	ai_strategy = {
		type = equipment_production_factor
		id = infantry
		value = 30
	}
	ai_strategy = {
		type = equipment_production_factor
		id = support
		value = 15
	}
	ai_strategy = {
		type = equipment_production_factor
		id = artillery
		value = 10
	}
}
