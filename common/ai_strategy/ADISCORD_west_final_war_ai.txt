# War for the West: champions commit their armies against the whole opposing
# side. The corridor separates the champions, so most fronts face the other
# side's regional members rather than the rival champion itself.

	allowed = { tag = WRK }
	enable = {
		ADISCORD_west_final_active = yes
		has_war_with = IVN

				tag = IVN

# Production weights share the existing factory budget; no factory minimums
# accumulate on top of the generic infantry and support-equipment guarantees.
ADISCORD_wrk_postwar_production = {
	allowed = { tag = WRK }
	enable = {
		is_ai = yes
		ADISCORD_vorkerland_pw_sovereign = yes
	}
	abort_when_not_enabled = yes
	ai_strategy = {
		type = equipment_production_factor
		id = infantry
		value = 30
	}
	ai_strategy = {
		type = equipment_production_factor
		id = support
		value = 15
	}
	ai_strategy = {
		type = equipment_production_factor
		id = artillery
		value = 10
	}
}

# The announced western crisis gives both sovereign champions the same preparation policy.
ADISCORD_difficulty_west_campaign_supply = {
	allowed = {
		OR = {
			tag = WRK
			tag = IVN
		}
	}
	enable = {
		ADISCORD_difficulty_ai_advanced = yes
		OR = {
			AND = {
				ADISCORD_west_crisis_open = yes
				has_global_flag = ADISCORD_west_crisis_warned
				WRK = {
					exists = yes
					has_capitulated = no
					is_subject = no
				}
				IVN = {
					exists = yes
					has_capitulated = no
					is_subject = no
					NOT = { is_in_faction_with = WRK }
				}
			}
			AND = {
				ADISCORD_west_final_active = yes
				OR = {
					has_war_with = WRK
					has_war_with = IVN
				}
			}
		}
	}
	abort_when_not_enabled = yes
	ai_strategy = {
		type = equipment_production_factor
		id = infantry
		value = 15
	}
	ai_strategy = {
		type = equipment_production_factor
		id = artillery
		value = 15
	}
	ai_strategy = {
		type = equipment_production_factor
		id = support
		value = 10
	}
}
