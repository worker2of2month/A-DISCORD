# Replacement demand follows the scheduled resource war or the current war of return.

	allowed = {
		OR = {
			tag = NAM
			tag = EFL
			tag = AZH
			tag = SLF
		}
	}
	enable = {
		ADISCORD_difficulty_ai_advanced = yes
		has_tech = ADISCORD_tech_restored_truck_fleets
		OR = {
			AND = {
				NOT = { tag = SLF }
				ADISCORD_nam_resource_war_preparation = yes
			}
			AND = {
				OR = {
					ADISCORD_nam_resource_war_active = yes
					NAM_war_of_return_active = yes
				}
				OR = {
					has_war_with = NAM
					has_war_with = EFL
					has_war_with = AZH
					has_war_with = SLF
				}
			}
		}
	}
	abort_when_not_enabled = yes
	ai_strategy = {
		type = equipment_variant_production_factor
		id = motorized_equipment
		value = 20
	}
	ai_strategy = {
		type = equipment_production_factor
		id = artillery
		value = 15
	}
}
