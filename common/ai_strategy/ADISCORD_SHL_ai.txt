# Preparation for an announced limited war, on both sides of the border.
ADISCORD_SHL_prepare_against_KYZ = {
	allowed = { tag = SHL }
	enable = {
		is_ai = yes
		has_capitulated = no
		NOT = { has_war_with = KYZ }
		KYZ = { exists = yes }
		OR = {
			check_variable = { var = SHL_keizan_state value = 5 compare = equals }
			has_country_flag = SHL_kyz_mobilizing
			AND = {
				SHL_qanat_dispute_open = yes
				NOT = { has_country_flag = SHL_neutral_to_KYZ }
				OR = {
					has_completed_focus = SHL_qanat_guard
					has_completed_focus = SHL_right_to_water
				}
			}
		}
	}
	abort_when_not_enabled = yes
	ai_strategy = { type = prepare_for_war id = KYZ value = 100 }
}
