# Country strength is a theatre-level signal, not local combat odds. Keep the
# native attack checks and every authored campaign order (priority >= 1000).
ADISCORD_difficulty_abyss_front_opportunities = {
	enable = {
		ADISCORD_difficulty_ai_abyss_front = yes
		ADISCORD_difficulty_ai_offensive_ready = yes
	}
	abort_when_not_enabled = yes
	ai_strategy = {
		type = force_concentration_front_factor
		country_trigger = {
			has_war_with = FROM
			fighting_army_strength_ratio = { tag = FROM ratio < 0.8 }
		}
		ratio = 0.01
		value = 30
	}
	ai_strategy = {
		type = front_control
		country_trigger = {
			has_war_with = FROM
			fighting_army_strength_ratio = { tag = FROM ratio < 0.8 }
		}
		ratio = 0.01
		priority = 950
		ordertype = front
		execution_type = balanced
		execute_order = yes
		manual_attack = yes
	}
}

# Careful execution still allows recovery and counterattacks; it never imposes
# a permanent halt. No extra front-unit demand is added to crowded theatres.
ADISCORD_difficulty_abyss_front_deliberate = {
	enable = { ADISCORD_difficulty_ai_abyss_front = yes }
	abort_when_not_enabled = yes
	ai_strategy = {
		type = front_control
		country_trigger = {
			has_war_with = FROM
			OR = {
				FROM = { ADISCORD_difficulty_ai_offensive_ready = no }
				fighting_army_strength_ratio = { tag = FROM ratio > 1.25 }
			}
		}
		ratio = 0.01
		priority = 950
		ordertype = front
		execution_type = careful
		execute_order = yes
		manual_attack = no
	}
	ai_strategy = {
		type = force_concentration_front_factor
		country_trigger = {
			has_war_with = FROM
			fighting_army_strength_ratio = { tag = FROM ratio > 1.25 }
		}
		ratio = 0.01
		value = -25
	}
}

# State targets are evaluated in state scope: FROM is the enemy and FROM.FROM
# is the allocating country. Ownership is insufficient after occupation.
# These modest weights complement explicit objectives such as the congress.
ADISCORD_difficulty_abyss_front_objectives = {
	enable = {
		ADISCORD_difficulty_ai_abyss_front = yes
		ADISCORD_difficulty_ai_offensive_ready = yes
	}
	abort_when_not_enabled = yes
	ai_strategy = {
		type = force_concentration_target_weight
		state_trigger = {
			FROM = { has_war_with = FROM.FROM }
			is_controlled_by = FROM
			is_capital = yes
		}
		value = 35
	}
	ai_strategy = {
		type = force_concentration_target_weight
		state_trigger = {
			FROM = { has_war_with = FROM.FROM }
			is_controlled_by = FROM
			any_province_building_level = {
				province = { all_provinces = yes }
				building = bunker
				level > 3
			}
		}
		value = -40
	}
}
