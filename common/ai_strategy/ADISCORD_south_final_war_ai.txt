# Every participant can attack or recover a recorded district. Champions
# concentrate on their paid order until settlement releases the state slot.
ADISCORD_south_district_objectives = {
	enable = {
		is_ai = yes
		has_capitulated = no
		has_war = yes
		ADISCORD_south_final_active = yes
		has_variable = ADISCORD_south_side
	}
	abort_when_not_enabled = yes
	ai_strategy = {
		type = force_concentration_target_weight
		state_trigger = {
			has_variable = ADISCORD_south_initial_side
			ADISCORD_south_enemy_front_state = yes
		}
		value = 65
	}
	ai_strategy = {
		type = force_concentration_target_weight
		state_trigger = {
			FROM.FROM = { has_variable = ADISCORD_south_order_state }
			state = var:FROM.FROM.ADISCORD_south_order_state
			ADISCORD_south_enemy_front_state = yes
		}
		value = 140
	}
	ai_strategy = {
		type = force_concentration_target_weight
		state_trigger = {
			is_capital = yes
			FROM.FROM = { difficulty > 2 }
			owner = { ADISCORD_south_war_champion = yes }
			ADISCORD_south_enemy_front_state = yes
		}
		value = 100
	}
}

# The desert countries choose sides at war entry. Until a direct border exists,
# each champion can stage a reserve on its own supply route.
ADISCORD_south_shl_staging_reserve = {
	allowed = { tag = SHL }
	enable = {
		difficulty > 2
		ADISCORD_south_ai_border_preparation = yes
		707 = {
			is_owned_by = SHL
			is_controlled_by = SHL
			supply_node > 0
		}
		NOT = { any_neighbor_country = { ADISCORD_south_declarer_valid = yes } }
	}
	abort_when_not_enabled = yes
	ai_strategy = {
		type = put_unit_buffers
		states = { 707 }
		ratio = 0.15
		subtract_fronts_from_need = yes
		subtract_invasions_from_need = yes
	}
}

# Production follows the selected champion through preparation and hostilities.
ADISCORD_difficulty_south_campaign_supply = {
	allowed = {
		OR = {
			tag = SHL
			tag = NAM
			tag = EFL
			tag = AZH
		}
	}
	enable = {
		ADISCORD_difficulty_ai_advanced = yes
		ADISCORD_south_final_war_enabled = yes
		OR = {
			ADISCORD_south_preparing_country = yes
			AND = {
				ADISCORD_south_final_active = yes
				OR = {
					tag = SHL
					has_country_flag = ADISCORD_south_north_leader
				}
				OR = {
					has_war_with = SHL
					has_war_with = NAM
					has_war_with = EFL
					has_war_with = AZH
				}
			}
		}
	}
	abort_when_not_enabled = yes
	ai_strategy = {
		type = equipment_production_factor
		id = infantry
		value = 15
	}
	ai_strategy = {
		type = equipment_production_factor
		id = artillery
		value = 15
	}
	ai_strategy = {
		type = equipment_production_factor
		id = support
		value = 10
	}
}
