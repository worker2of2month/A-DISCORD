# A-DISCORD doctrine AI. Tracks are parallel specializations, not the four
# mutually-exclusive vanilla land doctrines; bonuses therefore stay bounded.

ADISCORD_doctrine_general_staff_ai = {
	enable = { has_doctrine = ADISCORD_doctrine_restoration_general_staff }
	abort_when_not_enabled = yes
	ai_strategy = { type = force_concentration_factor value = 10 }
}

ADISCORD_doctrine_mass_restoration_ai = {
	enable = { has_completed_track = ADISCORD_land_mass_restoration }
	abort_when_not_enabled = yes
	ai_strategy = { type = role_ratio id = infantry value = 15 }
	ai_strategy = { type = ai_wanted_divisions_factor value = 10 }
	ai_strategy = { type = force_concentration_factor value = 5 }
}

ADISCORD_doctrine_platform_centric_ai = {
	enable = {
		has_completed_track = ADISCORD_land_platform_centric
		ADISCORD_economy_ai_can_fund_advanced_forces = yes
	}
	abort_when_not_enabled = yes
	ai_strategy = { type = role_ratio id = armor value = 12 }
	ai_strategy = { type = equipment_production_factor id = armor value = 15 }
	ai_strategy = { type = force_concentration_factor value = 15 }
}

ADISCORD_doctrine_networked_operations_ai = {
	enable = {
		has_completed_track = ADISCORD_land_networked_operations
		NOT = { ADISCORD_economy_ai_is_crisis = yes }
	}
	abort_when_not_enabled = yes
	ai_strategy = { type = equipment_production_factor id = support value = 15 }
	ai_strategy = { type = force_concentration_factor value = 20 }
}
