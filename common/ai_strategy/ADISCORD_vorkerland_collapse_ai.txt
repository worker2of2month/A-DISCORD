# Shared production priorities for every active Vorkerland successor. Keeping
# one profile avoids dozens of target-specific blocks that became stale as the
# political map changed.
ADISCORD_vorkerland_collapse_war_economy = {
	allowed = {
		OR = {
			tag = WRK tag = VAD tag = TVA tag = TGD tag = EYR tag = EGC
			tag = ZAO tag = WPA tag = WPS tag = PWR tag = PSD
			tag = VLA tag = EBA tag = ROM tag = DVA
			tag = SOL tag = SRA tag = TRU tag = ZTA
			tag = IBA tag = IBL tag = CSL
			tag = SLA tag = RZA tag = MLR tag = ERT tag = IRT tag = SCA
		}
	}
	enable = {
		has_global_flag = ADISCORD_vorkerland_collapse_wars_started
		NOT = { has_global_flag = ADISCORD_vorkerland_collapse_finished }
	}
	abort_when_not_enabled = yes
	ai_strategy = { type = ai_wanted_divisions_factor value = 8 }
	ai_strategy = { type = equipment_production_factor id = infantry value = 45 }
	ai_strategy = { type = equipment_production_factor id = support value = 20 }
	ai_strategy = { type = force_concentration_factor value = 35 }
}

# the same army to every possible successor at once.

# Afrelan support is political backing, not overlordship. PIV helps Macri only
# while his independent republic is actually fighting for survival.
ADISCORD_vorkerland_piv_support_macri = {
	allowed = { tag = PIV }
	enable = {
		has_global_flag = ADISCORD_vorkerland_collapse_wars_started
		country_exists = EBA
		EBA = {
			is_subject = no
			has_war = yes
			NOT = { has_capitulated = yes }
		}
		NOT = { has_war_with = EBA }
	}
	abort_when_not_enabled = yes
	ai_strategy = { type = send_volunteers_desire id = EBA value = 1000 }
}

# Preserve the separate Khan-border intervention, which is not part of the
# claimant decision chain but still needs an executable battle plan.
ADISCORD_vorkerland_khan_border_offensive = {
	allowed = { tag = RUS }
	enable = {
		has_global_flag = ADISCORD_vorkerland_khan_border_war_started
		country_exists = SLA
		has_war_with = SLA
	}
	abort_when_not_enabled = yes
	ai_strategy = {
		type = front_control
		tag = SLA
		ratio = 1.00
		priority = 1500
		ordertype = front
		execution_type = rush
		execute_order = yes
		manual_attack = yes
	}
	ai_strategy = { type = consider_weak id = SLA value = 100 }
	ai_strategy = { type = conquer id = SLA value = 150 }
}
