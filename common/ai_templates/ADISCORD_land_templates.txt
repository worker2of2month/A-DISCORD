# Staged total-conversion templates.  Cheap line troops, offensive assault
# groups, specialists, and tanks use separate roles so the AI keeps a balanced
# army instead of converting every formation to the most expensive design.

ADISCORD_garrison_templates = {
	role = garrison
	upgrade_prio = {
		base = 1
		modifier = { factor = 15 NOT = { ai_has_role_template = garrison } }
	}
	ADISCORD_garrison_levy = {
		reinforce_prio = 0
		custom_icon = 7
		upgrade_prio = { base = 2 }
		can_upgrade_in_field = { always = yes }
		target_template = { regiments = { ADISCORD_militia = 4 } }
	}
}

ADISCORD_infantry_templates = {
	role = infantry
	upgrade_prio = { base = 5 }

	ADISCORD_reconstruction_brigade = {
		reinforce_prio = 1
		upgrade_prio = { base = 5 }
		can_upgrade_in_field = { always = yes }
		target_template = { regiments = { infantry = 4 } }
		replace_at_match = 0.95
		replace_with = ADISCORD_line_brigade
	}

	ADISCORD_line_brigade = {
		enable = {
			num_of_military_factories > 3
			NOT = { ADISCORD_economy_ai_is_crisis = yes }

			has_equipment = { artillery_equipment > 250 }
		}
		reinforce_prio = 1
		upgrade_prio = { base = 6 }
		can_upgrade_in_field = {
			has_equipment = { support_equipment > 600 }
			has_equipment = { artillery_equipment > 400 }
		}
		target_template = {
			support = { engineer = 1 artillery = 1 }
			regiments = { infantry = 6 }
		}
		replace_at_match = 0.98
		replace_with = ADISCORD_defensive_line_brigade
	}

	ADISCORD_defensive_line_brigade = {
		enable = {
			has_tech = ADISCORD_tech_forward_supply_hubs

			num_of_military_factories > 6
			NOT = { ADISCORD_economy_ai_is_crisis = yes }
			has_equipment = { support_equipment > 900 }
			has_equipment = { artillery_equipment > 500 }
			has_equipment = { anti_tank_equipment > 160 }
			has_equipment = { anti_air_equipment > 160 }
		}
		reinforce_prio = 1
		upgrade_prio = { base = 8 }
		can_upgrade_in_field = {
			has_equipment = { support_equipment > 1300 }
			has_equipment = { artillery_equipment > 750 }
			has_equipment = { anti_tank_equipment > 240 }
			has_equipment = { anti_air_equipment > 240 }
		}
		target_template = {
			support = {
				engineer = 1
				artillery = 1
				anti_tank = 1
				anti_air = 1
				logistics_company = 1
			}
			regiments = { infantry = 7 ADISCORD_line_artillery = 1 }
		}
	}
}

# Assault formations use the mobile role.  This keeps them as a limited
# offensive reserve rather than replacing every ordinary infantry division.
ADISCORD_assault_templates = {
	role = mobile
	upgrade_prio = {
		base = 0
		modifier = {
			add = 9
			has_tech = ADISCORD_tech_remote_weapon_tripods
			num_of_military_factories > 9
			ADISCORD_economy_ai_can_fund_advanced_forces = yes
		}
	}

	ADISCORD_assault_group = {
		enable = {
			has_tech = ADISCORD_tech_remote_weapon_tripods
			has_tech = ADISCORD_tech_standardized_field_tool_chests
			has_tech = ADISCORD_tech_frequency_hopping_field_sets
			num_of_military_factories > 9
			ADISCORD_economy_ai_can_fund_advanced_forces = yes
			has_equipment = { ADISCORD_squad_weapons_equipment > 1600 }
			has_equipment = { support_equipment > 1200 }
			has_equipment = { artillery_equipment > 450 }
		}
		reinforce_prio = 2
		upgrade_prio = { base = 9 }
		can_upgrade_in_field = {
			NOT = { ADISCORD_economy_ai_is_crisis = yes }
			has_equipment = { ADISCORD_squad_weapons_equipment > 2200 }
			has_equipment = { support_equipment > 1600 }
			has_equipment = { artillery_equipment > 650 }
		}
		target_template = {
			support = {
				engineer = 1
				artillery = 1
				maintenance_company = 1
				signal_company = 1
			}
			regiments = {
				infantry = 5
				ADISCORD_assault_infantry = 3
				ADISCORD_line_artillery = 1
			}
		}
	}
}
