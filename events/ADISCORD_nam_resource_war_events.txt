add_namespace = ADISCORD_nam_resource_war

# The fresh collapse edge schedules this once as the backstop deadline; the
# restitution clock and a first strike dispatch it earlier. A signed Klyuchevsk
# protocol defers the backstop once by 365 days. A temporary third-party war
# gets one delayed retry. global.NAM_rw_entry_dispatches counts the queued
# copies: a failed check stops the entry and refunds the orders only when no
# other copy is queued, so a pending retry owns the final decision.
country_event = {
	id = ADISCORD_nam_resource_war.1
	hidden = yes
	is_triggered_only = yes

	immediate = {
		subtract_from_variable = { var = global.NAM_rw_entry_dispatches value = 1 }
		clamp_variable = {
			var = global.NAM_rw_entry_dispatches
			min = 0
			max = 100
		}
		if = {
			limit = {
				has_country_flag = NAM_rw_protocol_signed
				NOT = { has_country_flag = NAM_rw_war_dispatched }
				NOT = { has_global_flag = ADISCORD_nam_resource_war_protocol_deferred }
				NAM_rw_clock_running = yes
			}
			set_global_flag = ADISCORD_nam_resource_war_protocol_deferred
			NAM_rw_count_entry_dispatch = yes
			country_event = {
				id = ADISCORD_nam_resource_war.1
				days = 365
			}
		}
		# Azhar's bought delay: no war starts inside the 75 days, and the
		# first backstop after the payment moves 75 days later.
		else_if = {
			limit = {
				has_country_flag = NAM_rw_time_bought
				NOT = { has_country_flag = NAM_rw_first_strike }
				NAM_rw_clock_running = yes
				OR = {
					NAM_rw_time_bought_active = yes
					AND = {
						NOT = { has_country_flag = NAM_rw_war_dispatched }
						NOT = { has_global_flag = ADISCORD_nam_resource_war_time_deferred }
					}
				}
			}
			set_global_flag = ADISCORD_nam_resource_war_time_deferred
			NAM_rw_count_entry_dispatch = yes
			country_event = {
				id = ADISCORD_nam_resource_war.1
				days = 75
			}
		}
		else_if = {
			limit = {
				has_global_flag = ADISCORD_nam_resource_war_scheduled
				ADISCORD_nam_resource_war_ready = yes
			}
			ADISCORD_nam_resource_war_start = yes
		}
		else_if = {
			limit = {
				has_global_flag = ADISCORD_fresh_campaign_contract_v1
				has_global_flag = ADISCORD_nam_resource_war_scheduled
				NOT = { has_global_flag = ADISCORD_nam_resource_war_retry_used }
				NOT = { has_global_flag = ADISCORD_nam_resource_war_started }
				NOT = { has_global_flag = ADISCORD_nam_resource_war_resolved }
				country_exists = NAM
				country_exists = EFL
				country_exists = AZH
			}
			set_global_flag = ADISCORD_nam_resource_war_retry_used
			NAM_rw_count_entry_dispatch = yes
			country_event = {
				id = ADISCORD_nam_resource_war.1
				days = 120
				random_days = 90
			}
		}
		else_if = {
			limit = {
				has_global_flag = ADISCORD_nam_resource_war_scheduled
				NOT = { has_global_flag = ADISCORD_nam_resource_war_started }
				NOT = { has_global_flag = ADISCORD_nam_resource_war_resolved }
				check_variable = { var = global.NAM_rw_entry_dispatches value = 0 compare = less_than_or_equals }
			}
			clr_global_flag = ADISCORD_nam_resource_war_scheduled
			NAM = { NAM_rw_close_preparation = yes }
			log = "[ADISCORD][NAM][ENTRY] Resource-war entry stopped after its bounded readiness checks."
		}
	}
}
