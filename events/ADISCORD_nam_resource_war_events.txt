# The mission and a voluntary first strike queue the same bounded war entry.
# A failed check closes preparation only after every queued dispatch is settled.
country_event = {
	id = ADISCORD_nam_resource_war.1
	hidden = yes
	is_triggered_only = yes

	immediate = {
		subtract_from_variable = { var = global.NAM_rw_entry_dispatches value = 1 }
		clamp_variable = {
			var = global.NAM_rw_entry_dispatches
			min = 0
			max = 100
		}
		if = {
			limit = {
				has_global_flag = ADISCORD_nam_resource_war_scheduled
				ADISCORD_nam_resource_war_ready = yes
			}
			ADISCORD_nam_resource_war_start = yes
		}
		else_if = {
			limit = {
				has_global_flag = ADISCORD_fresh_campaign_contract_v1
				has_global_flag = ADISCORD_nam_resource_war_scheduled
				NOT = { has_global_flag = ADISCORD_nam_resource_war_retry_used }
				NOT = { has_global_flag = ADISCORD_nam_resource_war_started }
				NOT = { has_global_flag = ADISCORD_nam_resource_war_resolved }
				country_exists = NAM
				country_exists = EFL
				country_exists = AZH
			}
			set_global_flag = ADISCORD_nam_resource_war_retry_used
			NAM_rw_count_entry_dispatch = yes
			country_event = {
				id = ADISCORD_nam_resource_war.1
				days = 120
				random_days = 90
			}
		}
		else_if = {
			limit = {
				has_global_flag = ADISCORD_nam_resource_war_scheduled
				NOT = { has_global_flag = ADISCORD_nam_resource_war_started }
				NOT = { has_global_flag = ADISCORD_nam_resource_war_resolved }
				check_variable = { var = global.NAM_rw_entry_dispatches value = 0 compare = less_than_or_equals }
			}
			clr_global_flag = ADISCORD_nam_resource_war_scheduled
			NAM = { NAM_rw_close_preparation = yes }
			log = "[ADISCORD][NAM][ENTRY] Resource-war entry stopped after its bounded readiness checks."
		}
	}
}
