	id = ADISCORD_vorkerland_collapse.1
	hidden = yes
	is_triggered_only = yes
	fire_only_once = yes

	trigger = {
		tag = WRK
		ADISCORD_vorkerland_collapse_not_started = yes
	}

	immediate = {
		ADISCORD_vorkerland_begin_collapse = yes
		set_global_flag = ADISCORD_vorkerland_collapse_started
		# begin_collapse refreshes Vorkerland trees before this flag exists.
		# Shabrat's collapse-gated focuses read the flag, so dirty STP after it.
		if = {
			limit = { country_exists = STP }
			STP = { mark_focus_tree_layout_dirty = yes }
		}
		ADISCORD_vorkerland_expire_precollapse_itoran_armistice = yes
		random_list = {
			60 = { set_global_flag = ADISCORD_vorkerland_worker_safe_with_loyalists }
			25 = { set_global_flag = ADISCORD_vorkerland_worker_rescued_by_vlad }
			10 = { set_global_flag = ADISCORD_vorkerland_worker_missing }
			5 = { set_global_flag = ADISCORD_vorkerland_worker_killed }
		}
		# State 40 must already belong to WKR. A WRK-scoped actor created
		# before the partition is dropped when this event annexes WRK.
		ADISCORD_vorkerland_apply_initial_map = yes
		ADISCORD_vorkerland_prepare_claimant_characters = yes
		# change_tag_from transfers only human control. The preference event does
		# not alter the split, Worker's random fate, or any AI-controlled claimant.
		# Old queued collapse.1 events carry no preference and therefore keep WKR.
		if = {
			limit = { is_ai = no }
			if = {
				limit = {
					OR = {
						has_global_flag = ADISCORD_vorkerland_preference_worker
						AND = {
							NOT = { has_global_flag = ADISCORD_vorkerland_preference_worker }
							NOT = { has_global_flag = ADISCORD_vorkerland_preference_joint }
							NOT = { has_global_flag = ADISCORD_vorkerland_preference_utilitarian }
						}
						AND = {
							NOT = { has_global_flag = ADISCORD_vorkerland_preference_worker }
							has_global_flag = ADISCORD_vorkerland_preference_joint
							has_global_flag = ADISCORD_vorkerland_preference_utilitarian
						}
					}
				}
				WKR = { change_tag_from = WRK }
			}
			else_if = {
				limit = {
					has_global_flag = ADISCORD_vorkerland_preference_joint
					NOT = { has_global_flag = ADISCORD_vorkerland_preference_worker }
					NOT = { has_global_flag = ADISCORD_vorkerland_preference_utilitarian }
				}
				VAD = { change_tag_from = WRK }
			}
			else_if = {
				limit = {
					has_global_flag = ADISCORD_vorkerland_preference_utilitarian
					NOT = { has_global_flag = ADISCORD_vorkerland_preference_worker }
					NOT = { has_global_flag = ADISCORD_vorkerland_preference_joint }
				}
				TVA = { change_tag_from = WRK }
			}
		}
		clr_global_flag = ADISCORD_vorkerland_preference_worker
		clr_global_flag = ADISCORD_vorkerland_preference_joint
		clr_global_flag = ADISCORD_vorkerland_preference_utilitarian
		random_list = {
			7 = { ADISCORD_vorkerland_form_joint_government = yes }
			93 = { }
		}
		WKR = {
			inherit_technology = WRK
			load_focus_tree = {
				tree = ADISCORD_vorkerland_civil_war_focus
				keep_completed = yes
				copy_completed_from = WRK
			}
			annex_country = { target = WRK transfer_troops = yes }
			ADISCORD_vorkerland_relocate_legacy_armies = yes
			ADISCORD_vorkerland_teardown_confederation = yes
			ADISCORD_vorkerland_align_wtd_with_worx = yes
			ADISCORD_vorkerland_apply_claimant_cosmetics = yes
			ADISCORD_vorkerland_sync_all_party_identities = yes
			ADISCORD_vorkerland_repair_claimant_identities = yes
			focus_unlock = yes
			mark_focus_tree_layout_dirty = yes
			country_event = { id = ADISCORD_vorkerland_collapse.41 days = 1 }
			country_event = { id = ADISCORD_vorkerland_phase.2 days = 1 }
			# The dispatcher reads the fate roll made above and holds its own
			# guard, so calling it here cannot silence a later report.
			ADISCORD_vorkerland_story_report_worker_fate = yes
			if = {
				limit = {
					RUS = { exists = yes }
					NOT = { has_global_flag = ADISCORD_vorkerland_dirty_reveal_v2_scheduled }
				}
				set_global_flag = ADISCORD_vorkerland_dirty_reveal_v2_scheduled
				RUS = {
					country_event = {
						id = ADISCORD_vorkerland_collapse.85
						days = 1095
					}
				}
			}
		}
	}
}

		VAD = { ADISCORD_vorkerland_bootstrap_ai_force_designs = yes }
		TVA = { ADISCORD_vorkerland_bootstrap_ai_force_designs = yes }
		# The claimant spirits are asserted at the actual war-outbreak stage,
		# after the initial partition and political cache rebuild. This prevents
		# WRK's three obsolete starting spirits from resurfacing in live play.
		if = {
			limit = { WKR = { exists = yes } }
			WKR = { ADISCORD_vorkerland_repair_claimant_spirit_progression = yes }
		}
		if = {
			limit = { VAD = { exists = yes } }
			VAD = { ADISCORD_vorkerland_repair_claimant_spirit_progression = yes }
		}
		if = {
			limit = { TVA = { exists = yes } }
			TVA = { ADISCORD_vorkerland_repair_claimant_spirit_progression = yes }
		}
		set_global_flag = ADISCORD_vorkerland_claimant_spirit_progression_v3
		if = {
			limit = { VAL = { exists = yes } }
			VAL = { VAL_handle_vorkerland_war_outbreak = yes }
		}
		if = {
			limit = { WKR = { exists = yes } }
			WKR = {
				country_event = {
					id = ADISCORD_vorkerland_collapse.31
					days = 21
					random_days = 14
				}
			}
		}
	}

# A short mobilisation pause prevents the successor states from declaring war
# on the day the map is rebuilt. All later wars still come from visible/AI
# decisions and require a live common border.
country_event = {
	id = ADISCORD_vorkerland_collapse.31
	hidden = yes
	is_triggered_only = yes
	fire_only_once = yes
	trigger = {
		OR = { tag = WRK tag = WKR }
		has_global_flag = ADISCORD_vorkerland_collapse_wars_started
		NOT = { has_global_flag = ADISCORD_vorkerland_claim_wars_authorized }
	}
	immediate = {
		# Last guard before any AI can select a war decision. At this point every
		# successor has existed for several weeks, so faction removal is stable.
		ADISCORD_vorkerland_teardown_confederation = yes
		set_global_flag = ADISCORD_vorkerland_claim_wars_authorized
		# The central district bracket does not depend on a claimant or on a
		# theatre-anchor decision. It opens from the same authorisation so that
		# the nine districts are already fighting one another when the first
		# claimant reaches their border.
		country_event = { id = ADISCORD_vorkerland_collapse.90 days = 1 }
	}
}

# twice for presentation correctness, but a cache miss never freezes the war.
country_event = {
	id = ADISCORD_vorkerland_phase.14
	hidden = yes
	is_triggered_only = yes

	trigger = {
		tag = WKR
		has_global_flag = ADISCORD_vorkerland_collapse_started
		NOT = { has_global_flag = ADISCORD_vorkerland_collapse_finished }
		NOT = { has_global_flag = ADISCORD_vorkerland_claimant_identities_verified_v2 }
	}

	immediate = {
		clr_global_flag = ADISCORD_vorkerland_claimant_identity_check_pending_v2
		ADISCORD_vorkerland_repair_claimant_identities = yes
		if = {
			limit = { ADISCORD_vorkerland_claimant_identities_materialized = yes }
			set_global_flag = ADISCORD_vorkerland_claimant_identities_verified_v2
			clr_global_flag = ADISCORD_vorkerland_claimant_identity_retry_v2
			clr_global_flag = ADISCORD_vorkerland_claimant_identity_degraded_v2
