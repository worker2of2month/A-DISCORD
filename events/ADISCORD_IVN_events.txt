add_namespace = ADISCORD_ivn

# Revolt dispatcher. A busy Itora delays the revolt instead of discarding it;
# the retry stops once Itora is a subject or no longer exists.
country_event = {
	id = ADISCORD_ivn.1
	hidden = yes
	is_triggered_only = yes
	trigger = {
		tag = IVN
		has_global_flag = ADISCORD_vorkerland_ivanland_intervention_failed
	}

	immediate = {
		if = {
			limit = { IVN_marches_revolt_ready = yes }
			country_event = { id = ADISCORD_ivn.2 hours = 1 }
		}
		else_if = {
			limit = {
				exists = yes
				is_subject = no
				NOT = { has_global_flag = IVN_marches_revolt_active }
				NOT = { has_global_flag = IVN_marches_revolt_resolved }
			}
			country_event = { id = ADISCORD_ivn.1 days = 14 }
		}
	}
}

# The expedition comes back. The answer sets how the army and the marches
# meet the 30 days before the revolt.
country_event = {
	id = ADISCORD_ivn.6
	title = ADISCORD_ivn.6.t
	desc = ADISCORD_ivn.6.d
	picture = GFX_event_adiscord_military_headquarters
	is_triggered_only = yes
	trigger = {
		tag = IVN
		has_global_flag = ADISCORD_vorkerland_ivanland_intervention_failed
	}

	immediate = {
		hidden_effect = { IVN_marches_initialize = yes }
	}

	option = {
		name = ADISCORD_ivn.6.a
		add_power_balance_value = {
			id = IVN_command_bop
			value = -0.15
		}
		custom_effect_tooltip = ADISCORD_ivn.6.a_tt
		hidden_effect = {
			set_temp_variable = { var = IVN_trust_delta value = -10 }
			IVN_northern_marches_trust_change = yes
		}
		ai_chance = { base = 30 }
	}

	option = {
		name = ADISCORD_ivn.6.b
		add_stability = -0.10
		custom_effect_tooltip = ADISCORD_ivn.6.b_tt
		hidden_effect = {
			set_temp_variable = { var = IVN_trust_delta value = 10 }
			IVN_frontier_trust_change_all = yes
		}
		ai_chance = { base = 40 }
	}

	option = {
		name = ADISCORD_ivn.6.c
		trigger = {
			NOT = { has_political_power < 75 }
		}
		add_political_power = -75
		add_power_balance_value = {
			id = IVN_command_bop
			value = 0.10
		}
		custom_effect_tooltip = ADISCORD_ivn.6.c_tt
		hidden_effect = {
			set_temp_variable = { var = IVN_trust_delta value = 15 }
			IVN_northern_marches_trust_change = yes

	}
}
