add_namespace = stp

# Legacy side-choice IDs remain as hidden compatibility forwarders. Old saves
# that still have either event queued enter the canonical forced-choice event;
# if a side was already selected, the new event trigger makes this a no-op.
country_event = {
	id = stp.1
	hidden = yes
	is_triggered_only = yes
	trigger = {
		tag = STP
		has_country_flag = STP_main_campaign_side
		NOT = { has_country_flag = STP_ivanov_dead }
	}
	immediate = {
		country_event = { id = stp_crisis.6 }
	}
}
