# RUS. Each suspicion tier is announced once by RUS_suspicion_check.
country_event = {
	id = ADISCORD_rus_crisis.17
	title = {
		trigger = { check_variable = { var = RUS_suspicion_warning value = 2 compare = less_than } }
		text = ADISCORD_rus_crisis.17.rumours.t
	}
	title = {
		trigger = { check_variable = { var = RUS_suspicion_warning value = 2 compare = greater_than_or_equals } }
		text = ADISCORD_rus_crisis.17.observers.t
	}
	desc = {
		trigger = { check_variable = { var = RUS_suspicion_warning value = 2 compare = less_than } }
		text = ADISCORD_rus_crisis.17.rumours.d
	}
	desc = {
		trigger = { check_variable = { var = RUS_suspicion_warning value = 2 compare = greater_than_or_equals } }
		text = ADISCORD_rus_crisis.17.observers.d
	}
	picture = GFX_event_adiscord_negotiation_table
	is_triggered_only = yes
	trigger = { RUS_suspicion_active = yes }
	# The free answer comes first so an unattended event settles without a debit.
	option = {
		name = ADISCORD_rus_crisis.17.truth
		ai_chance = { base = 2 }
		custom_effect_tooltip = RUS_suspicion_warning_tt
		custom_effect_tooltip = RUS_suspicion_relief_8_tt
		if = {
			limit = { RUS_state_balance_operational = yes }
			add_power_balance_value = { id = RUS_state_balance value = -0.10 }
			custom_effect_tooltip = RUS_suspicion_energy_cover_support_tt
		}
		hidden_effect = {
			set_temp_variable = { var = RUS_suspicion_delta value = -8 }
			RUS_suspicion_add = yes
			if = {
				limit = { RUS_state_balance_operational = yes }
				set_temp_variable = { var = RUS_black_army_support_delta value = 0 }
				set_temp_variable = { var = RUS_scientists_support_delta value = 0.05 }
				RUS_faction_support_change = yes
			}
		}
	}
	option = {
		name = ADISCORD_rus_crisis.17.hide
		trigger = { RUS_inspection_hide_affordable = yes }
		ai_chance = { base = 1 }
		custom_effect_tooltip = RUS_inspection_hide_cost_tt
		custom_effect_tooltip = RUS_suspicion_relief_12_tt
		hidden_effect = {
			if = {
				limit = { RUS_inspection_hide_affordable = yes }
				subtract_from_variable = { var = RUS_bunker_food value = 20 }
				RUS_bunker_refresh = yes
				add_manpower = -500
				set_temp_variable = { var = RUS_suspicion_delta value = -12 }
				RUS_suspicion_add = yes
			}
		}
	}
	option = {
		name = ADISCORD_rus_crisis.17.bribe
		trigger = { RUS_inspection_bribe_affordable = yes }
		ai_chance = { base = 2 }
		custom_effect_tooltip = RUS_inspection_bribe_cost_tt
		custom_effect_tooltip = RUS_suspicion_relief_15_tt
		hidden_effect = {
			if = {
				limit = { RUS_inspection_bribe_affordable = yes }
				subtract_from_variable = { var = ADISCORD_economy_treasury value = 65 }
				add_to_variable = { var = ADISCORD_economy_current_month_action_costs value = 65 }
				ADISCORD_economy_mark_dirty = yes
				set_temp_variable = { var = RUS_suspicion_delta value = -15 }
				RUS_suspicion_add = yes
			}
		}
	}
	option = {
		name = ADISCORD_rus_crisis.17.expel
		ai_chance = { base = 1 }
		custom_effect_tooltip = RUS_suspicion_gain_10_tt
		add_war_support = 0.05
		if = {
			limit = { RUS_state_balance_operational = yes }
			add_power_balance_value = { id = RUS_state_balance value = 0.10 }
			custom_effect_tooltip = RUS_suspicion_disperse_musters_support_tt
		}
		hidden_effect = {
			set_temp_variable = { var = RUS_suspicion_delta value = 10 }
			RUS_suspicion_add = yes
			if = {
				limit = { RUS_state_balance_operational = yes }
				set_temp_variable = { var = RUS_black_army_support_delta value = 0.05 }
				set_temp_variable = { var = RUS_scientists_support_delta value = 0 }
				RUS_faction_support_change = yes
			}
		}
	}
}
