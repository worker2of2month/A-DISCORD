		text = ADISCORD_rus_crisis.4.t
	}
	title = {
		trigger = { NOT = { check_variable = { var = RUS_crisis_defeated_course value = 2 compare = less_than } } }
		text = ADISCORD_rus_crisis.4.government.t
	}
	desc = {
		trigger = {
			tag = VAL
			check_variable = { var = RUS_crisis_defeated_course value = 2 compare = less_than }
		}
		text = ADISCORD_rus_crisis.4.val
	}
	desc = {
		trigger = {
			OR = { tag = STS tag = STP }
			check_variable = { var = RUS_crisis_defeated_course value = 2 compare = less_than }
		}
		text = ADISCORD_rus_crisis.4.sts
	}
	desc = {
		trigger = {
			tag = NOD
			check_variable = { var = RUS_crisis_defeated_course value = 2 compare = less_than }
		}
		text = ADISCORD_rus_crisis.4.nod
	}
	desc = {
		trigger = {
			NOT = { OR = { tag = VAL tag = STS tag = STP tag = NOD } }
			check_variable = { var = RUS_crisis_defeated_course value = 2 compare = less_than }
		}
		text = ADISCORD_rus_crisis.4.other
	}
	desc = {
		trigger = { check_variable = { var = RUS_crisis_defeated_course value = 2 compare = equals } }
		text = ADISCORD_rus_crisis.4.army.d
	}
	desc = {
		trigger = { check_variable = { var = RUS_crisis_defeated_course value = 3 compare = equals } }
		text = ADISCORD_rus_crisis.4.reconstruction.d
	}
	picture = GFX_event_adiscord_parliament_chamber
	is_triggered_only = yes
	option = {
		name = ADISCORD_rus_crisis.4.a
		unlock_decision_tooltip = RUS_crisis_establish_special_zone
	}
}
