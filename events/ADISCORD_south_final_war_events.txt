# A regional state without a champion's subject or faction tie chooses its side.
# There is no neutral answer: the desert states are the only land routes.
country_event = {
	id = ADISCORD_south.10
	title = ADISCORD_south.10.t
	desc = ADISCORD_south.10.d
	picture = GFX_event_adiscord_military_planning
	is_triggered_only = yes
	trigger = {
		ADISCORD_south_final_active = yes
		ADISCORD_south_needs_side_choice = yes
	}

	option = {
		name = ADISCORD_south.10.a
		custom_effect_tooltip = ADISCORD_south_join_shl_tt
		hidden_effect = {
			ADISCORD_south_join_shl_side = yes
			ADISCORD_south_faction_partners_follow = yes
		}
		ai_chance = {
			base = 40
			modifier = {
				add = 40
				OR = {
					tag = EFL
					tag = AZH
				}
				ADISCORD_south_north_is_nam = yes
			}
			modifier = {
				add = 40
				tag = WEF
				EFL = { has_country_flag = ADISCORD_south_north_leader }
			}
			modifier = {
				add = 40
				tag = KDR
				SHL = { check_variable = { var = SHL_political_course value = 4 compare = equals } }
			}
			modifier = {
				add = 40
				tag = GLP
				SHL = { check_variable = { var = SHL_political_course value = 1 compare = equals } }
			}
			modifier = {
				add = 40
				tag = KYZ
				SHL = {
					OR = {
						check_variable = { var = SHL_political_course value = 2 compare = equals }
						check_variable = { var = SHL_political_course value = 5 compare = equals }
					}
				}
			}
			modifier = {
				add = 30
				tag = RHM
				SHL = {
					OR = {
						check_variable = { var = SHL_political_course value = 2 compare = equals }
						check_variable = { var = SHL_political_course value = 5 compare = equals }
					}
				}
			}
			modifier = {
				add = 15
				is_neighbor_of = SHL
			}
		}
	}

	option = {
		name = ADISCORD_south.10.b
		custom_effect_tooltip = ADISCORD_south_join_north_tt
		hidden_effect = {
			ADISCORD_south_join_north_side = yes
			ADISCORD_south_faction_partners_follow = yes
		}
		ai_chance = {
			base = 40
			modifier = {
				add = 30
				tag = WEF
				ADISCORD_south_north_is_nam = yes
			}
			modifier = {
				add = 30
				tag = MZR
			}
			modifier = {
				add = 30
				tag = KDR
				SHL = { NOT = { check_variable = { var = SHL_political_course value = 4 compare = equals } } }
			}
			modifier = {
				add = 30
				OR = {
					tag = RHM
					tag = SDR
				}
				NAM = {
					has_country_flag = ADISCORD_south_north_leader
					has_completed_focus = NAM_federal_charter
				}
			}
			modifier = {
				add = 30
				tag = AZH
				EFL = { has_country_flag = ADISCORD_south_north_leader }
			}
			modifier = {
				add = 15
				OR = {
					AND = {
						NAM = { has_country_flag = ADISCORD_south_north_leader }
						is_neighbor_of = NAM
					}
					AND = {
						EFL = { has_country_flag = ADISCORD_south_north_leader }
						is_neighbor_of = EFL
					}
					AND = {
						AZH = { has_country_flag = ADISCORD_south_north_leader }
						is_neighbor_of = AZH
					}
				}
			}
		}
	}
}

# Consolidation offer. ROOT is the target, FROM the champion whose focus sent it.
# Options clear the pending offer themselves, so a stale answer cannot block it.
country_event = {
	id = ADISCORD_south.40
	title = {
		text = ADISCORD_south.40.t_submit
		trigger = { check_variable = { var = ADISCORD_south_offer_kind value = 1 compare = equals } }
	}
	title = {
		text = ADISCORD_south.40.t_annex
		trigger = { check_variable = { var = ADISCORD_south_offer_kind value = 2 compare = equals } }
	}
	title = {
		text = ADISCORD_south.40.t_alliance
		trigger = { check_variable = { var = ADISCORD_south_offer_kind value = 3 compare = equals } }
	}
	desc = {
		text = ADISCORD_south.40.d_submit_shl
		trigger = {
			check_variable = { var = ADISCORD_south_offer_kind value = 1 compare = equals }
			FROM = {
				tag = SHL
				NOT = { check_variable = { var = SHL_political_course value = 2 compare = equals } }
			}
		}
	}
	desc = {
		text = ADISCORD_south.40.d_submit_shl_charter
		trigger = {
			check_variable = { var = ADISCORD_south_offer_kind value = 1 compare = equals }
			FROM = {
				tag = SHL
				check_variable = { var = SHL_political_course value = 2 compare = equals }
			}
		}
	}
	desc = {
		text = ADISCORD_south.40.d_submit_nam_namestnik
		trigger = {
			check_variable = { var = ADISCORD_south_offer_kind value = 1 compare = equals }
			FROM = {
				tag = NAM
				NOT = { has_completed_focus = NAM_federal_charter }
			}
		}
	}
	desc = {
		text = ADISCORD_south.40.d_submit_nam_charter
		trigger = {
			check_variable = { var = ADISCORD_south_offer_kind value = 1 compare = equals }
			FROM = {
				tag = NAM
				has_completed_focus = NAM_federal_charter
			}
		}
	}
	desc = {
		text = ADISCORD_south.40.d_annex
		trigger = { check_variable = { var = ADISCORD_south_offer_kind value = 2 compare = equals } }
	}
	desc = {
		text = ADISCORD_south.40.d_alliance_shl
		trigger = {
			check_variable = { var = ADISCORD_south_offer_kind value = 3 compare = equals }
			FROM = { tag = SHL }
		}
	}
	desc = {
		text = ADISCORD_south.40.d_alliance_nam
		trigger = {
			check_variable = { var = ADISCORD_south_offer_kind value = 3 compare = equals }
			FROM = { tag = NAM }
		}
	}
	picture = GFX_event_adiscord_negotiation_table
	is_triggered_only = yes

	option = {
		name = ADISCORD_south.40.a
		if = {
			limit = { check_variable = { var = ADISCORD_south_offer_kind value = 3 compare = equals } }
			custom_effect_tooltip = ADISCORD_south_offer_alliance_accept_tt
		}
		else_if = {
			limit = { check_variable = { var = ADISCORD_south_offer_kind value = 2 compare = equals } }
			custom_effect_tooltip = ADISCORD_south_offer_annex_accept_tt
		}
		else = {
			custom_effect_tooltip = ADISCORD_south_offer_submit_accept_tt
		}
		hidden_effect = {
			FROM = { country_event = { id = ADISCORD_south.42 hours = 1 } }
			ADISCORD_south_accept_offer = yes
		}
		ai_chance = {
			base = 10
			modifier = {
				add = 40
				strength_ratio = { tag = FROM ratio < 0.5 }
			}
			modifier = {
				add = 20
				strength_ratio = { tag = FROM ratio < 1 }
			}
			modifier = {
				add = 30
				has_country_flag = ADISCORD_south_indebted_to_SHL
				FROM = { tag = SHL }
			}
			modifier = {
				add = 40
				check_variable = { var = ADISCORD_south_offer_kind value = 3 compare = equals }
			}
			modifier = {
				add = 20
				FROM = {
					tag = NAM
					has_completed_focus = NAM_civil_dividend
				}
			}
			modifier = {
				factor = 0.5
				check_variable = { var = ADISCORD_south_offer_kind value = 2 compare = equals }
			}
		}
	}

	option = {
		name = ADISCORD_south.40.b
		if = {
			limit = { check_variable = { var = ADISCORD_south_offer_kind value = 3 compare = equals } }
			custom_effect_tooltip = ADISCORD_south_offer_alliance_refuse_tt
		}
		else = {
			custom_effect_tooltip = ADISCORD_south_offer_war_refuse_tt
		}
		hidden_effect = {
			FROM = { country_event = { id = ADISCORD_south.43 hours = 1 } }
			ADISCORD_south_refuse_offer = yes
		}
		ai_chance = {
			base = 40
			modifier = {
				factor = 0.5
				FROM = {
					tag = NAM
					has_completed_focus = NAM_armed_peace
				}
			}
		}
	}
}
