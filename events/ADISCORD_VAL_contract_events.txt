# Arms-debt instalments consume partner stocks before paying VAL.
country_event = {
	id = val_contract.30
	hidden = yes
	is_triggered_only = yes
	immediate = {
		if = {
			limit = {
				has_event_target = STP_postwar_country
				country_exists = event_target:STP_postwar_country
				event_target:STP_postwar_country = {
					num_equipment@infantry_equipment > 399
					num_equipment@support_equipment > 49
				}
			}
			event_target:STP_postwar_country = {
				add_equipment_to_stockpile = { type = infantry_equipment amount = -400 producer = ROOT }
				add_equipment_to_stockpile = { type = support_equipment amount = -50 producer = ROOT }
			}
			add_equipment_to_stockpile = { type = infantry_equipment amount = 400 producer = event_target:STP_postwar_country }
			add_equipment_to_stockpile = { type = support_equipment amount = 50 producer = event_target:STP_postwar_country }
		}
		else = {
			add_to_variable = { var = VAL_STP_missed_debt_payments value = 1 }
			VAL_change_stp_leverage = { value = 10 }
			if = { limit = { check_variable = { var = VAL_STP_missed_debt_payments value = 2 compare = greater_than_or_equals } } VAL_STP_trigger_breach = yes }
		}
	}
}
