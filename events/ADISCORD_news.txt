# Console compatibility gateways: event ADISCORD_superevent.1 / .2
# .1 starts the real Vorkerland collapse when possible, including the Unity
# Tower destruction. Once the collapse has already started it becomes a safe
# presentation replay. .2 is the Worker-victory presentation gateway used by
# the existing reunification effect.
country_event = {
	id = ADISCORD_superevent.1
	hidden = yes
	is_triggered_only = yes

	immediate = {
		if = {
			limit = {
				NOT = { has_global_flag = ADISCORD_vorkerland_collapse_started }
				WRK = { exists = yes }
			}
			WRK = { country_event = { id = ADISCORD_vorkerland_collapse.1 } }
		}
		else = {
			ADISCORD_vorkerland_clear_superevent_flags = yes
			set_global_flag = {
				flag = superevent_vorkerland_civilwar
				value = 1
				days = 20
			}
			if = {
				limit = { is_ai = no }
				scoped_sound_effect = superevent_vorkerland_civilwar_sound_e
				scoped_play_song = "one_minute_of_silence"
			}
		}
	}
}
country_event = {

news_event = { #vorkerland civilwar
	id = ADISCORD_superevent_news.1
	title = news.0.t
	desc = news.0.d

	major = yes

	# The collapse caller owns the save-safe one-shot guard. Keeping the event
	# reusable also permits a deterministic console/runtime audio check.
	fire_only_once = no

	is_triggered_only = yes

	picture = GFX_news_event_adiscord_vorkerland_explosion

	# Presentation must begin when the news is raised. Keeping this in the
	# option made the superevent flag and audio wait for a player click.
	immediate = {
		hidden_effect = {
			set_global_flag = {
				flag = superevent_vorkerland_civilwar
				value = 1
				days = 20
			}
			# A major news event keeps the country that fired it as ROOT. Dispatch
			# to the actual human country before using the scoped audio effects.
			every_country = {
				limit = { is_ai = no }
				country_event = { id = ADISCORD_superevent_audio.1 }
			}
		}
	}

	option = { name = news.0.a }
}
