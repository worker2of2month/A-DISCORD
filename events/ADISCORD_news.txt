add_namespace = ADISCORD_superevent_news
add_namespace = ADISCORD_superevent_audio

# Audio proxies for superevents raised from global scripted effects. The
# dispatching effect selects the human country; playback then happens inside
# that player's own country-event scope, matching the original working route.
country_event = {
	id = ADISCORD_superevent_audio.1
	hidden = yes
	is_triggered_only = yes

	immediate = {
		if = {
			limit = { is_ai = no }
			scoped_sound_effect = superevent_vorkerland_civilwar_sound_e
			scoped_play_song = "one_minute_of_silence"
		}
	}
}
country_event = {
	id = ADISCORD_superevent_audio.2
	hidden = yes
	is_triggered_only = yes

	immediate = {
		if = {
			limit = { is_ai = no }
			scoped_sound_effect = superevent_stelander_empire_sound_e
			scoped_play_song = "one_minute_of_silence"
		}
	}
}

news_event = { #vorkerland civilwar
	id = ADISCORD_superevent_news.1
	title = news.0.t
	desc = news.0.d

	major = yes
	
	# The collapse caller owns the save-safe one-shot guard. Keeping the event
	# reusable also permits a deterministic console/runtime audio check.
	fire_only_once = no

	is_triggered_only = yes

	picture = GFX_event_vorkerland_explosion

	# Presentation must begin when the news is raised. Keeping this in the
	# option made the superevent flag and audio wait for a player click.
	immediate = {
		hidden_effect = {
			set_global_flag = {
				flag = superevent_vorkerland_civilwar
				value = 1
				days = 20
			}
			# A major news event keeps the country that fired it as ROOT. Dispatch
			# to the actual human country before using the scoped audio effects.
			every_country = {
				limit = { is_ai = no }
				country_event = { id = ADISCORD_superevent_audio.1 }
			}
		}
	}

	option = { name = news.0.a }
}
