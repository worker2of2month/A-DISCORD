# Console compatibility gateways: event ADISCORD_superevent.1 / .2 / .3 / .4 / .5 / .6
# .1 starts the real Vorkerland collapse when possible, including the Unity
# Tower destruction. Once the collapse has already started it becomes a safe
# presentation replay. .2 is the Worker-victory presentation gateway used by
# the existing reunification effect. .3 is Anton's utilitarian victory
# presentation: picture and sound without locking the campaign ending. .4 is
# the Dirty Zone opening presentation and does not start the country cascade.
# .5 is Vlad's imperial restoration; .6 is Dorian's utilitarian victory.
# Audio is unscoped so observer/spectator hears the same track as a human.
country_event = {
	id = ADISCORD_superevent.1
	hidden = yes
	is_triggered_only = yes

	immediate = {
		if = {
			limit = {
				NOT = { has_global_flag = ADISCORD_vorkerland_collapse_started }
				WRK = { exists = yes }
			}
			WRK = { country_event = { id = ADISCORD_vorkerland_collapse.1 } }
		}
		else = {
			# superevent_vorkerland_civilwar
			set_temp_variable = { ADISCORD_superevent_request = 1 }
			ADISCORD_superevent_enqueue = yes
		}
	}
}

news_event = { #vorkerland civilwar
	id = ADISCORD_superevent_news.1
	title = news.0.t
	desc = news.0.d

	major = yes

	# The first-war announcer owns the save-safe one-shot guard. Keeping the
	# event reusable also permits a deterministic console/runtime audio check.
	fire_only_once = no

	is_triggered_only = yes

	picture = GFX_news_event_adiscord_vorkerland_explosion

	# Presentation must begin when the news is raised. Keeping this in the
	# option made the superevent flag and audio wait for a player click.
	immediate = {
		hidden_effect = {
			# superevent_vorkerland_civilwar
			set_temp_variable = { ADISCORD_superevent_request = 1 }
			ADISCORD_superevent_enqueue = yes
		}
	}

	option = {
		name = news.0.a
		# The paper covers the map. When the human closes it, pull their
		# camera to the Tower so the approach is visible.
		hidden_effect = {
			if = {
				limit = {
					is_ai = no
					has_global_flag = ADISCORD_vorkerland_unity_tower_destruction_resolved
				}
				goto_province = 16428
			}
		}
	}
}
