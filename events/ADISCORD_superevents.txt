# .1 starts the Vorkerland collapse when needed; .1-.4 otherwise only
# dispatch their presentation state.
country_event = {
	id = ADISCORD_superevent.1
	hidden = yes
	is_triggered_only = yes

	immediate = {
		if = {
			limit = {
				NOT = { has_global_flag = ADISCORD_vorkerland_collapse_started }
				WRK = { exists = yes }
			}
			WRK = { country_event = { id = ADISCORD_vorkerland_collapse.1 } }
		}
		else = {
			ADISCORD_vorkerland_clear_superevent_flags = yes
			set_global_flag = {
				flag = superevent_vorkerland_civilwar
				value = 1
				days = 20
			}
			if = {
				limit = { is_ai = no }
				scoped_sound_effect = superevent_vorkerland_civilwar_sound_e
				scoped_play_song = "one_minute_of_silence"
			}
		}
	}
}

# Vorkerland civil war
news_event = {
	id = ADISCORD_superevent_news.1
	title = news.0.t
	desc = news.0.d

	major = yes

	# The collapse subsystem owns the one-shot guard; this presentation is reusable.
	fire_only_once = no

	is_triggered_only = yes

	picture = GFX_news_event_adiscord_vorkerland_explosion

	immediate = {
		hidden_effect = {
			set_global_flag = {
				flag = superevent_vorkerland_civilwar
				value = 1
				days = 20
			}
			# Major news keeps the firing country as ROOT; audio must run in human scopes.
			every_country = {
				limit = { is_ai = no }
				country_event = { id = ADISCORD_superevent_audio.1 }
			}
		}
	}

	option = { name = news.0.a }
}
