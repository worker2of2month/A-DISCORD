# League invitation. ROOT is the invited state, FROM is Itora. The answer
# rechecks both sides because the faction may have changed in the meantime;
# the League is founded by its first member's signature.
country_event = {
	id = ADISCORD_west.40
	title = ADISCORD_west.40.t
	desc = ADISCORD_west.40.d
	picture = GFX_event_adiscord_negotiation_table
	is_triggered_only = yes

	option = {
		name = ADISCORD_west.40.a
		custom_effect_tooltip = ADISCORD_west_league_accept_tt
		hidden_effect = {
			clr_country_flag = ADISCORD_west_league_offer_pending
			if = {
				limit = {
					is_subject = no
					is_in_faction = no
					NOT = { has_war_with = FROM }
					NOT = { has_global_flag = ADISCORD_west_final_started }
					FROM = {
						exists = yes
						is_subject = no
						OR = {
							is_in_faction = no
							is_faction_leader = yes
						}
					}
				}
				FROM = {
					if = {
						limit = { is_in_faction = no }
						create_faction_from_template = {
							template = faction_template_ADISCORD_standard
							name = ADISCORD_west_ivn_faction
						}
					}
					add_to_faction = ROOT
					country_event = { id = ADISCORD_west.41 hours = 1 }
				}
				ADISCORD_west_crisis_add = yes
			}
		}
		ai_chance = {
			base = 10
			modifier = {
				add = 50
				ADISCORD_west_leans_to_ivn_ally = yes
			}
			modifier = {
				add = 30
				check_variable = { var = ADISCORD_west_lean value = -70 compare = less_than_or_equals }
			}
		}
	}

	option = {
		name = ADISCORD_west.40.b
		custom_effect_tooltip = ADISCORD_west_league_refuse_tt
		hidden_effect = {
			clr_country_flag = ADISCORD_west_league_offer_pending
			FROM = { country_event = { id = ADISCORD_west.42 hours = 1 } }
		}
		ai_chance = {
			base = 30
			modifier = {
				add = 60
				NOT = { ADISCORD_west_leans_to_ivn_ally = yes }
			}
		}
	}
}
