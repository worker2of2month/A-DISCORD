# A western state without a champion's subject or faction tie chooses its side.
# There is no neutral answer: the corridor is the only land route. AI weights
# mirror each other: every Vorkerland route has its own constituency, and the
# clearly stronger champion draws hesitant neighbours.
country_event = {
	id = ADISCORD_west.10
	title = ADISCORD_west.10.t
	desc = ADISCORD_west.10.d
	picture = GFX_event_adiscord_military_planning
	is_triggered_only = yes
	trigger = {
		ADISCORD_west_final_active = yes
		ADISCORD_west_needs_side_choice = yes
	}

	option = {
		name = ADISCORD_west.10.a
		custom_effect_tooltip = ADISCORD_west_join_wrk_tt
		hidden_effect = {
			ADISCORD_west_join_wrk_side = yes
			ADISCORD_west_faction_partners_follow = yes
		}
		ai_chance = {
			base = 40
			# The restored central command and the reconstruction directorate
			# both reattach the former administrative districts.
			modifier = {
				add = 40
				OR = {
					tag = PWR
					tag = ZAO
					tag = PSD
				}
				WRK = {
					OR = {
						has_country_flag = ADISCORD_vorkerland_route_utilitarian
						has_country_flag = ADISCORD_vorkerland_route_joint
					}
				}
			}
			modifier = {
				add = 30
				OR = {
					tag = WPA
					tag = WPS
				}
				WRK = { has_country_flag = ADISCORD_vorkerland_route_worker }
			}
			modifier = {
				add = 40
				OR = {
					tag = ROM
					tag = TRU
				}
				WRK = { has_country_flag = ADISCORD_vorkerland_free_republics_recognized }
			}
			modifier = {
				add = 20
				is_neighbor_of = WRK
			}
			modifier = {
				add = 30
				WRK = {
					strength_ratio = {
						tag = IVN
						ratio > 1.25
					}
				}
			}
			modifier = {
				factor = 0.5
				OR = {
					tag = ROM
					tag = TRU
				}
				has_global_flag = ADISCORD_vorkerland_free_republics_guaranteed_by_ivn
			}
		}
	}

	option = {
		name = ADISCORD_west.10.b
		custom_effect_tooltip = ADISCORD_west_join_ivn_tt
		hidden_effect = {
			ADISCORD_west_join_ivn_side = yes
			ADISCORD_west_faction_partners_follow = yes
		}
		ai_chance = {
			base = 40
			modifier = {
				add = 50
				OR = {
					tag = ROM
					tag = TRU
				}
				has_global_flag = ADISCORD_vorkerland_free_republics_guaranteed_by_ivn
			}
			modifier = {
				add = 40
				OR = {
					tag = WIT
					tag = IBA
					tag = IBL
				}
			}
			# Only the republics fear the military government more than the war.
			modifier = {
				add = 30
				OR = {
					tag = ROM
					tag = TRU
					tag = WPA
					tag = WPS
				}
				WRK = { has_country_flag = ADISCORD_vorkerland_route_joint }
			}
			modifier = {
				add = 20
				is_neighbor_of = IVN
			}
			modifier = {
				add = 30
				IVN = {
					strength_ratio = {
						tag = WRK
						ratio > 1.25
					}
				}
			}
		}
	}
}

# League invitation. ROOT is the invited state, FROM is Itora. The answer
# rechecks both sides because the faction may have changed in the meantime.
country_event = {
	id = ADISCORD_west.40
	title = ADISCORD_west.40.t
	desc = ADISCORD_west.40.d
	picture = GFX_event_adiscord_negotiation_table
	is_triggered_only = yes

	option = {
		name = ADISCORD_west.40.a
		custom_effect_tooltip = ADISCORD_west_league_accept_tt
		hidden_effect = {
			clr_country_flag = ADISCORD_west_league_offer_pending
			if = {
				limit = {
					is_subject = no
					is_in_faction = no
					FROM = {
						exists = yes
						is_faction_leader = yes
						NOT = { has_war_with = ROOT }
					}
				}
				FROM = {
					add_to_faction = ROOT
					country_event = { id = ADISCORD_west.41 hours = 1 }
				}
				ADISCORD_west_crisis_add = yes
			}
		}
		ai_chance = {
			base = 40
			modifier = {
				add = 40
				has_global_flag = ADISCORD_vorkerland_free_republics_guaranteed_by_ivn
				OR = {
					tag = ROM
					tag = TRU
				}
			}
			modifier = {
				add = 30
				tag = WIT
			}
		}
	}

	option = {
		name = ADISCORD_west.40.b
		custom_effect_tooltip = ADISCORD_west_league_refuse_tt
		hidden_effect = {
			clr_country_flag = ADISCORD_west_league_offer_pending
			FROM = { country_event = { id = ADISCORD_west.42 hours = 1 } }
		}
		ai_chance = {
			base = 20
			modifier = {
				add = 60
				OR = {
					tag = ROM
					tag = TRU
				}
				WRK = {
					exists = yes
					has_country_flag = ADISCORD_vorkerland_free_republics_recognized
				}
			}
			modifier = {
				add = 30
				WRK = {
					exists = yes
					strength_ratio = {
						tag = IVN
						ratio > 1.25
					}
				}
			}
		}
	}
}
