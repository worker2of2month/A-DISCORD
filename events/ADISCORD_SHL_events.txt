# Hidden dispatcher: one chapter every 90 days for a human ruler.
# AI or an inoperable country skips the chapter and keeps the clock running.
country_event = {
	id = ADISCORD_shl_book.99
	hidden = yes
	is_triggered_only = yes
	trigger = { tag = SHL }

	immediate = {
		if = {
			limit = {
				is_ai = no
				SHL_country_operable = yes
			}
			if = {
				limit = { check_variable = { var = SHL_book_next value = 1 compare = equals } }
				country_event = { id = ADISCORD_shl_book.1 }
			}
			else_if = {
				limit = { check_variable = { var = SHL_book_next value = 2 compare = equals } }
				country_event = { id = ADISCORD_shl_book.2 }
			}
			else_if = {
				limit = { check_variable = { var = SHL_book_next value = 3 compare = equals } }
				country_event = { id = ADISCORD_shl_book.3 }
			}
			else_if = {
				limit = { check_variable = { var = SHL_book_next value = 4 compare = equals } }
				country_event = { id = ADISCORD_shl_book.4 }
			}
			else_if = {
				limit = { check_variable = { var = SHL_book_next value = 5 compare = equals } }
				country_event = { id = ADISCORD_shl_book.5 }
			}
			add_to_variable = { var = SHL_book_next value = 1 }
		}
		if = {
			limit = { check_variable = { var = SHL_book_next value = 6 compare = less_than } }
			country_event = {
				id = ADISCORD_shl_book.99
				days = 90
			}
		}
	}
}
