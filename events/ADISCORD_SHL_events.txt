country_event = {
	id = ADISCORD_SHL.7
	title = ADISCORD_SHL.7.t
	desc = ADISCORD_SHL.7.d
	is_triggered_only = yes
	trigger = { tag = SHL }
	option = {
		name = ADISCORD_SHL.7.a
		default_option = yes
		custom_effect_tooltip = SHL_masters_wait_tt
		hidden_effect = {
			if = {
				limit = { NOT = { has_completed_focus = SHL_school_of_masters } }
				set_country_flag = SHL_master_dispute
			}
		}
	}
	option = {
		name = ADISCORD_SHL.7.b
		trigger = {
			NOT = { has_completed_focus = SHL_school_of_masters }
			check_variable = { var = ADISCORD_economy_treasury value = 40 compare = greater_than_or_equals }
		}
		custom_effect_tooltip = SHL_masters_pay_tt
		hidden_effect = {
			if = {
				limit = {
					NOT = { has_completed_focus = SHL_school_of_masters }
					check_variable = { var = ADISCORD_economy_treasury value = 40 compare = greater_than_or_equals }
				}
				subtract_from_variable = { var = ADISCORD_economy_treasury value = 40 }
				clr_country_flag = SHL_master_dispute
			}
		}
	}
}

			}
		}
	}
	option = {
		name = ADISCORD_SHL.8.b
		trigger = {
			NOT = { has_completed_focus = SHL_houses_railway }
			has_political_power = 15
		}
		custom_effect_tooltip = SHL_suppliers_pay_tt
		hidden_effect = {
			if = {
				limit = {
					NOT = { has_completed_focus = SHL_houses_railway }
					has_political_power = 15
				}
				add_political_power = -15
				clr_country_flag = SHL_supplier_premium
			}
		}
	}
}
