	id = stp_crisis.20
	hidden = yes
	is_triggered_only = yes

	trigger = { has_country_flag = STP_operation_token_palace_channel }

	immediate = {
		set_temp_variable = { var = STP_operation_result value = 1 }
		if = {
			limit = { check_variable = { var = STP_security_posture value = 1 compare = equals } }
			set_temp_variable = { var = STP_operation_result value = 2 }
		}
		else_if = {
			limit = { check_variable = { var = STP_security_posture value = 2 compare = equals } }
			if = {
				limit = { has_country_flag = STP_focus_palace_observed }
				set_temp_variable = { var = STP_operation_result value = 0 }
			}
			else = { set_temp_variable = { var = STP_operation_result value = -1 } }
		}

		if = {
			limit = { check_variable = { var = STP_security_adaptation_palace value = 1 compare = equals } }
			STP_change_suspicion = { value = 5 }
		}
		if = {
			limit = { check_variable = { var = STP_operation_result value = 2 compare = greater_than_or_equals } }
			STP_change_node_palace = { value = 2 }
			STP_change_readiness = { value = 8 }
		}
		else_if = {
			limit = { check_variable = { var = STP_operation_result value = 1 compare = greater_than_or_equals } }
			STP_change_node_palace = { value = 1 }
			STP_change_readiness = { value = 5 }
		}
		else_if = {
			limit = { check_variable = { var = STP_operation_result value = 0 compare = greater_than_or_equals } }
			STP_change_node_palace = { value = 1 }
			STP_change_suspicion = { value = 10 }
		}
		else = {
			STP_change_suspicion = { value = 12 }
			STP_change_readiness = { value = -5 }
		}

		if = {
			limit = {
				check_variable = { var = STP_operation_result value = 0 compare = greater_than_or_equals }
				has_country_flag = STP_focus_final_palace_move_open
				check_variable = { var = STP_node_palace value = 2 compare = equals }
				check_variable = { var = STP_party_suspicion value = 35 compare = less_than_or_equals }
			}
			set_country_flag = STP_capital_guard_loyal_to_resistance
		}
		if = {
			limit = {
				check_variable = { var = STP_operation_result value = 0 compare = less_than_or_equals }
				check_variable = { var = STP_party_suspicion value = 90 compare = greater_than_or_equals }
			}
			STP_lose_shabrat = yes
		}
		STP_change_security_adaptation_palace = { value = 1 }
		set_temp_variable = { var = STP_operation_cleanup_code value = 20 }
		STP_clear_operation_slot = yes
	}
}
