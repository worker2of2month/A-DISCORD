country_event = {
	id = ADISCORD_SLI.2
	title = ADISCORD_SLI.2.t
	desc = ADISCORD_SLI.2.d
	picture = GFX_event_adiscord_negotiation_table
	is_triggered_only = yes
	option = {
		name = ADISCORD_SLI.2.a
		trigger = { VAL = { VAL_island_treaty_available = yes } }
		ai_chance = { factor = 80 }
		custom_effect_tooltip = SLI_join_kefreyt_tt
		hidden_effect = {
			VAL = {
				clr_country_flag = VAL_island_treaty_rejected
				annex_country = { target = SLI transfer_troops = yes }
				ADISCORD_economy_mark_dirty = yes
				country_event = { id = ADISCORD_SLI.3 hours = 1 }
			}
		}
	}
	option = {
		name = ADISCORD_SLI.2.b
		trigger = { VAL = { VAL_island_treaty_available = yes } }
		ai_chance = { factor = 20 }
		custom_effect_tooltip = SLI_refuse_kefreyt_tt
		hidden_effect = {
			VAL = {
				set_country_flag = VAL_island_treaty_rejected
				country_event = { id = ADISCORD_SLI.3 hours = 1 }
			}
		}
	}
	option = {
		name = ADISCORD_SLI.2.c
		trigger = { VAL = { VAL_island_treaty_available = no } }
		hidden_effect = {
			if = {
				limit = {
					exists = yes
					VAL = { NOT = { owns_state = 709 } }
				}
				VAL = { set_country_flag = VAL_island_treaty_rejected }
			}
		}
	}
}
