# Fired in the already-normalized winning claimant's scope when the first rival
# claimant capitulates. Neither choice owns peace processing or the map.
country_event = {
	id = ADISCORD_vorkerland_story.5
	title = ADISCORD_vorkerland_story.5.t
	desc = ADISCORD_vorkerland_story.5.d
	picture = GFX_report_event_generic_political_reform
	is_triggered_only = yes
	fire_only_once = yes

	trigger = {
		OR = { tag = WKR tag = VAD tag = TVA tag = WRK }
		has_global_flag = ADISCORD_vorkerland_story_first_claimant_capitulated
		NOT = { has_country_flag = ADISCORD_vorkerland_story_command_choice_resolved }
	}

	option = {
		name = ADISCORD_vorkerland_story.5.a
		set_country_flag = ADISCORD_vorkerland_story_rival_headquarters_integrated
		set_country_flag = ADISCORD_vorkerland_story_command_choice_resolved
		add_command_power = 10
		army_experience = 5
		add_war_support = 0.01
		ai_chance = { factor = 55 }
	}
	option = {
		name = ADISCORD_vorkerland_story.5.b
		set_country_flag = ADISCORD_vorkerland_story_rival_apparatus_dissolved
		set_country_flag = ADISCORD_vorkerland_story_command_choice_resolved
		add_political_power = 25
		add_stability = 0.02
		add_war_support = -0.01
		ai_chance = { factor = 45 }
	}
}
