country_event = {
	id = ADISCORD_vorkerland_collapse.1
	hidden = yes
	is_triggered_only = yes
	fire_only_once = yes

	trigger = {
		tag = WRK
		ADISCORD_vorkerland_collapse_not_started = yes
	}

	immediate = {
		set_global_flag = ADISCORD_vorkerland_collapse_started
		goto_province = 6713
		launch_nuke = {
			province = 6713
			use_nuke = no
		}
		32 = {
			damage_building = {
				type = infrastructure
				damage = 0.1
			}
			damage_building = {
				type = rail_way
				damage = 0.1
			}
			damage_building = {
				type = anti_air_building
				damage = 0.1
			}
		}
		random_list = {
			60 = { set_global_flag = ADISCORD_vorkerland_worker_rescued_by_vlad }
			30 = { set_global_flag = ADISCORD_vorkerland_worker_missing }
			10 = { set_global_flag = ADISCORD_vorkerland_worker_killed }
		}
		ADISCORD_vorkerland_teardown_confederation = yes
		ADISCORD_vorkerland_apply_initial_map = yes
		country_event = {
			id = ADISCORD_vorkerland_collapse.2
			days = 1
		}
		if = {
			limit = { RUS = { exists = yes } }
			RUS = {
				country_event = {
					id = ADISCORD_vorkerland_collapse.10
					days = 60
					random_days = 30
				}
			}
		}
		news_event = {
			id = news.0
			hours = 1
		}
	}
}
