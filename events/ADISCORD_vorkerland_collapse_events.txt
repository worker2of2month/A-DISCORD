country_event = {
	id = ADISCORD_vorkerland_collapse.1
	hidden = yes
	is_triggered_only = yes
	fire_only_once = yes

	trigger = {
		tag = WRK
		ADISCORD_vorkerland_collapse_not_started = yes
	}

	immediate = {
		ADISCORD_vorkerland_begin_collapse = yes
		set_global_flag = ADISCORD_vorkerland_collapse_started
		goto_province = 6713
		launch_nuke = {
			province = 6713
			use_nuke = no
		}
		32 = {
			damage_building = {
				type = infrastructure
				damage = 0.1
			}
			damage_building = {
				type = rail_way
				damage = 0.1
			}
			damage_building = {
				type = anti_air_building
				damage = 0.1
			}
		}
		random_list = {
			60 = { set_global_flag = ADISCORD_vorkerland_worker_safe_with_loyalists }
			25 = { set_global_flag = ADISCORD_vorkerland_worker_rescued_by_vlad }
			10 = { set_global_flag = ADISCORD_vorkerland_worker_missing }
			5 = { set_global_flag = ADISCORD_vorkerland_worker_killed }
		}
		ADISCORD_vorkerland_apply_initial_map = yes
		ADISCORD_vorkerland_teardown_confederation = yes
		ADISCORD_vorkerland_prepare_claimant_characters = yes
		ADISCORD_vorkerland_align_wtd_with_worx = yes
		# Only the human player follows the live country tag. Observers and AI
		# keep WRK as an AI-side legacy object; both paths still materialize WKR.
		if = {
			limit = { is_ai = no country_exists = WKR }
			WKR = { change_tag_from = WRK }
		}
		if = {
			limit = { country_exists = WKR country_exists = WRK }
			WKR = {
				inherit_technology = WRK
				load_focus_tree = { tree = ADISCORD_vorkerland_civil_war_focus keep_completed = yes copy_completed_from = WRK }
				annex_country = { target = WRK transfer_troops = yes }
				ADISCORD_vorkerland_relocate_legacy_armies = yes
				ADISCORD_vorkerland_apply_claimant_cosmetics = yes
				country_event = { id = ADISCORD_vorkerland_collapse.41 days = 1 }
				country_event = { id = ADISCORD_vorkerland_collapse.2 days = 1 }
				country_event = { id = ADISCORD_vorkerland_phase.2 days = 1 }
			}
		}
		else = {
			WRK = {
				ADISCORD_vorkerland_apply_claimant_cosmetics = yes
				country_event = { id = ADISCORD_vorkerland_collapse.41 days = 1 }
				country_event = { id = ADISCORD_vorkerland_collapse.2 days = 1 }
			}
		}
		random_list = {
			7 = { ADISCORD_vorkerland_form_joint_government = yes }
			93 = { }
		}
		# The end of the unified state is the opening news of the collapse, not
		# a verdict delivered years into its wars. Keep an explicit save-safe
		# guard in addition to the news event's own fire_only_once contract.
		if = {
			limit = { NOT = { has_global_flag = ADISCORD_vorkerland_collapse_news_shown } }
			set_global_flag = ADISCORD_vorkerland_collapse_news_shown
			news_event = { id = news.0 }
		}
		if = {
			limit = { RUS = { exists = yes } }
			RUS = {
				country_event = {
					id = ADISCORD_vorkerland_collapse.10
					days = 60
					random_days = 30
				}
			}
		}
	}
}

	trigger = {
		ADISCORD_vorkerland_collapse_wars_not_started = yes
	}

	immediate = {
		# The first teardown runs in the same tick as the partition. Repeat it one
		# day later, after released-country faction membership has entered the cache.
		ADISCORD_vorkerland_teardown_confederation = yes
		set_global_flag = ADISCORD_vorkerland_collapse_wars_started
		# The claimant spirits are asserted at the actual war-outbreak stage,
		# after the initial partition and political cache rebuild. This prevents
		# WRK's three obsolete starting spirits from resurfacing in live play.
		if = {
			limit = { WRK = { exists = yes } }
			WRK = { ADISCORD_vorkerland_repair_claimant_spirit_progression = yes }
		}
		if = {
			limit = { WKR = { exists = yes } }
			WKR = { ADISCORD_vorkerland_repair_claimant_spirit_progression = yes }
		}
		if = {
			limit = { TVA = { exists = yes } }
			TVA = { ADISCORD_vorkerland_repair_claimant_spirit_progression = yes }
		}
		set_global_flag = ADISCORD_vorkerland_claimant_spirit_progression_v3
		if = {
			limit = { VAL = { exists = yes } }
			VAL = { VAL_handle_vorkerland_war_outbreak = yes }
		}
		if = {
			limit = { WKR = { exists = yes } }
			WKR = {
				country_event = {
					id = ADISCORD_vorkerland_collapse.31
					days = 21
					random_days = 14
				}
			}
		}
		else_if = {
			limit = { WRK = { exists = yes } }
			WRK = { country_event = { id = ADISCORD_vorkerland_collapse.31 days = 21 random_days = 14 } }
		}
	}

# The timed decision delegates the actual declaration here. If either claimant
# still has an inherited faction, detach both and retry after the cache advances
# by a day. The showdown flag is set only on the path that actually declares war.
country_event = {
	id = ADISCORD_vorkerland_collapse.48
	hidden = yes
	is_triggered_only = yes

	trigger = { tag = WRK }

	immediate = {
		if = {
			limit = {
				has_global_flag = ADISCORD_vorkerland_worker_doctor_front_preparation
				NOT = { has_global_flag = ADISCORD_vorkerland_central_war_finished }
				country_exists = TVA
				is_subject = no
				NOT = { has_capitulated = yes }
				NOT = { has_war_with = TVA }
				TVA = { is_subject = no NOT = { has_capitulated = yes } }
			}
			if = {
				limit = {
					OR = {
						is_in_faction = yes
						TVA = { is_in_faction = yes }
					}
				}
				ADISCORD_vorkerland_detach_worker_doctor_factions = yes
				# One bounded retry is enough for leave_faction/dismantle_faction to
				# enter the diplomatic cache; there is no permanent daily pulse.
				country_event = { id = ADISCORD_vorkerland_collapse.49 days = 1 }
			}
			else = {
				ADISCORD_vorkerland_launch_worker_doctor_war = yes
			}
		}
		else = {
			clr_global_flag = ADISCORD_vorkerland_worker_doctor_front_preparation
		}
	}
}
