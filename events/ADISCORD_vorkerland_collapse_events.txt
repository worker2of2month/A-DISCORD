add_namespace = ADISCORD_vorkerland_collapse

country_event = {
	id = ADISCORD_vorkerland_collapse.1
	hidden = yes
	is_triggered_only = yes
	fire_only_once = yes

	trigger = {
		tag = WRK
		ADISCORD_vorkerland_collapse_not_started = yes
	}

	immediate = {
		set_global_flag = ADISCORD_vorkerland_collapse_started
		goto_province = 6713
		launch_nuke = {
			province = 6713
			use_nuke = no
		}
		32 = {
			damage_building = {
				type = infrastructure
				damage = 0.1
			}
			damage_building = {
				type = rail_way
				damage = 0.1
			}
			damage_building = {
				type = anti_air_building
				damage = 0.1
			}
		}
		random_list = {
			60 = { set_global_flag = ADISCORD_vorkerland_worker_safe_with_loyalists }
			25 = { set_global_flag = ADISCORD_vorkerland_worker_rescued_by_vlad }
			10 = { set_global_flag = ADISCORD_vorkerland_worker_missing }
			5 = { set_global_flag = ADISCORD_vorkerland_worker_killed }
		}
		ADISCORD_vorkerland_apply_initial_map = yes
		ADISCORD_vorkerland_teardown_confederation = yes
		ADISCORD_vorkerland_apply_claimant_cosmetics = yes
		ROM = {
			set_country_leader_portrait = {
				ideology = humanism
				portrait = GFX_portrait_ROM_Erwin_Von_Romanovskiy_civilwar
			}
		}
		TRU = {
			set_country_leader_portrait = {
				ideology = pragmatism
				portrait = GFX_portrait_TRU_Nikita_Truman_civilwar
			}
		}
		random_list = {
			7 = { ADISCORD_vorkerland_form_joint_government = yes }
			93 = { }
		}
		# The end of the unified state is the opening news of the collapse, not
		# a verdict delivered years into its wars. Keep an explicit save-safe
		# guard in addition to the news event's own fire_only_once contract.
		if = {
			limit = { NOT = { has_global_flag = ADISCORD_vorkerland_collapse_news_shown } }
			set_global_flag = ADISCORD_vorkerland_collapse_news_shown
			news_event = { id = news.0 }
		}
		country_event = {
			id = ADISCORD_vorkerland_collapse.2
			days = 1
		}
		if = {
			limit = { RUS = { exists = yes } }
			RUS = {
				country_event = {
					id = ADISCORD_vorkerland_collapse.10
					days = 60
					random_days = 30
				}
			}
		}
	}
}

country_event = {
	id = ADISCORD_vorkerland_collapse.2
	hidden = yes
	is_triggered_only = yes
	fire_only_once = yes

	trigger = {
		ADISCORD_vorkerland_collapse_wars_not_started = yes
	}

	immediate = {
		set_global_flag = ADISCORD_vorkerland_collapse_wars_started
		if = {
			limit = { VAL = { exists = yes } }
			VAL = { VAL_handle_vorkerland_war_outbreak = yes }
		}
		if = {
			limit = { WRK = { exists = yes } }
			WRK = {
				country_event = {
					id = ADISCORD_vorkerland_collapse.31
					days = 21
					random_days = 14
				}
			}
		}
		if = {
			limit = { RUS = { exists = yes } }
			RUS = {
				country_event = {
					id = ADISCORD_vorkerland_collapse.30
					days = 90
					random_days = 180
				}
			}
		}
	}
}

country_event = {
	id = ADISCORD_vorkerland_collapse.30
	hidden = yes
	is_triggered_only = yes
	fire_only_once = yes
	trigger = {
		tag = RUS
		has_global_flag = ADISCORD_vorkerland_collapse_wars_started
		NOT = { has_global_flag = ADISCORD_vorkerland_ivanland_intervention_resolved }
		NOT = { has_global_flag = ADISCORD_vorkerland_ivanland_intervention_available }
	}
	immediate = {
		if = {
			limit = {
				IVN = { exists = yes }
				PWR = { exists = yes is_subject = no controls_state = 90 }
				IBL = { exists = yes is_subject = no controls_state = 91 }
			}
			random_list = {
				35 = { set_global_flag = ADISCORD_vorkerland_ivanland_intervention_available }
				65 = {
					set_global_flag = ADISCORD_vorkerland_ivanland_intervention_resolved
					set_global_flag = ADISCORD_vorkerland_ivanland_stayed_out
				}
			}
		}
		else = {
				set_global_flag = ADISCORD_vorkerland_ivanland_intervention_resolved
				set_global_flag = ADISCORD_vorkerland_ivanland_stayed_out
		}
	}
}

# creating two closed-zone countries in one effect can race that rebuild.
