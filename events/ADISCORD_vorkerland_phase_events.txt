country_event = {
	id = ADISCORD_vorkerland_phase.6
	hidden = yes
	is_triggered_only = yes

	trigger = {
		has_global_flag = ADISCORD_vorkerland_phase_reunification
		has_global_flag = ADISCORD_vorkerland_central_showdown_started
		ADISCORD_vorkerland_central_showdown_edges_verified = yes
		NOT = { has_global_flag = ADISCORD_vorkerland_central_showdown_launch_failed }
		ADISCORD_vorkerland_has_single_surviving_claimant = yes
	}

	immediate = {
		if = {
			limit = { WKR = { exists = yes is_subject = no NOT = { has_capitulated = yes } } }
			WRK = {
				ADISCORD_vorkerland_form_wrk_from_wkr = yes
				country_event = { id = ADISCORD_vorkerland_phase.7 days = 1 }
			}
		}
		else_if = {
			limit = { VAD = { exists = yes is_subject = no NOT = { has_capitulated = yes } } }
			WRK = {
				ADISCORD_vorkerland_form_wrk_from_vad = yes
				country_event = { id = ADISCORD_vorkerland_phase.7 days = 1 }
			}
		}
		else_if = {
			limit = { TVA = { exists = yes is_subject = no NOT = { has_capitulated = yes } } }
			WRK = {
				ADISCORD_vorkerland_form_wrk_from_tva = yes
				country_event = { id = ADISCORD_vorkerland_phase.7 days = 1 }
			}
		}
	}
}

country_event = {
	id = ADISCORD_vorkerland_phase.7
	hidden = yes
	is_triggered_only = yes

	immediate = {
		ADISCORD_vorkerland_verify_reunified_wrk = yes
		if = {
			limit = {
				NOT = { ADISCORD_vorkerland_reunification_verified = yes }
				NOT = { has_global_flag = ADISCORD_vorkerland_reunification_retry }
			}
			set_global_flag = ADISCORD_vorkerland_reunification_retry
			country_event = { id = ADISCORD_vorkerland_phase.7 days = 1 }
		}
		else_if = {
			limit = { NOT = { ADISCORD_vorkerland_reunification_verified = yes } }
			set_global_flag = ADISCORD_vorkerland_reunification_failed
			log = "[ADISCORD][VORKERLAND][RECOVERY] Reunified WRK failed formation postconditions after one retry."
		}
	}
}
