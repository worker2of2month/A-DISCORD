	hidden = yes
	is_triggered_only = yes

	immediate = {
		ADISCORD_vorkerland_verify_reunified_wrk = yes

			limit = {
				NOT = { ADISCORD_vorkerland_reunification_verified = yes }
				NOT = { has_global_flag = ADISCORD_vorkerland_reunification_retry }
			}
			set_global_flag = ADISCORD_vorkerland_reunification_retry
			country_event = { id = ADISCORD_vorkerland_phase.7 days = 1 }
		}
		else_if = {
			limit = { NOT = { ADISCORD_vorkerland_reunification_verified = yes } }
			set_global_flag = ADISCORD_vorkerland_reunification_failed
			log = "[ADISCORD][VORKERLAND][RECOVERY] Reunified WRK failed formation postconditions after one retry."
		}
	}
